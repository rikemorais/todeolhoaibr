1//! Renda domiciliar per capita dos municípios (IBGE/Censo via API SIDRA).
2//!
3//! A API SIDRA (`apisidra.ibge.gov.br/values`) publica o rendimento nominal médio
4//! e mediano mensal domiciliar per capita por município, em tabelas distintas por
5//! edição do Censo. Os censos de 2010 (tabela 3578) e 2022 (tabela 10295) estão
6//! disponíveis no nível municipal (N6); os de 1991 e 2000 não estão no SIDRA.
7//!
8//! Cada edição do censo é uma tarefa própria. A resposta JSON fica opaca na Raw;
9//! o refino tipará os valores na Silver.
10
11use std::time::Duration;
12
13use todeolho_nucleo::{FonteDeDados, LimiteHttp, PerfilHttp, TarefaPlanejada};
14
15use crate::ingestao_http::TarefaDeIngestaoHttp;
16
17const NOME: &str = "ibge-renda-per-capita";
18const FONTE_RAW: &str = "ibge";
19const BASE: &str = "https://apisidra.ibge.gov.br/values";
20
21/// Censo 2010 — tabela 3578 (rendimento domiciliar per capita, amostra).
22const TABELA_2010: u32 = 3578;
23/// Variáveis de 2010: média (2012), mediana (2013), moradores (2035).
24const VARIAVEIS_2010: &str = "2012,2013,2035";
25
26/// Censo 2022 — tabela 10295 (rendimento domiciliar per capita, amostra).
27const TABELA_2022: u32 = 10295;
28/// Variáveis de 2022: média (13431), mediana (13534), moradores (13604).
29const VARIAVEIS_2022: &str = "13431,13534,13604";
30/// Classificações fixadas em "Total" para 2022 (sexo, faixa etária, cor/raça).
31const CLASSIFICACOES_2022: &str = "/c2/6794/c287/95253/c86/95251";
32
33const TAMANHO_TIPICO: u64 = 2_000_000;
34
35fn url_2010() -> String {
36    format!("{BASE}/t/{TABELA_2010}/n6/all/v/{VARIAVEIS_2010}/p/all")
37}
38
39fn url_2022() -> String {
40    format!("{BASE}/t/{TABELA_2022}/n6/all/v/{VARIAVEIS_2022}/p/all{CLASSIFICACOES_2022}")
41}
42
43fn tarefa(censo: u16, url: String) -> TarefaPlanejada {
44    let recurso = format!("renda-per-capita/censo={censo}");
45    TarefaPlanejada::nova(
46        TarefaDeIngestaoHttp::sem_paginacao(FONTE_RAW, &recurso, url)
47            .com_tamanho_tipico(TAMANHO_TIPICO),
48    )
49    .com_etiquetas([
50        ("censo", censo.to_string().as_str()),
51        ("indicador", "renda-per-capita"),
52    ])
53}

55/// Renda domiciliar per capita dos municípios (IBGE/Censo).
56///
57/// Captura o rendimento nominal médio e mediano mensal domiciliar per capita, por
58/// município (N6), dos Censos 2010 e 2022 via API SIDRA. Os Censos de 1991 e 2000
59/// não estão disponíveis no SIDRA no nível municipal. Fonte separada e `sob_demanda`
60/// (duas tarefas). Os dados ficam opacos na Raw ([ADR 0014]); a tipagem e o
61/// cruzamento com dados eleitorais são do refino ([ADR 0015]).
62///
63/// [ADR 0014]: ../../../docs/adr/0014-raw-opaca-e-refino-em-estagio-separado.md
64/// [ADR 0015]: ../../../docs/adr/0015-arquitetura-de-dados-em-medalhao.md
65pub struct FonteIbgeRendaPerCapita;
66
67impl FonteDeDados for FonteIbgeRendaPerCapita {
68    fn nome(&self) -> &str {
69        NOME
70    }
71
72    fn descricao(&self) -> &str {
73        "IBGE - Rendimento domiciliar per capita por município (Censos 2010 e 2022, API SIDRA)"
74    }
75
76    fn sob_demanda(&self) -> bool {
77        true
78    }
79
80    fn perfil_http(&self) -> PerfilHttp {
81        PerfilHttp {
82            credencial: None,
83            limite: Some(LimiteHttp {
84                host: "apisidra.ibge.gov.br",
85                requisicoes: 2,
86                intervalo: Duration::from_secs(1),
87            }),
88        }
89    }
90
91    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
92        vec![tarefa(2010, url_2010()), tarefa(2022, url_2022())]
93    }
94}

96#[cfg(test)]
97mod testes {
98    use std::collections::BTreeSet;
99
100    use super::*;
101
102    #[test]
103    fn cobre_os_censos_2010_e_2022() {
104        let censos: BTreeSet<u16> = FonteIbgeRendaPerCapita
105            .criar_tarefas()
106            .iter()
107            .map(|t| t.etiquetas["censo"].parse().expect("censo numérico"))
108            .collect();
109        assert_eq!(censos, [2010, 2022].into());
110    }
111
112    #[test]
113    fn a_url_de_2010_pede_todos_os_municipios() {
114        let url = url_2010();
115        assert!(url.contains("/t/3578/"), "tabela 3578");
116        assert!(url.contains("/n6/all"), "nível municipal");
117        assert!(url.contains("/v/2012,2013,2035"), "três variáveis");
118    }
119
120    #[test]
121    fn a_url_de_2022_inclui_classificacoes_de_total() {
122        let url = url_2022();
123        assert!(url.contains("/t/10295/"), "tabela 10295");
124        assert!(url.contains("/n6/all"), "nível municipal");
125        assert!(url.contains("/v/13431,13534,13604"), "três variáveis");
126        assert!(url.contains("/c2/6794"), "sexo = Total");
127        assert!(url.contains("/c287/95253"), "faixa etária = Total");
128        assert!(url.contains("/c86/95251"), "cor/raça = Total");
129    }
130
131    #[test]
132    fn cada_tarefa_tem_censo_coerente_com_o_recurso() {
133        for tarefa in FonteIbgeRendaPerCapita.criar_tarefas() {
134            let nome = tarefa.tarefa.nome();
135            let censo = &tarefa.etiquetas["censo"];
136            assert_eq!(nome, format!("renda-per-capita/censo={censo}"));
137        }
138    }
139
140    #[test]
141    fn declara_o_limite_do_host_do_sidra() {
142        let limite = FonteIbgeRendaPerCapita
143            .perfil_http()
144            .limite
145            .expect("limite do SIDRA");
146        assert_eq!(limite.host, "apisidra.ibge.gov.br");
147    }
148
149    #[test]
150    fn nao_entra_em_todas_por_acidente() {
151        assert!(FonteIbgeRendaPerCapita.sob_demanda());
152    }
153}
