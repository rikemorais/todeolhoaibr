13/// Erros ao transformar um lote de despesas da Câmara.
14#[cfg(test)]
15#[derive(Debug, Error, PartialEq, Eq)]
16pub enum ErroCamaraDespesasSilver {
17    /// A origem não contém as colunas textuais necessárias.
18    #[error("schema Bronze de despesas da Câmara incompatível")]
19    Schema,
20    /// Um campo preenchido não corresponde ao formato documentado.
21    #[error("campo {campo} inválido na linha {linha} do lote")]
22    Valor {
23        /// Campo da origem que não pôde ser interpretado.
24        campo: &'static str,
25        /// Índice da linha inválida dentro do lote, começando em zero.
26        linha: usize,
27    },
28    /// Os campos da origem não puderam ser preservados em JSON.
29    #[error("não foi possível preservar os campos originais da despesa")]
30    Json,
31}

43        Field::new("valor_documento", DataType::Decimal128(18, 6), true),
44        Field::new("valor_glosa", DataType::Decimal128(18, 6), true),
45        Field::new("valor_liquido", DataType::Decimal128(18, 6), true),

51    ]))
52}

54/// Tipifica campos de consulta e preserva a linha original sem colapsar lançamentos.
55#[cfg(test)]
56pub fn transformar_camara_despesas(
57    lote: &RecordBatch,
58    ano: u16,
59) -> Result<RecordBatch, ErroCamaraDespesasSilver> {
60    if lote
61        .schema()
62        .fields()
63        .iter()
64        .any(|campo| campo.data_type() != &DataType::Utf8)
65    {
66        return Err(ErroCamaraDespesasSilver::Schema);
67    }
68    let campos = [
69        ("txNomeParlamentar", "nome_parlamentar"),
70        ("nuDeputadoId", "id_deputado"),
71        ("ideDocumento", "id_documento"),
72        ("txtCNPJCPF", "documento_fornecedor"),
73        ("datEmissao", "data_emissao"),
74        ("datPagamentoRestituicao", "data_pagamento_restituicao"),
75        ("vlrDocumento", "valor_documento"),
76        ("vlrGlosa", "valor_glosa"),
77        ("vlrLiquido", "valor_liquido"),
78        ("vlrRestituicao", "valor_restituicao"),
79    ];
80    let indices = campos
81        .iter()
82        .map(|(origem, _)| {
83            lote.schema()
84                .index_of(origem)
85                .map_err(|_| ErroCamaraDespesasSilver::Schema)
86        })
87        .collect::<Result<Vec<_>, _>>()?;
88    let colunas = indices
89        .iter()
90        .map(|indice| {
91            lote.column(*indice)
92                .as_any()
93                .downcast_ref::<StringArray>()
94                .ok_or(ErroCamaraDespesasSilver::Schema)
95        })
96        .collect::<Result<Vec<_>, _>>()?;
97
98    let mut emissao = Vec::with_capacity(lote.num_rows());
99    let mut pagamento = Vec::with_capacity(lote.num_rows());
100    let mut documento = Vec::with_capacity(lote.num_rows());
101    let mut glosa = Vec::with_capacity(lote.num_rows());
102    let mut liquido = Vec::with_capacity(lote.num_rows());
103    let mut restituicao = Vec::with_capacity(lote.num_rows());
104    let mut originais = Vec::with_capacity(lote.num_rows());
105    for linha in 0..lote.num_rows() {
106        emissao.push(parse_data(colunas[4], linha, "datEmissao")?);
107        pagamento.push(parse_data(colunas[5], linha, "datPagamentoRestituicao")?);
108        documento.push(parse_decimal(colunas[6], linha, "vlrDocumento")?);
109        glosa.push(parse_decimal(colunas[7], linha, "vlrGlosa")?);
110        liquido.push(parse_decimal(colunas[8], linha, "vlrLiquido")?);
111        restituicao.push(parse_decimal(colunas[9], linha, "vlrRestituicao")?);
112
113        let mut original = serde_json::Map::new();
114        for (indice, campo) in lote.schema().fields().iter().enumerate() {
115            let valores = lote
116                .column(indice)
117                .as_any()
118                .downcast_ref::<StringArray>()
119                .ok_or(ErroCamaraDespesasSilver::Schema)?;
120            let valor = if valores.is_null(linha) {
121                serde_json::Value::Null
122            } else {
123                serde_json::Value::String(valores.value(linha).to_owned())
124            };
125            original.insert(campo.name().clone(), valor);
126        }
127        originais.push(Some(
128            serde_json::to_string(&original).map_err(|_| ErroCamaraDespesasSilver::Json)?,
129        ));
130    }
131
132    let coluna_texto = |indice: usize| Arc::new(colunas[indice].clone()) as Arc<dyn Array>;
133    let mut saida: Vec<Arc<dyn Array>> = vec![
134        Arc::new(UInt16Array::from_value(ano, lote.num_rows())),
135        coluna_texto(0),
136        coluna_texto(1),
137        coluna_texto(2),
138        coluna_texto(3),
139        data_array(emissao)?,
140        data_array(pagamento)?,
141        decimal_array(documento)?,
142        decimal_array(glosa)?,
143        decimal_array(liquido)?,
144        decimal_array(restituicao)?,
145        Arc::new(StringArray::from(originais)),
146    ];
147    saida.extend(crate::executor_sql::colunas_linhagem_teste(lote.num_rows()));
148    RecordBatch::try_new(esquema_camara_despesas(), saida)
149        .map_err(|_| ErroCamaraDespesasSilver::Schema)
150}

152#[cfg(test)]
153fn parse_data(
154    valores: &StringArray,
155    linha: usize,
156    campo: &'static str,
157) -> Result<Option<i32>, ErroCamaraDespesasSilver> {
158    if valores.is_null(linha) || valores.value(linha).trim().is_empty() {
159        return Ok(None);
160    }
161    let texto = valores.value(linha).trim();
162    let data = texto
163        .get(..10)
164        .and_then(|parte| NaiveDate::parse_from_str(parte, "%Y-%m-%d").ok())
165        .ok_or(ErroCamaraDespesasSilver::Valor { campo, linha })?;
166    let epoch = NaiveDate::from_ymd_opt(1970, 1, 1).expect("epoch válida");
167    Ok(Some(data.signed_duration_since(epoch).num_days() as i32))
168}

170#[cfg(test)]
171fn parse_decimal(
172    valores: &StringArray,
173    linha: usize,
174    campo: &'static str,
175) -> Result<Option<i128>, ErroCamaraDespesasSilver> {
176    if valores.is_null(linha) || valores.value(linha).trim().is_empty() {
177        return Ok(None);
178    }
179    decimal_escalado(valores.value(linha).trim())
180        .map(Some)
181        .ok_or(ErroCamaraDespesasSilver::Valor { campo, linha })
182}
183
184#[cfg(test)]
185fn decimal_escalado(texto: &str) -> Option<i128> {
186    let (negativo, numero) = texto
187        .strip_prefix('-')
188        .map_or((false, texto), |resto| (true, resto));
189    let (inteiro, fracao) = numero.split_once('.').unwrap_or((numero, ""));
190    if inteiro.is_empty()
191        || !inteiro.bytes().all(|byte| byte.is_ascii_digit())
192        || !fracao.bytes().all(|byte| byte.is_ascii_digit())
193        || inteiro.len() > 12
194        || fracao.len() > 6
195    {
196        return None;
197    }
198    let magnitude = format!("{inteiro}{fracao:0<6}").parse::<i128>().ok()?;
199    Some(if negativo { -magnitude } else { magnitude })
200}

202#[cfg(test)]
203fn data_array(valores: Vec<Option<i32>>) -> Result<Arc<dyn Array>, ErroCamaraDespesasSilver> {
204    Ok(Arc::new(Date32Array::from(valores)))
205}

207#[cfg(test)]
208fn decimal_array(valores: Vec<Option<i128>>) -> Result<Arc<dyn Array>, ErroCamaraDespesasSilver> {
209    Ok(Arc::new(
210        Decimal128Array::from(valores)
211            .with_precision_and_scale(18, 6)
212            .map_err(|_| ErroCamaraDespesasSilver::Schema)?,
213    ))
214}

271    fn tipa_data_valores_e_preserva_identificadores_e_origem() {
272        let silver = transformar_camara_despesas(&lote(), 2024).unwrap();

274        assert_eq!(silver.num_rows(), 1);
275        assert_eq!(
276            silver
277                .column(0)
278                .as_any()
279                .downcast_ref::<UInt16Array>()
280                .unwrap()
281                .value(0),
282            2024
283        );
284        assert_eq!(
285            silver
286                .column(1)
287                .as_any()
288                .downcast_ref::<StringArray>()
289                .unwrap()
290                .value(0),
291            "Deputado Exemplo"
292        );

294            silver
295                .column(2)
296                .as_any()
297                .downcast_ref::<StringArray>()
298                .unwrap()
