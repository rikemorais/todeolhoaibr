22const TENTATIVAS: u32 = 3;
23const ESPERA_ENTRE_TENTATIVAS: Duration = Duration::from_secs(15);

25/// Baixa um arquivo grande para a camada Raw sem carregá-lo na memória.
26///
27/// Não baixa de novo o que não mudou na origem: compara a versão (ETag) com a da última
28/// ingestão concluída. Isso também permite retomar uma carga longa que foi interrompida.
29pub struct TarefaDeArquivo {
30    nome: String,
31    fonte: String,
32    recurso: String,
33    url: String,
34    espera_entre_tentativas: Duration,
35}
36
37struct ArquivoGravado {
38    parte: ParteRaw,
39    versao: Option<String>,
40}
41
42impl TarefaDeArquivo {
43    /// `recurso` pode conter subpastas, como `votacao_secao/ano=2024/uf=AC`.
44    pub fn nova(nome: String, fonte: &str, recurso: String, url: String) -> Self {
45        Self {
46            nome,
47            fonte: fonte.to_owned(),
48            recurso,
49            url,
50            espera_entre_tentativas: ESPERA_ENTRE_TENTATIVAS,
51        }
52    }
53
54    async fn transferir(
55        &self,
56        contexto: &ContextoDeExecucao,
57        local: &LocalizacaoRaw,
58    ) -> Result<ArquivoGravado, ErroDeIngestao> {
59        let download = contexto.http.baixar(&self.url).await?;
60        let chave = local.parte(1, extensao(&download.tipo_de_conteudo));
61        let mut leitor = LeitorComResumo::novo(download.fluxo);
62
63        contexto
64            .armazenamento
65            .gravar_fluxo(&chave, &download.tipo_de_conteudo, &mut leitor)
66            .await?;
67
68        let versao = download.versao;
69        Ok(ArquivoGravado {
70            parte: ParteRaw {
71                chave,
72                url: self.url.clone(),
73                bytes: leitor.bytes,
74                sha256: hex::encode(leitor.hash.finalize()),
75                tipo_de_conteudo: download.tipo_de_conteudo,
76                versao: versao.clone(),
77            },
78            versao,
79        })
80    }
81
82    fn erro_de_manifesto(&self, erro: serde_json::Error) -> ErroDeIngestao {
83        ErroDeIngestao::Manifesto {
84            recurso: self.recurso.clone(),
85            detalhe: erro.to_string(),
86        }
87    }
88}

90#[async_trait]
91impl Tarefa for TarefaDeArquivo {
92    fn nome(&self) -> &str {
93        &self.nome
94    }
95
96    /// O que já está atualizado na Raw não será baixado de novo, então não ocupa espaço novo.
97    async fn bytes_a_baixar(
98        &self,
99        contexto: &ContextoDeExecucao,
100    ) -> Result<Option<u64>, ErroDeIngestao> {
101        let informacoes = contexto.http.informacoes(&self.url).await?;
102        let estado =
103            estado_na_raw(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso).await?;
104        match decidir(&informacoes, estado.as_ref()) {
105            DecisaoDeIngestao::Reaproveitar => Ok(Some(0)),
106            DecisaoDeIngestao::Ingerir(_) => Ok(informacoes.tamanho),
107        }
108    }
109
110    async fn executar(
111        &self,
112        contexto: &ContextoDeExecucao,
113    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
114        let local = LocalizacaoRaw::nova(
115            &self.fonte,
116            &self.recurso,
117            contexto.iniciada_em,
118            &contexto.id_execucao,
119        );
120        let informacoes = contexto.http.informacoes(&self.url).await?;
121        let estado =
122            estado_na_raw(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso).await?;
123        let motivo = match decidir(&informacoes, estado.as_ref()) {
124            DecisaoDeIngestao::Reaproveitar => {
125                return Ok(ResumoDaTarefa {
126                    reaproveitada: true,
127                    ..ResumoDaTarefa::default()
128                });
129            }
130            DecisaoDeIngestao::Ingerir(motivo) => motivo,
131        };
132
133        // A conexão pode cair no meio de um arquivo grande. Nada fica visível na Raw
134        // antes do fim da transferência, então basta recomeçar.
135        let mut falhas = 0;
136        let arquivo = loop {
137            match self.transferir(contexto, &local).await {
138                Ok(arquivo) => break arquivo,
139                Err(erro) => {
140                    falhas += 1;
141                    if falhas >= TENTATIVAS {
142                        return Err(erro);
143                    }
144                    warn!(tarefa = %self.nome, %erro, "transferência falhou; recomeçando o arquivo");
145                    tokio::time::sleep(self.espera_entre_tentativas).await;
146                }
147            }
148        };
149
150        let resumo = ResumoDaTarefa {
151            partes: 1,
152            bytes: arquivo.parte.bytes,
153            reaproveitada: false,
154            motivo: Some(motivo),
155        };
156
157        let manifesto = ManifestoRaw {
158            fonte: self.fonte.clone(),
159            recurso: self.recurso.clone(),
160            id_execucao: contexto.id_execucao.clone(),
161            url_inicial: self.url.clone(),
162            iniciada_em: contexto.iniciada_em,
163            concluida_em: Utc::now(),
164            partes: vec![arquivo.parte],
165        };
166        gravar_json(
167            contexto,
168            local.manifesto(),
169            serde_json::to_vec_pretty(&manifesto).map_err(|e| self.erro_de_manifesto(e))?,
170        )
171        .await?;
172
173        // Grava o ponteiro mesmo sem validador (versao `None`): é o que permite
174        // a pré-checagem por tamanho reaproveitar numa próxima rodada. A
175        // publicação não deixa uma rodada mais antiga sobrescrever uma mais nova
176        // (ver issue #16).
177        let ultima = UltimaVersao {
178            versao: arquivo.versao,
179            manifesto: local.manifesto(),
180            iniciada_em: Some(contexto.iniciada_em),
181        };
182        let publicou = publicar_ponteiro(
183            contexto.armazenamento.as_ref(),
184            &self.fonte,
185            &self.recurso,
186            &ultima,
187        )
188        .await?;
189        if !publicou {
190            warn!(
191                tarefa = %self.nome,
192                "outra rodada já publicou um ponteiro mais novo; este manifesto não é o vigente"
193            );
194        }
195
196        Ok(resumo)
197    }
198}

200async fn gravar_json(
201    contexto: &ContextoDeExecucao,
202    chave: String,
203    conteudo: Vec<u8>,
204) -> Result<(), ErroDeIngestao> {
205    contexto
206        .armazenamento
207        .gravar(ObjetoRaw {
208            chave,
209            conteudo,
210            tipo_de_conteudo: "application/json".to_owned(),
211        })
212        .await
213}

215/// Calcula tamanho e hash enquanto os bytes passam.
216struct LeitorComResumo {
217    fluxo: FluxoDeBytes,
218    hash: Sha256,
219    bytes: u64,
220}
221
222impl LeitorComResumo {
223    fn novo(fluxo: FluxoDeBytes) -> Self {
224        Self {
225            fluxo,
226            hash: Sha256::new(),
227            bytes: 0,
228        }
229    }
230}
231
232impl AsyncRead for LeitorComResumo {
233    fn poll_read(
234        self: Pin<&mut Self>,
235        contexto: &mut Context<'_>,
236        destino: &mut ReadBuf<'_>,
237    ) -> Poll<io::Result<()>> {
238        let este = self.get_mut();
239        let antes = destino.filled().len();
240        let resultado = Pin::new(&mut este.fluxo).poll_read(contexto, destino);
241        if let Poll::Ready(Ok(())) = resultado {
242            let novos = &destino.filled()[antes..];
243            este.hash.update(novos);
244            este.bytes += novos.len() as u64;
245        }
246        resultado
247    }
248}

277                .lock()
278                .unwrap()

288            &self,
289            chave: &str,
290            _: &str,
291            fluxo: &mut (dyn AsyncRead + Send + Unpin),
292        ) -> Result<(), ErroDeIngestao> {
293            let mut conteudo = Vec::new();
294            fluxo.read_to_end(&mut conteudo).await.map_err(|erro| {
295                ErroDeIngestao::Armazenamento {
296                    chave: chave.to_owned(),
297                    detalhe: erro.to_string(),
298                }
299            })?;
300            self.objetos
301                .lock()
302                .unwrap()
303                .insert(chave.to_owned(), conteudo);
304            Ok(())
305        }

308    /// Falha nos `falhas` primeiros downloads, cortando o fluxo no meio.
309    struct HttpFalso {
310        versao: Mutex<Option<String>>,
311        falhas: Mutex<u32>,
312        downloads: Mutex<u32>,
313    }
314
315    impl HttpFalso {
316        fn novo(versao: Option<&str>, falhas: u32) -> Self {
317            Self {
318                versao: Mutex::new(versao.map(str::to_owned)),
319                falhas: Mutex::new(falhas),
320                downloads: Mutex::new(0),
321            }
322        }
323    }
324
325    #[async_trait]
326    impl ClienteHttp for HttpFalso {
327        async fn obter(&self, _: &str, _: &str) -> Result<RespostaHttp, ErroDeIngestao> {
328            unreachable!("a tarefa de arquivo não usa respostas em memória")
329        }
330
331        async fn informacoes(&self, _: &str) -> Result<InformacoesDoArquivo, ErroDeIngestao> {
332            Ok(InformacoesDoArquivo {
333                versao: self.versao.lock().unwrap().clone(),
334                tamanho: Some(CONTEUDO.len() as u64),
335            })
336        }
337
338        async fn baixar(&self, _: &str) -> Result<DownloadHttp, ErroDeIngestao> {
339            *self.downloads.lock().unwrap() += 1;
340            let mut falhas = self.falhas.lock().unwrap();
341            let fluxo: FluxoDeBytes = if *falhas > 0 {
342                *falhas -= 1;
343                Box::new(leitor_com_erro())
344            } else {
345                Box::new(CONTEUDO)
346            };
347            Ok(DownloadHttp {
348                tipo_de_conteudo: "application/zip".into(),
349                versao: self.versao.lock().unwrap().clone(),
350                fluxo,
351            })
352        }
353    }
354
355    fn leitor_com_erro() -> impl AsyncRead + Send + Unpin {
356        struct Quebrado;
357        impl AsyncRead for Quebrado {
358            fn poll_read(
359                self: Pin<&mut Self>,
360                _: &mut Context<'_>,
361                _: &mut ReadBuf<'_>,
362            ) -> Poll<io::Result<()>> {
363                Poll::Ready(Err(io::Error::other("conexão perdida")))
364            }
365        }
366        Quebrado
367    }

412    #[tokio::test]
413    async fn grava_o_arquivo_sem_alterar_com_manifesto_e_versao() {
414        let cenario = cenario(HttpFalso::novo(Some("v1"), 0));
415
416        let resumo = tarefa().executar(&cenario.contexto).await.unwrap();
417
418        assert_eq!(resumo.bytes, CONTEUDO.len() as u64);
419        assert!(!resumo.reaproveitada);
420
421        let parte = &chaves(&cenario.armazenamento, "/parte-00001.zip")[0];
422        assert!(parte.starts_with("tse/votacao_secao/ano=2024/uf=AC/data_ingestao="));
423        let objetos = cenario.armazenamento.objetos.lock().unwrap();
424        assert_eq!(objetos[parte], CONTEUDO);
425
426        let chave_do_manifesto = parte.replace("parte-00001.zip", "_manifesto.json");
427        let manifesto: ManifestoRaw =
428            serde_json::from_slice(&objetos[&chave_do_manifesto]).unwrap();
429        assert_eq!(
430            manifesto.partes[0].sha256,
431            hex::encode(Sha256::digest(CONTEUDO))
432        );
433    }
