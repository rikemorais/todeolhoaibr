189/// Unidades de Conservação do SNUC — ICMBio/CNUC via portal de dados do MMA.
190///
191/// Baixa CSVs semestrais (lista de UCs ativas com atributos) e SHPs dos
192/// polígonos dos limites, publicados em `dados.mma.gov.br`.
193///
194/// Série: 2018 a 2026, semestral. Inclui UCs federais, estaduais, municipais
195/// e RPPNs (particulares).
196pub struct FonteIcmbio;
197
198impl FonteDeDados for FonteIcmbio {
199    fn nome(&self) -> &str {
200        NOME
201    }
202
203    fn descricao(&self) -> &str {
204        "Unidades de Conservação do SNUC — CNUC/MMA (atributos e polígonos)"
205    }
206
207    fn sob_demanda(&self) -> bool {
208        true
209    }
210
211    fn perfil_http(&self) -> PerfilHttp {
212        PerfilHttp {
213            credencial: None,
214            limite: Some(LimiteHttp {
215                host: "dados.mma.gov.br",
216                requisicoes: 2,
217                intervalo: Duration::from_secs(1),
218            }),
219            agente: None,
220            redirecionamento: RedirecionamentoHttp::Seguir,
221            aceitar_certificado_invalido: false,
222        }
223    }
224
225    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
226        RECURSOS
227            .iter()
228            .map(|r| {
229                TarefaPlanejada::nova(TarefaDeArquivo::nova(
230                    r.nome.to_owned(),
231                    NOME,
232                    r.nome.to_owned(),
233                    url(r),
234                ))
235                .com_etiquetas([("periodo", r.periodo), ("formato", r.formato)])
236            })
237            .collect()
238    }
239}

241#[cfg(test)]
242mod testes {
243    use super::*;
244    use std::collections::HashSet;
245    use todeolho_nucleo::FiltroDeTarefas;
246
247    fn tarefas() -> Vec<TarefaPlanejada> {
248        FonteIcmbio.criar_tarefas()
249    }
250
251    #[test]
252    fn total_22_tarefas() {
253        assert_eq!(tarefas().len(), 22);
254    }
255
256    #[test]
257    fn nomes_sao_unicos() {
258        let t = tarefas();
259        let nomes: HashSet<_> = t.iter().map(|t| t.tarefa.nome()).collect();
260        assert_eq!(nomes.len(), t.len());
261    }
262
263    #[test]
264    fn filtra_por_periodo() {
265        let filtradas = FiltroDeTarefas::a_partir_de(&["periodo=2025-08".to_owned()])
266            .unwrap()
267            .aplicar(tarefas())
268            .unwrap();
269        assert_eq!(filtradas.len(), 2);
270        let nomes: HashSet<_> = filtradas.iter().map(|t| t.tarefa.nome()).collect();
271        assert!(nomes.contains("cnuc/2025-08"));
272        assert!(nomes.contains("poligono/2025-08"));
273    }
274
275    #[test]
276    fn filtra_por_formato_shapefile() {
277        let filtradas = FiltroDeTarefas::a_partir_de(&["formato=shapefile".to_owned()])
278            .unwrap()
279            .aplicar(tarefas())
280            .unwrap();
281        assert_eq!(filtradas.len(), 4);
282    }
