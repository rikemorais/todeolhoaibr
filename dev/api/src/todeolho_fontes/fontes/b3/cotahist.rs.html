1use todeolho_nucleo::{FonteDeDados, TarefaPlanejada};
2
3use crate::ingestao_arquivo::TarefaDeArquivo;
4
5use super::api;

7const NOME: &str = "b3";
8const BASE: &str = "https://bvmf.bmfbovespa.com.br/InstDados/SerHist";
9
10const PRIMEIRO_ANO: u16 = 1986;
11const ULTIMO_ANO: u16 = 2026;
12
13/// COTAHIST — cotações históricas diárias da B3 (ações, ETFs, BDRs, UNITs,
14/// FIIs, FIAGROs). Um ZIP por ano, formato fixed-width definido pela B3.
15///
16/// O vínculo geográfico (UF da sede) não vem neste arquivo: exige join com o
17/// cadastro de companhias da CVM (issue #37) ou a base CNPJ (issue #34) pelo
18/// CNPJ da empresa. Ver issue #53.
19pub struct FonteB3;
20
21impl FonteDeDados for FonteB3 {
22    fn nome(&self) -> &str {
23        NOME
24    }
25
26    fn descricao(&self) -> &str {
27        "B3 COTAHIST - cotações históricas diárias da bolsa, de 1986 a 2026"
28    }
29
30    fn perfil_http(&self) -> todeolho_nucleo::PerfilHttp {
31        api::perfil_http()
32    }
33
34    fn sob_demanda(&self) -> bool {
35        true
36    }
37
38    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
39        (PRIMEIRO_ANO..=ULTIMO_ANO)
40            .map(|ano| {
41                let mut tarefa = TarefaDeArquivo::nova(
42                    format!("cotahist_{ano}"),
43                    NOME,
44                    format!("cotahist/ano={ano}"),
45                    format!("{BASE}/COTAHIST_A{ano}.ZIP"),
46                );
47                if ano == ULTIMO_ANO {
48                    tarefa = tarefa.aceitar_ausencia();
49                }
50                TarefaPlanejada::nova(tarefa).com_etiquetas([("ano", ano.to_string().as_str())])
51            })
52            .collect()
53    }
54}

56#[cfg(test)]
57mod testes {

60    #[test]
61    fn gera_uma_tarefa_por_ano() {
62        let tarefas = FonteB3.criar_tarefas();
63        let total = (ULTIMO_ANO - PRIMEIRO_ANO + 1) as usize;
64        assert_eq!(tarefas.len(), total);
65    }
66
67    #[test]
68    fn primeira_tarefa_e_1986_e_ultima_e_o_ano_corrente() {
69        let tarefas = FonteB3.criar_tarefas();
70        assert_eq!(tarefas.first().unwrap().tarefa.nome(), "cotahist_1986");
71        assert_eq!(
72            tarefas.last().unwrap().tarefa.nome(),
73            format!("cotahist_{ULTIMO_ANO}")
74        );
75    }
76
77    #[test]
78    fn url_aponta_para_o_zip_anual_da_b3() {
79        let tarefas = FonteB3.criar_tarefas();
80        let etiqueta_2023 = tarefas
81            .iter()
82            .find(|t| t.tarefa.nome() == "cotahist_2023")
83            .unwrap();
84        assert_eq!(
85            etiqueta_2023.etiquetas.get("ano").map(String::as_str),
86            Some("2023")
87        );
88    }

95    #[test]

98    }
