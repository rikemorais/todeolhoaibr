1//! Arquivos mensais de remuneração dos servidores civis ativos do Executivo Federal.

3use std::time::Duration;
4
5use chrono::{Datelike, NaiveDate, Utc};
6use todeolho_nucleo::{FonteDeDados, LimiteHttp, PerfilHttp, TarefaPlanejada};

8use crate::ingestao_arquivo::TarefaDeArquivo;

10const NOME: &str = "transparencia-servidores-ativos";
11const FONTE_RAW: &str = "transparencia";
12const BASE: &str = "https://portaldatransparencia.gov.br/download-de-dados/servidores";
13const PRIMEIRA_COMPETENCIA: (i32, u32) = (2020, 1);
14
15// O CDN da CGU (dadosabertos-download.cgu.gov.br) usa AWS WAF que bloqueia
16// User-Agents de bots. Um agente genérico de navegador evita o bloqueio sem
17// violar os termos de uso dos dados abertos.
18const AGENTE: &str = "Mozilla/5.0 (X11; Linux x86_64; rv:128.0) Gecko/20100101 Firefox/128.0";
19
20const fn perfil_http_cgu() -> PerfilHttp {
21    PerfilHttp {
22        credencial: None,
23        limite: Some(LimiteHttp {
24            host: "portaldatransparencia.gov.br",
25            requisicoes: 2,
26            intervalo: Duration::from_secs(1),
27        }),
28        agente: Some(AGENTE),
29    }
30}

32#[derive(Debug, Clone, Copy)]
33enum Origem {
34    Siape,
35    Bacen,
36}
37
38impl Origem {
39    fn slug(self) -> &'static str {
40        match self {
41            Self::Siape => "siape",
42            Self::Bacen => "bacen",
43        }
44    }
45
46    fn arquivo(self) -> &'static str {
47        match self {
48            Self::Siape => "Servidores_SIAPE",
49            Self::Bacen => "Servidores_BACEN",
50        }
51    }
52}
53
54fn url(origem: Origem, competencia: &str) -> String {
55    format!(
56        "{BASE}/{}_{}",
57        competencia.replace('-', ""),
58        origem.arquivo()
59    )
60}
61
62fn ultima_competencia_publicada() -> (i32, u32) {
63    let mes = Utc::now().date_naive().with_day(1).expect("dia válido");
64    let mes = mes
65        .checked_sub_months(chrono::Months::new(2))
66        .unwrap_or(mes);
67    (mes.year(), mes.month())
68}
69
70fn competencias(inicio: (i32, u32), fim: (i32, u32)) -> Vec<String> {
71    let mut atual = NaiveDate::from_ymd_opt(inicio.0, inicio.1, 1).expect("competência válida");
72    let limite = NaiveDate::from_ymd_opt(fim.0, fim.1, 1).expect("competência válida");
73    let mut resultado = Vec::new();
74    while atual <= limite {
75        resultado.push(format!("{}-{:02}", atual.year(), atual.month()));
76        let Some(proximo) = atual.checked_add_months(chrono::Months::new(1)) else {
77            break;
78        };
79        atual = proximo;
80    }
81    resultado
82}

84fn tarefas_ate(fim: (i32, u32)) -> Vec<TarefaPlanejada> {
85    [Origem::Siape, Origem::Bacen]
86        .into_iter()
87        .flat_map(|origem| {
88            competencias(PRIMEIRA_COMPETENCIA, fim)
89                .into_iter()
90                .map(move |competencia| {
91                    let tarefa = TarefaDeArquivo::nova(
92                        format!("{}_{}", origem.slug(), competencia),
93                        FONTE_RAW,
94                        format!(
95                            "servidores-ativos/origem={}/competencia={competencia}",
96                            origem.slug()
97                        ),
98                        url(origem, &competencia),
99                    );
100                    TarefaPlanejada::nova(tarefa).com_etiquetas([
101                        ("origem", origem.slug()),
102                        ("competencia", competencia.as_str()),
103                    ])
104                })
105        })
106        .collect()
107}

109/// Preserva os arquivos oficiais mensais por origem e competência na camada Raw.
110pub struct FonteTransparenciaServidoresAtivos;
111
112impl FonteDeDados for FonteTransparenciaServidoresAtivos {
113    fn nome(&self) -> &str {
114        NOME
115    }
116
117    fn descricao(&self) -> &str {
118        "Portal da Transparência - remuneração de servidores civis ativos do Executivo Federal por competência"
119    }
120
121    fn sob_demanda(&self) -> bool {
122        true
123    }
124
125    fn perfil_http(&self) -> PerfilHttp {
126        perfil_http_cgu()
127    }
128
129    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
130        tarefas_ate(ultima_competencia_publicada())
131    }
132}

134#[cfg(test)]
135mod testes {
136    use super::*;
137
138    #[test]
139    fn gera_uma_tarefa_por_origem_e_competencia() {
140        let tarefas = tarefas_ate((2020, 2));
141        assert_eq!(tarefas.len(), 4);
142        assert_eq!(tarefas[0].tarefa.nome(), "siape_2020-01");
143        assert_eq!(tarefas[0].etiquetas["origem"], "siape");
144        assert_eq!(tarefas[0].etiquetas["competencia"], "2020-01");
145        assert_eq!(tarefas[1].tarefa.nome(), "siape_2020-02");
146        assert_eq!(tarefas[2].tarefa.nome(), "bacen_2020-01");
147        assert_eq!(tarefas[3].tarefa.nome(), "bacen_2020-02");
148    }
149
150    #[test]
151    fn usa_urls_oficiais_das_planilhas_mensais() {
152        assert_eq!(
153            url(Origem::Siape, "2020-01"),
154            "https://portaldatransparencia.gov.br/download-de-dados/servidores/202001_Servidores_SIAPE"
155        );
156        assert_eq!(
157            url(Origem::Bacen, "2020-01"),
158            "https://portaldatransparencia.gov.br/download-de-dados/servidores/202001_Servidores_BACEN"
159        );
160    }
161
162    #[test]
163    fn competencias_avancam_de_ano_sem_lacunas() {
164        assert_eq!(
165            competencias((2020, 11), (2021, 2)),
166            ["2020-11", "2020-12", "2021-01", "2021-02"]
167        );
168    }
169
170    #[test]
171    fn nao_entra_em_todas_por_acidente() {
172        assert!(FonteTransparenciaServidoresAtivos.sob_demanda());
173    }
174
175    #[test]
176    fn usa_agente_de_navegador_para_evitar_waf() {
177        let perfil = FonteTransparenciaServidoresAtivos.perfil_http();
178        assert!(perfil.agente.is_some());
179        assert!(perfil.agente.unwrap().contains("Mozilla"));
180    }
181
182    #[test]
183    fn limita_requisicoes_ao_portal() {
184        let limite = FonteTransparenciaServidoresAtivos
185            .perfil_http()
186            .limite
187            .expect("limite do portal");
188        assert_eq!(limite.host, "portaldatransparencia.gov.br");
189    }
