1//! Séries socioeconômicas regionais do Ipeadata (IPEA): IDHM, renda, desigualdade
2//! (Gini), pobreza e vulnerabilidade social (IVS), já consolidadas por território.
3//!
4//! Cada série é um recurso da fonte `ipeadata` que baixa **todos os valores** (todos os
5//! territórios e períodos) de um código do Ipeadata, em JSON (OData), gravado opaco na
6//! Raw. O recorte por município/UF e a interpretação ficam no refino.
7

17/// Uma série do Ipeadata: o `slug` (recurso na Raw e etiqueta), o código Ipeadata
18/// (`SERCODIGO`) e o tamanho típico da resposta, medido numa busca real.
19struct Serie {
20    slug: &'static str,
21    codigo: &'static str,
22    tamanho_tipico: u64,
23}

75fn url(serie: &Serie) -> String {
76    format!("{BASE}('{}')/Valores", serie.codigo)
77}

79/// Séries socioeconômicas regionais do Ipeadata.
80///
81/// Uma consulta por série traz todos os valores (territórios e períodos) em JSON OData,
82/// guardados opacos na Raw ([ADR 0014]). São séries já derivadas pelo IPEA a partir do
83/// Censo/PNAD; a procedência fica registrada no nome do recurso e o recorte por
84/// município/UF é do refino ([ADR 0015]). Ver issue #32.
85///
86/// Catálogo: <http://www.ipeadata.gov.br>
87///
88/// [ADR 0014]: ../../../docs/adr/0014-raw-opaca-e-refino-em-estagio-separado.md
89/// [ADR 0015]: ../../../docs/adr/0015-arquitetura-de-dados-em-medalhao.md
90pub struct FonteIpeadata;
91
92impl FonteDeDados for FonteIpeadata {
93    fn nome(&self) -> &str {
94        NOME
95    }
96
97    fn descricao(&self) -> &str {
98        "Ipeadata - séries socioeconômicas regionais (IDHM, renda, Gini, pobreza, IVS)"
99    }
100
101    fn perfil_http(&self) -> todeolho_nucleo::PerfilHttp {
102        api::perfil_http()
103    }
104
105    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
106        SERIES
107            .iter()
108            .map(|serie| {
109                let recurso = format!("series/{}", serie.slug);
110                TarefaPlanejada::nova(
111                    TarefaDeIngestaoHttp::sem_paginacao(NOME, &recurso, url(serie))
112                        .com_tamanho_tipico(serie.tamanho_tipico),
113                )
114                .com_etiquetas([("serie", serie.slug)])
115            })
116            .collect()
117    }
118}

120#[cfg(test)]
121mod testes {
122    use std::collections::BTreeSet;
123
124    use super::*;
125
126    #[test]
127    fn cada_serie_vira_uma_tarefa_com_recurso_e_etiqueta() {
128        let tarefas = FonteIpeadata.criar_tarefas();
129        assert_eq!(tarefas.len(), SERIES.len());
130        for (tarefa, serie) in tarefas.iter().zip(SERIES) {
131            assert_eq!(tarefa.tarefa.nome(), format!("series/{}", serie.slug));
132            assert_eq!(
133                tarefa.etiquetas.get("serie").map(String::as_str),
134                Some(serie.slug)
135            );
136        }
137    }
138
139    #[test]
140    fn slugs_e_codigos_sao_unicos() {
141        let slugs: BTreeSet<&str> = SERIES.iter().map(|s| s.slug).collect();
142        let codigos: BTreeSet<&str> = SERIES.iter().map(|s| s.codigo).collect();
143        assert_eq!(slugs.len(), SERIES.len(), "slug repetido");
144        assert_eq!(codigos.len(), SERIES.len(), "código repetido");
145    }
146
147    #[test]
148    fn a_url_aponta_para_os_valores_da_serie_no_odata() {
149        let gini = SERIES.iter().find(|s| s.slug == "gini").unwrap();
150        assert_eq!(
151            url(gini),
152            "https://www.ipeadata.gov.br/api/odata4/Metadados('ADH_GINI')/Valores"
153        );
154    }
155
156    #[test]
157    fn a_url_do_idhm_aponta_para_o_codigo_adh_idhm() {
158        let idhm = SERIES.iter().find(|s| s.slug == "idhm").unwrap();
159        assert_eq!(
160            url(idhm),
161            "https://www.ipeadata.gov.br/api/odata4/Metadados('ADH_IDHM')/Valores"
162        );
163    }
164
165    #[test]
166    fn as_quatro_dimensoes_do_idhm_estao_presentes() {
167        let slugs: Vec<&str> = SERIES.iter().map(|s| s.slug).collect();
168        assert!(slugs.contains(&"idhm"), "IDHM geral");
169        assert!(slugs.contains(&"idhm-educacao"), "IDHM educação");
170        assert!(slugs.contains(&"idhm-longevidade"), "IDHM longevidade");
171        assert!(slugs.contains(&"idhm-renda"), "IDHM renda");
172    }
173
174    #[test]
175    fn declara_o_limite_do_host_do_ipeadata() {
176        let limite = FonteIpeadata
177            .perfil_http()
178            .limite
179            .expect("limite do Ipeadata");
180        assert_eq!(limite.host, "www.ipeadata.gov.br");
181    }
