211fn url_do_recurso(ano: &AnoChegadas) -> String {

213        "https://{HOST}/dataset/{DATASET}/resource/{}/download/{}",
214        ano.recurso_id, ano.arquivo

218/// Chegada de turistas internacionais ao Brasil (Ministério do Turismo).
219pub struct FonteTurismoChegadas;
220
221impl FonteDeDados for FonteTurismoChegadas {
222    fn nome(&self) -> &str {
223        NOME
224    }
225
226    fn descricao(&self) -> &str {
227        "Turismo - chegada de turistas internacionais ao Brasil, por país de residência (1989–2026)"
228    }
229
230    fn sob_demanda(&self) -> bool {
231        true
232    }
233
234    fn perfil_http(&self) -> PerfilHttp {
235        PerfilHttp {
236            redirecionamento: todeolho_nucleo::RedirecionamentoHttp::Seguir,
237            credencial: None,
238            limite: Some(LimiteHttp {
239                host: HOST,
240                requisicoes: 2,
241                intervalo: Duration::from_secs(1),
242            }),
243            agente: None,
244            aceitar_certificado_invalido: false,
245        }
246    }
247
248    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
249        ANOS.iter()
250            .map(|ano| {
251                let ano_str = ano.ano.to_string();
252                let mut tarefa = TarefaDeArquivo::nova(
253                    ano_str.clone(),
254                    NOME,
255                    format!("ano={}", ano.ano),
256                    url_do_recurso(ano),
257                );
258                if ano.ano == 2026 {
259                    tarefa = tarefa.aceitar_ausencia();
260                }
261                TarefaPlanejada::nova(tarefa)
262                    .com_etiquetas([("dataset", "chegadas-internacionais"), ("ano", &ano_str)])
263            })
264            .collect()
265    }
266}

268#[cfg(test)]
269mod testes {
270    use super::*;
271    use todeolho_nucleo::FiltroDeTarefas;
272
273    #[test]
274    fn total_de_tarefas() {
275        assert_eq!(FonteTurismoChegadas.criar_tarefas().len(), ANOS.len());
276        assert_eq!(ANOS.len(), 38);
277    }
278
279    #[test]
280    fn nomes_sao_unicos() {
281        let tarefas = FonteTurismoChegadas.criar_tarefas();
282        let nomes: std::collections::HashSet<_> = tarefas.iter().map(|t| t.tarefa.nome()).collect();
283        assert_eq!(nomes.len(), tarefas.len());
284    }
285
286    #[test]
287    fn url_segue_padrao_ckan() {
288        let url = url_do_recurso(&ANOS[0]);
289        assert!(url.starts_with("https://dados.turismo.gov.br/dataset/"));
290        assert!(url.contains("/resource/"));
291        assert!(url.ends_with(".csv"));
292    }
293
294    #[test]
295    fn filtra_por_ano() {
296        let tarefas = FonteTurismoChegadas.criar_tarefas();
297        let filtro = FiltroDeTarefas::a_partir_de(&["ano=2019".to_owned()]).unwrap();
298        let filtradas = filtro.aplicar(tarefas).unwrap();
299        assert_eq!(filtradas.len(), 1);
300        assert_eq!(filtradas[0].tarefa.nome(), "2019");
301    }
302
303    #[test]
304    fn etiquetas_presentes() {
305        let tarefas = FonteTurismoChegadas.criar_tarefas();
306        for tarefa in &tarefas {
307            assert_eq!(
308                tarefa.etiquetas.get("dataset").unwrap(),
309                "chegadas-internacionais"
310            );
311            assert!(tarefa.etiquetas.contains_key("ano"));
312        }
313    }
314
315    #[test]
316    fn e_sob_demanda() {
317        assert!(FonteTurismoChegadas.sob_demanda());
318    }
319
320    #[test]
321    fn sem_credencial_e_segue_redirecionamento() {
322        let perfil = FonteTurismoChegadas.perfil_http();
323        assert!(perfil.credencial.is_none());
324        assert!(matches!(
325            perfil.redirecionamento,
326            todeolho_nucleo::RedirecionamentoHttp::Seguir
327        ));
328    }
329
330    #[test]
331    fn ano_2026_tem_nome_especial() {
332        let ultimo = ANOS.last().unwrap();
333        assert_eq!(ultimo.ano, 2026);
334        assert!(ultimo.arquivo.contains("acum"));
335    }
