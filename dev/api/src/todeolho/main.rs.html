42#[derive(Subcommand)]
43enum Comando {

57        /// Endereço em que o painel fica disponível

85#[tokio::main]
86async fn main() -> ExitCode {
87    tracing_subscriber::fmt()
88        .with_env_filter(
89            EnvFilter::try_from_env("TODEOLHO_LOG").unwrap_or_else(|_| EnvFilter::new("info")),
90        )
91        .with_target(false)
92        .with_writer(io::stderr)
93        .with_ansi(io::stderr().is_terminal())
94        .init();
95
96    match rodar(Cli::parse()).await {
97        Ok(true) => ExitCode::SUCCESS,
98        Ok(false) => ExitCode::FAILURE,
99        Err(erro) => {
100            eprintln!("erro: {erro:#}");
101            ExitCode::FAILURE
102        }
103    }
104}

106/// Devolve `false` quando alguma tarefa não foi concluída.
107async fn rodar(cli: Cli) -> Result<bool> {
108    match cli.comando {
109        Comando::Selecionar => selecionar().await,
110        Comando::Listar => {
111            let fontes = catalogo();
112            // Alinha a coluna pelo nome mais longo, para nomes de qualquer tamanho.
113            let largura = fontes.iter().map(|f| f.nome().len()).max().unwrap_or(0);
114            for fonte in &fontes {
115                println!("{:<largura$} {}", fonte.nome(), fonte.descricao());
116            }
117            Ok(true)
118        }
119        Comando::Plano { fonte } => {
120            let fonte = buscar_fonte(&fonte)?;
121            for (indice, nivel) in fonte.montar_dag()?.niveis()?.iter().enumerate() {
122                println!("nível {}: {}", indice + 1, nivel.join(", "));
123            }
124            Ok(true)
125        }
126        Comando::Executar(argumentos) => executar_sem_menu(argumentos).await,
127        Comando::Painel { endereco } => {
128            let historico = historico()?;
129            historico.preparar().await.context(ERRO_DE_ARMAZENAMENTO)?;
130            todeolho_painel::servir(endereco, Arc::new(historico)).await?;
131            Ok(true)
132        }
133    }
134}

136fn buscar_fonte(nome: &str) -> Result<Box<dyn FonteDeDados>> {
137    let fontes = catalogo();
138    let disponiveis: Vec<String> = fontes.iter().map(|f| f.nome().to_owned()).collect();
139    match fontes.into_iter().find(|fonte| fonte.nome() == nome) {
140        Some(fonte) => Ok(fonte),
141        None => bail!(
142            "fonte desconhecida: {nome}. Disponíveis: {}",
143            disponiveis.join(", ")
144        ),
145    }
146}

148async fn selecionar() -> Result<bool> {
149    if !io::stdin().is_terminal() {
150        bail!("o menu precisa de um terminal. Sem terminal, use: todeolho executar <fonte> --sim");
151    }
152
153    let contexto = preparar_contexto().await?;
154    println!("Consultando o tamanho dos dados...");
155
156    let fontes = catalogo();
157    let mut opcoes = Vec::with_capacity(fontes.len());
158    for fonte in &fontes {
159        let tarefas = fonte.criar_tarefas();
160        let opcao = montar_opcao_de_menu(fonte.as_ref(), &tarefas, async {
161            match contexto_para_fonte(&contexto, fonte.perfil_http()) {
162                Ok(contexto_da_fonte) => estimar(&tarefas_de(&tarefas), &contexto_da_fonte).await,
163                Err(_) => vec![None; tarefas.len()],
164            }
165        })
166        .await;
167        opcoes.push(opcao);
168    }
169
170    let mut entrada = io::stdin().lock();
171    let Some(escolha) = menu::escolher(&opcoes, &mut entrada, &mut io::stdout())? else {
172        println!("Nada foi executado.");
173        return Ok(true);
174    };
175    drop(entrada);
176
177    let mut selecionadas = Vec::with_capacity(escolha.len());
178    for (indice, filtro) in escolha {
179        let fonte = &fontes[indice];
180        selecionadas.push(FonteSelecionada {
181            nome: fonte.nome().to_owned(),
182            tarefas: filtro.aplicar(fonte.criar_tarefas())?,
183            perfil_http: fonte.perfil_http(),
184        });
185    }
186    executar(selecionadas, contexto, false).await
187}

189async fn montar_opcao_de_menu(
190    fonte: &dyn FonteDeDados,
191    tarefas: &[TarefaPlanejada],
192    consulta: impl std::future::Future<Output = Vec<Option<u64>>>,
193) -> menu::OpcaoDeFonte {
194    let estimativas = if fonte.sob_demanda() {
195        vec![None; tarefas.len()]
196    } else {
197        consulta.await
198    };
199    menu::OpcaoDeFonte {
200        nome: fonte.nome().to_owned(),
201        descricao: fonte.descricao().to_owned(),
202        sob_demanda: fonte.sob_demanda(),
203        tarefas: tarefas
204            .iter()
205            .zip(estimativas)
206            .map(|(tarefa, bytes)| menu::OpcaoDeTarefa {
207                etiquetas: tarefa.etiquetas.clone(),
208                bytes,
209            })
210            .collect(),
211    }
212}

214async fn executar_sem_menu(argumentos: ArgumentosDeExecucao) -> Result<bool> {
215    let selecionadas = match (argumentos.fonte, argumentos.todas) {
216        (Some(_), true) => bail!("informe uma fonte ou --todas, não os dois"),
217        (None, false) => bail!("informe uma fonte ou --todas"),
218        (None, true) => {
219            if !argumentos.filtros.is_empty() {
220                bail!("--filtro só vale para uma fonte específica");
221            }
222            catalogo()
223                .into_iter()
224                .filter(|fonte| !fonte.sob_demanda())
225                .map(|fonte| FonteSelecionada {
226                    nome: fonte.nome().to_owned(),
227                    tarefas: fonte.criar_tarefas(),
228                    perfil_http: fonte.perfil_http(),
229                })
230                .collect()
231        }
232        (Some(nome), false) => {
233            let fonte = buscar_fonte(&nome)?;
234            let filtro = FiltroDeTarefas::a_partir_de(&argumentos.filtros)?;
235            vec![FonteSelecionada {
236                nome,
237                tarefas: filtro.aplicar(fonte.criar_tarefas())?,
238                perfil_http: fonte.perfil_http(),
239            }]
240        }
241    };
242
243    let contexto = preparar_contexto().await?;
244    executar(selecionadas, contexto, argumentos.sim).await
245}

247async fn preparar_contexto() -> Result<ContextoDeExecucao> {
248    let armazenamento = Arc::new(ArmazenamentoS3::novo(configuracao_s3("raw")?)?);
249    armazenamento
250        .preparar()
251        .await
252        .context(ERRO_DE_ARMAZENAMENTO)?;
253
254    let iniciada_em = Utc::now();
255    Ok(ContextoDeExecucao {
256        id_execucao: format!(
257            "{}-{}",
258            iniciada_em.format("%Y%m%dT%H%M%SZ"),
259            &Uuid::new_v4().simple().to_string()[..8]
260        ),
261        iniciada_em,
262        armazenamento,
263        http: Arc::new(ClienteHttpComRetentativas::novo(
264            PoliticaDeRetentativas::default(),
265        )?),
266    })
267}

269fn contexto_para_fonte(
270    contexto: &ContextoDeExecucao,
271    perfil: PerfilHttp,
272) -> Result<ContextoDeExecucao> {
273    let segredo = perfil
274        .credencial
275        .map(|credencial| obrigatoria(credencial.variavel))
276        .transpose()?;
277    Ok(ContextoDeExecucao {
278        http: Arc::new(ClienteHttpComRetentativas::com_perfil(
279            PoliticaDeRetentativas::default(),
280            perfil,
281            segredo.as_deref(),
282        )?),
283        ..contexto.clone()
284    })
285}

287/// Mostra o espaço necessário, recusa o que não cabe, pede confirmação e executa.
288async fn executar(
289    selecionadas: Vec<FonteSelecionada>,
290    contexto: ContextoDeExecucao,
291    sem_confirmacao: bool,
292) -> Result<bool> {
293    let mut dags = Vec::with_capacity(selecionadas.len());
294    let mut contextos = Vec::with_capacity(selecionadas.len());
295    let mut estimativas = Vec::new();
296    for fonte in selecionadas {
297        let contexto_da_fonte = contexto_para_fonte(&contexto, fonte.perfil_http)?;
298        estimativas.extend(estimar(&tarefas_de(&fonte.tarefas), &contexto_da_fonte).await);
299        dags.push(DagDeIngestao::construir(&fonte.nome, fonte.tarefas)?);
300        contextos.push(contexto_da_fonte);
301    }
302
303    let quantidade = estimativas.len();
304    let espaco = EspacoNecessario::somar(&estimativas);
305    let livre = env::var("TODEOLHO_DIR_DADOS")
306        .ok()
307        .and_then(|caminho| espaco_livre_em(Path::new(&caminho)));
308
309    if !avisar_sobre_espaco(quantidade, &espaco, livre) {
310        return Ok(false);
311    }
312    if !sem_confirmacao && !confirmar()? {
313        println!("Nada foi executado.");
314        return Ok(true);
315    }
316
317    match contexto.armazenamento.descartar_envios_incompletos().await {
318        Ok(0) => {}
319        Ok(quantidade) => {
320            println!("Espaço recuperado de {quantidade} transferência(s) interrompida(s).")
321        }
322        Err(erro) => {
323            eprintln!("Atenção: não foi possível limpar transferências interrompidas: {erro}")
324        }
325    }
326
327    let historico = Arc::new(historico()?);
328    historico.preparar().await.context(ERRO_DE_ARMAZENAMENTO)?;
329
330    let registrador = Arc::new(RegistradorDeExecucao::novo(
331        RegistroDeExecucao::novo(&contexto.id_execucao, contexto.iniciada_em, &dags)?,
332        historico,
333    ));
334    registrador.publicar().await;
335    tokio::spawn({
336        let registrador = Arc::clone(&registrador);
337        async move { registrador.manter_batimento().await }
338    });
339
340    let executor = ExecutorDeDag::novo(concorrencia()?);
341
342    let mut sucesso = true;
343    for (dag, contexto_da_fonte) in dags.iter().zip(&contextos) {
344        let relatorio = executor
345            .executar(dag, contexto_da_fonte, registrador.as_ref())
346            .await?;
347        imprimir(&relatorio);
348        sucesso &= relatorio.sucesso();
349    }
350    registrador.concluir().await;
351
352    println!("\nexecução: {}", contexto.id_execucao);
353    Ok(sucesso)
354}

356/// Devolve `false` quando a seleção não cabe no disco.
357fn avisar_sobre_espaco(quantidade: usize, espaco: &EspacoNecessario, livre: Option<u64>) -> bool {
358    println!("\nSelecionado: {quantidade} tarefa(s)");
359    println!("Espaço necessário: {}", formatar_bytes(espaco.bytes));
360    if espaco.maior_arquivo > 0 {
361        println!(
362            "Pico durante a gravação: {} (o maior arquivo ocupa o dobro enquanto é gravado)",
363            formatar_bytes(espaco.pico())
364        );
365    }
366    if espaco.sem_estimativa > 0 {
367        println!(
368            "Atenção: {} tarefa(s) sem tamanho conhecido. O espaço real pode ser maior.",
369            espaco.sem_estimativa
370        );
371    }
372    if let Some(livre) = livre {
373        println!("Espaço livre: {}", formatar_bytes(livre));
374    }
375
376    match espaco.avaliar(livre) {
377        AvaliacaoDeEspaco::Cabe => true,
378        AvaliacaoDeEspaco::Desconhecido => {
379            println!("Atenção: não foi possível verificar o espaço livre em disco.");
380            true
381        }
382        AvaliacaoDeEspaco::NaoCabe { faltam } => {
383            eprintln!(
384                "\nNão cabe: faltam {}. Escolha menos dados ou libere espaço em disco.",
385                formatar_bytes(faltam)
386            );
387            false
388        }
389    }
390}

392fn confirmar() -> Result<bool> {
393    if !io::stdin().is_terminal() {
394        bail!("sem terminal para confirmar. Use --sim para executar sem a pergunta");
395    }
396
397    print!("\nContinuar? [s/N] ");
398    io::stdout().flush()?;
399    let mut resposta = String::new();
400    io::stdin().read_line(&mut resposta)?;
401    Ok(matches!(
402        resposta.trim().to_lowercase().as_str(),
403        "s" | "sim"
404    ))
405}

418fn imprimir(relatorio: &RelatorioDaExecucao) {
419    println!("\nfonte: {}", relatorio.fonte);
420    for resultado in &relatorio.resultados {
421        let situacao = match &resultado.estado {
422            EstadoDaTarefa::Concluida(resumo) if resumo.reaproveitada => {
423                "ok (sem alterações na origem)".to_owned()
424            }
425            EstadoDaTarefa::Concluida(resumo) => {
426                format!("ok ({} parte(s), {} bytes)", resumo.partes, resumo.bytes)
427            }
428            EstadoDaTarefa::Falhou(erro) => format!("FALHOU: {erro}"),
429            EstadoDaTarefa::Ignorada { causa } => format!("IGNORADA: depende de {causa}"),
430        };
431        println!(
432            "  {:<14} {:>6.1}s  {situacao}",
433            resultado.nome,
434            resultado.duracao.as_secs_f64()
435        );
436    }
437}

439/// Configuração do armazenamento para um dos buckets: `raw` ou `execucoes`.
440fn configuracao_s3(bucket: &str) -> Result<ConfiguracaoS3> {
441    let variavel_do_bucket = format!("TODEOLHO_S3_BUCKET_{}", bucket.to_uppercase());
442    Ok(ConfiguracaoS3 {
443        endpoint: obrigatoria("TODEOLHO_S3_ENDPOINT")?,
444        regiao: env::var("TODEOLHO_S3_REGIAO").unwrap_or_else(|_| "sa-east-1".to_owned()),
445        bucket: env::var(variavel_do_bucket).unwrap_or_else(|_| bucket.to_owned()),
446        chave_de_acesso: obrigatoria("TODEOLHO_S3_CHAVE_DE_ACESSO")?,
447        chave_secreta: obrigatoria("TODEOLHO_S3_CHAVE_SECRETA")?,
448    })
449}
450
451fn obrigatoria(nome: &str) -> Result<String> {
452    env::var(nome).with_context(|| format!("variável de ambiente ausente: {nome}"))
453}
454
455fn concorrencia() -> Result<usize> {
456    match env::var("TODEOLHO_CONCORRENCIA") {
457        Ok(valor) => valor
458            .parse()
459            .with_context(|| format!("TODEOLHO_CONCORRENCIA inválida: {valor}")),
460        Err(_) => Ok(2),
461    }
462}
