41#[derive(Subcommand)]
42enum Comando {
43    /// Abre o menu para escolher o que ingerir, mostrando o espaço necessário
44    Selecionar,
45    /// Lista as fontes de dados disponíveis
46    Listar,
47    /// Mostra a ordem de execução da DAG de uma fonte, sem acessar a rede
48    Plano {
49        /// Nome da fonte (veja `todeolho listar`)
50        fonte: String,
51    },
52    /// Executa a ingestão sem o menu e grava os dados na camada Raw
53    Executar(ArgumentosDeExecucao),
54    /// Inicia o painel web com o histórico das execuções
55    Painel {
56        /// Endereço em que o painel fica disponível
57        #[arg(long, default_value = "0.0.0.0:8080")]
58        endereco: SocketAddr,
59    },
60}

62#[derive(Args)]
63struct ArgumentosDeExecucao {

83#[tokio::main]
84async fn main() -> ExitCode {
85    tracing_subscriber::fmt()
86        .with_env_filter(
87            EnvFilter::try_from_env("TODEOLHO_LOG").unwrap_or_else(|_| EnvFilter::new("info")),
88        )
89        .with_target(false)
90        .with_writer(io::stderr)
91        .with_ansi(io::stderr().is_terminal())
92        .init();
93
94    match rodar(Cli::parse()).await {
95        Ok(true) => ExitCode::SUCCESS,
96        Ok(false) => ExitCode::FAILURE,
97        Err(erro) => {
98            eprintln!("erro: {erro:#}");
99            ExitCode::FAILURE
100        }
101    }
102}

104/// Devolve `false` quando alguma tarefa não foi concluída.
105async fn rodar(cli: Cli) -> Result<bool> {
106    match cli.comando {
107        Comando::Selecionar => selecionar().await,
108        Comando::Listar => {
109            let fontes = catalogo();
110            // Alinha a coluna pelo nome mais longo, para nomes de qualquer tamanho.
111            let largura = fontes.iter().map(|f| f.nome().len()).max().unwrap_or(0);
112            for fonte in &fontes {
113                println!("{:<largura$} {}", fonte.nome(), fonte.descricao());
114            }
115            Ok(true)
116        }
117        Comando::Plano { fonte } => {
118            let fonte = buscar_fonte(&fonte)?;
119            for (indice, nivel) in fonte.montar_dag()?.niveis()?.iter().enumerate() {
120                println!("nível {}: {}", indice + 1, nivel.join(", "));
121            }
122            Ok(true)
123        }
124        Comando::Executar(argumentos) => executar_sem_menu(argumentos).await,
125        Comando::Painel { endereco } => {
126            let historico = historico()?;
127            historico.preparar().await.context(ERRO_DE_ARMAZENAMENTO)?;
128            todeolho_painel::servir(endereco, Arc::new(historico)).await?;
129            Ok(true)
130        }
131    }
132}

134fn buscar_fonte(nome: &str) -> Result<Box<dyn FonteDeDados>> {
135    let fontes = catalogo();
136    let disponiveis: Vec<String> = fontes.iter().map(|f| f.nome().to_owned()).collect();
137    match fontes.into_iter().find(|fonte| fonte.nome() == nome) {
138        Some(fonte) => Ok(fonte),
139        None => bail!(
140            "fonte desconhecida: {nome}. Disponíveis: {}",
141            disponiveis.join(", ")
142        ),
143    }
144}

146async fn selecionar() -> Result<bool> {
147    if !io::stdin().is_terminal() {
148        bail!("o menu precisa de um terminal. Sem terminal, use: todeolho executar <fonte> --sim");
149    }
150
151    let contexto = preparar_contexto().await?;
152    println!("Consultando o tamanho dos dados...");
153
154    let fontes = catalogo();
155    let mut opcoes = Vec::with_capacity(fontes.len());
156    for fonte in &fontes {
157        let tarefas = fonte.criar_tarefas();
158        let estimativas = estimar(&tarefas_de(&tarefas), &contexto).await;
159        opcoes.push(menu::OpcaoDeFonte {
160            nome: fonte.nome().to_owned(),
161            descricao: fonte.descricao().to_owned(),
162            tarefas: tarefas
163                .iter()
164                .zip(estimativas)
165                .map(|(tarefa, bytes)| menu::OpcaoDeTarefa {
166                    etiquetas: tarefa.etiquetas.clone(),
167                    bytes,
168                })
169                .collect(),
170        });
171    }
172
173    let mut entrada = io::stdin().lock();
174    let Some(escolha) = menu::escolher(&opcoes, &mut entrada, &mut io::stdout())? else {
175        println!("Nada foi executado.");
176        return Ok(true);
177    };
178    drop(entrada);
179
180    let mut selecionadas = Vec::with_capacity(escolha.len());
181    for (indice, filtro) in escolha {
182        let fonte = &fontes[indice];
183        selecionadas.push(FonteSelecionada {
184            nome: fonte.nome().to_owned(),
185            tarefas: filtro.aplicar(fonte.criar_tarefas())?,
186        });
187    }
188    executar(selecionadas, contexto, false).await
189}

191async fn executar_sem_menu(argumentos: ArgumentosDeExecucao) -> Result<bool> {
192    let selecionadas = match (argumentos.fonte, argumentos.todas) {
193        (Some(_), true) => bail!("informe uma fonte ou --todas, não os dois"),
194        (None, false) => bail!("informe uma fonte ou --todas"),
195        (None, true) => {
196            if !argumentos.filtros.is_empty() {
197                bail!("--filtro só vale para uma fonte específica");
198            }
199            catalogo()
200                .into_iter()
201                .filter(|fonte| !fonte.sob_demanda())
202                .map(|fonte| FonteSelecionada {
203                    nome: fonte.nome().to_owned(),
204                    tarefas: fonte.criar_tarefas(),
205                })
206                .collect()
207        }
208        (Some(nome), false) => {
209            let fonte = buscar_fonte(&nome)?;
210            let filtro = FiltroDeTarefas::a_partir_de(&argumentos.filtros)?;
211            vec![FonteSelecionada {
212                nome,
213                tarefas: filtro.aplicar(fonte.criar_tarefas())?,
214            }]
215        }
216    };
217
218    let contexto = preparar_contexto().await?;
219    executar(selecionadas, contexto, argumentos.sim).await
220}

222async fn preparar_contexto() -> Result<ContextoDeExecucao> {
223    let armazenamento = Arc::new(ArmazenamentoS3::novo(configuracao_s3("raw")?)?);
224    armazenamento
225        .preparar()
226        .await
227        .context(ERRO_DE_ARMAZENAMENTO)?;
228
229    let iniciada_em = Utc::now();
230    Ok(ContextoDeExecucao {
231        id_execucao: format!(
232            "{}-{}",
233            iniciada_em.format("%Y%m%dT%H%M%SZ"),
234            &Uuid::new_v4().simple().to_string()[..8]
235        ),
236        iniciada_em,
237        armazenamento,
238        http: Arc::new(ClienteHttpComRetentativas::novo(
239            PoliticaDeRetentativas::default(),
240        )?),
241    })
242}

244/// Mostra o espaço necessário, recusa o que não cabe, pede confirmação e executa.
245async fn executar(
246    selecionadas: Vec<FonteSelecionada>,
247    contexto: ContextoDeExecucao,
248    sem_confirmacao: bool,
249) -> Result<bool> {
250    let mut dags = Vec::with_capacity(selecionadas.len());
251    let mut tarefas: Vec<Arc<dyn Tarefa>> = Vec::new();
252    for fonte in selecionadas {
253        tarefas.extend(tarefas_de(&fonte.tarefas));
254        dags.push(DagDeIngestao::construir(&fonte.nome, fonte.tarefas)?);
255    }
256
257    let estimativas = estimar(&tarefas, &contexto).await;
258    let espaco = EspacoNecessario::somar(&estimativas);
259    let livre = env::var("TODEOLHO_DIR_DADOS")
260        .ok()
261        .and_then(|caminho| espaco_livre_em(Path::new(&caminho)));
262
263    if !avisar_sobre_espaco(tarefas.len(), &espaco, livre) {
264        return Ok(false);
265    }
266    if !sem_confirmacao && !confirmar()? {
267        println!("Nada foi executado.");
268        return Ok(true);
269    }
270
271    match contexto.armazenamento.descartar_envios_incompletos().await {
272        Ok(0) => {}
273        Ok(quantidade) => {
274            println!("Espaço recuperado de {quantidade} transferência(s) interrompida(s).")
275        }
276        Err(erro) => {
277            eprintln!("Atenção: não foi possível limpar transferências interrompidas: {erro}")
278        }
279    }
280
281    let historico = Arc::new(historico()?);
282    historico.preparar().await.context(ERRO_DE_ARMAZENAMENTO)?;
283
284    let registrador = Arc::new(RegistradorDeExecucao::novo(
285        RegistroDeExecucao::novo(&contexto.id_execucao, contexto.iniciada_em, &dags)?,
286        historico,
287    ));
288    registrador.publicar().await;
289    tokio::spawn({
290        let registrador = Arc::clone(&registrador);
291        async move { registrador.manter_batimento().await }
292    });
293
294    let executor = ExecutorDeDag::novo(concorrencia()?);
295
296    let mut sucesso = true;
297    for dag in &dags {
298        let relatorio = executor
299            .executar(dag, &contexto, registrador.as_ref())
300            .await?;
301        imprimir(&relatorio);
302        sucesso &= relatorio.sucesso();
303    }
304    registrador.concluir().await;
305
306    println!("\nexecução: {}", contexto.id_execucao);
307    Ok(sucesso)
308}

310/// Devolve `false` quando a seleção não cabe no disco.
311fn avisar_sobre_espaco(quantidade: usize, espaco: &EspacoNecessario, livre: Option<u64>) -> bool {
312    println!("\nSelecionado: {quantidade} tarefa(s)");
313    println!("Espaço necessário: {}", formatar_bytes(espaco.bytes));
314    if espaco.maior_arquivo > 0 {
315        println!(
316            "Pico durante a gravação: {} (o maior arquivo ocupa o dobro enquanto é gravado)",
317            formatar_bytes(espaco.pico())
318        );
319    }
320    if espaco.sem_estimativa > 0 {
321        println!(
322            "Atenção: {} tarefa(s) sem tamanho conhecido. O espaço real pode ser maior.",
323            espaco.sem_estimativa
324        );
325    }
326    if let Some(livre) = livre {
327        println!("Espaço livre: {}", formatar_bytes(livre));
328    }
329
330    match espaco.avaliar(livre) {
331        AvaliacaoDeEspaco::Cabe => true,
332        AvaliacaoDeEspaco::Desconhecido => {
333            println!("Atenção: não foi possível verificar o espaço livre em disco.");
334            true
335        }
336        AvaliacaoDeEspaco::NaoCabe { faltam } => {
337            eprintln!(
338                "\nNão cabe: faltam {}. Escolha menos dados ou libere espaço em disco.",
339                formatar_bytes(faltam)
340            );
341            false
342        }
343    }
344}

346fn confirmar() -> Result<bool> {
347    if !io::stdin().is_terminal() {
348        bail!("sem terminal para confirmar. Use --sim para executar sem a pergunta");
349    }
350
351    print!("\nContinuar? [s/N] ");
352    io::stdout().flush()?;
353    let mut resposta = String::new();
354    io::stdin().read_line(&mut resposta)?;
355    Ok(matches!(
356        resposta.trim().to_lowercase().as_str(),
357        "s" | "sim"
358    ))
359}

372fn imprimir(relatorio: &RelatorioDaExecucao) {
373    println!("\nfonte: {}", relatorio.fonte);
374    for resultado in &relatorio.resultados {
375        let situacao = match &resultado.estado {
376            EstadoDaTarefa::Concluida(resumo) if resumo.reaproveitada => {
377                "ok (sem alterações na origem)".to_owned()
378            }
379            EstadoDaTarefa::Concluida(resumo) => {
380                format!("ok ({} parte(s), {} bytes)", resumo.partes, resumo.bytes)
381            }
382            EstadoDaTarefa::Falhou(erro) => format!("FALHOU: {erro}"),
383            EstadoDaTarefa::Ignorada { causa } => format!("IGNORADA: depende de {causa}"),
384        };
385        println!(
386            "  {:<14} {:>6.1}s  {situacao}",
387            resultado.nome,
388            resultado.duracao.as_secs_f64()
389        );
390    }
391}

393/// Configuração do armazenamento para um dos buckets: `raw` ou `execucoes`.
394fn configuracao_s3(bucket: &str) -> Result<ConfiguracaoS3> {
395    let variavel_do_bucket = format!("TODEOLHO_S3_BUCKET_{}", bucket.to_uppercase());
396    Ok(ConfiguracaoS3 {
397        endpoint: obrigatoria("TODEOLHO_S3_ENDPOINT")?,
398        regiao: env::var("TODEOLHO_S3_REGIAO").unwrap_or_else(|_| "sa-east-1".to_owned()),
399        bucket: env::var(variavel_do_bucket).unwrap_or_else(|_| bucket.to_owned()),
400        chave_de_acesso: obrigatoria("TODEOLHO_S3_CHAVE_DE_ACESSO")?,
401        chave_secreta: obrigatoria("TODEOLHO_S3_CHAVE_SECRETA")?,
402    })
403}
404
405fn obrigatoria(nome: &str) -> Result<String> {
406    env::var(nome).with_context(|| format!("variável de ambiente ausente: {nome}"))
407}

409fn concorrencia() -> Result<usize> {
410    match env::var("TODEOLHO_CONCORRENCIA") {
411        Ok(valor) => valor
412            .parse()
413            .with_context(|| format!("TODEOLHO_CONCORRENCIA inválida: {valor}")),
414        Err(_) => Ok(2),
415    }
