13/// Uma parte por vez em memória: limita o consumo, seja qual for o tamanho do arquivo.
14/// O `put_object_stream` do rust-s3 não serve: dimensiona a concorrência pela memória
15/// livre da máquina, ignorando o limite do container.
16const TAMANHO_DA_PARTE: usize = 16 * 1024 * 1024;
17const TENTATIVAS_DE_ESCRITA: u32 = 4;
18const ESPERA_BASE: std::time::Duration = std::time::Duration::from_millis(200);

42impl ArmazenamentoS3 {
43    async fn repetir_escrita<T, F, Fut>(
44        &self,
45        chave: &str,
46        mut operacao: F,
47    ) -> Result<T, ErroDeIngestao>
48    where
49        F: FnMut() -> Fut,
50        Fut: Future<Output = Result<T, S3Error>>,
51    {
52        for tentativa in 1..=TENTATIVAS_DE_ESCRITA {
53            match operacao().await {
54                Ok(resultado) => return Ok(resultado),
55                Err(erro) if eh_transitorio(&erro) && tentativa < TENTATIVAS_DE_ESCRITA => {
56                    let exponencial = ESPERA_BASE.saturating_mul(1 << (tentativa - 1));
57                    let espera = exponencial + jitter(tentativa);
58                    warn!(%chave, %erro, tentativa, ?espera, "gravação S3 falhou; tentando novamente");
59                    tokio::time::sleep(espera).await;
60                }
61                Err(erro) => return Err(self.erro(chave, erro)),
62            }
63        }
64        unreachable!("ao menos uma tentativa de gravação é feita")
65    }

67    /// Conecta ao serviço S3 com a configuração dada.
68    pub fn novo(configuracao: ConfiguracaoS3) -> Result<Self, ErroDeIngestao> {
69        let erro = |detalhe: String| ErroDeIngestao::Armazenamento {
70            chave: configuracao.bucket.clone(),
71            detalhe,
72        };
73
74        let regiao = Region::Custom {
75            region: configuracao.regiao.clone(),
76            endpoint: configuracao.endpoint.clone(),
77        };
78        let credenciais = Credentials::new(
79            Some(&configuracao.chave_de_acesso),
80            Some(&configuracao.chave_secreta),
81            None,
82            None,
83            None,
84        )
85        .map_err(|e| erro(e.to_string()))?;
86        // Timeout generoso: baixar/subir objetos de vários GB (ZIPs do TSE) passa do padrão.
87        let bucket = Bucket::new(&configuracao.bucket, regiao.clone(), credenciais.clone())
88            .map_err(|e| erro(e.to_string()))?
89            .with_path_style()
90            .with_request_timeout(Duration::from_secs(1800))
91            .map_err(|e| erro(e.to_string()))?;
92
93        Ok(Self {
94            bucket,
95            regiao,
96            credenciais,
97        })
98    }

107    /// Lista todas as chaves sob um prefixo (percorrendo as páginas). Usado pelo refino
108    /// para descobrir quais recursos já foram ingeridos no bucket.
109    pub async fn listar_chaves(&self, prefixo: &str) -> Result<Vec<String>, ErroDeIngestao> {
110        let paginas = self

124impl ArmazenamentoRaw for ArmazenamentoS3 {
125    async fn preparar(&self) -> Result<(), ErroDeIngestao> {
126        let nome = self.bucket.name();
127        if self
128            .bucket
129            .exists()
130            .await
131            .map_err(|e| self.erro(&nome, e))?
132        {
133            return Ok(());
134        }
135
136        let criacao = Bucket::create_with_path_style(
137            &nome,
138            self.regiao.clone(),
139            self.credenciais.clone(),
140            BucketConfiguration::default(),
141        )
142        .await;
143
144        match criacao {
145            Ok(_) => {
146                info!(bucket = %nome, "bucket criado");
147                Ok(())
148            }
149            // Outro processo criou o bucket entre a verificação e a criação.
150            Err(S3Error::HttpFailWithBody(409, _)) => Ok(()),
151            Err(erro) => Err(self.erro(&nome, erro)),
152        }
153    }
154
155    async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao> {
156        self.repetir_escrita(&objeto.chave, || {
157            self.bucket.put_object_with_content_type(
158                &objeto.chave,
159                &objeto.conteudo,
160                &objeto.tipo_de_conteudo,
161            )
162        })
163        .await?;
164        Ok(())
165    }
166
167    async fn remover(&self, chave: &str) -> Result<(), ErroDeIngestao> {
168        self.bucket
169            .delete_object(chave)
170            .await
171            .map_err(|e| self.erro(chave, e))?;
172        Ok(())
173    }
174
175    async fn gravar_fluxo(
176        &self,
177        chave: &str,
178        tipo_de_conteudo: &str,
179        fluxo: &mut (dyn AsyncRead + Send + Unpin),
180    ) -> Result<(), ErroDeIngestao> {
181        let primeira = ler_parte(fluxo).await.map_err(|e| self.erro(chave, e))?;
182        if primeira.len() < TAMANHO_DA_PARTE {
183            self.repetir_escrita(chave, || {
184                self.bucket
185                    .put_object_with_content_type(chave, &primeira, tipo_de_conteudo)
186            })
187            .await?;
188            return Ok(());
189        }
190
191        let envio = self
192            .repetir_escrita(chave, || {
193                self.bucket
194                    .initiate_multipart_upload(chave, tipo_de_conteudo)
195            })
196            .await?;
197
198        let resultado = self
199            .enviar_partes(chave, tipo_de_conteudo, &envio.upload_id, primeira, fluxo)
200            .await;
201        if resultado.is_err() {
202            // Sem isso, as partes já enviadas ficariam ocupando espaço no volume.
203            if let Err(erro) = self.bucket.abort_upload(chave, &envio.upload_id).await {
204                warn!(%chave, %erro, "não foi possível cancelar o envio");
205            }
206        }
207        resultado
208    }
209
210    async fn descartar_envios_incompletos(&self) -> Result<usize, ErroDeIngestao> {
211        let nome = self.bucket.name();
212        let paginas = self
213            .bucket
214            .list_multiparts_uploads(None, None)
215            .await
216            .map_err(|e| self.erro(&nome, e))?;
217
218        let mut descartados = 0;
219        for envio in paginas.iter().flat_map(|pagina| &pagina.uploads) {
220            self.bucket
221                .abort_upload(&envio.key, &envio.id)
222                .await
223                .map_err(|e| self.erro(&envio.key, e))?;
224            descartados += 1;
225        }
226        Ok(descartados)
227    }
228
229    async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao> {
230        match self.bucket.get_object(chave).await {
231            Ok(resposta) => Ok(Some(resposta.bytes().to_vec())),
232            Err(S3Error::HttpFailWithBody(404, _)) => Ok(None),
233            Err(erro) => Err(self.erro(chave, erro)),
234        }
235    }
236
237    async fn ler_fluxo(
238        &self,
239        chave: &str,
240        escrita: &mut (dyn tokio::io::AsyncWrite + Send + Unpin),
241    ) -> Result<bool, ErroDeIngestao> {
242        match self.bucket.get_object_to_writer(chave, escrita).await {
243            Ok(200) => Ok(true),
244            Ok(404) => Ok(false),
245            Ok(status) => Err(self.erro(
246                chave,
247                S3Error::HttpFailWithBody(status, "status inesperado".into()),
248            )),
249            Err(S3Error::HttpFailWithBody(404, _)) => Ok(false),
250            Err(erro) => Err(self.erro(chave, erro)),
251        }
252    }
253
254    /// Um `HEAD`, sem baixar o objeto: a pré-checagem confere o tamanho de
255    /// arquivos grandes sem carregá-los na memória (ver `docs/adr/0012`).
256    async fn tamanho(&self, chave: &str) -> Result<Option<u64>, ErroDeIngestao> {
257        match self.bucket.head_object(chave).await {
258            Ok((cabecalho, _)) => Ok(cabecalho
259                .content_length
260                .and_then(|bytes| u64::try_from(bytes).ok())),
261            Err(S3Error::HttpFailWithBody(404, _)) => Ok(None),
262            Err(erro) => Err(self.erro(chave, erro)),
263        }
264    }

267impl ArmazenamentoS3 {
268    async fn enviar_partes(
269        &self,
270        chave: &str,
271        tipo_de_conteudo: &str,
272        id_do_envio: &str,
273        primeira: Vec<u8>,
274        fluxo: &mut (dyn AsyncRead + Send + Unpin),
275    ) -> Result<(), ErroDeIngestao> {
276        let mut partes = Vec::new();
277        let mut atual = primeira;
278        loop {
279            let parte = self
280                .bucket
281                .put_multipart_chunk(
282                    atual,
283                    chave,
284                    partes.len() as u32 + 1,
285                    id_do_envio,
286                    tipo_de_conteudo,
287                )
288                .await
289                .map_err(|e| self.erro(chave, e))?;
290            partes.push(parte);
291
292            atual = ler_parte(fluxo).await.map_err(|e| self.erro(chave, e))?;
293            if atual.is_empty() {
294                break;
295            }
296        }
297
298        self.repetir_escrita(chave, || {
299            self.bucket
300                .complete_multipart_upload(chave, id_do_envio, partes.clone())
301        })
302        .await?;
303        Ok(())
304    }
305}

307/// Lê até completar uma parte; menos que isso só no fim do fluxo.
308async fn ler_parte(fluxo: &mut (dyn AsyncRead + Send + Unpin)) -> std::io::Result<Vec<u8>> {
309    let mut parte = Vec::with_capacity(TAMANHO_DA_PARTE);
310    fluxo
311        .take(TAMANHO_DA_PARTE as u64)
312        .read_to_end(&mut parte)
313        .await?;
314    Ok(parte)
315}

317fn eh_transitorio(erro: &S3Error) -> bool {
318    match erro {
319        S3Error::HttpFail => true,
320        S3Error::HttpFailWithBody(status, detalhe) => {
321            (500..=599).contains(status)
322                || *status == 408
323                || *status == 429
324                || detalhe
325                    .to_ascii_lowercase()
326                    .contains("end of file before message length reached")
327        }
328        S3Error::Reqwest(erro) => {
329            erro.is_timeout()
330                || erro.is_connect()
331                || erro.is_body()
332                || erro
333                    .to_string()
334                    .to_ascii_lowercase()
335                    .contains("end of file before message length reached")
336        }
337        S3Error::Io(erro) => matches!(
338            erro.kind(),
339            std::io::ErrorKind::TimedOut
340                | std::io::ErrorKind::ConnectionReset
341                | std::io::ErrorKind::ConnectionAborted
342                | std::io::ErrorKind::UnexpectedEof
343        ),
344        _ => false,
345    }
346}

348fn jitter(tentativa: u32) -> std::time::Duration {
349    static SEQUENCIA: AtomicU64 = AtomicU64::new(0x9e37_79b9_7f4a_7c15);
350    let agora = std::time::SystemTime::now()
351        .duration_since(std::time::UNIX_EPOCH)
352        .map_or(0, |duracao| duracao.subsec_nanos() as u64);
353    let valor = SEQUENCIA
354        .fetch_add(0x9e37_79b9_7f4a_7c15, Ordering::Relaxed)
355        .wrapping_add(agora)
356        .wrapping_add(u64::from(tentativa));
357    std::time::Duration::from_millis(valor % 200)
358}

417    /// Simula somente o protocolo HTTP que o SDK usa nos cenários exercitados.
418    /// Cada resposta fecha a conexão para manter o teste independente de estado.
419    fn servidor_s3(respostas: Vec<RespostaS3>) -> (String, JoinHandle<Vec<String>>) {
420        let listener = TcpListener::bind("127.0.0.1:0").expect("bind loopback");
421        listener
422            .set_nonblocking(true)
423            .expect("listener não bloqueante");
424        let endpoint = format!("http://{}", listener.local_addr().unwrap());
425        let servidor = std::thread::spawn(move || {
426            let mut requisicoes = Vec::new();
427            for resposta in respostas {
428                let prazo = Instant::now() + Duration::from_secs(10);
429                let mut stream = loop {
430                    match listener.accept() {
431                        Ok((stream, _)) => break stream,
432                        Err(erro) if erro.kind() == std::io::ErrorKind::WouldBlock => {
433                            if Instant::now() >= prazo {
434                                return requisicoes;
435                            }
436                            std::thread::sleep(Duration::from_millis(5));
437                        }
438                        Err(erro) => panic!("falha ao aceitar conexão do SDK: {erro}"),
439                    }
440                };
441                stream
442                    .set_read_timeout(Some(Duration::from_secs(10)))
443                    .unwrap();
444
445                let mut leitor = BufReader::new(stream.try_clone().unwrap());
446                let requisicao = ler_requisicao(&mut leitor, &mut stream).unwrap();
447                requisicoes.push(requisicao);
448
449                let texto_status = match resposta.status {
450                    200 => "OK",
451                    204 => "No Content",
452                    404 => "Not Found",
453                    status => panic!("status de mock não previsto: {status}"),
454                };
455                let tamanho_corpo = resposta
456                    .cabecalhos
457                    .lines()
458                    .find_map(|linha| {
459                        let (nome, valor) = linha.split_once(':')?;
460                        nome.eq_ignore_ascii_case("content-length")
461                            .then(|| valor.trim().to_owned())
462                    })
463                    .unwrap_or_else(|| resposta.corpo.len().to_string());
464                let content_length = if resposta
465                    .cabecalhos
466                    .lines()
467                    .any(|linha| linha.to_ascii_lowercase().starts_with("content-length:"))
468                {
469                    String::new()
470                } else {
471                    format!("Content-Length: {tamanho_corpo}\r\n")
472                };
473                let cabecalho = format!(
474                    "HTTP/1.1 {} {}\r\n{}Connection: close\r\n{}\r\n",
475                    resposta.status, texto_status, content_length, resposta.cabecalhos,
476                );
477                stream.write_all(cabecalho.as_bytes()).unwrap();
478                stream.write_all(resposta.corpo.as_bytes()).unwrap();
479                stream.flush().unwrap();
480            }
481            requisicoes
482        });
483
484        (endpoint, servidor)
485    }

487    fn ler_requisicao(
488        leitor: &mut BufReader<TcpStream>,
489        stream: &mut TcpStream,
490    ) -> std::io::Result<String> {
491        let mut texto = String::new();
492        let mut linha = String::new();
493        leitor.read_line(&mut linha)?;
494        texto.push_str(&linha);
495        let mut tamanho = 0u64;
496        let mut chunked = false;
497        let mut espera = false;
498        loop {
499            linha.clear();
500            leitor.read_line(&mut linha)?;
501            if linha == "\r\n" || linha.is_empty() {
502                break;
503            }
504            let (nome, valor) = linha.split_once(':').unwrap_or(("", ""));
505            match nome.to_ascii_lowercase().as_str() {
506                "content-length" => tamanho = valor.trim().parse().unwrap_or(0),
507                "transfer-encoding" => chunked = valor.to_ascii_lowercase().contains("chunked"),
508                "expect" => espera = valor.trim().eq_ignore_ascii_case("100-continue"),
509                _ => {}
510            }
511            texto.push_str(&linha);
512        }
513        if espera {
514            stream.write_all(b"HTTP/1.1 100 Continue\r\n\r\n")?;
515        }
516
517        if chunked {
518            loop {
519                linha.clear();
520                leitor.read_line(&mut linha)?;
521                let tamanho_chunk =
522                    u64::from_str_radix(linha.trim().split(';').next().unwrap_or("0"), 16)
523                        .unwrap_or(0);
524                if tamanho_chunk == 0 {
525                    loop {
526                        linha.clear();
527                        leitor.read_line(&mut linha)?;
528                        if linha == "\r\n" || linha.is_empty() {
529                            break;
530                        }
531                    }
532                    break;
533                }
534                std::io::copy(
535                    &mut (&mut *leitor).take(tamanho_chunk),
536                    &mut std::io::sink(),
537                )?;
538                let mut crlf = [0; 2];
539                leitor.read_exact(&mut crlf)?;
540            }
541        } else if tamanho > 0 {
542            std::io::copy(&mut (&mut *leitor).take(tamanho), &mut std::io::sink())?;
543        }
544        Ok(texto)
545    }

642        let concluida = "<CompleteMultipartUploadResult><Location>http://localhost/grande.bin</Location><Bucket>teste-raw-79</Bucket><Key>grande.bin</Key><ETag>\"final\"</ETag></CompleteMultipartUploadResult>";
643        let (endpoint, servidor) = servidor_s3(vec![
644            resposta_s3(200, "Content-Type: application/xml\r\n", iniciada),
645            resposta_s3(200, "ETag: \"parte-1\"\r\n", ""),
646            resposta_s3(200, "ETag: \"parte-2\"\r\n", ""),
647            resposta_s3(200, "Content-Type: application/xml\r\n", concluida),
648        ]);
649        let s3 = armazenamento_local(endpoint);
650        let dados = vec![7u8; TAMANHO_DA_PARTE + 1];
651        let mut fluxo: &[u8] = &dados;
652
653        s3.gravar_fluxo("grande.bin", "application/octet-stream", &mut fluxo)
654            .await
655            .unwrap();
656
657        let requisicoes = servidor.join().unwrap();
658        assert_eq!(requisicoes.len(), 4, "inicia, envia duas partes e conclui");
659        assert!(requisicoes[0].starts_with("POST /teste-raw-79/grande.bin?uploads"));
660        assert!(requisicoes[1].starts_with("PUT /teste-raw-79/grande.bin?partNumber=1"));
661        assert!(
662            requisicoes[1]
663                .to_ascii_lowercase()
664                .contains(&format!("content-length: {TAMANHO_DA_PARTE}"))
665        );
666        assert!(requisicoes[2].starts_with("PUT /teste-raw-79/grande.bin?partNumber=2"));
667        assert!(
668            requisicoes[2]
669                .to_ascii_lowercase()
670                .contains("content-length: 1")
671        );
672        assert!(requisicoes[3].starts_with("POST /teste-raw-79/grande.bin?uploadId=upload-1"));
673    }
