9/// Uma parte por vez em memória: limita o consumo, seja qual for o tamanho do arquivo.
10/// O `put_object_stream` do rust-s3 não serve: dimensiona a concorrência pela memória
11/// livre da máquina, ignorando o limite do container.
12const TAMANHO_DA_PARTE: usize = 16 * 1024 * 1024;
13
14/// Dados de conexão a um serviço compatível com S3.
15#[derive(Debug, Clone)]
16pub struct ConfiguracaoS3 {
17    /// Endpoint do serviço (ex.: `http://armazenamento:7070`).
18    pub endpoint: String,
19    /// Região a enviar nas requisições.
20    pub regiao: String,
21    /// Nome do bucket.
22    pub bucket: String,
23    /// Chave de acesso (credencial).
24    pub chave_de_acesso: String,
25    /// Chave secreta (credencial).
26    pub chave_secreta: String,
27}

29/// Camada Raw em qualquer serviço compatível com a API S3 (ver `docs/adr/0005`).
30pub struct ArmazenamentoS3 {
31    pub(crate) bucket: Box<Bucket>,
32    regiao: Region,
33    credenciais: Credentials,
34}
35
36impl ArmazenamentoS3 {
37    /// Conecta ao serviço S3 com a configuração dada.
38    pub fn novo(configuracao: ConfiguracaoS3) -> Result<Self, ErroDeIngestao> {
39        let erro = |detalhe: String| ErroDeIngestao::Armazenamento {
40            chave: configuracao.bucket.clone(),
41            detalhe,
42        };
43
44        let regiao = Region::Custom {
45            region: configuracao.regiao.clone(),
46            endpoint: configuracao.endpoint.clone(),
47        };
48        let credenciais = Credentials::new(
49            Some(&configuracao.chave_de_acesso),
50            Some(&configuracao.chave_secreta),
51            None,
52            None,
53            None,
54        )
55        .map_err(|e| erro(e.to_string()))?;
56        let bucket = Bucket::new(&configuracao.bucket, regiao.clone(), credenciais.clone())
57            .map_err(|e| erro(e.to_string()))?
58            .with_path_style();
59
60        Ok(Self {
61            bucket,
62            regiao,
63            credenciais,
64        })
65    }
66
67    pub(crate) fn erro(&self, chave: &str, detalhe: impl ToString) -> ErroDeIngestao {
68        ErroDeIngestao::Armazenamento {
69            chave: chave.to_owned(),
70            detalhe: detalhe.to_string(),
71        }
72    }
73}

75#[async_trait]
76impl ArmazenamentoRaw for ArmazenamentoS3 {
77    async fn preparar(&self) -> Result<(), ErroDeIngestao> {
78        let nome = self.bucket.name();
79        if self
80            .bucket
81            .exists()
82            .await
83            .map_err(|e| self.erro(&nome, e))?
84        {
85            return Ok(());
86        }
87
88        let criacao = Bucket::create_with_path_style(
89            &nome,
90            self.regiao.clone(),
91            self.credenciais.clone(),
92            BucketConfiguration::default(),
93        )
94        .await;
95
96        match criacao {
97            Ok(_) => {
98                info!(bucket = %nome, "bucket criado");
99                Ok(())
100            }
101            // Outro processo criou o bucket entre a verificação e a criação.
102            Err(S3Error::HttpFailWithBody(409, _)) => Ok(()),
103            Err(erro) => Err(self.erro(&nome, erro)),
104        }
105    }
106
107    async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao> {
108        self.bucket
109            .put_object_with_content_type(&objeto.chave, &objeto.conteudo, &objeto.tipo_de_conteudo)
110            .await
111            .map_err(|e| self.erro(&objeto.chave, e))?;
112        Ok(())
113    }
114
115    async fn gravar_fluxo(
116        &self,
117        chave: &str,
118        tipo_de_conteudo: &str,
119        fluxo: &mut (dyn AsyncRead + Send + Unpin),
120    ) -> Result<(), ErroDeIngestao> {
121        let primeira = ler_parte(fluxo).await.map_err(|e| self.erro(chave, e))?;
122        if primeira.len() < TAMANHO_DA_PARTE {
123            self.bucket
124                .put_object_with_content_type(chave, &primeira, tipo_de_conteudo)
125                .await
126                .map_err(|e| self.erro(chave, e))?;
127            return Ok(());
128        }
129
130        let envio = self
131            .bucket
132            .initiate_multipart_upload(chave, tipo_de_conteudo)
133            .await
134            .map_err(|e| self.erro(chave, e))?;
135
136        let resultado = self
137            .enviar_partes(chave, tipo_de_conteudo, &envio.upload_id, primeira, fluxo)
138            .await;
139        if resultado.is_err() {
140            // Sem isso, as partes já enviadas ficariam ocupando espaço no volume.
141            if let Err(erro) = self.bucket.abort_upload(chave, &envio.upload_id).await {
142                warn!(%chave, %erro, "não foi possível cancelar o envio");
143            }
144        }
145        resultado
146    }
147
148    async fn descartar_envios_incompletos(&self) -> Result<usize, ErroDeIngestao> {
149        let nome = self.bucket.name();
150        let paginas = self
151            .bucket
152            .list_multiparts_uploads(None, None)
153            .await
154            .map_err(|e| self.erro(&nome, e))?;
155
156        let mut descartados = 0;
157        for envio in paginas.iter().flat_map(|pagina| &pagina.uploads) {
158            self.bucket
159                .abort_upload(&envio.key, &envio.id)
160                .await
161                .map_err(|e| self.erro(&envio.key, e))?;
162            descartados += 1;
163        }
164        Ok(descartados)
165    }
166
167    async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao> {
168        match self.bucket.get_object(chave).await {
169            Ok(resposta) => Ok(Some(resposta.bytes().to_vec())),
170            Err(S3Error::HttpFailWithBody(404, _)) => Ok(None),
171            Err(erro) => Err(self.erro(chave, erro)),
172        }
173    }
174
175    /// Um `HEAD`, sem baixar o objeto: a pré-checagem confere o tamanho de
176    /// arquivos grandes sem carregá-los na memória (ver `docs/adr/0012`).
177    async fn tamanho(&self, chave: &str) -> Result<Option<u64>, ErroDeIngestao> {
178        match self.bucket.head_object(chave).await {
179            Ok((cabecalho, _)) => Ok(cabecalho
180                .content_length
181                .and_then(|bytes| u64::try_from(bytes).ok())),
182            Err(S3Error::HttpFailWithBody(404, _)) => Ok(None),
183            Err(erro) => Err(self.erro(chave, erro)),
184        }
185    }
186}

188impl ArmazenamentoS3 {
189    async fn enviar_partes(
190        &self,
191        chave: &str,
192        tipo_de_conteudo: &str,
193        id_do_envio: &str,
194        primeira: Vec<u8>,
195        fluxo: &mut (dyn AsyncRead + Send + Unpin),
196    ) -> Result<(), ErroDeIngestao> {
197        let mut partes = Vec::new();
198        let mut atual = primeira;
199        loop {
200            let parte = self
201                .bucket
202                .put_multipart_chunk(
203                    atual,
204                    chave,
205                    partes.len() as u32 + 1,
206                    id_do_envio,
207                    tipo_de_conteudo,
208                )
209                .await
210                .map_err(|e| self.erro(chave, e))?;
211            partes.push(parte);
212
213            atual = ler_parte(fluxo).await.map_err(|e| self.erro(chave, e))?;
214            if atual.is_empty() {
215                break;
216            }
217        }
218
219        self.bucket
220            .complete_multipart_upload(chave, id_do_envio, partes)
221            .await
222            .map_err(|e| self.erro(chave, e))?;
223        Ok(())
224    }
225}
226
227/// Lê até completar uma parte; menos que isso só no fim do fluxo.
228async fn ler_parte(fluxo: &mut (dyn AsyncRead + Send + Unpin)) -> std::io::Result<Vec<u8>> {
229    let mut parte = Vec::with_capacity(TAMANHO_DA_PARTE);
230    fluxo
231        .take(TAMANHO_DA_PARTE as u64)
232        .read_to_end(&mut parte)
233        .await?;
234    Ok(parte)
235}

241    /// Lê `nome`, caindo para `alternativa` (as mesmas variáveis do serviço local,
242    /// `TODEOLHO_S3_*`) e, por fim, para `padrao`.
243    fn var(nome: &str, alternativa: &str, padrao: &str) -> String {
244        std::env::var(nome)
245            .or_else(|_| std::env::var(alternativa))
246            .unwrap_or_else(|_| padrao.to_owned())
247    }

271    /// Conecta a um **bucket dedicado e único** por teste, para que um teste não
272    /// interfira em outro nem em dados reais — `descartar_envios_incompletos`, por
273    /// exemplo, cancela todos os multiparts do bucket. Devolve `None` (pula) quando
274    /// `TODEOLHO_TESTE_S3_ENDPOINT` não está definido; mesmo assim os testes de
275    /// integração são `#[ignore]`, então só rodam com `--ignored` e o versitygw no ar.
276    fn s3_em_bucket_novo(prefixo: &str) -> Option<ArmazenamentoS3> {
277        let endpoint = std::env::var("TODEOLHO_TESTE_S3_ENDPOINT").ok()?;
278        let nanos = std::time::SystemTime::now()
279            .duration_since(std::time::UNIX_EPOCH)
280            .unwrap()
281            .as_nanos();
282        let bucket = format!("teste-{prefixo}-{nanos}");
283        Some(
284            ArmazenamentoS3::novo(config_de_teste(endpoint, bucket))
285                .expect("config de teste válida"),
286        )
287    }
