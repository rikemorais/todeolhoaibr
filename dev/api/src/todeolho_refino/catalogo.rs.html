1//! Catálogo de refino: liga cada fonte do bronze ao seu decodificador e ao recurso no raw.
2//!
3//! Espelha o papel do catálogo da ingestão (`fontes::catalogo`), mas para a decodificação:
4//! diz de qual fonte/prefixo do raw cada refino lê e qual [`Decodificador`] usa.
5
6use std::sync::Arc;
7
8use crate::decodificador::Decodificador;
9use crate::fontes::DecodificadorCamaraDespesas;
10
11/// Uma fonte do bronze: o nome no catálogo, de onde lê no raw e como decodifica.
12pub struct FonteDeRefino {
13    /// Nome usado na CLI e no painel (ex.: `camara-despesas`).
14    pub nome: &'static str,
15    /// Fonte no raw de onde os recursos são lidos (ex.: `camara`).
16    pub fonte_raw: &'static str,
17    /// Prefixo dos recursos desta fonte dentro do raw (ex.: `despesas/`).
18    pub prefixo_recurso: &'static str,
19    cria_decodificador: fn() -> Arc<dyn Decodificador>,
20}
21
22impl FonteDeRefino {
23    /// Cria o decodificador desta fonte.
24    pub fn decodificador(&self) -> Arc<dyn Decodificador> {
25        (self.cria_decodificador)()
26    }
27}
28
29/// Fontes do bronze disponíveis. Cresce conforme novos decodificadores entram.
30pub fn catalogo_refino() -> Vec<FonteDeRefino> {
31    vec![FonteDeRefino {
32        nome: "camara-despesas",
33        fonte_raw: "camara",
34        prefixo_recurso: "despesas/",
35        cria_decodificador: || Arc::new(DecodificadorCamaraDespesas),
36    }]
37}
38
39#[cfg(test)]
40mod testes {
41    use super::*;
42
43    #[test]
44    fn o_catalogo_tem_a_camara_despesas_apontando_para_o_raw_camara() {
45        let fonte = catalogo_refino()
46            .into_iter()
47            .find(|f| f.nome == "camara-despesas")
48            .expect("camara-despesas no catálogo");
49        assert_eq!(fonte.fonte_raw, "camara");
50        assert_eq!(fonte.prefixo_recurso, "despesas/");
51        // O decodificador é construível.
52        let _ = fonte.decodificador();
53    }
54}
