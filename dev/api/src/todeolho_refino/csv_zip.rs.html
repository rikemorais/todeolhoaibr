1//! Helper compartilhado: decodificar um ZIP de CSVs (reencodando para UTF-8) em Parquet.
2//!
3//! Boa parte das fontes de arquivo grava, no raw, um ZIP com um ou mais CSVs — às vezes em
4//! Latin-1 (CEAPS do Senado, TSE), às vezes em UTF-8 (CEAP da Câmara). O bronze só traduz:
5//! descompacta, reencoda e põe em colunas, mantendo os valores como texto.
6//!
7//! A escrita é **em lotes** (row groups): as linhas são agrupadas e gravadas aos poucos, de
8//! modo que o pico de memória fique no tamanho de um lote, não no arquivo inteiro.
9
10use std::io::{Cursor, Read, Seek};

12use crate::decodificador::ParteBronze;
13use crate::escritor::{ErroDeRefino, EscritorParquet};

15/// Quantas linhas acumular antes de gravar um row group.
16const LINHAS_POR_LOTE: usize = 50_000;
17
18/// Decodifica, para UTF-8, os bytes de um CSV que pode estar em UTF-8 ou em Windows-1252
19/// (Latin-1). Tenta UTF-8 primeiro; se não for válido, assume Windows-1252, que nunca falha.
20/// Remove o BOM de UTF-8 quando presente (ex.: arquivos da Câmara).
21fn para_utf8(bytes: &[u8]) -> String {
22    let mut texto = match std::str::from_utf8(bytes) {
23        Ok(texto) => texto.to_owned(),
24        Err(_) => encoding_rs::WINDOWS_1252.decode(bytes).0.into_owned(),
25    };
26    if texto.starts_with('\u{feff}') {
27        texto.drain(..'\u{feff}'.len_utf8());
28    }
29    texto
30}

32/// Descarta `n` linhas físicas do início do texto (preâmbulo antes do cabeçalho, ex.: a
33/// linha "ULTIMA ATUALIZACAO" do CEAPS do Senado).
34fn descartar_linhas(texto: &str, n: usize) -> &str {
35    let mut resto = texto;
36    for _ in 0..n {
37        match resto.find('\n') {
38            Some(fim) => resto = &resto[fim + 1..],
39            None => return "",
40        }
41    }
42    resto
43}

45/// Abre um CSV e separa o cabeçalho (colunas) do leitor de registros.
46fn colunas_e_registros<R: Read>(
47    leitor: R,
48    delimitador: u8,
49) -> Result<(Vec<String>, csv::Reader<R>), ErroDeRefino> {
50    let mut leitor = csv::ReaderBuilder::new()
51        .delimiter(delimitador)
52        .flexible(true)
53        .from_reader(leitor);
54    let colunas = leitor
55        .headers()?
56        .iter()
57        .map(|c| c.trim().to_owned())
58        .collect();
59    Ok((colunas, leitor))
60}

62/// Lê os registros em lotes e os grava no escritor dado (memória ou arquivo).
63fn alimentar<R: Read, W: std::io::Write + Send>(
64    mut registros: csv::Reader<R>,
65    escritor: &mut EscritorParquet<W>,
66    num_colunas: usize,
67) -> Result<(), ErroDeRefino> {
68    let mut lote: Vec<Vec<Option<String>>> = Vec::with_capacity(LINHAS_POR_LOTE);
69    for registro in registros.records() {
70        let registro = registro?;
71        // Mantém o número de colunas do cabeçalho: completa linhas curtas com nulos e
72        // ignora células excedentes, sem interpretar o conteúdo (fidelidade do bronze).
73        let linha: Vec<Option<String>> = (0..num_colunas)
74            .map(|i| registro.get(i).map(|v| v.to_owned()))
75            .collect();
76        lote.push(linha);
77        if lote.len() >= LINHAS_POR_LOTE {
78            escritor.escrever_lote(&lote)?;
79            lote.clear();
80        }
81    }
82    escritor.escrever_lote(&lote)
83}

85/// Lê um CSV de um `Read` e devolve o Parquet **em memória** (para arquivos modestos).
86fn parquet_de_leitor<R: Read>(leitor: R, delimitador: u8) -> Result<Vec<u8>, ErroDeRefino> {
87    let (colunas, registros) = colunas_e_registros(leitor, delimitador)?;
88    let mut escritor = EscritorParquet::novo(&colunas)?;
89    alimentar(registros, &mut escritor, colunas.len())?;
90    escritor.finalizar()
91}

93/// Lê um CSV de um `Read` e grava o Parquet **direto em um arquivo temporário**, sem manter
94/// o Parquet inteiro em memória — para arquivos grandes (#171).
95fn parte_em_arquivo<R: Read>(
96    nome: &str,
97    leitor: R,
98    delimitador: u8,
99) -> Result<ParteBronze, ErroDeRefino> {
100    let (colunas, registros) = colunas_e_registros(leitor, delimitador)?;
101    let arquivo = tempfile::NamedTempFile::new()?;
102    let mut escritor = EscritorParquet::novo_em(arquivo.reopen()?, &colunas)?;
103    alimentar(registros, &mut escritor, colunas.len())?;
104    escritor.fechar()?;
105    Ok(ParteBronze::de_arquivo(nome, arquivo.into_temp_path()))
106}

108/// Lê um CSV (já em UTF-8, em memória) pulando `linhas_a_pular` de preâmbulo.
109fn parquet_de_csv(
110    texto: &str,
111    delimitador: u8,
112    linhas_a_pular: usize,
113) -> Result<Vec<u8>, ErroDeRefino> {
114    parquet_de_leitor(
115        descartar_linhas(texto, linhas_a_pular).as_bytes(),
116        delimitador,
117    )
118}

120/// Extensões tabulares aceitas dentro do ZIP (o TSE usa `.csv` nos anos recentes e `.txt`
121/// nos antigos, com o mesmo formato `;`/Latin-1).
122fn e_tabular(nome: &str) -> bool {
123    let n = nome.to_ascii_lowercase();
124    n.ends_with(".csv") || n.ends_with(".txt")
125}
126
127/// Nome lógico de um arquivo interno do ZIP: o basename sem a extensão `.csv`/`.txt`.
128fn nome_logico(nome_interno: &str) -> String {
129    let base = nome_interno.rsplit('/').next().unwrap_or(nome_interno);
130    match base.len() {
131        n if n >= 4
132            && (base[n - 4..].eq_ignore_ascii_case(".csv")
133                || base[n - 4..].eq_ignore_ascii_case(".txt")) =>
134        {
135            base[..n - 4].to_owned()
136        }
137        _ => base.to_owned(),
138    }
139}

141/// Descompacta um ZIP em memória e decodifica cada CSV em uma parte Parquet, nomeada pelo
142/// arquivo interno (sem a extensão). Entradas que não sejam `.csv` são ignoradas.
143pub fn partes_de_zip_csv(bytes: &[u8], delimitador: u8) -> Result<Vec<ParteBronze>, ErroDeRefino> {
144    let mut arquivo = zip::ZipArchive::new(Cursor::new(bytes))?;
145    let mut partes = Vec::new();
146
147    for indice in 0..arquivo.len() {
148        let mut entrada = arquivo.by_index(indice)?;
149        if !entrada.is_file() {
150            continue;
151        }
152        let nome_interno = entrada.name()?.into_owned();
153        if !e_tabular(&nome_interno) {
154            continue;
155        }
156        let mut conteudo = Vec::new();
157        entrada.read_to_end(&mut conteudo)?;
158        let texto = para_utf8(&conteudo);
159        drop(conteudo);
160        let parquet = parquet_de_csv(&texto, delimitador, 0)?;
161        partes.push(ParteBronze::nova(nome_logico(&nome_interno), &parquet)?);
162    }
163
164    if partes.is_empty() {
165        return Err(ErroDeRefino::Decodificacao(
166            "nenhum CSV encontrado no ZIP".to_owned(),
167        ));
168    }
169    Ok(partes)
170}

172/// Como [`partes_de_zip_csv`], mas decodifica cada CSV **em fluxo**, direto da entrada do
173/// ZIP, transcodificando de Latin-1 na hora — para arquivos grandes (ex.: CSVs consolidados
174/// do TSE, de vários GB), em que materializar o arquivo inteiro estouraria a memória (#171).
175pub fn partes_de_zip_csv_em_fluxo<F: Read + Seek>(
176    fonte: F,
177    delimitador: u8,
178) -> Result<Vec<ParteBronze>, ErroDeRefino> {
179    let mut arquivo = zip::ZipArchive::new(fonte)?;
180    let mut partes = Vec::new();
181
182    for indice in 0..arquivo.len() {
183        let entrada = arquivo.by_index(indice)?;
184        if !entrada.is_file() {
185            continue;
186        }
187        let nome_interno = entrada.name()?.into_owned();
188        if !e_tabular(&nome_interno) {
189            continue;
190        }
191        // Transcodifica Latin-1 -> UTF-8 em fluxo (os ZIPs em fluxo são do TSE, Latin-1).
192        let leitor = encoding_rs_io::DecodeReaderBytesBuilder::new()
193            .encoding(Some(encoding_rs::WINDOWS_1252))
194            .build(entrada);
195        partes.push(parte_em_arquivo(
196            &nome_logico(&nome_interno),
197            leitor,
198            delimitador,
199        )?);
200    }
201
202    if partes.is_empty() {
203        return Err(ErroDeRefino::Decodificacao(
204            "nenhum CSV encontrado no ZIP".to_owned(),
205        ));
206    }
207    Ok(partes)
208}

210/// Decodifica um CSV "solto" (não zipado) em uma única parte Parquet, com o `nome` lógico
211/// dado. `linhas_a_pular` descarta um preâmbulo antes do cabeçalho (ex.: CEAPS do Senado).
212pub fn partes_de_csv(
213    bytes: &[u8],
214    nome: &str,
215    delimitador: u8,
216    linhas_a_pular: usize,
217) -> Result<Vec<ParteBronze>, ErroDeRefino> {
218    let texto = para_utf8(bytes);
219    let parquet = parquet_de_csv(&texto, delimitador, linhas_a_pular)?;
220    Ok(vec![ParteBronze::nova(nome, &parquet)?])
221}

231    fn zip_com(nome: &str, conteudo: &[u8]) -> Vec<u8> {
232        let mut buffer = Vec::new();
233        {
234            let mut zw = zip::ZipWriter::new(Cursor::new(&mut buffer));
235            zw.start_file(nome, SimpleFileOptions::default()).unwrap();
236            zw.write_all(conteudo).unwrap();
237            zw.finish().unwrap();
238        }
239        buffer
240    }

242    /// Lê o Parquet de uma [`ParteBronze`] (em disco) de volta em (colunas, linhas).
243    pub(crate) fn ler_parte(parte: &ParteBronze) -> (Vec<String>, Vec<Vec<Option<String>>>) {
244        ler_parquet(&std::fs::read(&parte.parquet).unwrap())
245    }
246
247    /// Lê um Parquet (bytes) de volta em (colunas, linhas) para as asserções.
248    pub(crate) fn ler_parquet(bytes: &[u8]) -> (Vec<String>, Vec<Vec<Option<String>>>) {
249        let builder =
250            ParquetRecordBatchReaderBuilder::try_new(bytes::Bytes::from(bytes.to_vec())).unwrap();
251        let colunas: Vec<String> = builder
252            .schema()
253            .fields()
254            .iter()
255            .map(|f| f.name().to_owned())
256            .collect();
257        let mut linhas = Vec::new();
258        for lote in builder.build().unwrap() {
259            let lote = lote.unwrap();
260            for linha in 0..lote.num_rows() {
261                let celulas = (0..lote.num_columns())
262                    .map(|coluna| {
263                        let arr = lote
264                            .column(coluna)
265                            .as_any()
266                            .downcast_ref::<StringArray>()
267                            .unwrap();
268                        if arr.is_null(linha) {
269                            None
270                        } else {
271                            Some(arr.value(linha).to_owned())
272                        }
273                    })
274                    .collect();
275                linhas.push(celulas);
276            }
277        }
278        (colunas, linhas)
279    }

281    fn c(s: &str) -> Option<String> {
282        Some(s.to_owned())

285    #[test]
286    fn decodifica_csv_utf8_separado_por_ponto_e_virgula() {
287        let csv = "id;nome\n1;São Paulo\n2;\n";
288        let partes = partes_de_zip_csv(&zip_com("Ano-2024.csv", csv.as_bytes()), b';').unwrap();
289        assert_eq!(partes.len(), 1);
290        assert_eq!(partes[0].nome, "Ano-2024");
291        let (colunas, linhas) = ler_parte(&partes[0]);
292        assert_eq!(colunas, vec!["id", "nome"]);
293        assert_eq!(
294            linhas,
295            vec![vec![c("1"), c("São Paulo")], vec![c("2"), c("")],]
296        );
297    }

301        // "João" em Windows-1252: 0x4A 0x6F 0xE3 0x6F.
302        let mut csv = b"cidade\n".to_vec();
303        csv.extend_from_slice(&[0x4A, 0x6F, 0xE3, 0x6F]);
304        csv.push(b'\n');
305        let partes = partes_de_zip_csv(&zip_com("x.csv", &csv), b';').unwrap();
306        let (_, linhas) = ler_parte(&partes[0]);
307        assert_eq!(linhas, vec![vec![c("João")]]);

336    #[test]
337    fn grava_em_varios_lotes_sem_perder_linhas() {
338        // Mais linhas que um lote, para exercitar múltiplos row groups.
339        let mut csv = String::from("n\n");
340        let total = LINHAS_POR_LOTE + 123;
341        for i in 0..total {
342            csv.push_str(&format!("{i}\n"));
343        }
344        let partes = partes_de_zip_csv(&zip_com("g.csv", csv.as_bytes()), b';').unwrap();
345        let (_, linhas) = ler_parte(&partes[0]);
346        assert_eq!(linhas.len(), total);
347        assert_eq!(linhas[0], vec![c("0")]);
348        assert_eq!(linhas[total - 1], vec![c(&(total - 1).to_string())]);
349    }
