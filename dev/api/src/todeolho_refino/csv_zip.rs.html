1//! Helper compartilhado: decodificar um ZIP de CSVs (reencodando para UTF-8) em Parquet.
2//!
3//! Boa parte das fontes de arquivo grava, no raw, um ZIP com um ou mais CSVs — às vezes em
4//! Latin-1 (CEAPS do Senado, TSE), às vezes em UTF-8 (CEAP da Câmara). O bronze só traduz:
5//! descompacta, reencoda e põe em colunas, mantendo os valores como texto.
6//!
7//! A escrita é **em lotes** (row groups): as linhas são agrupadas e gravadas aos poucos, de
8//! modo que o pico de memória fique no tamanho de um lote, não no arquivo inteiro.
9

13use crate::escritor::{ErroDeRefino, EscritorParquet};

15/// Quantas linhas acumular antes de gravar um row group.
16const LINHAS_POR_LOTE: usize = 50_000;
17
18/// Leitor que auto-detecta encoding: UTF-8 passa direto (sem BOM), senão transcodifica
19/// de Latin-1. Usa `encoding_rs_io` em ambos os caminhos para garantir remoção de BOM e
20/// tratamento correto na fronteira de buffer.
21enum LeitorAutoEncoding<R> {
22    Utf8(encoding_rs_io::DecodeReaderBytes<BufReader<R>, Vec<u8>>),
23    Latin1(encoding_rs_io::DecodeReaderBytes<BufReader<R>, Vec<u8>>),
24}
25
26impl<R: Read> Read for LeitorAutoEncoding<R> {
27    fn read(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
28        match self {
29            Self::Utf8(r) => r.read(buf),
30            Self::Latin1(r) => r.read(buf),
31        }
32    }
33}
34
35/// Verifica se a amostra é UTF-8 válido, tolerando uma sequência multi-byte incompleta
36/// na fronteira do buffer. `error_len() == None` indica truncamento, não byte inválido.
37fn amostra_e_utf8(amostra: &[u8]) -> bool {
38    match std::str::from_utf8(amostra) {
39        Ok(_) => true,
40        Err(e) => e.error_len().is_none(),
41    }
42}
43
44/// Envolve um `Read` num leitor que auto-detecta o encoding a partir dos primeiros bytes
45/// buffered. Se o conteúdo disponível no buffer for UTF-8 válido, os bytes passam direto
46/// (BOM removido); senão, são transcodificados de Windows-1252 — o mesmo critério de
47/// [`para_utf8`], mas em fluxo.
48fn leitor_auto_encoding<R: Read>(fonte: R) -> LeitorAutoEncoding<R> {
49    let mut buf = BufReader::with_capacity(16_384, fonte);
50    let e_utf8 = buf.fill_buf().map_or(true, amostra_e_utf8);
51    if e_utf8 {
52        LeitorAutoEncoding::Utf8(
53            encoding_rs_io::DecodeReaderBytesBuilder::new()
54                .strip_bom(true)
55                .build(buf),
56        )
57    } else {
58        LeitorAutoEncoding::Latin1(
59            encoding_rs_io::DecodeReaderBytesBuilder::new()
60                .encoding(Some(encoding_rs::WINDOWS_1252))
61                .strip_bom(true)
62                .build(buf),
63        )
64    }
65}

67/// Descarta uma linha do `BufRead` usando apenas o buffer interno (sem alocar a linha
68/// descartada em memória), resistindo a preâmbulos arbitrariamente longos.
69fn descartar_uma_linha(leitor: &mut impl BufRead) -> std::io::Result<()> {
70    loop {
71        let buf = leitor.fill_buf()?;
72        if buf.is_empty() {
73            return Ok(());
74        }
75        let (consumido, achou_nl) = match buf.iter().position(|&b| b == b'\n') {
76            Some(pos) => (pos + 1, true),
77            None => (buf.len(), false),
78        };
79        leitor.consume(consumido);
80        if achou_nl {
81            return Ok(());
82        }
83    }
84}

86/// Decodifica, para UTF-8, os bytes de um CSV que pode estar em UTF-8 ou em Windows-1252
87/// (Latin-1). Tenta UTF-8 primeiro; se não for válido, assume Windows-1252, que nunca falha.
88/// Remove o BOM de UTF-8 quando presente (ex.: arquivos da Câmara).
89fn para_utf8(bytes: &[u8]) -> String {
90    let mut texto = match std::str::from_utf8(bytes) {
91        Ok(texto) => texto.to_owned(),
92        Err(_) => encoding_rs::WINDOWS_1252.decode(bytes).0.into_owned(),
93    };
94    if texto.starts_with('\u{feff}') {
95        texto.drain(..'\u{feff}'.len_utf8());
96    }
97    texto
98}

100/// Descarta `n` linhas físicas do início do texto (preâmbulo antes do cabeçalho, ex.: a
101/// linha "ULTIMA ATUALIZACAO" do CEAPS do Senado).
102fn descartar_linhas(texto: &str, n: usize) -> &str {
103    let mut resto = texto;
104    for _ in 0..n {
105        match resto.find('\n') {
106            Some(fim) => resto = &resto[fim + 1..],
107            None => return "",
108        }
109    }
110    resto
111}

113/// Abre um CSV e separa o cabeçalho (colunas) do leitor de registros.
114fn colunas_e_registros<R: Read>(
115    leitor: R,
116    delimitador: u8,
117    sem_cabecalho: bool,
118) -> Result<(Vec<String>, csv::Reader<R>), ErroDeRefino> {
119    let mut leitor = csv::ReaderBuilder::new()
120        .delimiter(delimitador)
121        .flexible(true)
122        .has_headers(!sem_cabecalho)
123        .from_reader(leitor);
124    let primeiro = leitor.headers()?;
125    let colunas = if sem_cabecalho {
126        (1..=primeiro.len())
127            .map(|indice| format!("coluna_{indice:03}"))
128            .collect()
129    } else {
130        primeiro.iter().map(|c| c.trim().to_owned()).collect()
131    };
132    Ok((colunas, leitor))
133}

135/// Lê os registros em lotes e os grava no escritor dado (memória ou arquivo).
136fn alimentar<R: Read, W: std::io::Write + Send>(
137    mut registros: csv::Reader<R>,
138    escritor: &mut EscritorParquet<W>,
139    num_colunas: usize,
140) -> Result<(), ErroDeRefino> {
141    let mut lote: Vec<Vec<Option<String>>> = Vec::with_capacity(LINHAS_POR_LOTE);
142    for registro in registros.records() {
143        let registro = registro?;
144        // Mantém o número de colunas do cabeçalho: completa linhas curtas com nulos e
145        // ignora células excedentes, sem interpretar o conteúdo (fidelidade do bronze).
146        let linha: Vec<Option<String>> = (0..num_colunas)
147            .map(|i| registro.get(i).map(|v| v.to_owned()))
148            .collect();
149        lote.push(linha);
150        if lote.len() >= LINHAS_POR_LOTE {
151            escritor.escrever_lote(&lote)?;
152            lote.clear();
153        }
154    }
155    escritor.escrever_lote(&lote)
156}

158/// Lê um CSV de um `Read` e devolve o Parquet **em memória** (para arquivos modestos).
159fn parquet_de_leitor<R: Read>(
160    leitor: R,
161    delimitador: u8,
162    sem_cabecalho: bool,
163) -> Result<Vec<u8>, ErroDeRefino> {
164    let (colunas, registros) = colunas_e_registros(leitor, delimitador, sem_cabecalho)?;
165    let mut escritor = EscritorParquet::novo(&colunas)?;
166    alimentar(registros, &mut escritor, colunas.len())?;
167    escritor.finalizar()
168}

170/// Lê um CSV de um `Read` e grava o Parquet **direto em um arquivo temporário**, sem manter
171/// o Parquet inteiro em memória — para arquivos grandes (#171).
172fn parte_em_arquivo<R: Read>(
173    nome: &str,
174    leitor: R,
175    delimitador: u8,
176    sem_cabecalho: bool,
177) -> Result<ParteBronze, ErroDeRefino> {
178    let (colunas, registros) = colunas_e_registros(leitor, delimitador, sem_cabecalho)?;
179    let arquivo = tempfile::NamedTempFile::new()?;
180    let mut escritor = EscritorParquet::novo_em(arquivo.reopen()?, &colunas)?;
181    alimentar(registros, &mut escritor, colunas.len())?;
182    escritor.fechar()?;
183    Ok(ParteBronze::de_arquivo(nome, arquivo.into_temp_path()))
184}

186/// Lê um CSV (já em UTF-8, em memória) pulando `linhas_a_pular` de preâmbulo.
187fn parquet_de_csv(
188    texto: &str,
189    delimitador: u8,
190    linhas_a_pular: usize,
191    sem_cabecalho: bool,
192) -> Result<Vec<u8>, ErroDeRefino> {
193    parquet_de_leitor(
194        descartar_linhas(texto, linhas_a_pular).as_bytes(),
195        delimitador,
196        sem_cabecalho,
197    )
198}

200/// Extensões tabulares aceitas dentro do ZIP (o TSE usa `.csv` nos anos recentes e `.txt`
201/// nos antigos, com o mesmo formato `;`/Latin-1).
202fn e_tabular(nome: &str) -> bool {
203    let n = nome.to_ascii_lowercase();
204    n.ends_with(".csv") || n.ends_with(".txt")
205}
206
207/// Nome lógico de um arquivo interno do ZIP: o basename sem a extensão `.csv`/`.txt`.
208fn nome_logico(nome_interno: &str) -> String {
209    let base = nome_interno.rsplit('/').next().unwrap_or(nome_interno);
210    match base.len() {
211        n if n >= 4
212            && (base[n - 4..].eq_ignore_ascii_case(".csv")
213                || base[n - 4..].eq_ignore_ascii_case(".txt")) =>
214        {
215            base[..n - 4].to_owned()
216        }
217        _ => base.to_owned(),
218    }
219}

221/// Os 27 TXT de receitas partidárias de 2012 com sufixo de duas letras começam por um registro; o
222/// consolidado `_brasil` do mesmo ZIP tem cabeçalho e não entra nesta exceção.
223fn sem_cabecalho_tse_2012(nome: &str) -> bool {
224    nome.strip_prefix("receitas_partidos_2012_")
225        .is_some_and(|uf| uf.len() == 2 && uf.bytes().all(|byte| byte.is_ascii_uppercase()))
226}

228/// Descompacta um ZIP e decodifica cada CSV em uma parte Parquet. Delega para
229/// [`partes_de_zip_csv_em_fluxo`] — mantida para testes e chamadores com `&[u8]`.
230pub fn partes_de_zip_csv(bytes: &[u8], delimitador: u8) -> Result<Vec<ParteBronze>, ErroDeRefino> {
231    partes_de_zip_csv_em_fluxo(Cursor::new(bytes), delimitador)
232}

234/// Decodifica cada CSV de um ZIP **em fluxo**: lê a entrada do ZIP, transcodifica
235/// encoding e parseia CSV direto para lotes Parquet em disco, sem materializar o
236/// arquivo inteiro em memória. Usa `utf8_passthru`: se o CSV for UTF-8 válido,
237/// passa direto; senão, transcodifica de Windows-1252 (Latin-1). Remove BOM quando
238/// presente.
239pub fn partes_de_zip_csv_em_fluxo<F: Read + Seek>(
240    fonte: F,
241    delimitador: u8,
242) -> Result<Vec<ParteBronze>, ErroDeRefino> {
243    let mut arquivo = zip::ZipArchive::new(fonte)?;
244    let mut partes = Vec::new();
245
246    for indice in 0..arquivo.len() {
247        let entrada = arquivo.by_index(indice)?;
248        if !entrada.is_file() {
249            continue;
250        }
251        let nome_interno = entrada.name()?.into_owned();
252        if !e_tabular(&nome_interno) {
253            continue;
254        }
255        let leitor = leitor_auto_encoding(entrada);
256        let nome = nome_logico(&nome_interno);
257        partes.push(parte_em_arquivo(
258            &nome,
259            leitor,
260            delimitador,
261            sem_cabecalho_tse_2012(&nome),
262        )?);
263    }
264
265    if partes.is_empty() {
266        return Err(ErroDeRefino::Decodificacao(
267            "nenhum CSV encontrado no ZIP".to_owned(),
268        ));
269    }
270    Ok(partes)
271}

273/// Decodifica um CSV "solto" (não zipado) em uma única parte Parquet, com o `nome` lógico
274/// dado. `linhas_a_pular` descarta um preâmbulo antes do cabeçalho (ex.: CEAPS do Senado).
275pub fn partes_de_csv(
276    bytes: &[u8],
277    nome: &str,
278    delimitador: u8,
279    linhas_a_pular: usize,
280) -> Result<Vec<ParteBronze>, ErroDeRefino> {
281    let texto = para_utf8(bytes);
282    let parquet = parquet_de_csv(&texto, delimitador, linhas_a_pular, false)?;
283    Ok(vec![ParteBronze::nova(nome, &parquet)?])
284}

286/// Como [`partes_de_csv`], mas lê o CSV **em fluxo** a partir de um `Read`, transcodificando
287/// encoding e gravando Parquet em disco sem materializar o CSV inteiro em memória.
288pub fn partes_de_csv_em_fluxo<R: Read>(
289    fonte: R,
290    nome: &str,
291    delimitador: u8,
292    linhas_a_pular: usize,
293) -> Result<Vec<ParteBronze>, ErroDeRefino> {
294    let leitor = leitor_auto_encoding(fonte);
295    let mut leitor_buf = BufReader::new(leitor);
296    for _ in 0..linhas_a_pular {
297        descartar_uma_linha(&mut leitor_buf)?;
298    }
299    Ok(vec![parte_em_arquivo(
300        nome,
301        leitor_buf,
302        delimitador,
303        false,
304    )?])
305}

315    fn zip_com(nome: &str, conteudo: &[u8]) -> Vec<u8> {
316        let mut buffer = Vec::new();
317        {
318            let mut zw = zip::ZipWriter::new(Cursor::new(&mut buffer));
319            zw.start_file(nome, SimpleFileOptions::default()).unwrap();
320            zw.write_all(conteudo).unwrap();
321            zw.finish().unwrap();
322        }
323        buffer
324    }

326    /// Lê o Parquet de uma [`ParteBronze`] (em disco) de volta em (colunas, linhas).
327    pub(crate) fn ler_parte(parte: &ParteBronze) -> (Vec<String>, Vec<Vec<Option<String>>>) {
328        ler_parquet(&std::fs::read(&parte.parquet).unwrap())
329    }
330
331    /// Lê um Parquet (bytes) de volta em (colunas, linhas) para as asserções.
332    pub(crate) fn ler_parquet(bytes: &[u8]) -> (Vec<String>, Vec<Vec<Option<String>>>) {
333        let builder =
334            ParquetRecordBatchReaderBuilder::try_new(bytes::Bytes::from(bytes.to_vec())).unwrap();
335        let colunas: Vec<String> = builder
336            .schema()
337            .fields()
338            .iter()
339            .map(|f| f.name().to_owned())
340            .collect();
341        let mut linhas = Vec::new();
342        for lote in builder.build().unwrap() {
343            let lote = lote.unwrap();
344            for linha in 0..lote.num_rows() {
345                let celulas = (0..lote.num_columns())
346                    .map(|coluna| {
347                        let arr = lote
348                            .column(coluna)
349                            .as_any()
350                            .downcast_ref::<StringArray>()
351                            .unwrap();
352                        if arr.is_null(linha) {
353                            None
354                        } else {
355                            Some(arr.value(linha).to_owned())
356                        }
357                    })
358                    .collect();
359                linhas.push(celulas);
360            }
361        }
362        (colunas, linhas)
363    }
