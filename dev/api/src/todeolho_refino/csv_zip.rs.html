1//! Helper compartilhado: decodificar um ZIP de CSVs (reencodando para UTF-8) em tabelas.
2//!
3//! Boa parte das fontes de arquivo grava, no raw, um ZIP com um ou mais CSVs — às vezes em
4//! Latin-1 (CEAPS do Senado, TSE), às vezes em UTF-8 (CEAP da Câmara). O bronze só traduz:
5//! descompacta, reencoda e põe em colunas, mantendo os valores como texto.
6
7use std::io::{Cursor, Read};
8
9use crate::decodificador::TabelaNomeada;
10use crate::escritor::{ErroDeRefino, Tabela};
11
12/// Decodifica, para UTF-8, os bytes de um CSV que pode estar em UTF-8 ou em Windows-1252
13/// (Latin-1). Tenta UTF-8 primeiro; se não for válido, assume Windows-1252, que nunca falha.
14fn para_utf8(bytes: &[u8]) -> String {
15    match std::str::from_utf8(bytes) {
16        Ok(texto) => texto.to_owned(),
17        Err(_) => encoding_rs::WINDOWS_1252.decode(bytes).0.into_owned(),
18    }
19}

21/// Lê um CSV (já em UTF-8) com o delimitador dado, em uma [`Tabela`] de texto.
22fn tabela_de_csv(texto: &str, delimitador: u8) -> Result<Tabela, ErroDeRefino> {
23    let mut leitor = csv::ReaderBuilder::new()
24        .delimiter(delimitador)
25        .flexible(true)
26        .from_reader(texto.as_bytes());
27
28    let colunas: Vec<String> = leitor
29        .headers()?
30        .iter()
31        .map(|c| c.trim().to_owned())
32        .collect();
33    let num_colunas = colunas.len();
34
35    let mut tabela = Tabela::nova(colunas);
36    for registro in leitor.records() {
37        let registro = registro?;
38        // Mantém o número de colunas do cabeçalho: completa com nulos ou ignora excedentes,
39        // sem interpretar o conteúdo (fidelidade do bronze).
40        let mut linha: Vec<Option<String>> = (0..num_colunas)
41            .map(|i| registro.get(i).map(|v| v.to_owned()))
42            .collect();
43        linha.truncate(num_colunas);
44        tabela.linhas.push(linha);
45    }
46    Ok(tabela)
47}

49/// Descompacta um ZIP em memória e decodifica cada CSV em uma tabela, nomeada pelo arquivo
50/// interno (sem a extensão). Entradas que não sejam `.csv` são ignoradas.
51pub fn tabelas_de_zip_csv(
52    bytes: &[u8],
53    delimitador: u8,
54) -> Result<Vec<TabelaNomeada>, ErroDeRefino> {
55    let mut arquivo = zip::ZipArchive::new(Cursor::new(bytes))?;
56    let mut tabelas = Vec::new();
57
58    for indice in 0..arquivo.len() {
59        let mut entrada = arquivo.by_index(indice)?;
60        if !entrada.is_file() {
61            continue;
62        }
63        let nome_interno = entrada.name()?.into_owned();
64        if !nome_interno.to_ascii_lowercase().ends_with(".csv") {
65            continue;
66        }
67        let mut conteudo = Vec::new();
68        entrada.read_to_end(&mut conteudo)?;
69        let texto = para_utf8(&conteudo);
70        let tabela = tabela_de_csv(&texto, delimitador)?;
71        let base = nome_interno.rsplit('/').next().unwrap_or(&nome_interno);
72        let nome = match base.len() {
73            n if n >= 4 && base[n - 4..].eq_ignore_ascii_case(".csv") => base[..n - 4].to_owned(),
74            _ => base.to_owned(),
75        };
76        tabelas.push(TabelaNomeada::nova(nome, tabela));
77    }
78
79    if tabelas.is_empty() {
80        return Err(ErroDeRefino::Decodificacao(
81            "nenhum CSV encontrado no ZIP".to_owned(),
82        ));
83    }
84    Ok(tabelas)
85}

87#[cfg(test)]
88mod testes {
89    use super::*;
90    use std::io::Write;
91    use zip::write::SimpleFileOptions;
92
93    fn zip_com(nome: &str, conteudo: &[u8]) -> Vec<u8> {
94        let mut buffer = Vec::new();
95        {
96            let mut zw = zip::ZipWriter::new(Cursor::new(&mut buffer));
97            zw.start_file(nome, SimpleFileOptions::default()).unwrap();
98            zw.write_all(conteudo).unwrap();
99            zw.finish().unwrap();
100        }
101        buffer
102    }
103
104    #[test]
105    fn decodifica_csv_utf8_separado_por_ponto_e_virgula() {
106        let csv = "id;nome\n1;São Paulo\n2;\n";
107        let tabelas = tabelas_de_zip_csv(&zip_com("Ano-2024.csv", csv.as_bytes()), b';').unwrap();
108        assert_eq!(tabelas.len(), 1);
109        assert_eq!(tabelas[0].nome, "Ano-2024");
110        assert_eq!(tabelas[0].tabela.colunas, vec!["id", "nome"]);
111        assert_eq!(
112            tabelas[0].tabela.linhas,
113            vec![
114                vec![Some("1".into()), Some("São Paulo".into())],
115                vec![Some("2".into()), Some("".into())],
116            ]
117        );
118    }
119
120    #[test]
121    fn reencoda_latin1_para_utf8() {
122        // "João" em Windows-1252: 0x4A 0x6F 0xE3 0x6F.
123        let mut csv = b"cidade\n".to_vec();
124        csv.extend_from_slice(&[0x4A, 0x6F, 0xE3, 0x6F]);
125        csv.push(b'\n');
126        let tabelas = tabelas_de_zip_csv(&zip_com("x.csv", &csv), b';').unwrap();
127        assert_eq!(tabelas[0].tabela.linhas, vec![vec![Some("João".into())]]);
128    }
129
130    #[test]
131    fn ignora_entradas_que_nao_sao_csv() {
132        let mut buffer = Vec::new();
133        {
134            let mut zw = zip::ZipWriter::new(Cursor::new(&mut buffer));
135            zw.start_file("leia-me.txt", SimpleFileOptions::default())
136                .unwrap();
137            zw.write_all(b"ignore").unwrap();
138            zw.start_file("dados.csv", SimpleFileOptions::default())
139                .unwrap();
140            zw.write_all(b"a;b\n1;2\n").unwrap();
141            zw.finish().unwrap();
142        }
143        let tabelas = tabelas_de_zip_csv(&buffer, b';').unwrap();
144        assert_eq!(tabelas.len(), 1);
145        assert_eq!(tabelas[0].nome, "dados");
146    }
