18const TIPO_PARQUET: &str = "application/vnd.apache.parquet";

20/// Tarefa Silver de uma série BCB, utilizável pelo executor e pelo painel.
21pub struct TarefaDeSilverBcb {
22    bronze: Arc<dyn ArmazenamentoRaw>,
23    snapshot: Arc<SnapshotBronze>,
24    recurso: String,
25}
26
27impl TarefaDeSilverBcb {
28    /// Cria a tarefa para um snapshot já selecionado e verificado.
29    pub fn nova(
30        bronze: Arc<dyn ArmazenamentoRaw>,
31        snapshot: Arc<SnapshotBronze>,
32        recurso: impl Into<String>,
33    ) -> Self {
34        Self {
35            bronze,
36            snapshot,
37            recurso: recurso.into(),
38        }
39    }
40
41    /// Indica se a versão Silver precisa acompanhar o snapshot Bronze atual.
42    pub async fn pendente(&self, silver: &dyn ArmazenamentoRaw) -> Result<bool, ErroDeIngestao> {
43        let versao = format!("{}|regra={REGRA}", self.snapshot.manifesto().id_execucao);
44        Ok(
45            !matches!(estado_na_raw(silver, "bcb", &self.recurso).await?, Some(ref estado)
46            if estado.completa && estado.integra && estado.versao.as_deref() == Some(versao.as_str())),
47        )
48    }
49}
50
51#[async_trait]
52impl Tarefa for TarefaDeSilverBcb {
53    fn nome(&self) -> &str {
54        &self.recurso
55    }
56
57    async fn executar(
58        &self,
59        contexto: &ContextoDeExecucao,
60    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
61        publicar_bcb_sgs(
62            self.bronze.as_ref(),
63            contexto.armazenamento.as_ref(),
64            &self.snapshot,
65            &self.recurso,
66            contexto.iniciada_em,
67            &contexto.id_execucao,
68        )
69        .await
70    }
71}

73/// Transforma e publica uma versão Silver BCB a partir do snapshot Bronze recebido.
74///
75/// O manifesto e o ponteiro Silver só são escritos depois de todas as partes.
76pub async fn publicar_bcb_sgs(
77    bronze: &dyn ArmazenamentoRaw,
78    silver: &dyn ArmazenamentoRaw,
79    snapshot: &SnapshotBronze,
80    recurso: &str,
81    iniciada_em: DateTime<Utc>,
82    id_execucao: &str,
83) -> Result<ResumoDaTarefa, ErroDeIngestao> {
84    if snapshot.manifesto().fonte != "bcb" || snapshot.manifesto().recurso != recurso {
85        return Err(ErroDeIngestao::Manifesto {
86            recurso: recurso.to_owned(),
87            detalhe: "snapshot não pertence ao recurso BCB solicitado".into(),
88        });
89    }
90    let versao_silver = format!("{}|regra={REGRA}", snapshot.manifesto().id_execucao);
91    if let Some(estado) = estado_na_raw(silver, "bcb", recurso).await?
92        && estado.completa
93        && estado.integra
94        && estado.versao.as_deref() == Some(versao_silver.as_str())
95    {
96        return Ok(ResumoDaTarefa {
97            reaproveitada: true,
98            ..ResumoDaTarefa::default()
99        });
100    }
101    let local = LocalizacaoRaw::nova("bcb", recurso, iniciada_em, id_execucao);
102    let mut partes = Vec::new();
103    let mut total_bytes = 0u64;
104    for entrada in &snapshot.manifesto().partes {
105        let baixada = snapshot
106            .baixar_parte(bronze, &entrada.chave)
107            .await
108            .map_err(|erro| ErroDeIngestao::Manifesto {
109                recurso: recurso.to_owned(),
110                detalhe: erro.to_string(),
111            })?;
112        let mut leitor = baixada
113            .abrir_lotes(65_536)
114            .map_err(|erro| ErroDeIngestao::Manifesto {
115                recurso: recurso.to_owned(),
116                detalhe: erro.to_string(),
117            })?;
118        let arquivo = tempfile::NamedTempFile::new().map_err(|erro| ErroDeIngestao::Manifesto {
119            recurso: recurso.to_owned(),
120            detalhe: erro.to_string(),
121        })?;
122        let mut escritor = EscritorParquetTipado::novo_em(
123            arquivo.reopen().map_err(|erro| ErroDeIngestao::Manifesto {
124                recurso: recurso.to_owned(),
125                detalhe: erro.to_string(),
126            })?,
127            esquema_bcb_sgs(),
128        )
129        .map_err(|erro| ErroDeIngestao::Manifesto {
130            recurso: recurso.to_owned(),
131            detalhe: erro.to_string(),
132        })?;
133        for lote in &mut leitor {
134            let transformado = transformar_bcb_sgs(
135                &lote
136                    .map_err(|erro| ErroDeIngestao::Manifesto {
137                        recurso: recurso.to_owned(),
138                        detalhe: erro.to_string(),
139                    })?
140                    .registros,
141            )
142            .map_err(|erro| ErroDeIngestao::Manifesto {
143                recurso: recurso.to_owned(),
144                detalhe: erro.to_string(),
145            })?;
146            escritor
147                .escrever_lote(&transformado)
148                .map_err(|erro| ErroDeIngestao::Manifesto {
149                    recurso: recurso.to_owned(),
150                    detalhe: erro.to_string(),
151                })?;
152        }
153        escritor
154            .fechar()
155            .map_err(|erro| ErroDeIngestao::Manifesto {
156                recurso: recurso.to_owned(),
157                detalhe: erro.to_string(),
158            })?;
159        let bytes = arquivo
160            .as_file()
161            .metadata()
162            .map_err(|erro| ErroDeIngestao::Manifesto {
163                recurso: recurso.to_owned(),
164                detalhe: erro.to_string(),
165            })?
166            .len();
167        total_bytes = total_bytes.saturating_add(bytes);
168        let mut hash = Sha256::new();
169        let mut leitura = arquivo.reopen().map_err(|erro| ErroDeIngestao::Manifesto {
170            recurso: recurso.to_owned(),
171            detalhe: erro.to_string(),
172        })?;
173        let mut bloco = [0u8; 64 * 1024];
174        loop {
175            let lidos = leitura
176                .read(&mut bloco)
177                .map_err(|erro| ErroDeIngestao::Manifesto {
178                    recurso: recurso.to_owned(),
179                    detalhe: erro.to_string(),
180                })?;
181            if lidos == 0 {
182                break;
183            }
184            hash.update(&bloco[..lidos]);
185        }
186        let chave = local.parte(partes.len() + 1, "parquet");
187        let mut fluxo = tokio::fs::File::from_std(arquivo.reopen().map_err(|erro| {
188            ErroDeIngestao::Manifesto {
189                recurso: recurso.to_owned(),
190                detalhe: erro.to_string(),
191            }
192        })?);
193        silver
194            .gravar_fluxo(&chave, TIPO_PARQUET, &mut fluxo)
195            .await?;
196        partes.push(ParteRaw {
197            chave,
198            url: entrada.url.clone(),
199            bytes,
200            sha256: hex::encode(hash.finalize()),
201            tipo_de_conteudo: TIPO_PARQUET.into(),
202            versao: Some(format!(
203                "{}|regra={REGRA}",
204                entrada.versao.as_deref().unwrap_or("desconhecida")
205            )),
206        });
207    }
208    let snapshot_atual = SnapshotBronze::vigente(bronze, "bcb", recurso)
209        .await
210        .map_err(|erro| ErroDeIngestao::Manifesto {
211            recurso: recurso.to_owned(),
212            detalhe: erro.to_string(),
213        })?;
214    if snapshot_atual
215        .as_ref()
216        .is_none_or(|atual| atual.chave_manifesto() != snapshot.chave_manifesto())
217    {
218        return Err(ErroDeIngestao::Manifesto {
219            recurso: recurso.to_owned(),
220            detalhe: "ponteiro Bronze mudou durante a transformação; Silver não publicada".into(),
221        });
222    }
223    let agora = Utc::now();
224    let manifesto = ManifestoRaw {
225        fonte: "bcb".into(),
226        recurso: recurso.into(),
227        id_execucao: id_execucao.into(),
228        url_inicial: format!("bronze/{recurso}"),
229        iniciada_em,
230        concluida_em: agora,
231        partes,
232    };
233    let quantidade = manifesto.partes.len();
234    silver
235        .gravar(ObjetoRaw {
236            chave: local.manifesto(),
237            conteudo: serde_json::to_vec_pretty(&manifesto).map_err(|erro| {
238                ErroDeIngestao::Manifesto {
239                    recurso: recurso.into(),
240                    detalhe: erro.to_string(),
241                }
242            })?,
243            tipo_de_conteudo: "application/json".into(),
244        })
245        .await?;
246    let ponteiro = UltimaVersao {
247        versao: Some(versao_silver),
248        manifesto: local.manifesto(),
249        iniciada_em: Some(iniciada_em),
250    };
251    publicar_ponteiro(silver, "bcb", recurso, &ponteiro).await?;
252    Ok(ResumoDaTarefa {
253        partes: quantidade,
254        bytes: total_bytes,
255        ..ResumoDaTarefa::default()
256    })
257}
