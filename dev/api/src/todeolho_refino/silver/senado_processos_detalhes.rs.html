14/// Erros ao validar detalhes de processo legislativo do Senado.
15#[cfg(test)]
16#[derive(Debug, Error, PartialEq, Eq)]
17pub enum ErroSenadoProcessosDetalhesSilver {
18    /// O Bronze não tem schema textual com `id` e `ano`.
19    #[error("schema Bronze de processos-detalhes incompatível")]
20    Schema,
21    /// A linha não tem ID de processo.
22    #[error("ID de processo inválido na linha {linha} do lote")]
23    Identidade {
24        /// Índice da linha inválida dentro do lote, começando em zero.
25        linha: usize,
26    },
27    /// O ano não pode ser convertido para UInt16.
28    #[error("ano inválido na linha {linha} do lote")]
29    Ano {
30        /// Índice da linha inválida dentro do lote, começando em zero.
31        linha: usize,
32    },
33    /// O ID já apareceu em uma parte anterior do mesmo recurso Bronze.
34    #[error("ID de processo duplicado na linha {linha} do lote")]
35    Duplicada {
36        /// Índice da linha duplicada dentro do lote, começando em zero.
37        linha: usize,
38    },
39    /// Não foi possível serializar os campos da origem em JSON.
40    #[error("não foi possível preservar os campos originais do processo")]
41    Json,
42}

44/// Schema estável do contrato `senado.processo.detalhe.v1`.
45pub fn esquema_senado_processos_detalhes() -> Arc<Schema> {
46    Arc::new(Schema::new(vec![
47        Field::new("id", DataType::Utf8, false),
48        Field::new("ano", DataType::UInt16, false),
49        Field::new("processo_original", DataType::Utf8, false),
50    ]))
51}

53/// Tipifica identidade e ano sem explodir ou perder campos do detalhe do processo.
54#[cfg(test)]
55pub fn transformar_senado_processos_detalhes(
56    lote: &RecordBatch,
57    ids_vistos: &mut HashSet<String>,
58) -> Result<RecordBatch, ErroSenadoProcessosDetalhesSilver> {
59    let schema = lote.schema();
60    let indice_id = schema
61        .index_of("id")
62        .map_err(|_| ErroSenadoProcessosDetalhesSilver::Schema)?;
63    let indice_ano = schema
64        .index_of("ano")
65        .map_err(|_| ErroSenadoProcessosDetalhesSilver::Schema)?;
66    if schema
67        .fields()
68        .iter()
69        .any(|campo| campo.data_type() != &DataType::Utf8)
70    {
71        return Err(ErroSenadoProcessosDetalhesSilver::Schema);
72    }
73    let ids = lote
74        .column(indice_id)
75        .as_any()
76        .downcast_ref::<StringArray>()
77        .ok_or(ErroSenadoProcessosDetalhesSilver::Schema)?;
78    let anos = lote
79        .column(indice_ano)
80        .as_any()
81        .downcast_ref::<StringArray>()
82        .ok_or(ErroSenadoProcessosDetalhesSilver::Schema)?;
83
84    let mut ids_silver = Vec::with_capacity(lote.num_rows());
85    let mut anos_silver = Vec::with_capacity(lote.num_rows());
86    let mut originais = Vec::with_capacity(lote.num_rows());
87    for linha in 0..lote.num_rows() {
88        if ids.is_null(linha) || ids.value(linha).trim().is_empty() {
89            return Err(ErroSenadoProcessosDetalhesSilver::Identidade { linha });
90        }
91        let id = ids.value(linha).trim();
92        if !ids_vistos.insert(id.to_owned()) {
93            return Err(ErroSenadoProcessosDetalhesSilver::Duplicada { linha });
94        }
95        if anos.is_null(linha) {
96            return Err(ErroSenadoProcessosDetalhesSilver::Ano { linha });
97        }
98        let ano_texto = anos.value(linha).trim();
99        if ano_texto.len() != 4 || !ano_texto.bytes().all(|byte| byte.is_ascii_digit()) {
100            return Err(ErroSenadoProcessosDetalhesSilver::Ano { linha });
101        }
102        let ano = ano_texto
103            .parse::<u16>()
104            .map_err(|_| ErroSenadoProcessosDetalhesSilver::Ano { linha })?;
105
106        let mut original = serde_json::Map::new();
107        for (indice, campo) in schema.fields().iter().enumerate() {
108            let valores = lote
109                .column(indice)
110                .as_any()
111                .downcast_ref::<StringArray>()
112                .ok_or(ErroSenadoProcessosDetalhesSilver::Schema)?;
113            let valor = if valores.is_null(linha) {
114                serde_json::Value::Null
115            } else {
116                valor_original(valores.value(linha))
117            };
118            original.insert(campo.name().clone(), valor);
119        }
120        ids_silver.push(id.to_owned());
121        anos_silver.push(ano);
122        originais.push(Some(
123            serde_json::to_string(&original)
124                .map_err(|_| ErroSenadoProcessosDetalhesSilver::Json)?,
125        ));
126    }
127
128    RecordBatch::try_new(
129        esquema_senado_processos_detalhes(),
130        vec![
131            Arc::new(StringArray::from(ids_silver)),
132            Arc::new(UInt16Array::from(anos_silver)),
133            Arc::new(StringArray::from(originais)),
134        ],
135    )
136    .map_err(|_| ErroSenadoProcessosDetalhesSilver::Schema)
137}

139#[cfg(test)]
140fn valor_original(texto: &str) -> serde_json::Value {
141    let texto_aparado = texto.trim_start();
142    if texto_aparado.starts_with('{') || texto_aparado.starts_with('[') {
143        serde_json::from_str(texto).unwrap_or_else(|_| serde_json::Value::String(texto.to_owned()))
144    } else {
145        serde_json::Value::String(texto.to_owned())
146    }
147}

160    fn lote(ids: Vec<Option<&str>>, anos: Vec<Option<&str>>) -> RecordBatch {
161        let despachos = vec![Some(r#"[{"ordem":1}]"#); ids.len()];
162        RecordBatch::try_from_iter(vec![
163            ("id", Arc::new(StringArray::from(ids)) as _),
164            ("ano", Arc::new(StringArray::from(anos)) as _),
165            ("despachos", Arc::new(StringArray::from(despachos)) as _),
166        ])
167        .unwrap()
168    }
169
170    #[test]
171    fn tipa_chave_e_ano_e_preserva_todos_os_campos_no_json_original() {
172        let bronze = lote(vec![Some("00825")], vec![Some("2024")]);
173        let silver = transformar_senado_processos_detalhes(&bronze, &mut HashSet::new()).unwrap();
174
175        assert_eq!(silver.schema(), esquema_senado_processos_detalhes());
176        assert_eq!(silver.schema().field(0).data_type(), &DataType::Utf8);
177        assert_eq!(silver.schema().field(1).data_type(), &DataType::UInt16);
178        assert_eq!(
179            silver
180                .column(0)
181                .as_any()
182                .downcast_ref::<StringArray>()
183                .unwrap()
184                .value(0),
185            "00825"
186        );
187        assert_eq!(
188            silver
189                .column(1)
190                .as_any()
191                .downcast_ref::<UInt16Array>()
192                .unwrap()
193                .value(0),
194            2024
195        );
196        let original = silver
197            .column(2)
198            .as_any()
199            .downcast_ref::<StringArray>()
200            .unwrap();
201        let original: serde_json::Value = serde_json::from_str(original.value(0)).unwrap();
202        assert_eq!(original["id"], "00825");
203        assert_eq!(original["ano"], "2024");
204        assert_eq!(original["despachos"][0]["ordem"], 1);
205    }
