15/// Erros encontrados ao validar a tabela Bronze de legislaturas.
16#[cfg(test)]
17#[derive(Debug, Error, PartialEq, Eq)]
18pub enum ErroCamaraLegislaturasSilver {
19    /// A parte não contém as colunas textuais esperadas.
20    #[error("schema Bronze de legislaturas da Câmara incompatível")]
21    Schema,
22    /// Um ID, uma data ou o ano eleitoral não pode ser interpretado.
23    #[error("campo inválido na linha {linha} do lote")]
24    Valor {
25        /// Índice da linha inválida dentro do lote, começando em zero.
26        linha: usize,
27    },
28    /// O mesmo ID ocorre mais de uma vez na parte.
29    #[error("ID de legislatura duplicado na linha {linha} do lote")]
30    IdDuplicado {
31        /// Índice da segunda ocorrência dentro do lote, começando em zero.
32        linha: usize,
33    },
34}

36/// Schema versionado do contrato `camara.legislatura.periodo.v1`.
37pub fn esquema_camara_legislaturas() -> Arc<Schema> {
38    Arc::new(Schema::new(vec![
39        Field::new("id", DataType::UInt16, false),
40        Field::new("data_inicio", DataType::Date32, false),
41        Field::new("data_fim", DataType::Date32, false),
42        Field::new("ano_eleicao", DataType::UInt16, false),
43        Field::new("uri", DataType::Utf8, true),
44        Field::new("_tdoa_origem_manifesto", DataType::Utf8, false),
45        Field::new("_tdoa_origem_parte", DataType::Utf8, false),
46        Field::new("_tdoa_origem_ordinal", DataType::UInt64, false),
47    ]))
48}

50/// Interpreta o ID sequencial, o período e o ano eleitoral da API da Câmara.
51#[cfg(test)]
52pub fn transformar_camara_legislaturas(
53    lote: &RecordBatch,
54) -> Result<RecordBatch, ErroCamaraLegislaturasSilver> {
55    let schema = lote.schema();
56    let indices: Vec<usize> = ["id", "dataInicio", "dataFim", "dataEleicao", "uri"]
57        .iter()
58        .map(|nome| {
59            schema
60                .index_of(nome)
61                .map_err(|_| ErroCamaraLegislaturasSilver::Schema)
62        })
63        .collect::<Result<_, _>>()?;
64    if indices
65        .iter()
66        .any(|indice| schema.field(*indice).data_type() != &DataType::Utf8)
67    {
68        return Err(ErroCamaraLegislaturasSilver::Schema);
69    }
70    let arrays: Vec<&StringArray> = indices
71        .iter()
72        .map(|indice| {
73            lote.column(*indice)
74                .as_any()
75                .downcast_ref::<StringArray>()
76                .ok_or(ErroCamaraLegislaturasSilver::Schema)
77        })
78        .collect::<Result<_, _>>()?;
79    let mut ids = HashSet::with_capacity(lote.num_rows());
80    let mut legislaturas = Vec::with_capacity(lote.num_rows());
81    let mut inicios = Vec::with_capacity(lote.num_rows());
82    let mut fins = Vec::with_capacity(lote.num_rows());
83    let mut anos = Vec::with_capacity(lote.num_rows());
84    for linha in 0..lote.num_rows() {
85        let parse =
86            |indice: usize| (!arrays[indice].is_null(linha)).then(|| arrays[indice].value(linha));
87        let id = parse(0)
88            .and_then(|texto| texto.parse::<u16>().ok())
89            .ok_or(ErroCamaraLegislaturasSilver::Valor { linha })?;
90        if !ids.insert(id) {
91            return Err(ErroCamaraLegislaturasSilver::IdDuplicado { linha });
92        }
93        let inicio = parse(1)
94            .and_then(parse_data)
95            .ok_or(ErroCamaraLegislaturasSilver::Valor { linha })?;
96        let fim = parse(2)
97            .and_then(parse_data)
98            .ok_or(ErroCamaraLegislaturasSilver::Valor { linha })?;
99        let ano = parse(3)
100            .and_then(|texto| texto.parse::<u16>().ok())
101            .ok_or(ErroCamaraLegislaturasSilver::Valor { linha })?;
102        if fim < inicio {
103            return Err(ErroCamaraLegislaturasSilver::Valor { linha });
104        }
105        legislaturas.push(id);
106        inicios.push(inicio);
107        fins.push(fim);
108        anos.push(ano);
109    }
110    let uri = arrays[4].clone();
111    let para_dias = |data: NaiveDate| {
112        data.signed_duration_since(NaiveDate::from_ymd_opt(1970, 1, 1).expect("epoch válida"))
113            .num_days() as i32
114    };
115    let mut colunas: Vec<Arc<dyn Array>> = vec![
116        Arc::new(UInt16Array::from(legislaturas)),
117        Arc::new(Date32Array::from(
118            inicios.into_iter().map(para_dias).collect::<Vec<_>>(),
119        )),
120        Arc::new(Date32Array::from(
121            fins.into_iter().map(para_dias).collect::<Vec<_>>(),
122        )),
123        Arc::new(UInt16Array::from(anos)),
124        Arc::new(uri.clone()),
125    ];
126    colunas.extend(crate::executor_sql::colunas_linhagem_teste(lote.num_rows()));
127    RecordBatch::try_new(esquema_camara_legislaturas(), colunas)
128        .map_err(|_| ErroCamaraLegislaturasSilver::Schema)
129}

131#[cfg(test)]
132fn parse_data(texto: &str) -> Option<NaiveDate> {
133    NaiveDate::parse_from_str(texto, "%Y-%m-%d").ok()
134}

136#[cfg(test)]
137mod testes {
138    use super::*;
139
140    fn lote(
141        ids: Vec<Option<&str>>,
142        inicio: Vec<Option<&str>>,
143        fim: Vec<Option<&str>>,
144        ano: Vec<Option<&str>>,
145    ) -> RecordBatch {
146        let quantidade = ids.len();
147        let uris =
148            vec![Some("https://dadosabertos.camara.leg.br/api/v2/legislaturas/57"); quantidade];
149        RecordBatch::try_from_iter(vec![
150            ("id", Arc::new(StringArray::from(ids)) as _),
151            ("dataInicio", Arc::new(StringArray::from(inicio)) as _),
152            ("dataFim", Arc::new(StringArray::from(fim)) as _),
153            ("dataEleicao", Arc::new(StringArray::from(ano)) as _),
154            ("uri", Arc::new(StringArray::from(uris)) as _),
155        ])
156        .unwrap()
157    }
158
159    #[test]
160    fn tipa_id_periodo_e_ano_eleitoral() {
161        let bronze = lote(
162            vec![Some("57")],
163            vec![Some("2023-02-01")],
164            vec![Some("2027-01-31")],
165            vec![Some("2022")],
166        );
167        let silver = transformar_camara_legislaturas(&bronze).unwrap();
168        assert_eq!(silver.schema(), esquema_camara_legislaturas());
169        assert_eq!(
170            silver
171                .column(0)
172                .as_any()
173                .downcast_ref::<UInt16Array>()
174                .unwrap()
175                .value(0),
176            57
177        );
178        assert_eq!(
179            silver
180                .column(1)
181                .as_any()
182                .downcast_ref::<Date32Array>()
183                .unwrap()
184                .value(0),
185            19389
186        );
187    }
188
189    #[test]
190    fn rejeita_periodo_invalido_e_id_duplicado() {
191        let duplicada = lote(
192            vec![Some("57"), Some("57")],
193            vec![Some("2023-02-01"); 2],
194            vec![Some("2027-01-31"); 2],
195            vec![Some("2022"); 2],
196        );
197        assert_eq!(
198            transformar_camara_legislaturas(&duplicada),
199            Err(ErroCamaraLegislaturasSilver::IdDuplicado { linha: 1 })
200        );
201        let invertida = lote(
202            vec![Some("57")],
203            vec![Some("2027-01-31")],
204            vec![Some("2023-02-01")],
205            vec![Some("2022")],
206        );
207        assert_eq!(
208            transformar_camara_legislaturas(&invertida),
209            Err(ErroCamaraLegislaturasSilver::Valor { linha: 0 })
210        );
211    }
