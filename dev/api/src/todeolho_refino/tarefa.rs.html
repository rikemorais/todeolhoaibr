1//! A `TarefaDeBronze`: lê um recurso do `raw`, decodifica e grava Parquet no `bronze`.
2//!
3//! Implementa o trait [`Tarefa`] do núcleo, então roda no mesmo executor de DAG e aparece
4//! no mesmo painel da ingestão. Reaproveita os helpers de procedência do `raw` (manifesto,
5//! ponteiro) contra o bucket `bronze` — o layout é o mesmo, só muda o bucket.
6//!
7//! Procedência: cada part do bronze registra, no manifesto, a tabela de origem e o
8//! **ID do manifesto Raw** (`versao`) e, quando necessária, a versão da regra:
9//! reexecutar só reprocessa quando uma nova captura ou regra for publicada.
10

26/// `Content-Type` registrado para as partes em Parquet do bronze.
27const TIPO_PARQUET: &str = "application/vnd.apache.parquet";
28const LIMITE_JSON_DECODIFICADO: u64 = 256 * 1024 * 1024;
29
30fn json_gzip_decodificado(
31    caminho: &std::path::Path,
32) -> std::io::Result<Option<tempfile::NamedTempFile>> {
33    let mut entrada = std::fs::File::open(caminho)?;
34    let mut assinatura = [0u8; 2];
35    if entrada.read_exact(&mut assinatura).is_err() || assinatura != [0x1f, 0x8b] {
36        return Ok(None);
37    }
38    let entrada = std::fs::File::open(caminho)?;
39    let mut saida = tempfile::NamedTempFile::new()?;
40    let mut decoder = flate2::read::MultiGzDecoder::new(entrada).take(LIMITE_JSON_DECODIFICADO + 1);
41    let bytes = std::io::copy(&mut decoder, saida.as_file_mut())?;
42    if bytes > LIMITE_JSON_DECODIFICADO {
43        return Err(std::io::Error::other(
44            "JSON gzip excede 256 MiB após decodificação",
45        ));
46    }
47    Ok(Some(saida))
48}

50/// Decodifica um recurso do `raw` para o `bronze` (Parquet).
51///
52/// Lê do bucket `raw` pela porta `origem`; grava no bucket `bronze` pela porta
53/// `contexto.armazenamento` injetada pelo executor.
54pub struct TarefaDeBronze {
55    origem: Arc<dyn ArmazenamentoRaw>,
56    fonte: String,
57    recurso: String,
58    decodificador: Arc<dyn Decodificador>,
59}
60
61impl TarefaDeBronze {
62    /// Cria a tarefa. `origem` é o armazenamento do `raw` (leitura); o `bronze` (escrita)
63    /// vem no contexto do executor.
64    pub fn nova(
65        origem: Arc<dyn ArmazenamentoRaw>,
66        fonte: impl Into<String>,
67        recurso: impl Into<String>,
68        decodificador: Arc<dyn Decodificador>,
69    ) -> Self {
70        Self {
71            origem,
72            fonte: fonte.into(),
73            recurso: recurso.into(),
74            decodificador,
75        }
76    }
77
78    fn erro_manifesto(&self, detalhe: String) -> ErroDeIngestao {
79        ErroDeIngestao::Manifesto {
80            recurso: self.recurso.clone(),
81            detalhe,
82        }
83    }
84
85    /// Versão do bronze: id do manifesto Raw selecionado e versão de regra opcional.
86    fn versao_de_origem(&self, manifesto_raw: &ManifestoRaw) -> String {
87        let versao_raw = &manifesto_raw.id_execucao;
88        match self.decodificador.versao_regra(&self.fonte, &self.recurso) {
89            Some(regra) => format!("{versao_raw}|regra={regra}"),
90            None => versao_raw.clone(),
91        }
92    }
93
94    /// Lê a procedência do `raw` e decide se o `bronze` precisa reprocessar.
95    ///
96    /// Devolve `None` quando não há recurso vigente no `raw` (nada a refinar) e, caso
97    /// contrário, `(manifesto do raw, versão de origem, reaproveitar)` — `reaproveitar` é
98    /// `true` quando o `bronze` já está íntegro e na mesma versão do `raw`.
99    async fn avaliar(
100        &self,
101        bronze: &dyn ArmazenamentoRaw,
102    ) -> Result<Option<(ManifestoRaw, String, bool)>, ErroDeIngestao> {
103        let Some(manifesto_raw) =
104            manifesto_vigente(self.origem.as_ref(), &self.fonte, &self.recurso).await?
105        else {
106            return Ok(None);
107        };
108        let versao_origem = self.versao_de_origem(&manifesto_raw);
109        let reaproveitar = match estado_na_raw(bronze, &self.fonte, &self.recurso).await? {
110            Some(estado) => {
111                estado.completa
112                    && estado.integra
113                    && estado.versao.as_deref() == Some(versao_origem.as_str())
114            }
115            None => false,
116        };
117        Ok(Some((manifesto_raw, versao_origem, reaproveitar)))
118    }
119
120    /// Há trabalho a fazer? `true` quando o `raw` tem recurso vigente e o `bronze` ainda
121    /// não está nessa versão. Usado pela reconciliação para pular passadas sem mudança.
122    pub async fn pendente(&self, bronze: &dyn ArmazenamentoRaw) -> Result<bool, ErroDeIngestao> {
123        Ok(matches!(self.avaliar(bronze).await?, Some((_, _, false))))
124    }
125}

127#[async_trait]
128impl Tarefa for TarefaDeBronze {
129    fn nome(&self) -> &str {
130        &self.recurso
131    }
132
133    async fn executar(
134        &self,
135        contexto: &ContextoDeExecucao,
136    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
137        let bronze = contexto.armazenamento.as_ref();
138        let local = LocalizacaoRaw::nova(
139            &self.fonte,
140            &self.recurso,
141            contexto.iniciada_em,
142            &contexto.id_execucao,
143        );
144        marcar_execucao_ativa(bronze, &local.manifesto()).await?;
145        if let Err(erro) = limpar_orfas_do_recurso(
146            bronze,
147            &format!("{}/{}/", self.fonte, self.recurso),
148            &contexto.id_execucao,
149        )
150        .await
151        {
152            tracing::warn!(%erro, fonte = %self.fonte, recurso = %self.recurso, "falha ao varrer snapshots Bronze pendentes");
153        }
154
155        // 1-2. Procedência do raw + idempotência. Sem ingestão na origem não há o que
156        // refinar; bronze íntegro e na mesma versão do raw é reaproveitado.
157        let Some((manifesto_raw, versao_origem, reaproveitar)) = self.avaliar(bronze).await? else {
158            return Err(ErroDeIngestao::NaoEncontrado {
159                url: format!("raw/{}/{}", self.fonte, self.recurso),
160            });
161        };
162        if reaproveitar {
163            return Ok(ResumoDaTarefa {
164                reaproveitada: true,
165                ..ResumoDaTarefa::default()
166            });
167        }
168
169        // 3. Baixa cada part para um arquivo temporário (em fluxo, sem carregar o part — que
170        // pode ter vários GB — na memória) e decodifica a partir do disco. Os arquivos são
171        // decodificados em bloco, para o decodificador poder agregar quando houver muitos.
172        let mut arquivos = Vec::with_capacity(manifesto_raw.partes.len());
173        let tipo_de_conteudo = manifesto_raw
174            .partes
175            .first()
176            .map(|p| p.tipo_de_conteudo.clone())
177            .unwrap_or_default();
178        for parte in &manifesto_raw.partes {
179            let arquivo = tempfile::NamedTempFile::new()
180                .map_err(|erro| self.erro_manifesto(erro.to_string()))?;
181            let mut escrita = tokio::fs::File::from_std(
182                arquivo
183                    .reopen()
184                    .map_err(|erro| self.erro_manifesto(erro.to_string()))?,
185            );
186            let existe = self.origem.ler_fluxo(&parte.chave, &mut escrita).await?;
187            tokio::io::AsyncWriteExt::flush(&mut escrita)
188                .await
189                .map_err(|erro| self.erro_manifesto(erro.to_string()))?;
190            drop(escrita);
191            if !existe {
192                return Err(ErroDeIngestao::NaoEncontrado {
193                    url: parte.chave.clone(),
194                });
195            }
196            if tipo_de_conteudo.to_ascii_lowercase().contains("json") {
197                match json_gzip_decodificado(arquivo.path())
198                    .map_err(|erro| self.erro_manifesto(erro.to_string()))?
199                {
200                    Some(decodificado) => arquivos.push(decodificado.into_temp_path()),
201                    None => arquivos.push(arquivo.into_temp_path()),
202                }
203            } else {
204                arquivos.push(arquivo.into_temp_path());
205            }
206        }
207        let caminhos: Vec<std::path::PathBuf> = arquivos.iter().map(|a| a.to_path_buf()).collect();
208        let conteudos = self
209            .decodificador
210            .decodificar_arquivos(&caminhos, &tipo_de_conteudo)
211            .map_err(|erro| self.erro_manifesto(erro.to_string()))?;
212
213        // 4. Grava uma parte Parquet por tabela no bronze.
214        let mut partes = Vec::with_capacity(conteudos.len());
215        let mut total_bytes = 0u64;
216        for (indice, conteudo) in conteudos.into_iter().enumerate() {
217            let chave = local.parte(indice + 1, "parquet");
218            let arquivo = std::fs::File::open(&conteudo.parquet)
219                .map_err(|erro| self.erro_manifesto(erro.to_string()))?;
220            let bytes = arquivo
221                .metadata()
222                .map_err(|erro| self.erro_manifesto(erro.to_string()))?
223                .len();
224            total_bytes += bytes;
225            let mut digest = Sha256::new();
226            let mut leitor = BufReader::new(arquivo);
227            let mut buffer = [0u8; 64 * 1024];
228            loop {
229                let lidos = leitor
230                    .read(&mut buffer)
231                    .map_err(|erro| self.erro_manifesto(erro.to_string()))?;
232                if lidos == 0 {
233                    break;
234                }
235                digest.update(&buffer[..lidos]);
236            }
237            let sha256 = hex::encode(digest.finalize());
238            let mut fluxo = tokio::fs::File::open(&conteudo.parquet)
239                .await
240                .map_err(|erro| self.erro_manifesto(erro.to_string()))?;
241            bronze
242                .gravar_fluxo(&chave, TIPO_PARQUET, &mut fluxo)
243                .await?;
244            partes.push(ParteRaw {
245                chave,
246                url: conteudo.nome,
247                bytes,
248                sha256,
249                tipo_de_conteudo: TIPO_PARQUET.to_owned(),
250                versao: Some(versao_origem.clone()),
251            });
252        }
253
254        // 5. Manifesto do bronze (procedência aponta para o recurso do raw).
255        let manifesto = ManifestoRaw {
256            fonte: self.fonte.clone(),
257            recurso: self.recurso.clone(),
258            id_execucao: contexto.id_execucao.clone(),
259            url_inicial: format!("raw/{}/{}", self.fonte, self.recurso),
260            iniciada_em: contexto.iniciada_em,
261            concluida_em: Utc::now(),
262            partes: partes.clone(),
263        };
264        let conteudo = serde_json::to_vec_pretty(&manifesto)
265            .map_err(|erro| self.erro_manifesto(erro.to_string()))?;
266        bronze
267            .gravar(ObjetoRaw {
268                chave: local.manifesto(),
269                conteudo,
270                tipo_de_conteudo: "application/json".to_owned(),
271            })
272            .await?;
273
274        // 6. Confere a geração Raw e publica Bronze sob a trava distribuída
275        // compartilhada com os publicadores Raw do mesmo recurso.
276        let ponteiro = UltimaVersao {
277            versao: Some(versao_origem),
278            manifesto: local.manifesto(),
279            iniciada_em: Some(contexto.iniciada_em),
280        };
281        let resultado_publicacao = publicar_ponteiro_bronze_se_raw_vigente(
282            Arc::clone(&self.origem),
283            bronze,
284            &self.fonte,
285            &self.recurso,
286            &manifesto_raw.id_execucao,
287            &ponteiro,
288        )
289        .await;
290        if matches!(&resultado_publicacao, Ok(true)) {
291            return Ok(ResumoDaTarefa {
292                partes: partes.len(),
293                bytes: total_bytes,
294                reaproveitada: false,
295                motivo: None,
296            });
297        }
298
299        let manifesto_bronze_atual = manifesto_vigente(bronze, &self.fonte, &self.recurso).await;
300        if manifesto_bronze_atual.as_ref().is_ok_and(|manifesto| {
301            manifesto
302                .as_ref()
303                .is_some_and(|vigente| vigente.id_execucao == contexto.id_execucao)
304        }) {
305            return Ok(ResumoDaTarefa {
306                partes: partes.len(),
307                bytes: total_bytes,
308                reaproveitada: false,
309                motivo: None,
310            });
311        }
312        if let Err(erro) = &manifesto_bronze_atual {
313            tracing::warn!(%erro, recurso = %self.recurso, "falha ao conferir se a publicação Bronze foi confirmada");
314        }
315        if let Err(erro) = agendar_limpeza_snapshot(bronze, &local.manifesto()).await {
316            tracing::warn!(%erro, manifesto = %local.manifesto(), "não foi possível agendar a limpeza tardia do snapshot Bronze");
317        }
318        match resultado_publicacao {
319            Err(erro) => Err(erro),
320            Ok(false) => Err(self.erro_manifesto(
321                "ponteiro Bronze mais recente preservado; esta execução não foi publicada".into(),
322            )),
323            Ok(true) => unreachable!("publicação bem-sucedida já retornou acima"),
324        }
325    }
326}

366    #[async_trait]
367    impl ArmazenamentoRaw for RawEmMemoria {
368        async fn preparar(&self) -> Result<(), ErroDeIngestao> {
369            Ok(())
370        }
371        async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao> {
372            if objeto.chave.ends_with("/_execucao_ativa.json") {
373                return Ok(());
374            }
375            self.objetos
376                .lock()
377                .unwrap()
378                .insert(objeto.chave, objeto.conteudo);
379            Ok(())
380        }
381        async fn gravar_fluxo(
382            &self,
383            chave: &str,
384            tipo_de_conteudo: &str,
385            fluxo: &mut (dyn tokio::io::AsyncRead + Send + Unpin),
386        ) -> Result<(), ErroDeIngestao> {
387            let mut conteudo = Vec::new();
388            tokio::io::AsyncReadExt::read_to_end(fluxo, &mut conteudo)
389                .await
390                .map_err(|erro| ErroDeIngestao::Armazenamento {
391                    chave: chave.to_owned(),
392                    detalhe: erro.to_string(),
393                })?;
394            self.gravar(ObjetoRaw {
395                chave: chave.to_owned(),
396                conteudo,
397                tipo_de_conteudo: tipo_de_conteudo.to_owned(),
398            })
399            .await
400        }
401        async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao> {
402            if chave.ends_with("_ultima_versao.json")
403                && self.leituras_de_ponteiro.fetch_add(1, Ordering::Relaxed) == 1
404                && let Some(mutacoes) = self.mutacao_na_segunda_leitura.lock().unwrap().take()
405            {
406                let mut objetos = self.objetos.lock().unwrap();
407                for (chave, conteudo) in mutacoes {
408                    objetos.insert(chave, conteudo);
409                }
410            }
411            Ok(self.objetos.lock().unwrap().get(chave).cloned())
412        }
413
414        async fn gravar_condicional(
415            &self,
416            objeto: ObjetoRaw,
417            conteudo_vigente: Option<&[u8]>,
418        ) -> Result<bool, ErroDeIngestao> {
419            let mut objetos = self.objetos.lock().unwrap();
420            if objetos.get(&objeto.chave).map(Vec::as_slice) != conteudo_vigente {
421                return Ok(false);
422            }
423            objetos.insert(objeto.chave, objeto.conteudo);
424            Ok(true)
425        }
426
427        async fn remover(&self, chave: &str) -> Result<(), ErroDeIngestao> {
428            self.objetos.lock().unwrap().remove(chave);
429            Ok(())
430        }
431    }

537            iniciada_em: Some(dt),

541            (
542                LocalizacaoRaw::ultima_versao(FONTE, RECURSO),
543                serde_json::to_vec(&ponteiro).unwrap(),

566        let raw = Arc::new(RawEmMemoria::default());
567        semear_raw(&raw, "etag-1");
568        let bronze = Arc::new(RawEmMemoria::default());
569
570        let resumo = tarefa(raw.clone())

572            .await
573            .unwrap();
574
575        assert!(!resumo.reaproveitada);
576        assert_eq!(resumo.partes, 1);
577        assert!(resumo.bytes > 0);
578
579        let objetos = bronze.objetos.lock().unwrap();
580        assert!(
581            objetos.keys().any(|c| c.ends_with("parte-00001.parquet")),
582            "faltou a parte parquet: {:?}",
583            objetos.keys().collect::<Vec<_>>()
584        );
585        let manifesto = objetos
586            .iter()
587            .find(|(chave, _)| chave.ends_with("/_manifesto.json"))
588            .map(|(_, bytes)| serde_json::from_slice::<ManifestoRaw>(bytes).unwrap())
589            .unwrap();
590        let parte = &manifesto.partes[0];
591        let parquet = objetos.get(&parte.chave).unwrap();
592        assert_eq!(parte.bytes, parquet.len() as u64);
593        assert_eq!(parte.sha256, hex::encode(Sha256::digest(parquet)));
594        assert!(objetos.contains_key(&LocalizacaoRaw::ultima_versao(FONTE, RECURSO)));
595    }
