1//! Decodificador do bronze para fontes que gravam **JSON de API** no raw.
2//!
3//! Cobre os casos de "array de registros": no topo (`bcb`), sob uma chave (`value` no
4//! ipeadata, `dados` nas APIs da Câmara) ou um array de objetos (listas do IBGE, processos e
5//! votações do Senado). Cada registro vira uma linha; os campos viram colunas de **texto**,
6//! e valores aninhados (objeto/array) são guardados como **JSON compacto** — o bronze
7//! mantém a fidelidade, a interpretação é do silver.
8
9use serde_json::Value;
10
11use crate::decodificador::{Decodificador, ParteBronze};
12use crate::escritor::{ErroDeRefino, Tabela};
13
14/// Decodifica um JSON de API em uma parte Parquet, a partir do array de registros.
15pub struct DecodificadorJson {
16    nome: &'static str,
17    /// Caminho de chaves até o array de registros; vazio = array no topo.
18    caminho: &'static [&'static str],
19}
20
21impl DecodificadorJson {
22    /// Para JSON cujo array de registros está no **topo** (ex.: `bcb`).
23    pub const fn novo(nome: &'static str) -> Self {
24        Self { nome, caminho: &[] }
25    }
26
27    /// Para JSON cujo array está sob um **caminho** de chaves (ex.: `["value"]`, `["dados"]`,
28    /// ou aninhado como `["ListaParlamentarLegislatura", "Parlamentares", "Parlamentar"]`).
29    pub const fn com_caminho(nome: &'static str, caminho: &'static [&'static str]) -> Self {
30        Self { nome, caminho }
31    }
32}
33
34/// Converte um valor JSON em texto: escalares viram o texto natural; objeto/array viram
35/// JSON compacto; `null` vira ausência (nulo no Parquet).
36fn texto_de(valor: &Value) -> Option<String> {
37    match valor {
38        Value::Null => None,
39        Value::String(s) => Some(s.clone()),
40        Value::Bool(b) => Some(b.to_string()),
41        Value::Number(n) => Some(n.to_string()),
42        outro => Some(outro.to_string()),
43    }
44}
45
46impl Decodificador for DecodificadorJson {
47    fn decodificar(
48        &self,
49        bytes: &[u8],
50        _tipo_de_conteudo: &str,
51    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
52        let raiz: Value = serde_json::from_slice(bytes)
53            .map_err(|erro| ErroDeRefino::Decodificacao(format!("JSON inválido: {erro}")))?;
54        let mut atual = Some(&raiz);
55        for chave in self.caminho {
56            atual = atual.and_then(|v| v.get(chave));
57        }
58        let Some(Value::Array(registros)) = atual else {
59            let onde = if self.caminho.is_empty() {
60                String::new()
61            } else {
62                format!(" em '{}'", self.caminho.join("."))
63            };
64            return Err(ErroDeRefino::Decodificacao(format!(
65                "esperava um array de registros{onde}"
66            )));
67        };
68
69        // Colunas: a união das chaves dos registros, na ordem de primeira aparição.
70        let mut colunas: Vec<String> = Vec::new();
71        for registro in registros {
72            if let Value::Object(mapa) = registro {
73                for chave in mapa.keys() {
74                    if !colunas.iter().any(|c| c == chave) {
75                        colunas.push(chave.clone());
76                    }
77                }
78            }
79        }
80
81        // Sem colunas (array vazio): não gera Parquet — um Parquet de 0 colunas é ilegível.
82        if colunas.is_empty() {
83            return Ok(Vec::new());
84        }
85
86        let mut tabela = Tabela::nova(colunas.clone());
87        for registro in registros {
88            let objeto = registro.as_object();
89            let linha: Vec<Option<String>> = colunas
90                .iter()
91                .map(|coluna| objeto.and_then(|mapa| mapa.get(coluna)).and_then(texto_de))
92                .collect();
93            tabela = tabela.com_linha(linha);
94        }
95
96        Ok(vec![ParteBronze::de_tabela(self.nome, &tabela)?])
97    }
98}

100#[cfg(test)]
101mod testes {
102    use super::*;
103    use crate::csv_zip::testes::ler_parte;
104
105    fn c(s: &str) -> Option<String> {
106        Some(s.to_owned())
107    }
108
109    #[test]
110    fn decodifica_array_no_topo() {
111        let json =
112            br#"[{"data":"01/06/1989","valor":"19.68"},{"data":"01/07/1989","valor":"35.90"}]"#;
113        let partes = DecodificadorJson::novo("serie")
114            .decodificar(json, "application/json")
115            .unwrap();
116        let (colunas, linhas) = ler_parte(&partes[0]);
117        assert_eq!(colunas, vec!["data", "valor"]);
118        assert_eq!(
119            linhas,
120            vec![
121                vec![c("01/06/1989"), c("19.68")],
122                vec![c("01/07/1989"), c("35.90")]
123            ]
124        );
125    }
126
127    #[test]
128    fn decodifica_array_sob_uma_chave_com_numeros_e_nulos() {
129        let json = br#"{"value":[{"a":5.78,"b":"x"},{"a":null,"b":"y","c":true}]}"#;
130        let partes = DecodificadorJson::com_caminho("v", &["value"])
131            .decodificar(json, "application/json")
132            .unwrap();
133        let (colunas, linhas) = ler_parte(&partes[0]);
134        assert_eq!(colunas, vec!["a", "b", "c"]);
135        assert_eq!(
136            linhas,
137            vec![vec![c("5.78"), c("x"), None], vec![None, c("y"), c("true")]]
138        );
139    }
140
141    #[test]
142    fn aninhados_viram_json_compacto() {
143        let json = br#"[{"id":11,"regiao":{"id":1,"sigla":"N"}}]"#;
144        let partes = DecodificadorJson::novo("estados")
145            .decodificar(json, "application/json")
146            .unwrap();
147        let (_, linhas) = ler_parte(&partes[0]);
148        assert_eq!(linhas, vec![vec![c("11"), c(r#"{"id":1,"sigla":"N"}"#)]]);
149    }
150
151    #[test]
152    fn decodifica_array_em_caminho_profundo() {
153        let json = br#"{"raiz":{"meio":{"itens":[{"a":"1"},{"a":"2"}]}}}"#;
154        let partes = DecodificadorJson::com_caminho("x", &["raiz", "meio", "itens"])
155            .decodificar(json, "application/json")
156            .unwrap();
157        let (colunas, linhas) = ler_parte(&partes[0]);
158        assert_eq!(colunas, vec!["a"]);
159        assert_eq!(linhas, vec![vec![c("1")], vec![c("2")]]);
160    }
161
162    #[test]
163    fn falha_quando_nao_ha_array() {
164        let erro = DecodificadorJson::com_caminho("x", &["dados"])
165            .decodificar(b"{\"dados\":{}}", "application/json")
166            .unwrap_err();
167        assert!(matches!(erro, ErroDeRefino::Decodificacao(_)), "{erro:?}");
168    }
