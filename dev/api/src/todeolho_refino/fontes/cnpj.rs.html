81fn colunas_da_tabela(nome_interno: &str) -> Result<&'static [&'static str], ErroDeRefino> {
82    let nome = nome_interno.rsplit('/').next().unwrap_or(nome_interno);
83    let nome = nome.to_ascii_uppercase();
84    if nome.starts_with("EMPRESA") || nome.ends_with(".EMPRECSV") {
85        Ok(COLUNAS_EMPRESAS)
86    } else if nome.starts_with("ESTABELECIMENTO") || nome.ends_with(".ESTABELE") {
87        Ok(COLUNAS_ESTABELECIMENTOS)
88    } else if nome.starts_with("SOCIO") || nome.ends_with(".SOCIOCSV") {
89        Ok(COLUNAS_SOCIOS)
90    } else if nome.starts_with("SIMPLES") || nome.contains(".SIMPLES.CSV.") {
91        Ok(COLUNAS_SIMPLES)
92    } else if nome.starts_with("CNAE")
93        || nome.starts_with("MUNICIPIO")
94        || nome.starts_with("NATUREZA")
95        || nome.starts_with("MOTIVO")
96        || nome.starts_with("QUALIFICACO") // Qualificacoes → QUALIFICACO...
97        || nome.starts_with("PAIS")
98        || [".CNAECSV", ".MUNICCSV", ".NATJUCSV", ".MOTICSV", ".QUALSCSV", ".PAISCSV"]
99            .iter()
100            .any(|sufixo| nome.ends_with(sufixo))
101    {
102        Ok(COLUNAS_AUXILIAR)
103    } else {
104        Err(ErroDeRefino::Decodificacao(format!(
105            "tabela CNPJ desconhecida: {nome_interno}"
106        )))
107    }
108}

110fn nome_logico(nome_interno: &str) -> String {
111    let base = nome_interno.rsplit('/').next().unwrap_or(nome_interno);
112    match base.rfind('.') {
113        Some(pos) => base[..pos].to_owned(),
114        None => base.to_owned(),
115    }
116}

118fn decodificar_zip<F: Read + Seek>(fonte: F) -> Result<Vec<ParteBronze>, ErroDeRefino> {
119    let mut arquivo = zip::ZipArchive::new(fonte)?;
120    let mut partes = Vec::new();
121
122    for indice in 0..arquivo.len() {
123        let entrada = arquivo.by_index(indice)?;
124        if !entrada.is_file() {
125            continue;
126        }
127        let nome_interno = entrada.name()?.into_owned();
128        // A Receita também publica os CSVs com extensões próprias, como
129        // `.EMPRECSV`, `.ESTABELE` e `.CSV.D61010`.
130        let colunas = colunas_da_tabela(&nome_interno)?;
131        let leitor = leitor_windows1252(entrada);
132        let nome = nome_logico(&nome_interno);
133        partes.push(parte_de_csv_com_colunas(&nome, leitor, b';', colunas)?);
134    }
135
136    if partes.is_empty() {
137        return Err(ErroDeRefino::Decodificacao(
138            "nenhum CSV encontrado no ZIP do CNPJ".to_owned(),
139        ));
140    }
141    Ok(partes)
142}

144/// Decodifica ZIPs de CSVs do CNPJ (Receita Federal) em Parquet com colunas nomeadas.
145///
146/// Cada ZIP contém um CSV sem cabeçalho, com separador `;` e encoding Latin-1/Windows-1252.
147/// O nome do CSV interno determina o esquema de colunas (layout do PDF da Receita).
148pub struct DecodificadorCnpj;
149
150impl Decodificador for DecodificadorCnpj {
151    fn decodificar(
152        &self,
153        bytes: &[u8],
154        _tipo_de_conteudo: &str,
155    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
156        decodificar_zip(Cursor::new(bytes))
157    }
158
159    fn decodificar_arquivo(
160        &self,
161        caminho: &std::path::Path,
162        _tipo_de_conteudo: &str,
163    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
164        decodificar_zip(std::fs::File::open(caminho)?)
165    }
166}

175    fn zip_com_csv(nome_csv: &str, conteudo: &[u8]) -> Vec<u8> {
176        let mut buffer = Vec::new();
177        {
178            let mut zw = zip::ZipWriter::new(Cursor::new(&mut buffer));
179            zw.start_file(nome_csv, SimpleFileOptions::default())
180                .unwrap();
181            zw.write_all(conteudo).unwrap();
182            zw.finish().unwrap();
183        }
184        buffer
185    }

187    #[test]
188    fn decodifica_empresas_com_colunas_nomeadas() {
189        let csv = b"12345678;Empresa Teste;2062;50;1000.00;05;";
190        let zip = zip_com_csv("Empresas0.csv", csv);
191        let partes = DecodificadorCnpj
192            .decodificar(&zip, "application/zip")
193            .unwrap();
194        assert_eq!(partes.len(), 1);
195        assert_eq!(partes[0].nome, "Empresas0");
196        let (colunas, linhas) = ler_parte(&partes[0]);
197        assert_eq!(colunas, COLUNAS_EMPRESAS);
198        assert_eq!(linhas.len(), 1);
199        assert_eq!(linhas[0][0], Some("12345678".to_owned()));
200        assert_eq!(linhas[0][1], Some("Empresa Teste".to_owned()));
201    }

203    #[test]
204    fn reconhece_nomes_internos_publicados_pela_receita() {
205        let casos = [
206            ("K3241.K03200Y0.D61010.EMPRECSV", COLUNAS_EMPRESAS),
207            ("K3241.K03200Y0.D61010.ESTABELE", COLUNAS_ESTABELECIMENTOS),
208            ("K3241.K03200Y0.D61010.SOCIOCSV", COLUNAS_SOCIOS),
209            ("F.K03200$W.SIMPLES.CSV.D61010", COLUNAS_SIMPLES),
210            ("F.K03200$Z.D61010.CNAECSV", COLUNAS_AUXILIAR),
211            ("F.K03200$Z.D61010.MUNICCSV", COLUNAS_AUXILIAR),
212            ("F.K03200$Z.D61010.NATJUCSV", COLUNAS_AUXILIAR),
213            ("F.K03200$Z.D61010.MOTICSV", COLUNAS_AUXILIAR),
214            ("F.K03200$Z.D61010.QUALSCSV", COLUNAS_AUXILIAR),
215            ("F.K03200$Z.D61010.PAISCSV", COLUNAS_AUXILIAR),
216        ];
217        for (nome, colunas) in casos {
218            assert_eq!(colunas_da_tabela(nome).unwrap(), colunas, "{nome}");
219            let linha = vec!["valor"; colunas.len()].join(";");
220            let zip = zip_com_csv(nome, linha.as_bytes());
221            let partes = DecodificadorCnpj
222                .decodificar(&zip, "application/zip")
223                .unwrap();
224            assert_eq!(partes.len(), 1, "{nome}");
225            let (obtidas, linhas) = ler_parte(&partes[0]);
226            assert_eq!(obtidas, colunas, "{nome}");
227            assert_eq!(linhas.len(), 1, "{nome}");
228        }
229    }
