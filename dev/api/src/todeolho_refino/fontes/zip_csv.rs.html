1//! Decodificador do bronze para fontes que gravam um **ZIP de CSVs** no raw.
2//!
3//! Cobre a CEAP da Câmara (um CSV anual) e os pacotes do TSE (prestação de contas,
4//! candidaturas, bens), que trazem vários CSVs por ZIP — tipicamente um por UF, em Latin-1
5//! e com `;` como separador. O bronze só descompacta, reencoda para UTF-8 e põe em colunas;
6//! a interpretação dos campos é do silver.
7
8use crate::csv_zip::{partes_de_zip_csv, partes_de_zip_csv_em_fluxo};
9use crate::decodificador::{Decodificador, ParteBronze};
10use crate::escritor::ErroDeRefino;
11
12/// Decodifica um ZIP de CSVs em uma parte Parquet por CSV (entradas não-CSV são ignoradas).
13pub struct DecodificadorZipCsv {
14    separador: u8,
15    /// Decodifica cada CSV em fluxo (para arquivos grandes do TSE; assume Latin-1). Ver #171.
16    em_fluxo: bool,
17}
18
19impl DecodificadorZipCsv {
20    /// CSVs lidos inteiros na memória (auto-detecta UTF-8/Latin-1). Bom para arquivos modestos.
21    pub const fn novo(separador: u8) -> Self {
22        Self {
23            separador,
24            em_fluxo: false,
25        }
26    }
27
28    /// CSVs lidos **em fluxo** (Latin-1), para pacotes grandes do TSE que estourariam a memória.
29    pub const fn em_fluxo(separador: u8) -> Self {
30        Self {
31            separador,
32            em_fluxo: true,
33        }
34    }
35}
36
37impl Decodificador for DecodificadorZipCsv {
38    fn decodificar(
39        &self,
40        bytes: &[u8],
41        _tipo_de_conteudo: &str,
42    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
43        if self.em_fluxo {
44            partes_de_zip_csv_em_fluxo(std::io::Cursor::new(bytes), self.separador)
45        } else {
46            partes_de_zip_csv(bytes, self.separador)
47        }
48    }
49
50    fn decodificar_arquivo(
51        &self,
52        caminho: &std::path::Path,
53        tipo_de_conteudo: &str,
54    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
55        if self.em_fluxo {
56            // Abre o ZIP direto do disco: não carrega o arquivo (de vários GB) na memória.
57            partes_de_zip_csv_em_fluxo(std::fs::File::open(caminho)?, self.separador)
58        } else {
59            self.decodificar(&std::fs::read(caminho)?, tipo_de_conteudo)
60        }
61    }
62}

64#[cfg(test)]
65mod testes {
66    use super::*;
67    use crate::csv_zip::testes::ler_parte;
68    use std::io::{Cursor, Write};
69    use zip::write::SimpleFileOptions;
70
71    fn zip_com_varios(arquivos: &[(&str, &[u8])]) -> Vec<u8> {
72        let mut buffer = Vec::new();
73        {
74            let mut zw = zip::ZipWriter::new(Cursor::new(&mut buffer));
75            for (nome, conteudo) in arquivos {
76                zw.start_file(*nome, SimpleFileOptions::default()).unwrap();
77                zw.write_all(conteudo).unwrap();
78            }
79            zw.finish().unwrap();
80        }
81        buffer
82    }

84    #[test]
85    fn decodifica_zip_com_um_csv() {
86        let csv = "txNomeParlamentar;vlrLiquido\nFulano;100,50\n";
87        let partes = DecodificadorZipCsv::novo(b';')
88            .decodificar(
89                &zip_com_varios(&[("Ano-2024.csv", csv.as_bytes())]),
90                "application/zip",
91            )
92            .unwrap();
93        assert_eq!(partes.len(), 1);
94        let (colunas, linhas) = ler_parte(&partes[0]);
95        assert_eq!(colunas, vec!["txNomeParlamentar", "vlrLiquido"]);
96        assert_eq!(
97            linhas,
98            vec![vec![Some("Fulano".to_owned()), Some("100,50".to_owned())]]
99        );
100    }

102    #[test]
103    fn decodifica_zip_com_varios_csv_por_uf_ignorando_pdf() {
104        // Como os pacotes do TSE: um CSV por UF + um leiame.pdf.
105        let ac = "SG_UF;NM\nAC;Acre\n";
106        let am = "SG_UF;NM\nAM;Amazonas\n";
107        let zip = zip_com_varios(&[
108            ("leiame.pdf", b"ignore"),
109            ("bem_candidato_2026_AC.csv", ac.as_bytes()),
110            ("bem_candidato_2026_AM.csv", am.as_bytes()),
111        ]);
112        let partes = DecodificadorZipCsv::novo(b';')
113            .decodificar(&zip, "application/zip")
114            .unwrap();
115        assert_eq!(partes.len(), 2, "um Parquet por CSV, sem o PDF");
116        let nomes: Vec<&str> = partes.iter().map(|p| p.nome.as_str()).collect();
117        assert!(nomes.contains(&"bem_candidato_2026_AC"));
118        assert!(nomes.contains(&"bem_candidato_2026_AM"));
119    }
