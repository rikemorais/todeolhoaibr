10const ABA: &str = "Brasil - Total por Produto";
11const COLUNAS: [&str; 5] = [
12    "produto",
13    "safra",
14    "area_mil_ha",
15    "produtividade_kg_ha",
16    "producao_mil_t",
17];

19/// Decodifica o XLSX de safra de grãos da CONAB (aba "Brasil - Total por Produto").
20///
21/// Cada linha da aba contém um produto com 3 métricas (área, produtividade, produção)
22/// para duas safras. O decodificador emite duas linhas Bronze por produto (uma por safra).
23pub struct DecodificadorConabSafra;
24
25impl Decodificador for DecodificadorConabSafra {
26    fn decodificar(
27        &self,
28        bytes: &[u8],
29        _tipo_de_conteudo: &str,
30    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
31        let mut workbook: Xlsx<_> = open_workbook_from_rs(Cursor::new(bytes)).map_err(|erro| {
32            ErroDeRefino::Decodificacao(format!("XLSX inválido da CONAB: {erro}"))
33        })?;
34        let planilha = workbook.worksheet_range(ABA).map_err(|erro| {
35            ErroDeRefino::Decodificacao(format!("aba '{ABA}' ausente ou inválida: {erro}"))
36        })?;
37
38        let linhas: Vec<Vec<Data>> = planilha.rows().map(|r| r.to_vec()).collect();
39
40        let (safra_ant, safra_atual) = extrair_safras(&linhas)?;
41        let cabecalho = encontrar_cabecalho(&linhas)?;
42        let inicio_dados = cabecalho + 1;
43
44        let mut tabela = Tabela::nova(COLUNAS);
45
46        for linha in &linhas[inicio_dados..] {
47            if linha.len() < 10 {
48                continue;
49            }
50            let Some(produto) = texto(&linha[0]) else {
51                continue;
52            };
53            let produto = produto.trim().to_owned();
54            if produto.is_empty() || eh_subtotal(&produto) {
55                continue;
56            }
57
58            let area_ant = texto_numerico(&linha[1]);
59            let area_atual = texto_numerico(&linha[2]);
60            let prod_ant = texto_numerico(&linha[4]);
61            let prod_atual = texto_numerico(&linha[5]);
62            let producao_ant = texto_numerico(&linha[7]);
63            let producao_atual = texto_numerico(&linha[8]);
64
65            tabela = tabela.com_linha([
66                Some(produto.clone()),
67                Some(safra_ant.clone()),
68                area_ant.clone(),
69                prod_ant,
70                producao_ant,
71            ]);
72            tabela = tabela.com_linha([
73                Some(produto),
74                Some(safra_atual.clone()),
75                area_atual,
76                prod_atual,
77                producao_atual,
78            ]);
79        }
80
81        Ok(vec![ParteBronze::de_tabela("produtos", &tabela)?])
82    }
83}

85fn extrair_safras(linhas: &[Vec<Data>]) -> Result<(String, String), ErroDeRefino> {
86    for linha in linhas.iter().take(10) {
87        for celula in linha {
88            if let Some(t) = texto(celula)
89                && t.starts_with("Safra ")
90                && t.len() >= 10
91            {
92                let safra = t.strip_prefix("Safra ").unwrap().to_owned();
93                for outra in linha {
94                    if let Some(t2) = texto(outra)
95                        && t2.starts_with("Safra ")
96                        && t2 != t
97                    {
98                        let s2 = t2.strip_prefix("Safra ").unwrap().to_owned();
99                        return Ok((safra, s2));
100                    }
101                }
102            }
103        }
104    }
105    Err(ErroDeRefino::Decodificacao(
106        "não encontrou os rótulos de safra no cabeçalho da CONAB".into(),
107    ))
108}

110fn encontrar_cabecalho(linhas: &[Vec<Data>]) -> Result<usize, ErroDeRefino> {
111    for (i, linha) in linhas.iter().enumerate().take(15) {
112        for celula in linha {
113            if let Some(t) = texto(celula)
114                && (t.contains("REGIÃO/UF") || t.contains("PRODUÇÃO"))
115            {
116                return Ok(i);
117            }
118        }
119    }
120    Err(ErroDeRefino::Decodificacao(
121        "não encontrou a linha de cabeçalho (REGIÃO/UF) na aba da CONAB".into(),
122    ))
123}

125fn eh_subtotal(produto: &str) -> bool {
126    let p = produto.to_uppercase();
127    p.starts_with("NORTE")
128        || p.starts_with("NORDESTE")
129        || p.starts_with("CENTRO")
130        || p.starts_with("SUDESTE")
131        || p.starts_with("SUL ")
132        || p.starts_with("BRASIL")
133        || p.starts_with("TOTAL")
134        || p.starts_with("GRÃOS")
135        || p.starts_with("GRAOS")
136        || p.starts_with("Fonte:")
137}
138
139fn texto(celula: &Data) -> Option<String> {
140    (!matches!(celula, Data::Empty)).then(|| celula.to_string())
141}
142
143fn texto_numerico(celula: &Data) -> Option<String> {
144    match celula {
145        Data::Empty => None,
146        Data::Float(f) => Some(f.to_string()),
147        Data::Int(i) => Some(i.to_string()),
148        outro => texto(outro),
149    }
150}

251    #[test]
252    fn extrai_produtos_em_duas_safras() {
253        let bytes = xlsx_safra(&["SOJA", "MILHO TOTAL"]);
254        let partes = DecodificadorConabSafra.decodificar(&bytes, "").unwrap();
255        let (colunas, linhas) = ler_parte(&partes[0]);
256
257        assert_eq!(colunas, COLUNAS);
258        assert_eq!(linhas.len(), 4);
259        assert_eq!(linhas[0][0], Some("SOJA".into()));
260        assert_eq!(linhas[0][1], Some("24/25".into()));
261        assert_eq!(linhas[1][0], Some("SOJA".into()));
262        assert_eq!(linhas[1][1], Some("25/26".into()));
263        assert_eq!(linhas[2][0], Some("MILHO TOTAL".into()));
264    }

266    #[test]
267    fn filtra_subtotais() {
268        let bytes = xlsx_safra(&["SOJA", "NORTE", "BRASIL", "MILHO TOTAL"]);
269        let partes = DecodificadorConabSafra.decodificar(&bytes, "").unwrap();
270        let (_, linhas) = ler_parte(&partes[0]);
271
272        let produtos: Vec<_> = linhas.iter().filter_map(|l| l[0].as_deref()).collect();
273        assert!(!produtos.contains(&"NORTE"));
274        assert!(!produtos.contains(&"BRASIL"));
275        assert!(produtos.contains(&"SOJA"));
276        assert!(produtos.contains(&"MILHO TOTAL"));
277    }
