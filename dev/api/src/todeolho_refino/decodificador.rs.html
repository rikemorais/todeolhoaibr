1//! O contrato que cada fonte implementa para traduzir seus bytes do raw em Parquet.
2//!
3//! O [`Decodificador`] é o único lugar que conhece o **formato** de uma fonte (ZIP, CSV
4//! em Latin-1, JSON de API) — o conhecimento que o [ADR 0014] mantém fora da ingestão.
5//! Ele devolve o Parquet **já pronto** (uma [`ParteBronze`] por tabela), para poder
6//! escrever em lotes e não manter arquivos grandes inteiros em memória.
7//!
8//! [ADR 0014]: ../../docs/adr/0014-raw-opaca-e-refino-em-estagio-separado.md
9
10use crate::escritor::{ErroDeRefino, Tabela};
11
12/// Uma tabela do bronze já serializada em Parquet, com o nome lógico que terá.
13#[derive(Debug, Clone, PartialEq, Eq)]
14pub struct ParteBronze {
15    /// Nome lógico (ex.: o CSV interno de um ZIP; vira parte da procedência).
16    pub nome: String,
17    /// Bytes do Parquet.
18    pub parquet: Vec<u8>,
19}
20
21impl ParteBronze {
22    /// Cria a parte a partir de bytes de Parquet já prontos.
23    pub fn nova(nome: impl Into<String>, parquet: Vec<u8>) -> Self {
24        Self {
25            nome: nome.into(),
26            parquet,
27        }
28    }
29
30    /// Conveniência para decodificadores que montam uma [`Tabela`] pequena (ex.: JSON):
31    /// serializa a tabela em Parquet e empacota como parte.
32    pub fn de_tabela(nome: impl Into<String>, tabela: &Tabela) -> Result<Self, ErroDeRefino> {
33        Ok(Self::nova(nome, tabela.para_parquet()?))
34    }
35}
36
37/// Traduz os bytes de **um part** do raw em uma ou mais partes de bronze (Parquet).
38///
39/// Um recurso de API costuma render uma parte; um ZIP com vários CSVs rende uma por CSV.
40pub trait Decodificador: Send + Sync {
41    /// Decodifica os bytes de um part do raw, usando o `tipo_de_conteudo` registrado no
42    /// manifesto quando precisar distinguir formatos.
43    fn decodificar(
44        &self,
45        bytes: &[u8],
46        tipo_de_conteudo: &str,
47    ) -> Result<Vec<ParteBronze>, ErroDeRefino>;
48}
