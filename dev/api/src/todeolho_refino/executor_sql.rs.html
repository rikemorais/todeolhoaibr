26/// Limites operacionais aplicados a uma execução SQL Silver.
27#[derive(Debug, Clone)]
28pub struct ConfiguracaoExecutorSql {
29    /// Tamanho máximo das partes Bronze a preparar em disco, em bytes.
30    pub limite_entrada_bytes: u64,
31    /// Memória máxima que o pool do DataFusion tentará controlar, em bytes.
32    pub limite_memoria_bytes: usize,
33    /// Espaço máximo para spill e arquivos temporários do DataFusion, em bytes.
34    pub limite_spill_bytes: u64,
35    /// Quantidade de partições alvo do plano, usada também para limitar paralelismo.
36    pub particoes: usize,
37    /// Tamanho máximo dos batches emitidos pelo plano.
38    pub tamanho_lote: usize,
39}
40
41impl Default for ConfiguracaoExecutorSql {
42    fn default() -> Self {
43        Self {
44            limite_entrada_bytes: 100 * 1024 * 1024 * 1024,
45            limite_memoria_bytes: 512 * 1024 * 1024,
46            limite_spill_bytes: 2 * 1024 * 1024 * 1024,
47            particoes: 1,
48            tamanho_lote: 8_192,
49        }
50    }
51}

53/// Falhas ao preparar ou planejar uma consulta Silver.
54#[derive(Debug, Error)]
55pub enum ErroExecutorSql {
56    /// A configuração não permite uma execução limitada.
57    #[error("configuração inválida do executor SQL Silver: {0}")]
58    Configuracao(String),
59    /// Falha ao baixar, validar ou preparar as partes do snapshot.
60    #[error("não foi possível preparar o snapshot Bronze para SQL: {0}")]
61    Snapshot(String),
62    /// Falha ao criar a tabela temporária Parquet com linhagem explícita.
63    #[error("não foi possível preparar Parquet temporário para SQL: {0}")]
64    Parquet(#[from] ErroDeRefino),
65    /// Falha de I/O ao preparar diretórios ou arquivos temporários.
66    #[error("falha de I/O no executor SQL: {0}")]
67    Io(#[from] std::io::Error),
68    /// Falha de registro de arquivos, análise ou planejamento SQL.
69    #[error("DataFusion não conseguiu planejar o contrato Silver: {0}")]
70    DataFusion(#[from] DataFusionError),
71}

94/// Baixa somente as partes do snapshot e regrava batches com identidade de origem.
95pub async fn preparar_snapshot(
96    bronze: &dyn ArmazenamentoRaw,
97    snapshot: &SnapshotBronze,
98    limite_entrada_bytes: u64,

101}

103/// Prepara o snapshot e acrescenta uma coluna auxiliar com cada registro Bronze serializado
104/// como JSON. É destinada a contratos SQL que precisam preservar campos de origem dinâmicos.
105pub async fn preparar_snapshot_com_json_original(
106    bronze: &dyn ArmazenamentoRaw,
107    snapshot: &SnapshotBronze,
108    limite_entrada_bytes: u64,
109) -> Result<EntradaSqlPreparada, ErroExecutorSql> {
110    preparar_snapshot_interno(bronze, snapshot, limite_entrada_bytes, true).await
111}
112
113async fn preparar_snapshot_interno(
114    bronze: &dyn ArmazenamentoRaw,
115    snapshot: &SnapshotBronze,
116    limite_entrada_bytes: u64,
117    incluir_json_original: bool,
118) -> Result<EntradaSqlPreparada, ErroExecutorSql> {
119    let total = snapshot
120        .manifesto()
121        .partes
122        .iter()
123        .try_fold(0u64, |total, parte| total.checked_add(parte.bytes))
124        .ok_or_else(|| {
125            ErroExecutorSql::Configuracao("tamanho agregado do snapshot excede u64".into())
126        })?;
127    if limite_entrada_bytes == 0 || total > limite_entrada_bytes {
128        return Err(ErroExecutorSql::Configuracao(format!(
129            "snapshot tem {total} bytes e excede o limite de entrada {limite_entrada_bytes}"
130        )));
131    }
132
133    let temporario = tempfile::tempdir()?;
134    let diretorio_bronze = temporario.path().join("bronze");
135    std::fs::create_dir(&diretorio_bronze)?;
136    for (indice, parte) in snapshot.manifesto().partes.iter().enumerate() {
137        let baixada = snapshot
138            .baixar_parte(bronze, &parte.chave)
139            .await
140            .map_err(|erro| ErroExecutorSql::Snapshot(erro.to_string()))?;
141        let leitor = baixada.abrir_lotes(65_536)?;
142        let caminho = diretorio_bronze.join(format!("parte-{indice:08}.parquet"));
143        let mut escritor = None;
144        for lote in leitor {
145            let lote = lote?;
146            let registros = anexar_linhagem(
147                &lote.registros,
148                &lote.origem.manifesto,
149                &lote.origem.parte,
150                lote.primeiro_ordinal,
151                incluir_json_original,
152            )?;
153            if escritor.is_none() {
154                escritor = Some(EscritorParquetTipado::novo_em(
155                    File::create(&caminho)?,
156                    registros.schema(),
157                )?);
158            }
159            escritor
160                .as_mut()
161                .expect("writer initialized")
162                .escrever_lote(&registros)?;
163        }
164        if let Some(escritor) = escritor {
165            escritor.fechar()?;
166        } else {
167            return Err(ErroExecutorSql::Snapshot(format!(
168                "a parte {} não contém batches para inferir o schema",
169                parte.chave
170            )));
171        }
172    }
173    Ok(EntradaSqlPreparada {
174        temporario,
175        diretorio_bronze,
176    })
177}

179fn anexar_linhagem(
180    lote: &RecordBatch,
181    manifesto: &str,
182    parte: &str,
183    primeiro_ordinal: u64,
184    incluir_json_original: bool,
185) -> Result<RecordBatch, ErroDeRefino> {
186    let schema_origem = lote.schema();
187    if [COL_MANIFESTO, COL_PARTE, COL_ORDINAL, COL_JSON_ORIGINAL]
188        .iter()
189        .any(|nome| schema_origem.index_of(nome).is_ok())
190    {
191        return Err(ErroDeRefino::Decodificacao(
192            "o Parquet Bronze usa um nome reservado para linhagem SQL".into(),
193        ));
194    }
195    let mut campos = schema_origem.fields().iter().cloned().collect::<Vec<_>>();
196    let mut colunas = lote.columns().to_vec();
197    if incluir_json_original {
198        let originais = (0..lote.num_rows())
199            .map(|linha| json_original(lote, linha))
200            .collect::<Result<Vec<_>, _>>()?;
201        campos.push(Arc::new(Field::new(
202            COL_JSON_ORIGINAL,
203            arrow::datatypes::DataType::Utf8,
204            false,
205        )));
206        colunas.push(Arc::new(StringArray::from(originais)));
207    }
208    campos.extend([
209        Arc::new(Field::new(
210            COL_MANIFESTO,
211            arrow::datatypes::DataType::Utf8,
212            false,
213        )),
214        Arc::new(Field::new(
215            COL_PARTE,
216            arrow::datatypes::DataType::Utf8,
217            false,
218        )),
219        Arc::new(Field::new(
220            COL_ORDINAL,
221            arrow::datatypes::DataType::UInt64,
222            false,
223        )),
224    ]);
225    colunas.extend([
226        Arc::new(StringArray::from_iter_values(std::iter::repeat_n(
227            manifesto,
228            lote.num_rows(),
229        ))) as _,
230        Arc::new(StringArray::from_iter_values(std::iter::repeat_n(
231            parte,
232            lote.num_rows(),
233        ))) as _,
234        Arc::new(UInt64Array::from_iter_values(
235            (0..lote.num_rows()).map(|offset| primeiro_ordinal + offset as u64),
236        )) as _,
237    ]);
238    RecordBatch::try_new(Arc::new(Schema::new(campos)), colunas).map_err(ErroDeRefino::from)
239}

241fn json_original(lote: &RecordBatch, linha: usize) -> Result<String, ErroDeRefino> {
242    let mut original = serde_json::Map::new();
243    for (indice, campo) in lote.schema().fields().iter().enumerate() {
244        if campo.data_type() != &arrow::datatypes::DataType::Utf8 {
245            return Err(ErroDeRefino::Decodificacao(format!(
246                "a serialização JSON original requer colunas UTF-8; '{}' tem tipo {:?}",
247                campo.name(),
248                campo.data_type()
249            )));
250        }
251        let valores = lote
252            .column(indice)
253            .as_any()
254            .downcast_ref::<StringArray>()
255            .ok_or_else(|| {
256                ErroDeRefino::Decodificacao(format!(
257                    "a coluna UTF-8 '{}' não pode ser lida",
258                    campo.name()
259                ))
260            })?;
261        let valor = if valores.is_null(linha) {
262            serde_json::Value::Null
263        } else {
264            let texto = valores.value(linha);
265            let aparado = texto.trim_start();
266            if aparado.starts_with('{') || aparado.starts_with('[') {
267                serde_json::from_str(texto)
268                    .unwrap_or_else(|_| serde_json::Value::String(texto.to_owned()))
269            } else {
270                serde_json::Value::String(texto.to_owned())
271            }
272        };
273        original.insert(campo.name().clone(), valor);
274    }
275    serde_json::to_string(&original)
276        .map_err(|erro| ErroDeRefino::Decodificacao(format!("JSON original inválido: {erro}")))
277}

279/// Executa SQL versionado sobre Parquets preparados em um diretório de snapshot.
280///
281/// A tabela disponível ao SQL chama-se `bronze`. Os arquivos devem ser somente
282/// aqueles enumerados pelo manifesto vigente e devem conter colunas explícitas
283/// de linhagem. A função devolve um stream; não use `collect` para materializar
284/// a saída inteira. Descartar o stream cancela a execução DataFusion.
285pub async fn executar_sql(
286    diretorio_bronze: &Path,
287    diretorio_spill: &Path,
288    sql: &str,
289    configuracao: &ConfiguracaoExecutorSql,
290) -> Result<SendableRecordBatchStream, ErroExecutorSql> {
291    if configuracao.limite_memoria_bytes == 0
292        || configuracao.limite_spill_bytes == 0
293        || configuracao.particoes == 0
294        || configuracao.tamanho_lote == 0
295    {
296        return Err(ErroExecutorSql::Configuracao(
297            "memória, spill, partições e tamanho de lote devem ser maiores que zero".into(),
298        ));
299    }
300
301    let mut config = SessionConfig::new()
302        .with_target_partitions(configuracao.particoes)
303        .with_batch_size(configuracao.tamanho_lote);
304    // `Utf8View` otimiza scans do DataFusion, mas os contratos existentes usam
305    // `Utf8`. A saída precisa corresponder ao schema Arrow/Parquet declarado.
306    config.options_mut().sql_parser.map_string_types_to_utf8view = false;
307    config
308        .options_mut()
309        .execution
310        .parquet
311        .schema_force_view_types = false;
312    let runtime = RuntimeEnvBuilder::new()
313        .with_temp_file_path(diretorio_spill)
314        .with_max_temp_directory_size(configuracao.limite_spill_bytes)
315        .with_memory_limit(configuracao.limite_memoria_bytes, 1.0)
316        .build_arc()?;
317    let contexto = SessionContext::new_with_config_rt(config, runtime);
318    contexto
319        .register_parquet(
320            "bronze",
321            diretorio_bronze.to_string_lossy().as_ref(),
322            ParquetReadOptions::default(),
323        )
324        .await?;
325    let dataframe = contexto.sql(sql).await?;
326    Ok(dataframe.execute_stream().await?)
327}

329#[cfg(test)]
330mod testes_json_original {
331    use super::*;
332
333    #[test]
334    fn preserva_valores_escalares_aninhados_nulos_e_caracteres_json() {
335        let lote = RecordBatch::try_from_iter(vec![
336            ("id", Arc::new(StringArray::from(vec![Some("00825")])) as _),
337            (
338                "despachos",
339                Arc::new(StringArray::from(vec![Some(r#"[{"ordem":1}]"#)])) as _,
340            ),
341            (
342                "ementa",
343                Arc::new(StringArray::from(vec![Some("Aspas: \"sim\" e acento é")])) as _,
344            ),
345            (
346                "campo_ausente",
347                Arc::new(StringArray::from(vec![None::<&str>])) as _,
348            ),
349        ])
350        .unwrap();
351        let original: serde_json::Value =
352            serde_json::from_str(&json_original(&lote, 0).unwrap()).unwrap();
353        assert_eq!(original["id"], "00825");
354        assert_eq!(original["despachos"][0]["ordem"], 1);
355        assert_eq!(original["ementa"], "Aspas: \"sim\" e acento é");
356        assert_eq!(original["campo_ausente"], serde_json::Value::Null);
357    }
