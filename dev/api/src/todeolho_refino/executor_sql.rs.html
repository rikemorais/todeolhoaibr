25/// Limites operacionais aplicados a uma execução SQL Silver.
26#[derive(Debug, Clone)]
27pub struct ConfiguracaoExecutorSql {
28    /// Tamanho máximo das partes Bronze a preparar em disco, em bytes.
29    pub limite_entrada_bytes: u64,
30    /// Memória máxima que o pool do DataFusion tentará controlar, em bytes.
31    pub limite_memoria_bytes: usize,
32    /// Espaço máximo para spill e arquivos temporários do DataFusion, em bytes.
33    pub limite_spill_bytes: u64,
34    /// Quantidade de partições alvo do plano, usada também para limitar paralelismo.
35    pub particoes: usize,
36    /// Tamanho máximo dos batches emitidos pelo plano.
37    pub tamanho_lote: usize,
38}
39
40impl Default for ConfiguracaoExecutorSql {
41    fn default() -> Self {
42        Self {
43            limite_entrada_bytes: 100 * 1024 * 1024 * 1024,
44            limite_memoria_bytes: 512 * 1024 * 1024,
45            limite_spill_bytes: 2 * 1024 * 1024 * 1024,
46            particoes: 1,
47            tamanho_lote: 8_192,
48        }
49    }
50}

52/// Falhas ao preparar ou planejar uma consulta Silver.
53#[derive(Debug, Error)]
54pub enum ErroExecutorSql {
55    /// A configuração não permite uma execução limitada.
56    #[error("configuração inválida do executor SQL Silver: {0}")]
57    Configuracao(String),
58    /// Falha ao baixar, validar ou preparar as partes do snapshot.
59    #[error("não foi possível preparar o snapshot Bronze para SQL: {0}")]
60    Snapshot(String),
61    /// Falha ao criar a tabela temporária Parquet com linhagem explícita.
62    #[error("não foi possível preparar Parquet temporário para SQL: {0}")]
63    Parquet(#[from] ErroDeRefino),
64    /// Falha de I/O ao preparar diretórios ou arquivos temporários.
65    #[error("falha de I/O no executor SQL: {0}")]
66    Io(#[from] std::io::Error),
67    /// Falha de registro de arquivos, análise ou planejamento SQL.
68    #[error("DataFusion não conseguiu planejar o contrato Silver: {0}")]
69    DataFusion(#[from] DataFusionError),
70}

72/// Arquivos Parquet do snapshot com as colunas internas de proveniência anexadas.
73///
74/// O diretório temporário é removido ao descartar este valor, inclusive quando
75/// a consulta falha ou seu stream é cancelado.
76pub struct EntradaSqlPreparada {
77    temporario: tempfile::TempDir,
78    diretorio_bronze: PathBuf,
79}

81impl EntradaSqlPreparada {
82    /// Diretório lido pela tabela SQL `bronze`.
83    pub fn diretorio_bronze(&self) -> &Path {
84        &self.diretorio_bronze
85    }

87    /// Raiz temporária desta execução, usada também como base do spill.

90    }

93/// Baixa somente as partes do snapshot e regrava batches com identidade de origem.
94pub async fn preparar_snapshot(
95    bronze: &dyn ArmazenamentoRaw,
96    snapshot: &SnapshotBronze,
97    limite_entrada_bytes: u64,
98) -> Result<EntradaSqlPreparada, ErroExecutorSql> {
99    let total = snapshot
100        .manifesto()
101        .partes
102        .iter()
103        .try_fold(0u64, |total, parte| total.checked_add(parte.bytes))
104        .ok_or_else(|| {
105            ErroExecutorSql::Configuracao("tamanho agregado do snapshot excede u64".into())
106        })?;
107    if limite_entrada_bytes == 0 || total > limite_entrada_bytes {
108        return Err(ErroExecutorSql::Configuracao(format!(
109            "snapshot tem {total} bytes e excede o limite de entrada {limite_entrada_bytes}"
110        )));
111    }
112
113    let temporario = tempfile::tempdir()?;
114    let diretorio_bronze = temporario.path().join("bronze");
115    std::fs::create_dir(&diretorio_bronze)?;
116    for (indice, parte) in snapshot.manifesto().partes.iter().enumerate() {
117        let baixada = snapshot
118            .baixar_parte(bronze, &parte.chave)
119            .await
120            .map_err(|erro| ErroExecutorSql::Snapshot(erro.to_string()))?;
121        let leitor = baixada.abrir_lotes(65_536)?;
122        let caminho = diretorio_bronze.join(format!("parte-{indice:08}.parquet"));
123        let mut escritor = None;
124        for lote in leitor {
125            let lote = lote?;
126            let registros = anexar_linhagem(
127                &lote.registros,
128                &lote.origem.manifesto,
129                &lote.origem.parte,
130                lote.primeiro_ordinal,
131            )?;
132            if escritor.is_none() {
133                escritor = Some(EscritorParquetTipado::novo_em(
134                    File::create(&caminho)?,
135                    registros.schema(),
136                )?);
137            }
138            escritor
139                .as_mut()
140                .expect("writer initialized")
141                .escrever_lote(&registros)?;
142        }
143        if let Some(escritor) = escritor {
144            escritor.fechar()?;
145        } else {
146            return Err(ErroExecutorSql::Snapshot(format!(
147                "a parte {} não contém batches para inferir o schema",
148                parte.chave
149            )));
150        }
151    }
152    Ok(EntradaSqlPreparada {
153        temporario,
154        diretorio_bronze,
155    })
156}

158fn anexar_linhagem(
159    lote: &RecordBatch,
160    manifesto: &str,
161    parte: &str,
162    primeiro_ordinal: u64,
163) -> Result<RecordBatch, ErroDeRefino> {
164    let schema_origem = lote.schema();
165    if [COL_MANIFESTO, COL_PARTE, COL_ORDINAL]
166        .iter()
167        .any(|nome| schema_origem.index_of(nome).is_ok())
168    {
169        return Err(ErroDeRefino::Decodificacao(
170            "o Parquet Bronze usa um nome reservado para linhagem SQL".into(),
171        ));
172    }
173    let mut campos = schema_origem.fields().iter().cloned().collect::<Vec<_>>();
174    campos.extend([
175        Arc::new(Field::new(
176            COL_MANIFESTO,
177            arrow::datatypes::DataType::Utf8,
178            false,
179        )),
180        Arc::new(Field::new(
181            COL_PARTE,
182            arrow::datatypes::DataType::Utf8,
183            false,
184        )),
185        Arc::new(Field::new(
186            COL_ORDINAL,
187            arrow::datatypes::DataType::UInt64,
188            false,
189        )),
190    ]);
191    let mut colunas = lote.columns().to_vec();
192    colunas.extend([
193        Arc::new(StringArray::from_iter_values(std::iter::repeat_n(
194            manifesto,
195            lote.num_rows(),
196        ))) as _,
197        Arc::new(StringArray::from_iter_values(std::iter::repeat_n(
198            parte,
199            lote.num_rows(),
200        ))) as _,
201        Arc::new(UInt64Array::from_iter_values(
202            (0..lote.num_rows()).map(|offset| primeiro_ordinal + offset as u64),
203        )) as _,
204    ]);
205    RecordBatch::try_new(Arc::new(Schema::new(campos)), colunas).map_err(ErroDeRefino::from)
206}

208/// Executa SQL versionado sobre Parquets preparados em um diretório de snapshot.
209///
210/// A tabela disponível ao SQL chama-se `bronze`. Os arquivos devem ser somente
211/// aqueles enumerados pelo manifesto vigente e devem conter colunas explícitas
212/// de linhagem. A função devolve um stream; não use `collect` para materializar
213/// a saída inteira. Descartar o stream cancela a execução DataFusion.
214pub async fn executar_sql(
215    diretorio_bronze: &Path,
216    diretorio_spill: &Path,
217    sql: &str,
218    configuracao: &ConfiguracaoExecutorSql,
219) -> Result<SendableRecordBatchStream, ErroExecutorSql> {
220    if configuracao.limite_memoria_bytes == 0
221        || configuracao.limite_spill_bytes == 0
222        || configuracao.particoes == 0
223        || configuracao.tamanho_lote == 0
224    {
225        return Err(ErroExecutorSql::Configuracao(
226            "memória, spill, partições e tamanho de lote devem ser maiores que zero".into(),
227        ));
228    }
229
230    let mut config = SessionConfig::new()
231        .with_target_partitions(configuracao.particoes)
232        .with_batch_size(configuracao.tamanho_lote);
233    // `Utf8View` otimiza scans do DataFusion, mas os contratos existentes usam
234    // `Utf8`. A saída precisa corresponder ao schema Arrow/Parquet declarado.
235    config.options_mut().sql_parser.map_string_types_to_utf8view = false;
236    config
237        .options_mut()
238        .execution
239        .parquet
240        .schema_force_view_types = false;
241    let runtime = RuntimeEnvBuilder::new()
242        .with_temp_file_path(diretorio_spill)
243        .with_max_temp_directory_size(configuracao.limite_spill_bytes)
244        .with_memory_limit(configuracao.limite_memoria_bytes, 1.0)
245        .build_arc()?;
246    let contexto = SessionContext::new_with_config_rt(config, runtime);
247    contexto
248        .register_parquet(
249            "bronze",
250            diretorio_bronze.to_string_lossy().as_ref(),
251            ParquetReadOptions::default(),
252        )
253        .await?;
254    let dataframe = contexto.sql(sql).await?;
255    Ok(dataframe.execute_stream().await?)
256}
