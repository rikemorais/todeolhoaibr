17/// Como uma tarefa terminou.
18#[derive(Debug, Clone, PartialEq, Eq)]
19pub enum EstadoDaTarefa {
20    /// Concluída, com o resumo do que produziu.
21    Concluida(ResumoDaTarefa),
22    /// Falhou, com a mensagem de erro.
23    Falhou(String),
24    /// Não executada porque a tarefa `causa`, da qual ela depende, falhou.
25    Ignorada {
26        /// Nome da tarefa que falhou e causou a ignorada.
27        causa: String,
28    },
29}
30
31/// Resultado de uma tarefa: nome, estado final e duração.
32#[derive(Debug, Clone)]
33pub struct ResultadoDaTarefa {
34    /// Nome da tarefa.
35    pub nome: String,
36    /// Como ela terminou.
37    pub estado: EstadoDaTarefa,
38    /// Quanto tempo levou.
39    pub duracao: Duration,
40}
41
42/// Resultado agregado da execução de uma fonte.
43#[derive(Debug, Clone)]
44pub struct RelatorioDaExecucao {
45    /// Fonte executada.
46    pub fonte: String,
47    /// Resultado de cada tarefa.
48    pub resultados: Vec<ResultadoDaTarefa>,
49}
50
51impl RelatorioDaExecucao {
52    /// `true` se todas as tarefas concluíram (nenhuma falhou nem foi ignorada).
53    pub fn sucesso(&self) -> bool {
54        self.resultados
55            .iter()
56            .all(|r| matches!(r.estado, EstadoDaTarefa::Concluida(_)))
57    }
58}

60/// Recebe os acontecimentos de uma execução (padrão Observer).
61///
62/// O executor não sabe quem observa nem para quê. Os métodos não devolvem erro de
63/// propósito: quem observa não pode interferir na ingestão.
64#[async_trait]
65pub trait ObservadorDeExecucao: Send + Sync {
66    /// Avisado quando uma tarefa começa, com o acompanhador de progresso que ela
67    /// vai alimentar enquanto baixa.
68    async fn tarefa_iniciada(&self, fonte: &str, tarefa: &str, progresso: Arc<ProgressoDaTarefa>);
69    /// Avisado quando uma tarefa termina, com o resultado.
70    async fn tarefa_finalizada(&self, fonte: &str, resultado: &ResultadoDaTarefa);
71}
72
73/// Executa uma DAG nível a nível, com concorrência limitada dentro de cada nível.
74///
75/// A falha de uma tarefa não interrompe as demais: apenas seus descendentes são
76/// ignorados. O Dagron só fornece o grafo; a execução é nossa (ver `docs/adr/0003`).
77pub struct ExecutorDeDag {
78    concorrencia: usize,
79}
80
81impl ExecutorDeDag {
82    /// Cria o executor; `concorrencia` é o máximo de tarefas simultâneas por
83    /// nível (no mínimo 1).
84    pub fn novo(concorrencia: usize) -> Self {
85        Self {
86            concorrencia: concorrencia.max(1),
87        }
88    }
89
90    /// Executa a DAG nível a nível e devolve o relatório com o resultado de cada
91    /// tarefa. A falha de uma tarefa ignora apenas seus descendentes.
92    pub async fn executar(
93        &self,
94        dag: &DagDeIngestao,
95        contexto: &ContextoDeExecucao,
96        observador: &dyn ObservadorDeExecucao,
97    ) -> Result<RelatorioDaExecucao, ErroDeDag> {
98        let mut resultados = Vec::new();
99        let mut causa_de_ignorar: HashMap<String, String> = HashMap::new();
100
101        for nivel in dag.niveis()? {
102            let mut a_executar = Vec::new();
103            for nome in nivel {
104                match causa_de_ignorar.get(&nome) {
105                    Some(causa) => {
106                        warn!(fonte = dag.fonte(), tarefa = %nome, %causa, "tarefa ignorada");
107                        let resultado = ResultadoDaTarefa {
108                            nome,
109                            estado: EstadoDaTarefa::Ignorada {
110                                causa: causa.clone(),
111                            },
112                            duracao: Duration::ZERO,
113                        };
114                        observador.tarefa_finalizada(dag.fonte(), &resultado).await;
115                        resultados.push(resultado);
116                    }
117                    None => a_executar.push((dag.tarefa(&nome)?, nome)),
118                }
119            }
120
121            let concluidas: Vec<ResultadoDaTarefa> = stream::iter(a_executar)
122                .map(|(tarefa, nome)| async move {
123                    info!(fonte = dag.fonte(), tarefa = %nome, "tarefa iniciada");
124                    let progresso = Arc::new(ProgressoDaTarefa::default());
125                    observador
126                        .tarefa_iniciada(dag.fonte(), &nome, Arc::clone(&progresso))
127                        .await;
128                    let contexto_da_tarefa = ContextoDeExecucao {
129                        progresso: Some(progresso),
130                        ..contexto.clone()
131                    };
132                    let inicio = Instant::now();
133                    let estado = match tarefa.executar(&contexto_da_tarefa).await {
134                        Ok(resumo) => {
135                            info!(
136                                fonte = dag.fonte(),
137                                tarefa = %nome,
138                                partes = resumo.partes,
139                                bytes = resumo.bytes,
140                                "tarefa concluída"
141                            );
142                            EstadoDaTarefa::Concluida(resumo)
143                        }
144                        Err(erro) => {
145                            error!(fonte = dag.fonte(), tarefa = %nome, %erro, "tarefa falhou");
146                            EstadoDaTarefa::Falhou(erro.to_string())
147                        }
148                    };
149                    let resultado = ResultadoDaTarefa {
150                        nome,
151                        estado,
152                        duracao: inicio.elapsed(),
153                    };
154                    observador.tarefa_finalizada(dag.fonte(), &resultado).await;
155                    resultado
156                })
157                .buffer_unordered(self.concorrencia)
158                .collect()
159                .await;
160
161            for resultado in &concluidas {
162                if matches!(resultado.estado, EstadoDaTarefa::Falhou(_)) {
163                    for descendente in dag.descendentes(&resultado.nome)? {
164                        causa_de_ignorar
165                            .entry(descendente)
166                            .or_insert_with(|| resultado.nome.clone());
167                    }
168                }
169            }
170            resultados.extend(concluidas);
171        }
172
173        Ok(RelatorioDaExecucao {
174            fonte: dag.fonte().to_owned(),
175            resultados,
176        })
177    }
178}

291    struct TarefaFalsa {
292        nome: &'static str,
293        falha: bool,
294        ordem: Arc<Mutex<Vec<&'static str>>>,
295    }
296
297    #[async_trait]
298    impl Tarefa for TarefaFalsa {
299        fn nome(&self) -> &str {
300            self.nome
301        }
302
303        async fn executar(&self, _: &ContextoDeExecucao) -> Result<ResumoDaTarefa, ErroDeIngestao> {
304            self.ordem.lock().unwrap().push(self.nome);
305            if self.falha {
306                return Err(ErroDeIngestao::Http {
307                    url: "http://exemplo".into(),
308                    detalhe: "fora do ar".into(),
309                });
310            }
311            Ok(ResumoDaTarefa::default())
312        }
313    }

375                cenario.tarefa("independente", false),
376            ],
377        )
378        .unwrap();
379

402        assert!(!cenario.ordem.lock().unwrap().contains(&"filha"));
