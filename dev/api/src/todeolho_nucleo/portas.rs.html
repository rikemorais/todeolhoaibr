8/// Objeto pronto para ser gravado na camada Raw.
9#[derive(Debug, Clone)]
10pub struct ObjetoRaw {
11    /// Chave de destino na camada Raw.
12    pub chave: String,
13    /// Conteúdo a gravar.
14    pub conteudo: Vec<u8>,
15    /// Tipo de conteúdo (`Content-Type`) a registrar.
16    pub tipo_de_conteudo: String,
17}

19/// Destino dos dados brutos.
20#[async_trait]
21pub trait ArmazenamentoRaw: Send + Sync {
22    /// Garante que o destino existe. Precisa ser idempotente.
23    async fn preparar(&self) -> Result<(), ErroDeIngestao>;
24
25    /// Grava um objeto por inteiro na camada Raw.
26    async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao>;
27
28    /// Grava sem carregar tudo na memória. O objeto só fica visível ao final da transferência.
29    async fn gravar_fluxo(
30        &self,
31        chave: &str,
32        tipo_de_conteudo: &str,
33        fluxo: &mut (dyn AsyncRead + Send + Unpin),
34    ) -> Result<(), ErroDeIngestao>;
35
36    /// Apaga transferências que nunca terminaram, como as de um processo interrompido,
37    /// para que não ocupem disco. Uma ingestão em andamento perde a transferência atual e
38    /// a recomeça, então nenhum dado se perde.
39    async fn descartar_envios_incompletos(&self) -> Result<usize, ErroDeIngestao> {
40        Ok(0)
41    }
42
43    /// Conteúdo do objeto, ou `None` se ele não existir.
44    async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao>;
45
46    /// Tamanho do objeto em bytes, sem baixá-lo (um `HEAD`), ou `None` se não
47    /// existir. A implementação padrão lê o objeto; as de infraestrutura devem
48    /// sobrescrever com um `HEAD` para não carregar arquivos grandes na memória.
49    async fn tamanho(&self, chave: &str) -> Result<Option<u64>, ErroDeIngestao> {
50        Ok(self.ler(chave).await?.map(|conteudo| conteudo.len() as u64))
51    }
52}

54/// Resposta bem-sucedida de uma API.
55#[derive(Debug, Clone)]
56pub struct RespostaHttp {
57    /// Tipo de conteúdo (`Content-Type`) devolvido.
58    pub tipo_de_conteudo: String,
59    /// Corpo da resposta.
60    pub corpo: Vec<u8>,
61}
62
63/// Resultado de um GET condicional.
64#[derive(Debug, Clone)]
65pub enum RespostaCondicional {
66    /// 304: o conteúdo não mudou desde o validador enviado.
67    NaoModificado,
68    /// A origem devolveu um corpo novo.
69    Modificado {
70        /// Tipo de conteúdo (`Content-Type`) do novo corpo.
71        tipo_de_conteudo: String,
72        /// Validador do novo conteúdo (ETag), quando informado.
73        versao: Option<String>,
74        /// Corpo novo.
75        corpo: Vec<u8>,
76    },
77}
78
79/// O que a origem informa sobre um arquivo sem precisar baixá-lo.
80#[derive(Debug, Clone, Default, PartialEq, Eq)]
81pub struct InformacoesDoArquivo {
82    /// Identifica a versão do arquivo (ETag).
83    pub versao: Option<String>,
84    /// Tamanho do arquivo em bytes (`Content-Length`), quando informado.
85    pub tamanho: Option<u64>,
86}
87
88/// Fluxo de bytes lido aos poucos, sem carregar tudo na memória.
89pub type FluxoDeBytes = Box<dyn AsyncRead + Send + Unpin>;
90
91/// Resposta bem-sucedida cujo corpo ainda será lido aos poucos.
92pub struct DownloadHttp {
93    /// Tipo de conteúdo (`Content-Type`) do arquivo.
94    pub tipo_de_conteudo: String,
95    /// Identifica a versão do arquivo na origem (ETag), quando informada.
96    pub versao: Option<String>,
97    /// Fluxo do corpo, lido sob demanda.
98    pub fluxo: FluxoDeBytes,
99}

101/// Acesso às APIs públicas. Falhas transitórias são tratadas pela implementação.
102#[async_trait]
103pub trait ClienteHttp: Send + Sync {
104    /// GET simples: devolve o corpo inteiro em memória.
105    async fn obter(&self, url: &str, aceita: &str) -> Result<RespostaHttp, ErroDeIngestao>;
106
107    /// GET condicional: envia o validador (If-None-Match) e devolve
108    /// `NaoModificado` em caso de 304. A implementação padrão ignora o validador
109    /// e sempre baixa, para fontes que não suportam requisição condicional.
110    async fn obter_condicional(
111        &self,
112        url: &str,
113        aceita: &str,
114        _validador: Option<&str>,
115    ) -> Result<RespostaCondicional, ErroDeIngestao> {
116        let resposta = self.obter(url, aceita).await?;
117        Ok(RespostaCondicional::Modificado {
118            tipo_de_conteudo: resposta.tipo_de_conteudo,
119            versao: None,
120            corpo: resposta.corpo,
121        })
122    }
123
124    /// Consulta o arquivo na origem, sem baixá-lo.
125    async fn informacoes(&self, url: &str) -> Result<InformacoesDoArquivo, ErroDeIngestao>;
126
127    /// Baixa um arquivo como fluxo, para gravá-lo sem carregá-lo na memória.
128    async fn baixar(&self, url: &str) -> Result<DownloadHttp, ErroDeIngestao>;
129}
