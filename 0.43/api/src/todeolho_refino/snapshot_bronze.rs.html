1//! Seleção e verificação de uma parte do manifesto bronze vigente.
2//!
3//! O snapshot fixa o ponteiro lido no início. A tarefa silver futura deverá
4//! conferir novamente o ponteiro antes de publicar sua própria versão.
5

16/// Erros ao selecionar ou verificar uma parte bronze para a silver.
17#[derive(Debug, Error)]
18pub enum ErroDeSnapshotBronze {
19    /// Falha ao acessar o bucket bronze.
20    #[error(transparent)]
21    Armazenamento(#[from] ErroDeIngestao),
22    /// Falha ao criar ou ler o arquivo temporário.
23    #[error(transparent)]
24    Io(#[from] std::io::Error),
25    /// Falha ao abrir o Parquet verificado.
26    #[error(transparent)]
27    Refino(#[from] ErroDeRefino),
28    /// Ponteiro, manifesto ou parte inconsistentes.
29    #[error("snapshot bronze inválido: {0}")]
30    Invalido(String),
31}
32
33/// Manifesto selecionado pelo ponteiro vigente de um recurso bronze.
34#[derive(Debug)]
35pub struct SnapshotBronze {
36    chave_manifesto: String,
37    manifesto: ManifestoRaw,
38}
39
40impl SnapshotBronze {
41    /// Consulta o ponteiro vigente e valida o manifesto apontado.
42    ///
43    /// `None` significa que ainda não há dados bronze para o recurso, seja por ausência
44    /// de ponteiro ou por um manifesto válido sem partes. Ponteiro ou manifesto ausente
45    /// após a leitura do ponteiro é um erro.
46    pub async fn vigente(
47        armazenamento: &dyn ArmazenamentoRaw,
48        fonte: &str,
49        recurso: &str,
50    ) -> Result<Option<Self>, ErroDeSnapshotBronze> {
51        let chave_ponteiro = LocalizacaoRaw::ultima_versao(fonte, recurso);
52        let Some(bytes) = armazenamento.ler(&chave_ponteiro).await? else {
53            return Ok(None);
54        };
55        let ponteiro: UltimaVersao = serde_json::from_slice(&bytes)
56            .map_err(|erro| ErroDeSnapshotBronze::Invalido(format!("{chave_ponteiro}: {erro}")))?;
57        if ponteiro.manifesto.is_empty() {
58            return Err(ErroDeSnapshotBronze::Invalido(format!(
59                "{chave_ponteiro}: chave de manifesto vazia"
60            )));
61        }
62        let bytes_manifesto = armazenamento
63            .ler(&ponteiro.manifesto)
64            .await?
65            .ok_or_else(|| {
66                ErroDeSnapshotBronze::Invalido(format!(
67                    "manifesto {} apontado por {chave_ponteiro} não encontrado",
68                    ponteiro.manifesto
69                ))
70            })?;
71        let manifesto: ManifestoRaw = serde_json::from_slice(&bytes_manifesto).map_err(|erro| {
72            ErroDeSnapshotBronze::Invalido(format!("{}: {erro}", ponteiro.manifesto))
73        })?;
74        if manifesto.fonte != fonte || manifesto.recurso != recurso {
75            return Err(ErroDeSnapshotBronze::Invalido(format!(
76                "{} pertence a {}/{}, esperado {fonte}/{recurso}",
77                ponteiro.manifesto, manifesto.fonte, manifesto.recurso
78            )));
79        }
80        if manifesto.partes.is_empty() {
81            return Ok(None);
82        }
83        Ok(Some(Self {
84            chave_manifesto: ponteiro.manifesto,
85            manifesto,
86        }))
87    }
88
89    /// Chave do manifesto que identifica esta versão bronze.
90    pub fn chave_manifesto(&self) -> &str {
91        &self.chave_manifesto
92    }
93
94    /// Conteúdo do manifesto para selecionar tabelas e partes.
95    pub fn manifesto(&self) -> &ManifestoRaw {
96        &self.manifesto
97    }

99    /// Baixa uma parte listada no manifesto e valida tamanho e SHA-256.
100    ///
101    /// O download e a verificação usam arquivo temporário e memória limitada.
102    pub async fn baixar_parte(
103        &self,
104        armazenamento: &dyn ArmazenamentoRaw,
105        chave_parte: &str,
106    ) -> Result<ParteBronzeBaixada, ErroDeSnapshotBronze> {
107        let parte = self
108            .manifesto
109            .partes
110            .iter()
111            .find(|parte| parte.chave == chave_parte)
112            .ok_or_else(|| {
113                ErroDeSnapshotBronze::Invalido(format!(
114                    "parte {chave_parte} não pertence a {}",
115                    self.chave_manifesto
116                ))
117            })?;
118        let arquivo = tempfile::NamedTempFile::new()?;
119        let mut escrita = tokio::fs::File::from_std(arquivo.reopen()?);
120        let existe = armazenamento.ler_fluxo(chave_parte, &mut escrita).await?;
121        tokio::io::AsyncWriteExt::flush(&mut escrita).await?;
122        drop(escrita);
123        if !existe {
124            return Err(ErroDeSnapshotBronze::Invalido(format!(
125                "parte {chave_parte} do manifesto não encontrada"
126            )));
127        }
128        let tamanho = arquivo.as_file().metadata()?.len();
129        if tamanho != parte.bytes {
130            return Err(ErroDeSnapshotBronze::Invalido(format!(
131                "parte {chave_parte}: {tamanho} bytes, manifesto declara {}",
132                parte.bytes
133            )));
134        }
135        let mut leitura = arquivo.reopen()?;
136        let mut digest = Sha256::new();
137        let mut bloco = [0u8; 64 * 1024];
138        loop {
139            let lidos = leitura.read(&mut bloco)?;
140            if lidos == 0 {
141                break;
142            }
143            digest.update(&bloco[..lidos]);
144        }
145        let hash = hex::encode(digest.finalize());
146        if hash != parte.sha256 {
147            return Err(ErroDeSnapshotBronze::Invalido(format!(
148                "parte {chave_parte}: SHA-256 diverge do manifesto"
149            )));
150        }
151        Ok(ParteBronzeBaixada {
152            arquivo,
153            chave_manifesto: self.chave_manifesto.clone(),
154            parte: parte.clone(),
155        })
156    }
157}

159/// Arquivo temporário verificado contra uma parte do snapshot bronze.
160pub struct ParteBronzeBaixada {
161    arquivo: tempfile::NamedTempFile,
162    chave_manifesto: String,
163    parte: ParteRaw,
164}
165
166impl ParteBronzeBaixada {
167    /// Abre a parte verificada para leitura sequencial em lotes.
168    ///
169    /// Mantenha este valor vivo até terminar de consumir o leitor.
170    pub fn abrir_lotes(&self, tamanho_lote: usize) -> Result<LeitorDeParteBronze, ErroDeRefino> {
171        LeitorDeParteBronze::abrir(
172            self.arquivo.path(),
173            &self.chave_manifesto,
174            &self.parte,
175            tamanho_lote,
176        )
177    }
178}

198    #[async_trait]
199    impl ArmazenamentoRaw for ArmazenamentoFalso {
200        async fn preparar(&self) -> Result<(), ErroDeIngestao> {
201            Ok(())
202        }
203
204        async fn gravar(&self, _: ObjetoRaw) -> Result<(), ErroDeIngestao> {
205            unreachable!()
206        }
207
208        async fn gravar_fluxo(
209            &self,
210            _: &str,
211            _: &str,
212            _: &mut (dyn AsyncRead + Send + Unpin),
213        ) -> Result<(), ErroDeIngestao> {
214            unreachable!()
215        }
216
217        async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao> {
218            Ok(self.objetos.get(chave).cloned())
219        }
220
221        async fn ler_fluxo(
222            &self,
223            chave: &str,
224            escrita: &mut (dyn AsyncWrite + Send + Unpin),
225        ) -> Result<bool, ErroDeIngestao> {
226            self.leituras_em_fluxo.fetch_add(1, Ordering::Relaxed);
227            let Some(bytes) = self.objetos.get(chave) else {
228                return Ok(false);
229            };
230            for bloco in bytes.chunks(17) {
231                escrita
232                    .write_all(bloco)
233                    .await
234                    .map_err(|erro| ErroDeIngestao::Armazenamento {
235                        chave: chave.to_owned(),
236                        detalhe: erro.to_string(),
237                    })?;
238            }
239            Ok(true)
240        }
241    }

293    #[tokio::test]
294    async fn baixa_parte_do_manifesto_vigente_e_preserva_linhas() {
295        let (armazenamento, chave) = fixture();
296        let snapshot = SnapshotBronze::vigente(&armazenamento, "tse", "prestacao-contas")
297            .await
298            .unwrap()
299            .unwrap();
300        assert_eq!(snapshot.manifesto().partes.len(), 1);
301        let parte = snapshot.baixar_parte(&armazenamento, &chave).await.unwrap();
302        let lote = parte
303            .abrir_lotes(1)
304            .unwrap()
305            .map(Result::unwrap)
306            .collect::<Vec<_>>();
307        assert_eq!(lote.len(), 2);
308        assert_eq!(lote[0].referencia(0).unwrap().ordinal, 1);
309        assert_eq!(lote[1].referencia(0).unwrap().ordinal, 2);
310        assert_eq!(lote[0].origem.manifesto, snapshot.chave_manifesto());
311        assert_eq!(lote[0].origem.parte, chave);
312        assert_eq!(armazenamento.leituras_em_fluxo.load(Ordering::Relaxed), 1);
313    }
