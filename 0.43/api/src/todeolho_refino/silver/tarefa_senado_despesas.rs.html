20/// Tarefa de transformação de um recurso anual de despesas do Senado.
21pub struct TarefaDeSilverSenadoDespesas {
22    bronze: Arc<dyn ArmazenamentoRaw>,
23    snapshot: Arc<SnapshotBronze>,
24    recurso: String,
25    ano: u16,
26}
27
28impl TarefaDeSilverSenadoDespesas {
29    /// Cria a tarefa para um recurso CEAPS que contém o ano no caminho.
30    pub fn nova(
31        bronze: Arc<dyn ArmazenamentoRaw>,
32        snapshot: Arc<SnapshotBronze>,
33        recurso: impl Into<String>,
34    ) -> Option<Self> {
35        let recurso = recurso.into();
36        let ano = ano_do_recurso(&recurso)?;
37        Some(Self {
38            bronze,
39            snapshot,
40            recurso,
41            ano,
42        })
43    }
44
45    /// Indica se a Silver precisa acompanhar o snapshot Bronze e a regra atuais.
46    pub async fn pendente(&self, silver: &dyn ArmazenamentoRaw) -> Result<bool, ErroDeIngestao> {
47        let versao = format!("{}|regra={REGRA}", self.snapshot.manifesto().id_execucao);
48        Ok(!matches!(
49            estado_na_raw(silver, "senado", &self.recurso).await?,
50            Some(ref estado)
51                if estado.completa && estado.integra && estado.versao.as_deref() == Some(versao.as_str())
52        ))
53    }
54}

76    }

79/// Publica despesas tipadas e valida códigos de documento sem descartar linhas incompletas.
80pub async fn publicar_senado_despesas(
81    bronze: &dyn ArmazenamentoRaw,
82    silver: &dyn ArmazenamentoRaw,
83    snapshot: &SnapshotBronze,
84    recurso: &str,
85    ano: u16,
86    iniciada_em: DateTime<Utc>,
87    id_execucao: &str,
88) -> Result<ResumoDaTarefa, ErroDeIngestao> {
89    if ano_do_recurso(recurso) != Some(ano) {
90        return Err(erro_recurso(recurso));
91    }
92    let mut codigos_vistos = HashSet::new();
93    publicar_transformacao(
94        bronze,
95        silver,
96        snapshot,
97        ConfiguracaoDePublicacao {
98            fonte: "senado",
99            recurso,
100            recurso_destino: None,
101            regra: REGRA,
102            esquema: Some(esquema_senado_despesas()),
103            iniciada_em,
104            id_execucao,
105        },
106        |lote| {
107            transformar_senado_despesas(lote, ano, &mut codigos_vistos)
108                .map_err(|erro| erro.to_string())
109        },
110    )
111    .await
112}

114/// Extrai o ano de um recurso `despesas/ano=AAAA`.
115fn ano_do_recurso(recurso: &str) -> Option<u16> {
116    let ano = recurso.strip_prefix("despesas/ano=")?;
117    if ano.len() != 4 || !ano.bytes().all(|byte| byte.is_ascii_digit()) {
118        return None;
119    }
120    ano.parse().ok()
121}
122
123fn erro_recurso(recurso: &str) -> ErroDeIngestao {
124    ErroDeIngestao::Manifesto {
125        recurso: recurso.to_owned(),
126        detalhe: "recurso de despesas deve conter um ano válido".into(),
127    }
128}

147    #[async_trait]
148    impl ArmazenamentoRaw for Memoria {
149        async fn preparar(&self) -> Result<(), ErroDeIngestao> {
150            Ok(())
151        }
152        async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao> {
153            self.0.lock().unwrap().insert(objeto.chave, objeto.conteudo);
154            Ok(())
155        }
156        async fn gravar_fluxo(
157            &self,
158            chave: &str,
159            _: &str,
160            fluxo: &mut (dyn AsyncRead + Send + Unpin),
161        ) -> Result<(), ErroDeIngestao> {
162            let mut bytes = Vec::new();
163            fluxo
164                .read_to_end(&mut bytes)
165                .await
166                .map_err(|erro| ErroDeIngestao::Armazenamento {
167                    chave: chave.into(),
168                    detalhe: erro.to_string(),
169                })?;
170            self.0.lock().unwrap().insert(chave.into(), bytes);
171            Ok(())
172        }
173        async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao> {
174            Ok(self.0.lock().unwrap().get(chave).cloned())
175        }
176    }

275    #[tokio::test]
276    async fn publica_recurso_anual_e_reaproveita_versao_atual() {
277        let (bronze, snapshot) = snapshot().await;
278        let silver = Memoria::default();
279        let tarefa = TarefaDeSilverSenadoDespesas::nova(
280            Arc::new(Memoria::default()),
281            Arc::clone(&snapshot),
282            "despesas/ano=2024",
283        )
284        .unwrap();
285        assert!(tarefa.pendente(&silver).await.unwrap());
286        let resumo = publicar_senado_despesas(
287            &bronze,
288            &silver,
289            &snapshot,
290            "despesas/ano=2024",
291            2024,
292            Utc::now(),
293            "silver-1",
294        )
295        .await
296        .unwrap();
297        assert_eq!(resumo.partes, 1);
298        assert!(!tarefa.pendente(&silver).await.unwrap());
299        assert!(
300            publicar_senado_despesas(
301                &bronze,
302                &silver,
303                &snapshot,
304                "despesas/ano=2024",
305                2024,
306                Utc::now(),
307                "silver-2",
308            )
309            .await
310            .unwrap()
311            .reaproveitada
312        );
313    }
