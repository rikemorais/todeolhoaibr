43    const COLUNAS_BRONZE: &[&str] = &[
44        "cnpj_basico",
45        "opcao_simples",
46        "data_opcao_simples",
47        "data_exclusao_simples",
48        "opcao_mei",

53    fn parquet_bronze(linhas: &[Vec<&str>]) -> Vec<u8> {
54        let mut campos: Vec<Field> = COLUNAS_BRONZE
55            .iter()
56            .map(|nome| Field::new(*nome, DataType::Utf8, true))
57            .collect();
58        campos.extend([
59            Field::new("_tdoa_origem_manifesto", DataType::Utf8, false),
60            Field::new("_tdoa_origem_parte", DataType::Utf8, false),
61            Field::new("_tdoa_origem_ordinal", DataType::UInt64, false),
62        ]);
63        let schema = Arc::new(Schema::new(campos));
64        let n = linhas.len();
65        let mut arrays: Vec<Arc<dyn Array>> = (0..COLUNAS_BRONZE.len())
66            .map(|i| {
67                Arc::new(StringArray::from(
68                    linhas.iter().map(|r| r[i]).collect::<Vec<_>>(),
69                )) as _
70            })
71            .collect();
72        arrays.extend([
73            Arc::new(StringArray::from(vec!["m"; n])) as _,
74            Arc::new(StringArray::from(vec!["p"; n])) as _,
75            Arc::new(arrow::array::UInt64Array::from_iter_values(0..n as u64)) as _,
76        ]);
77        let batch = RecordBatch::try_new(schema.clone(), arrays).unwrap();
78        let mut bytes = Vec::new();
79        let mut writer = ArrowWriter::try_new(&mut bytes, schema, None).unwrap();
80        writer.write(&batch).unwrap();
81        writer.close().unwrap();
82        bytes
83    }
84
85    async fn consultar(dir: &Path, linhas: &[Vec<&str>]) -> Vec<RecordBatch> {
86        fs::write(dir.join("parte.parquet"), parquet_bronze(linhas)).unwrap();
87        let spill = dir.join("spill");
88        fs::create_dir(&spill).unwrap();
89        let mut fluxo: SendableRecordBatchStream = executar_sql(
90            dir,
91            &spill,
92            SQL_CNPJ_SIMPLES,
93            &ConfiguracaoExecutorSql {
94                limite_entrada_bytes: 1024 * 1024,
95                limite_memoria_bytes: 64 * 1024 * 1024,
96                limite_spill_bytes: 64 * 1024 * 1024,
97                particoes: 1,
98                tamanho_lote: 8192,
99            },
100        )
101        .await
102        .unwrap();
103        let mut saida = Vec::new();
104        while let Some(lote) = fluxo.next().await {
105            saida.push(lote.unwrap());
106        }
107        saida
108    }

110    #[tokio::test]
111    async fn converte_datas_yyyymmdd_e_preserva_nulos() {
112        let dir = tempfile::tempdir().unwrap();
113        let lotes = consultar(
114            dir.path(),
115            &[vec![
116                "12345678", "S", "20200315", "", "N", "20210101", "20221231",
117            ]],
118        )
119        .await;
120        let lote = &lotes[0];
121
122        let esperado = esquema_cnpj_simples();
123        assert_eq!(
124            lote.schema()
125                .fields()
126                .iter()
127                .map(|f| (f.name().as_str(), f.data_type().clone()))
128                .collect::<Vec<_>>(),
129            esperado
130                .fields()
131                .iter()
132                .map(|f| (f.name().as_str(), f.data_type().clone()))
133                .collect::<Vec<_>>(),
134        );
135
136        let data_opcao = lote
137            .column(2)
138            .as_any()
139            .downcast_ref::<Date32Array>()
140            .unwrap();
141        let esperada = NaiveDate::from_ymd_opt(2020, 3, 15)
142            .unwrap()
143            .signed_duration_since(NaiveDate::from_ymd_opt(1970, 1, 1).unwrap())
144            .num_days() as i32;
145        assert_eq!(data_opcao.value(0), esperada);
146
147        let data_exclusao_simples = lote
148            .column(3)
149            .as_any()
150            .downcast_ref::<Date32Array>()
151            .unwrap();
152        assert!(data_exclusao_simples.is_null(0));
153
154        let data_exclusao_mei = lote
155            .column(6)
156            .as_any()
157            .downcast_ref::<Date32Array>()
158            .unwrap();
159        let esperada_mei = NaiveDate::from_ymd_opt(2022, 12, 31)
160            .unwrap()
161            .signed_duration_since(NaiveDate::from_ymd_opt(1970, 1, 1).unwrap())
162            .num_days() as i32;
163        assert_eq!(data_exclusao_mei.value(0), esperada_mei);
164    }
