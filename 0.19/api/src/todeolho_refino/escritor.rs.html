1//! Escrita de uma tabela colunar simples em Parquet.
2//!
3//! O bronze mantém fidelidade máxima: cada coluna é **texto** (`Utf8`), nomeada como na
4//! origem (cabeçalho do CSV ou campo do JSON). Tipos, chaves e regra de negócio ficam no
5//! silver ([ADR 0015]). Manter tudo como texto evita qualquer interpretação do conteúdo
6//! nesta camada.
7//!
8//! [ADR 0015]: ../../docs/adr/0015-arquitetura-de-dados-em-medalhao.md
9
10use std::io::Write;
11use std::sync::Arc;
12
13use arrow::array::{ArrayRef, StringArray};
14use arrow::datatypes::{DataType, Field, Schema};
15use arrow::record_batch::RecordBatch;
16use parquet::arrow::ArrowWriter;
17use parquet::basic::{Compression, ZstdLevel};
18use parquet::file::properties::WriterProperties;
19use thiserror::Error;

21/// Monta o esquema (todas as colunas `Utf8`, anuláveis) a partir dos nomes.
22fn esquema_de(colunas: &[String]) -> Arc<Schema> {
23    let campos: Vec<Field> = colunas
24        .iter()
25        .map(|nome| Field::new(nome, DataType::Utf8, true))
26        .collect();
27    Arc::new(Schema::new(campos))
28}
29
30/// Converte um lote de linhas de texto em arrays Arrow (um `StringArray` por coluna).
31fn arrays_de_linhas(linhas: &[Vec<Option<String>>], num_colunas: usize) -> Vec<ArrayRef> {
32    (0..num_colunas)
33        .map(|c| {
34            let coluna: StringArray = linhas.iter().map(|linha| linha[c].clone()).collect();
35            Arc::new(coluna) as ArrayRef
36        })
37        .collect()
38}

40/// Escreve Parquet **em lotes** (um row group por lote), sem manter a tabela inteira em
41/// memória. É o caminho do bronze para arquivos grandes: o decodificador alimenta lotes de
42/// algumas dezenas de milhares de linhas e o pico de memória fica no tamanho de um lote.
43pub struct EscritorParquet<W: Write + Send> {
44    escritor: ArrowWriter<W>,
45    esquema: Arc<Schema>,
46    num_colunas: usize,
47}
48
49fn propriedades() -> WriterProperties {
50    WriterProperties::builder()
51        .set_compression(Compression::ZSTD(ZstdLevel::default()))
52        .build()
53}
54
55impl EscritorParquet<Vec<u8>> {
56    /// Cria um escritor **em memória** (devolve os bytes). Bom para tabelas pequenas.
57    pub fn novo(colunas: &[String]) -> Result<Self, ErroDeRefino> {
58        let esquema = esquema_de(colunas);
59        let escritor =
60            ArrowWriter::try_new(Vec::new(), Arc::clone(&esquema), Some(propriedades()))?;
61        Ok(Self {
62            escritor,
63            esquema,
64            num_colunas: colunas.len(),
65        })
66    }
67
68    /// Fecha o Parquet e devolve os bytes.
69    pub fn finalizar(self) -> Result<Vec<u8>, ErroDeRefino> {
70        Ok(self.escritor.into_inner()?)
71    }
72}
73
74impl<W: Write + Send> EscritorParquet<W> {
75    /// Cria um escritor para um destino qualquer (ex.: um arquivo), sem manter o Parquet
76    /// inteiro em memória — para arquivos grandes.
77    pub fn novo_em(destino: W, colunas: &[String]) -> Result<Self, ErroDeRefino> {
78        let esquema = esquema_de(colunas);
79        let escritor = ArrowWriter::try_new(destino, Arc::clone(&esquema), Some(propriedades()))?;
80        Ok(Self {
81            escritor,
82            esquema,
83            num_colunas: colunas.len(),
84        })
85    }
86
87    /// Fecha o Parquet, liberando o destino (os bytes já foram escritos nele).
88    pub fn fechar(self) -> Result<(), ErroDeRefino> {
89        self.escritor.close()?;
90        Ok(())
91    }
92
93    /// Escreve um lote de linhas como um row group. Linhas devem ter uma célula por coluna.
94    pub fn escrever_lote(&mut self, linhas: &[Vec<Option<String>>]) -> Result<(), ErroDeRefino> {
95        if linhas.is_empty() {
96            return Ok(());
97        }
98        if let Some((indice, linha)) = linhas
99            .iter()
100            .enumerate()
101            .find(|(_, l)| l.len() != self.num_colunas)
102        {
103            return Err(ErroDeRefino::LinhaForaDoFormato {
104                linha: indice,
105                celulas: linha.len(),
106                colunas: self.num_colunas,
107            });
108        }
109        let colunas = arrays_de_linhas(linhas, self.num_colunas);
110        let lote = RecordBatch::try_new(Arc::clone(&self.esquema), colunas)?;
111        self.escritor.write(&lote)?;
112        Ok(())
113    }
114}

116/// Falhas ao converter uma [`Tabela`] em Parquet.
117#[derive(Debug, Error)]
118pub enum ErroDeRefino {
119    /// Alguma linha tem um número de células diferente do número de colunas.
120    #[error("linha {linha} tem {celulas} células, mas a tabela tem {colunas} colunas")]
121    LinhaForaDoFormato {
122        /// Índice da linha problemática (base 0).
123        linha: usize,
124        /// Quantas células a linha trouxe.
125        celulas: usize,
126        /// Quantas colunas a tabela declara.
127        colunas: usize,
128    },
129    /// Erro vindo do Arrow ao montar o lote de registros.
130    #[error("erro do Arrow: {0}")]
131    Arrow(#[from] arrow::error::ArrowError),
132    /// Erro vindo do escritor Parquet.
133    #[error("erro do Parquet: {0}")]
134    Parquet(#[from] parquet::errors::ParquetError),
135    /// Erro ao abrir ou percorrer um arquivo ZIP.
136    #[error("erro no ZIP: {0}")]
137    Zip(#[from] zip::result::ZipError),
138    /// Erro ao ler um CSV.
139    #[error("erro no CSV: {0}")]
140    Csv(#[from] csv::Error),
141    /// Erro de I/O ao decodificar o conteúdo.
142    #[error("erro de I/O: {0}")]
143    Io(#[from] std::io::Error),
144    /// Conteúdo inesperado para o decodificador (estrutura ou formato inválido).
145    #[error("conteúdo inválido: {0}")]
146    Decodificacao(String),
147}

149/// Uma tabela colunar simples: nomes de colunas e linhas de texto.
150///
151/// Cada célula é `Option<String>`: `None` vira nulo no Parquet (campo ausente na origem),
152/// `Some` vira o texto exato. Todas as colunas são gravadas como `Utf8`.
153#[derive(Debug, Clone, Default, PartialEq, Eq)]
154pub struct Tabela {
155    /// Nomes das colunas, na ordem.
156    pub colunas: Vec<String>,
157    /// Linhas; cada linha tem uma célula por coluna, na mesma ordem.
158    pub linhas: Vec<Vec<Option<String>>>,
159}
160
161impl Tabela {
162    /// Cria uma tabela a partir dos nomes de coluna.
163    pub fn nova(colunas: impl IntoIterator<Item = impl Into<String>>) -> Self {
164        Self {
165            colunas: colunas.into_iter().map(Into::into).collect(),
166            linhas: Vec::new(),
167        }
168    }
169
170    /// Acrescenta uma linha. A validação do tamanho acontece em [`Self::para_parquet`].
171    pub fn com_linha(mut self, linha: impl IntoIterator<Item = Option<String>>) -> Self {
172        self.linhas.push(linha.into_iter().collect());
173        self
174    }
175
176    /// Serializa a tabela em Parquet (bytes), com todas as colunas como texto.
177    ///
178    /// Caminho conveniente para tabelas pequenas (JSON, testes). Arquivos grandes usam o
179    /// [`EscritorParquet`] em lotes, para não manter tudo em memória.
180    ///
181    /// # Erros
182    ///
183    /// Falha se alguma linha não tiver exatamente uma célula por coluna.
184    pub fn para_parquet(&self) -> Result<Vec<u8>, ErroDeRefino> {
185        let mut escritor = EscritorParquet::novo(&self.colunas)?;
186        escritor.escrever_lote(&self.linhas)?;
187        escritor.finalizar()
188    }
189}

197    fn c(s: &str) -> Option<String> {
198        Some(s.to_owned())
199    }

201    fn ler_de_volta(bytes: Vec<u8>) -> (Vec<String>, Vec<Vec<Option<String>>>) {
202        let builder = ParquetRecordBatchReaderBuilder::try_new(bytes::Bytes::from(bytes)).unwrap();
203        let colunas: Vec<String> = builder
204            .schema()
205            .fields()
206            .iter()
207            .map(|f| f.name().to_owned())
208            .collect();
209        let leitor = builder.build().unwrap();
210        let mut linhas: Vec<Vec<Option<String>>> = Vec::new();
211        for lote in leitor {
212            let lote = lote.unwrap();
213            for linha in 0..lote.num_rows() {
214                let mut celulas = Vec::new();
215                for coluna in 0..lote.num_columns() {
216                    let arr = lote
217                        .column(coluna)
218                        .as_any()
219                        .downcast_ref::<StringArray>()
220                        .unwrap();
221                    celulas.push(if arr.is_null(linha) {
222                        None
223                    } else {
224                        Some(arr.value(linha).to_owned())
225                    });
226                }
227                linhas.push(celulas);
228            }
229        }
230        (colunas, linhas)
231    }

233    #[test]
234    fn grava_e_le_de_volta_mantendo_colunas_linhas_e_nulos() {
235        let tabela = Tabela::nova(["sgCodigo", "nome"])
236            .com_linha([c("1"), c("São Paulo")])
237            .com_linha([c("2"), None]);
238
239        let bytes = tabela.para_parquet().unwrap();
240        let (colunas, linhas) = ler_de_volta(bytes);
241
242        assert_eq!(colunas, vec!["sgCodigo", "nome"]);
243        assert_eq!(
244            linhas,
245            vec![vec![c("1"), c("São Paulo")], vec![c("2"), None],]
246        );
247    }
