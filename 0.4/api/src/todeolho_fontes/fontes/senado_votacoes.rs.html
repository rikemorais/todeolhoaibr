1use chrono::{Datelike, Utc};
2use todeolho_nucleo::{FonteDeDados, TarefaPlanejada};
3
4use crate::ingestao_http::TarefaDeIngestaoHttp;
5
6const NOME: &str = "senado-votacoes";
7const FONTE_RAW: &str = "senado";
8const BASE: &str = "https://legis.senado.leg.br/dadosabertos";
9/// Primeiro ano com votação registrada na API (1990 e antes retornam vazio).
10const PRIMEIRO_ANO: u16 = 1991;
11
12/// Último ano coberto: o **ano corrente**, derivado do relógio, para a fonte não
13/// deixar de cobrir o ano novo sem edição manual. Um ano ainda sem votações volta
14/// vazio da API, o que é inofensivo.
15fn ultimo_ano() -> u16 {
16    Utc::now().year() as u16
17}
18
19/// Matérias e votações nominais do Senado Federal — "quais projetos foram votados
20/// e quem votou em quê".
21///
22/// Usa o serviço `GET /dadosabertos/votacao?dataInicio=&dataFim=`, que devolve um
23/// array de votações **com `votos[]` embutido** (o voto nominal de cada senador) e a
24/// matéria votada (`identificacao`, `ementa`, `codigoMateria`). Como o voto já vem
25/// embutido, basta **uma requisição por ano** ([`TarefaDeIngestaoHttp::sem_paginacao`]) —
26/// sem o fan-out por id que a antiga API por votação exigiria. É o par, no Senado, da
27/// fonte `camara-votacoes`. Ver issue #87.
28///
29/// A resposta é guardada **opaca** na Raw ([ADR 0014]); cruzar matéria ↔ votação ↔ voto
30/// ↔ senador e análises ficam para o refino ([ADR 0015]). Fonte separada e `sob_demanda`
31/// para não entrar em `--todas` (é um backfill anual). O dado mora em `raw/senado/votacoes/`.
32///
33/// Documentação: <https://www12.senado.leg.br/dados-abertos/legislativo/plenario/votacoes-nominais>
34///
35/// [ADR 0014]: ../../../docs/adr/0014-raw-opaca-e-refino-em-estagio-separado.md
36/// [ADR 0015]: ../../../docs/adr/0015-arquitetura-de-dados-em-medalhao.md
37pub struct FonteSenadoVotacoes;
38
39impl FonteDeDados for FonteSenadoVotacoes {
40    fn nome(&self) -> &str {
41        NOME
42    }
43
44    fn descricao(&self) -> &str {
45        "Senado Federal - matérias e votações nominais, uma consulta por ano desde 1991"
46    }
47
48    /// São dezenas de anos em lote: não devem entrar em `--todas` por acidente.
49    fn sob_demanda(&self) -> bool {
50        true
51    }
52
53    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
54        (PRIMEIRO_ANO..=ultimo_ano())
55            .map(|ano| {
56                let url = format!("{BASE}/votacao?dataInicio={ano}-01-01&dataFim={ano}-12-31");
57                TarefaPlanejada::nova(
58                    TarefaDeIngestaoHttp::sem_paginacao(
59                        FONTE_RAW,
60                        &format!("votacoes/ano={ano}"),
61                        url,
62                    )
63                    // A API não informa o tamanho de antemão; valor típico de um ano recente.
64                    .com_tamanho_tipico(1_500_000),
65                )
66                .com_etiquetas([("ano", ano.to_string().as_str())])
67            })
68            .collect()
69    }
70}

72#[cfg(test)]
73mod testes {
74    use std::collections::BTreeSet;
75
76    use super::*;
77
78    /// Anos cobertos pelas tarefas, lidos da etiqueta `ano`.
79    fn anos() -> BTreeSet<u16> {
80        FonteSenadoVotacoes
81            .criar_tarefas()
82            .iter()
83            .map(|t| {
84                t.etiquetas
85                    .get("ano")
86                    .and_then(|ano| ano.parse().ok())
87                    .expect("etiqueta `ano` numérica")
88            })
89            .collect()
90    }
91
92    #[test]
93    fn cobre_exatamente_um_ano_por_tarefa_de_1991_ate_o_ultimo() {
94        let esperado: BTreeSet<u16> = (PRIMEIRO_ANO..=ultimo_ano()).collect();
95        // A contagem bruta pega um ano duplicado que o conjunto esconderia.
96        assert_eq!(
97            FonteSenadoVotacoes.criar_tarefas().len(),
98            esperado.len(),
99            "número de tarefas diverge do esperado (duplicata ou lacuna)"
100        );
101        assert_eq!(anos(), esperado);
102    }
103
104    #[test]
105    fn comeca_em_1991_quando_a_serie_passa_a_ter_dados() {
106        // 1990 e antes retornam vazio na API: o floor não é arbitrário.
107        assert_eq!(anos().iter().min().copied(), Some(1991));
108    }
109
110    #[test]
111    fn nao_entra_em_todas_por_acidente() {
112        assert!(FonteSenadoVotacoes.sob_demanda());
113    }
114}
