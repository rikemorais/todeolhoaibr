1//! Contrato da camada Raw: onde os dados ficam e como cada ingestão é descrita.
2//! As futuras etapas de transformação vão ler a partir deste contrato (ver `docs/adr/0006`).
3
4use chrono::{DateTime, Utc};
5use serde::{Deserialize, Serialize};
6
7/// Prefixo de um recurso dentro de uma execução.
8#[derive(Debug, Clone, PartialEq, Eq)]
9pub struct LocalizacaoRaw {
10    prefixo: String,
11}
12
13impl LocalizacaoRaw {
14    /// Monta o prefixo de um recurso dentro de uma execução, particionado por data.
15    pub fn nova(fonte: &str, recurso: &str, iniciada_em: DateTime<Utc>, id_execucao: &str) -> Self {
16        let data = iniciada_em.format("%Y-%m-%d");
17        Self {
18            prefixo: format!("{fonte}/{recurso}/data_ingestao={data}/execucao={id_execucao}"),
19        }
20    }
21
22    /// Chave que aponta para a última versão ingerida de um recurso (ver `docs/adr/0010`).
23    ///
24    /// # Exemplos
25    ///
26    /// ```
27    /// use todeolho_nucleo::LocalizacaoRaw;
28    ///
29    /// assert_eq!(
30    ///     LocalizacaoRaw::ultima_versao("ibge", "estados"),
31    ///     "ibge/estados/_ultima_versao.json",
32    /// );
33    /// ```
34    pub fn ultima_versao(fonte: &str, recurso: &str) -> String {
35        format!("{fonte}/{recurso}/_ultima_versao.json")
36    }
37
38    /// Chave da parte de número `numero` (a partir de 1).
39    pub fn parte(&self, numero: usize, extensao: &str) -> String {
40        format!("{}/parte-{numero:05}.{extensao}", self.prefixo)
41    }
42
43    /// Chave do manifesto, gravado por último: sua presença indica ingestão completa.
44    pub fn manifesto(&self) -> String {
45        format!("{}/_manifesto.json", self.prefixo)
46    }
47}

49/// Uma parte gravada de um recurso (uma página de API ou o arquivo baixado).
50#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
51pub struct ParteRaw {
52    /// Chave do objeto na camada Raw.
53    pub chave: String,
54    /// URL de origem desta parte.
55    pub url: String,
56    /// Tamanho do conteúdo, em bytes.
57    pub bytes: u64,
58    /// Assinatura `sha256` do conteúdo (procedência; ver `docs/adr/0012`).
59    pub sha256: String,
60    /// Tipo de conteúdo (`Content-Type`) como veio da origem.
61    pub tipo_de_conteudo: String,
62    /// Validador da origem (ETag) desta página, usado na requisição condicional
63    /// da próxima rodada. `None` quando a origem não o informou.
64    #[serde(default)]
65    pub versao: Option<String>,
66}
67
68/// Procedência de uma ingestão completa de um recurso.
69#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
70pub struct ManifestoRaw {
71    /// Fonte à qual o recurso pertence (ex.: `ibge`).
72    pub fonte: String,
73    /// Recurso ingerido (ex.: `estados`).
74    pub recurso: String,
75    /// Execução que produziu esta ingestão.
76    pub id_execucao: String,
77    /// Primeira URL buscada (início da paginação ou o próprio arquivo).
78    pub url_inicial: String,
79    /// Quando a ingestão começou.
80    pub iniciada_em: DateTime<Utc>,
81    /// Quando a ingestão terminou (o manifesto é gravado por último).
82    pub concluida_em: DateTime<Utc>,
83    /// Partes que compõem o recurso, na ordem em que foram buscadas.
84    pub partes: Vec<ParteRaw>,
85}
86
87/// Ponteiro para a última ingestão completa de um recurso (`_ultima_versao.json`).
88/// Gravado por último, depois do manifesto, e lido pela pré-checagem (ver `docs/adr/0010`).
89#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
90pub struct UltimaVersao {
91    /// Validador da origem (ETag/Last-Modified) daquela ingestão, se houver.
92    #[serde(default)]
93    pub versao: Option<String>,
94    /// Chave do manifesto vigente.
95    pub manifesto: String,
96    /// Início da execução que publicou este ponteiro. Usado para não deixar uma
97    /// rodada mais antiga sobrescrever uma mais nova (ver issue #16). `None` nos
98    /// ponteiros gravados antes deste campo existir, tratados como os mais antigos.
99    #[serde(default)]
100    pub iniciada_em: Option<DateTime<Utc>>,
101}

103#[cfg(test)]
104mod testes {

107    use super::*;

109    #[test]
110    fn monta_as_chaves_no_layout_documentado() {
111        let instante = Utc.with_ymd_and_hms(2026, 10, 31, 23, 59, 0).unwrap();
112        let local = LocalizacaoRaw::nova("ibge", "estados", instante, "abc");
113
114        assert_eq!(
115            local.parte(1, "json"),
116            "ibge/estados/data_ingestao=2026-10-31/execucao=abc/parte-00001.json"
117        );
118        assert_eq!(
119            local.manifesto(),
120            "ibge/estados/data_ingestao=2026-10-31/execucao=abc/_manifesto.json"
121        );
122    }
