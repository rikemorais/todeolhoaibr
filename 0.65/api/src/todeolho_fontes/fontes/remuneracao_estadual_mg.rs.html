16const NOME: &str = "remuneracao-estadual-mg";
17const FONTE_RAW: &str = "remuneracao-estadual";
18const CATALOGO: &str =
19    "https://www.dados.mg.gov.br/api/3/action/package_show?id=remuneracao-servidores-ativos";
20const PREFIXO_DOWNLOAD: &str = "https://www.dados.mg.gov.br/dataset/";
21const TAMANHO_MAXIMO: u64 = 1_073_741_824;
22const HOST: &str = "www.dados.mg.gov.br";
23
24#[derive(Debug, Deserialize)]
25struct RespostaCatalogo {
26    success: bool,
27    result: Option<Pacote>,
28}
29
30#[derive(Debug, Deserialize)]
31struct Pacote {
32    #[serde(default)]
33    resources: Vec<Recurso>,
34}
35
36#[derive(Debug, Deserialize)]
37struct Recurso {
38    format: String,
39    url: String,
40    size: Option<u64>,
41}
42
43#[derive(Debug, Clone, PartialEq, Eq)]
44struct ArquivoMensal {
45    competencia: String,
46    url: String,
47}
48
49fn competencia_do_url(url: &str) -> Option<(i32, u32, String)> {
50    let nome = url.rsplit('/').next()?;
51    let data = nome.strip_prefix("servidores_")?.strip_suffix(".csv")?;
52    if data.len() != 6 || !data.bytes().all(|byte| byte.is_ascii_digit()) {
53        return None;
54    }
55    let mes: u32 = data[..2].parse().ok()?;
56    let ano: i32 = data[2..].parse().ok()?;
57    if !(1..=12).contains(&mes) || !(2012..=2099).contains(&ano) {
58        return None;
59    }
60    Some((ano, mes, format!("{ano}-{mes:02}")))
61}
62
63fn validar_url(url: &str) -> bool {
64    url.starts_with(PREFIXO_DOWNLOAD)
65        && !url.contains(['?', '#', '@'])
66        && competencia_do_url(url).is_some()
67}
68
69fn selecionar_arquivo(pacote: Pacote) -> Result<ArquivoMensal, ErroDeIngestao> {
70    let escolhido = pacote
71        .resources
72        .into_iter()
73        .filter(|recurso| recurso.format.eq_ignore_ascii_case("csv"))
74        .filter(|recurso| validar_url(&recurso.url))
75        .filter(|recurso| {
76            recurso
77                .size
78                .is_some_and(|tamanho| tamanho > 0 && tamanho <= TAMANHO_MAXIMO)
79        })
80        .filter_map(|recurso| {
81            competencia_do_url(&recurso.url).map(|(ano, mes, competencia)| {
82                (
83                    (ano, mes),
84                    ArquivoMensal {
85                        competencia,
86                        url: recurso.url,
87                    },
88                )
89            })
90        })
91        .max_by_key(|(data, _)| *data)
92        .map(|(_, arquivo)| arquivo);
93
94    escolhido.ok_or_else(|| ErroDeIngestao::Http {
95        url: CATALOGO.to_owned(),
96        detalhe: "o catálogo não contém CSV mensal recente com URL e tamanho válidos".into(),
97    })
98}

100async fn arquivo_atual(contexto: &ContextoDeExecucao) -> Result<ArquivoMensal, ErroDeIngestao> {
101    let resposta = contexto.http.obter(CATALOGO, "application/json").await?;
102    let catalogo: RespostaCatalogo =
103        serde_json::from_slice(&resposta.corpo).map_err(|erro| ErroDeIngestao::Http {
104            url: CATALOGO.to_owned(),
105            detalhe: format!("resposta CKAN inválida: {erro}"),
106        })?;
107    if !catalogo.success {
108        return Err(ErroDeIngestao::Http {
109            url: CATALOGO.to_owned(),
110            detalhe: "o catálogo CKAN informou falha ao consultar o conjunto".into(),
111        });
112    }
113    selecionar_arquivo(catalogo.result.ok_or_else(|| ErroDeIngestao::Http {
114        url: CATALOGO.to_owned(),
115        detalhe: "resposta CKAN sem o conjunto solicitado".into(),
116    })?)
117}

119struct TarefaRemuneracaoMg;
120
121fn tarefa_de_arquivo(arquivo: ArquivoMensal) -> TarefaDeArquivo {
122    TarefaDeArquivo::nova(
123        "ultima-competencia".to_owned(),
124        FONTE_RAW,
125        format!("uf=MG/competencia={}", arquivo.competencia),
126        arquivo.url,
127    )
128}
129
130#[async_trait]
131impl Tarefa for TarefaRemuneracaoMg {
132    fn nome(&self) -> &str {
133        "ultima-competencia"
134    }
135
136    async fn bytes_a_baixar(
137        &self,
138        contexto: &ContextoDeExecucao,
139        precisao: PrecisaoDaEstimativa,
140    ) -> Result<Option<u64>, ErroDeIngestao> {
141        // Evita consultar o catálogo e fazer HEAD do arquivo no menu de seleção.
142        if precisao == PrecisaoDaEstimativa::Rapida {
143            return Ok(None);
144        }
145        let arquivo = arquivo_atual(contexto).await?;
146        tarefa_de_arquivo(arquivo)
147            .bytes_a_baixar(contexto, precisao)
148            .await
149    }
150
151    async fn executar(
152        &self,
153        contexto: &ContextoDeExecucao,
154    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
155        let arquivo = arquivo_atual(contexto).await?;
156        tarefa_de_arquivo(arquivo).executar(contexto).await
157    }
158}

160/// Ingestão sob demanda do CSV mensal mais recente de remuneração do Executivo de MG.
161pub struct FonteRemuneracaoEstadualMg;
162
163impl FonteDeDados for FonteRemuneracaoEstadualMg {
164    fn nome(&self) -> &str {
165        NOME
166    }
167
168    fn descricao(&self) -> &str {
169        "Catálogo de Dados Abertos de MG - remuneração do Executivo estadual por competência"
170    }
171
172    fn sob_demanda(&self) -> bool {
173        true
174    }
175
176    fn perfil_http(&self) -> PerfilHttp {
177        PerfilHttp {
178            credencial: None,
179            limite: Some(LimiteHttp {
180                host: HOST,
181                requisicoes: 2,
182                intervalo: Duration::from_secs(1),
183            }),
184        }
185    }
186
187    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
188        vec![TarefaPlanejada::nova(TarefaRemuneracaoMg)]
189    }
190}

198        let pacote: Pacote = serde_json::from_str(
199            r#"{"resources":[

201                {"format":"CSV","url":"https://www.dados.mg.gov.br/dataset/abc/resource/456/download/servidores_092026.csv","size":200},
