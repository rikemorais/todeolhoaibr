13/// Erros ao transformar os votos do TSE por seção.
14#[derive(Debug, Error, PartialEq, Eq)]
15pub enum ErroTseVotacaoSecaoSilver {
16    /// O Parquet Bronze não contém os campos textuais esperados.
17    #[error("schema Bronze de votação por seção do TSE incompatível")]
18    Schema,
19    /// Um dos campos da chave está ausente, vazio ou fora do domínio.
20    #[error("campo {campo} inválido na linha {linha} do lote")]
21    Obrigatorio {
22        /// Campo que não pôde ser validado.
23        campo: &'static str,
24        /// Índice da linha do lote, começando em zero.
25        linha: usize,
26    },
27    /// O ano registrado na linha diverge do ano do recurso.
28    #[error("ANO_ELEICAO diverge do recurso na linha {linha} do lote")]
29    AnoDivergente {
30        /// Índice da linha do lote, começando em zero.
31        linha: usize,
32    },
33    /// A chave da apuração repete uma linha no mesmo recurso.
34    #[error("chave de votação por seção duplicada na linha {linha} do lote")]
35    ChaveDuplicada {
36        /// Índice da linha do lote, começando em zero.
37        linha: usize,
38    },
39    /// A data preenchida não pôde ser interpretada.
40    #[error("DT_ELEICAO inválida na linha {linha} do lote")]
41    Data {
42        /// Índice da linha do lote, começando em zero.
43        linha: usize,
44    },
45    /// A quantidade de votos preenchida não é inteira ou excede UInt32.
46    #[error("QT_VOTOS inválida na linha {linha} do lote")]
47    Votos {
48        /// Índice da linha do lote, começando em zero.
49        linha: usize,
50    },
51    /// Os campos de origem não puderam ser preservados como JSON.
52    #[error("não foi possível preservar os campos originais da votação")]
53    Json,
54}

86/// Tipa a chave da apuração, turno e quantidade de votos sem alterar códigos textuais.
87pub fn transformar_tse_votacao_secao(
88    lote: &RecordBatch,
89    ano: u16,
90    uf_recurso: &str,
91    chaves_vistas: &mut HashSet<String>,
92) -> Result<RecordBatch, ErroTseVotacaoSecaoSilver> {
93    if lote
94        .schema()
95        .fields()
96        .iter()
97        .any(|campo| campo.data_type() != &DataType::Utf8)
98    {
99        return Err(ErroTseVotacaoSecaoSilver::Schema);
100    }
101
102    let nomes = [
103        "ANO_ELEICAO",
104        "CD_ELEICAO",
105        "NR_TURNO",
106        "SG_UF",
107        "SG_UE",
108        "CD_MUNICIPIO",
109        "NR_ZONA",
110        "NR_SECAO",
111        "CD_CARGO",
112        "NR_VOTAVEL",
113        "QT_VOTOS",
114    ];
115    let colunas = nomes
116        .iter()
117        .map(|nome| coluna_obrigatoria(lote, nome))
118        .collect::<Result<Vec<_>, _>>()?;
119    let data_eleicao = coluna_opcional(lote, "DT_ELEICAO")?;
120    let tipo_eleicao = coluna_opcional(lote, "NM_TIPO_ELEICAO")?;
121    let nome_eleicao = coluna_opcional(lote, "DS_ELEICAO")?;
122    let abrangencia = coluna_opcional(lote, "TP_ABRANGENCIA")?;
123    let nome_unidade = coluna_opcional(lote, "NM_UE")?;
124    let nome_municipio = coluna_opcional(lote, "NM_MUNICIPIO")?;
125    let descricao_cargo = coluna_opcional(lote, "DS_CARGO")?;
126    let nome_votavel = coluna_opcional(lote, "NM_VOTAVEL")?;
127    let numero_local = coluna_opcional(lote, "NR_LOCAL_VOTACAO")?;
128    let sequencia_candidato = coluna_opcional(lote, "SQ_CANDIDATO")?;
129    let nome_local = coluna_opcional(lote, "NM_LOCAL_VOTACAO")?;
130    let endereco_local = coluna_opcional(lote, "DS_LOCAL_VOTACAO_ENDERECO")?;
131
132    let mut turnos = Vec::with_capacity(lote.num_rows());
133    let mut votos = Vec::with_capacity(lote.num_rows());
134    let mut datas = Vec::with_capacity(lote.num_rows());
135    let mut originais = Vec::with_capacity(lote.num_rows());
136    for linha in 0..lote.num_rows() {
137        if texto_obrigatorio(colunas[0], linha, "ANO_ELEICAO")?
138            .parse::<u16>()
139            .ok()
140            != Some(ano)
141        {
142            return Err(ErroTseVotacaoSecaoSilver::AnoDivergente { linha });
143        }
144        let uf = texto_obrigatorio(colunas[3], linha, "SG_UF")?;
145        if uf_recurso != "BR" && uf != uf_recurso {
146            return Err(ErroTseVotacaoSecaoSilver::Obrigatorio {
147                campo: "SG_UF",
148                linha,
149            });
150        }
151        let turno = texto_obrigatorio(colunas[2], linha, "NR_TURNO")?
152            .parse::<u8>()
153            .ok()
154            .filter(|turno| *turno > 0)
155            .ok_or(ErroTseVotacaoSecaoSilver::Obrigatorio {
156                campo: "NR_TURNO",
157                linha,
158            })?;
159        let quantidade_votos = texto_obrigatorio(colunas[10], linha, "QT_VOTOS")?
160            .parse::<u32>()
161            .map_err(|_| ErroTseVotacaoSecaoSilver::Votos { linha })?;
162        let chave = [
163            ano.to_string(),
164            texto_obrigatorio(colunas[1], linha, "CD_ELEICAO")?.to_owned(),
165            uf.to_owned(),
166            turno.to_string(),
167            texto_obrigatorio(colunas[5], linha, "CD_MUNICIPIO")?.to_owned(),
168            texto_obrigatorio(colunas[6], linha, "NR_ZONA")?.to_owned(),
169            texto_obrigatorio(colunas[7], linha, "NR_SECAO")?.to_owned(),
170            texto_obrigatorio(colunas[8], linha, "CD_CARGO")?.to_owned(),
171            texto_obrigatorio(colunas[9], linha, "NR_VOTAVEL")?.to_owned(),
172        ]
173        .join("\0");
174        if !chaves_vistas.insert(chave) {
175            return Err(ErroTseVotacaoSecaoSilver::ChaveDuplicada { linha });
176        }
177        datas.push(parse_data(data_eleicao, linha)?);
178        turnos.push(turno);
179        votos.push(quantidade_votos);
180        originais.push(Some(registro_original(lote, linha)?));
181    }
182
183    let texto_preenchido = |coluna: Option<&StringArray>| -> Arc<dyn Array> {
184        coluna.map_or_else(
185            || Arc::new(StringArray::from(vec![None::<&str>; lote.num_rows()])) as Arc<dyn Array>,
186            |coluna| Arc::new(coluna.clone()),
187        )
188    };
189    RecordBatch::try_new(
190        esquema_tse_votacao_secao(),
191        vec![
192            Arc::new(UInt16Array::from_value(ano, lote.num_rows())),
193            Arc::new(colunas[1].clone()),
194            Arc::new(UInt8Array::from(turnos)),
195            Arc::new(colunas[3].clone()),
196            Arc::new(colunas[4].clone()),
197            Arc::new(colunas[5].clone()),
198            Arc::new(colunas[6].clone()),
199            Arc::new(colunas[7].clone()),
200            Arc::new(colunas[8].clone()),
201            Arc::new(colunas[9].clone()),
202            Arc::new(UInt32Array::from(votos)),
203            Arc::new(Date32Array::from(datas)),
204            texto_preenchido(tipo_eleicao),
205            texto_preenchido(nome_eleicao),
206            texto_preenchido(abrangencia),
207            texto_preenchido(nome_unidade),
208            texto_preenchido(nome_municipio),
209            texto_preenchido(descricao_cargo),
210            texto_preenchido(nome_votavel),
211            texto_preenchido(numero_local),
212            texto_preenchido(sequencia_candidato),
213            texto_preenchido(nome_local),
214            texto_preenchido(endereco_local),
215            Arc::new(StringArray::from(originais)),
216        ],
217    )
218    .map_err(|_| ErroTseVotacaoSecaoSilver::Schema)
219}

221fn coluna_obrigatoria<'a>(
222    lote: &'a RecordBatch,
223    nome: &'static str,
224) -> Result<&'a StringArray, ErroTseVotacaoSecaoSilver> {
225    coluna_opcional(lote, nome)?.ok_or(ErroTseVotacaoSecaoSilver::Schema)
226}
227
228fn coluna_opcional<'a>(
229    lote: &'a RecordBatch,
230    nome: &str,
231) -> Result<Option<&'a StringArray>, ErroTseVotacaoSecaoSilver> {
232    let Ok(indice) = lote.schema().index_of(nome) else {
233        return Ok(None);
234    };
235    lote.column(indice)
236        .as_any()
237        .downcast_ref::<StringArray>()
238        .map(Some)
239        .ok_or(ErroTseVotacaoSecaoSilver::Schema)
240}
241
242fn texto_obrigatorio<'a>(
243    coluna: &'a StringArray,
244    linha: usize,
245    campo: &'static str,
246) -> Result<&'a str, ErroTseVotacaoSecaoSilver> {
247    if coluna.is_null(linha) || coluna.value(linha).trim().is_empty() {
248        return Err(ErroTseVotacaoSecaoSilver::Obrigatorio { campo, linha });
249    }
250    Ok(coluna.value(linha).trim())
251}
252
253fn parse_data(
254    coluna: Option<&StringArray>,
255    linha: usize,
256) -> Result<Option<i32>, ErroTseVotacaoSecaoSilver> {
257    let Some(coluna) = coluna else {
258        return Ok(None);
259    };
260    if coluna.is_null(linha) || coluna.value(linha).trim().is_empty() {
261        return Ok(None);
262    }
263    let texto = coluna.value(linha).trim();
264    let data = ["%d/%m/%Y", "%Y-%m-%d", "%Y%m%d"]
265        .iter()
266        .find_map(|formato| NaiveDate::parse_from_str(texto, formato).ok())
267        .ok_or(ErroTseVotacaoSecaoSilver::Data { linha })?;
268    let epoch = NaiveDate::from_ymd_opt(1970, 1, 1).expect("epoch válida");
269    Ok(Some(data.signed_duration_since(epoch).num_days() as i32))
270}
271
272fn registro_original(
273    lote: &RecordBatch,
274    linha: usize,
275) -> Result<String, ErroTseVotacaoSecaoSilver> {
276    let mut original = serde_json::Map::new();
277    for (indice, campo) in lote.schema().fields().iter().enumerate() {
278        let coluna = lote
279            .column(indice)
280            .as_any()
281            .downcast_ref::<StringArray>()
282            .ok_or(ErroTseVotacaoSecaoSilver::Schema)?;
283        original.insert(
284            campo.name().clone(),
285            if coluna.is_null(linha) {
286                serde_json::Value::Null
287            } else {
288                serde_json::Value::String(coluna.value(linha).to_owned())
289            },
290        );
291    }
292    serde_json::to_string(&original).map_err(|_| ErroTseVotacaoSecaoSilver::Json)
293}
