19/// Destino dos dados brutos.
20#[async_trait]
21pub trait ArmazenamentoRaw: Send + Sync {
22    /// Garante que o destino existe. Precisa ser idempotente.
23    async fn preparar(&self) -> Result<(), ErroDeIngestao>;
24
25    /// Grava um objeto por inteiro na camada Raw.
26    async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao>;
27
28    /// Grava sem carregar tudo na memória. O objeto só fica visível ao final da transferência.
29    async fn gravar_fluxo(
30        &self,
31        chave: &str,
32        tipo_de_conteudo: &str,
33        fluxo: &mut (dyn AsyncRead + Send + Unpin),
34    ) -> Result<(), ErroDeIngestao>;
35
36    /// Remove um objeto publicado na camada Raw.
37    async fn remover(&self, chave: &str) -> Result<(), ErroDeIngestao> {
38        Err(ErroDeIngestao::Armazenamento {
39            chave: chave.to_owned(),
40            detalhe: "remoção não implementada pelo adaptador".into(),
41        })
42    }
43
44    /// Apaga transferências que nunca terminaram, como as de um processo interrompido,
45    /// para que não ocupem disco. Uma ingestão em andamento perde a transferência atual e
46    /// a recomeça, então nenhum dado se perde.
47    async fn descartar_envios_incompletos(&self) -> Result<usize, ErroDeIngestao> {
48        Ok(0)
49    }
50
51    /// Conteúdo do objeto, ou `None` se ele não existir.
52    async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao>;
53
54    /// Baixa o objeto escrevendo-o em `escrita`, sem carregar tudo na memória; devolve `false`
55    /// se não existir. Útil para objetos grandes (ex.: ZIPs de vários GB do TSE). O padrão usa
56    /// [`Self::ler`] (carrega na memória); a infraestrutura deve transmitir em fluxo.
57    async fn ler_fluxo(
58        &self,
59        chave: &str,
60        escrita: &mut (dyn AsyncWrite + Send + Unpin),
61    ) -> Result<bool, ErroDeIngestao> {
62        use tokio::io::AsyncWriteExt;
63        match self.ler(chave).await? {
64            Some(bytes) => {
65                escrita
66                    .write_all(&bytes)
67                    .await
68                    .map_err(|erro| ErroDeIngestao::Armazenamento {
69                        chave: chave.to_owned(),
70                        detalhe: erro.to_string(),
71                    })?;
72                Ok(true)
73            }
74            None => Ok(false),
75        }
76    }
77
78    /// Tamanho do objeto em bytes, sem baixá-lo (um `HEAD`), ou `None` se não
79    /// existir. A implementação padrão lê o objeto; as de infraestrutura devem
80    /// sobrescrever com um `HEAD` para não carregar arquivos grandes na memória.
81    async fn tamanho(&self, chave: &str) -> Result<Option<u64>, ErroDeIngestao> {
82        Ok(self.ler(chave).await?.map(|conteudo| conteudo.len() as u64))
83    }
84}

86/// Resposta bem-sucedida de uma API.

88pub struct RespostaHttp {
89    /// Tipo de conteúdo (`Content-Type`) devolvido.
90    pub tipo_de_conteudo: String,

92    pub corpo: Vec<u8>,

95/// Resultado de um GET condicional.
96#[derive(Debug, Clone)]
97pub enum RespostaCondicional {
98    /// 304: o conteúdo não mudou desde o validador enviado.
99    NaoModificado,
100    /// A origem devolveu um corpo novo.
101    Modificado {
102        /// Tipo de conteúdo (`Content-Type`) do novo corpo.
103        tipo_de_conteudo: String,
104        /// Validador do novo conteúdo (ETag), quando informado.
105        versao: Option<String>,
106        /// Corpo novo.
107        corpo: Vec<u8>,
108    },
109}

111/// O que a origem informa sobre um arquivo sem precisar baixá-lo.
112#[derive(Debug, Clone, Default, PartialEq, Eq)]
113pub struct InformacoesDoArquivo {
114    /// Identifica a versão do arquivo (ETag).
115    pub versao: Option<String>,
116    /// Tamanho do arquivo em bytes (`Content-Length`), quando informado.
117    pub tamanho: Option<u64>,
118}

120/// Fluxo de bytes lido aos poucos, sem carregar tudo na memória.

123/// Resposta bem-sucedida cujo corpo ainda será lido aos poucos.
124pub struct DownloadHttp {
125    /// Tipo de conteúdo (`Content-Type`) do arquivo.
126    pub tipo_de_conteudo: String,
127    /// Identifica a versão do arquivo na origem (ETag), quando informada.
128    pub versao: Option<String>,
129    /// Deslocamento, em bytes, onde este fluxo começa no arquivo. `0` num
130    /// download do início; numa retomada (`206 Partial Content`) é o início do
131    /// `Content-Range`. Quem retoma compara com o offset pedido: se não bater, a
132    /// origem não honrou a retomada e o download deve recomeçar do zero.
133    pub inicio: u64,
134    /// Tamanho total do arquivo em bytes, quando a origem informa
135    /// (`Content-Length` de um `200`, ou o total do `Content-Range` de um `206`).
136    pub tamanho_total: Option<u64>,
137    /// Fluxo do corpo, lido sob demanda.
138    pub fluxo: FluxoDeBytes,
139}

141/// Acesso às APIs públicas. Falhas transitórias são tratadas pela implementação.
142#[async_trait]
143pub trait ClienteHttp: Send + Sync {
144    /// GET simples: devolve o corpo inteiro em memória.
145    async fn obter(&self, url: &str, aceita: &str) -> Result<RespostaHttp, ErroDeIngestao>;
146
147    /// GET condicional: envia o validador (If-None-Match) e devolve
148    /// `NaoModificado` em caso de 304. A implementação padrão ignora o validador
149    /// e sempre baixa, para fontes que não suportam requisição condicional.
150    async fn obter_condicional(
151        &self,
152        url: &str,
153        aceita: &str,
154        _validador: Option<&str>,
155    ) -> Result<RespostaCondicional, ErroDeIngestao> {
156        let resposta = self.obter(url, aceita).await?;
157        Ok(RespostaCondicional::Modificado {
158            tipo_de_conteudo: resposta.tipo_de_conteudo,
159            versao: None,
160            corpo: resposta.corpo,
161        })
162    }
163
164    /// Consulta o arquivo na origem, sem baixá-lo.
165    async fn informacoes(&self, url: &str) -> Result<InformacoesDoArquivo, ErroDeIngestao>;
166
167    /// Baixa um arquivo como fluxo, para gravá-lo sem carregá-lo na memória.
168    async fn baixar(&self, url: &str) -> Result<DownloadHttp, ErroDeIngestao>;
169
170    /// Baixa a partir de `inicio` (em bytes; `0` é do começo), para **retomar**
171    /// um download interrompido sem perder nem duplicar byte. `versao` é o
172    /// validador (ETag) da primeira resposta: a implementação deve enviar
173    /// `If-Range`, de modo a só retomar se o arquivo não mudou. O
174    /// [`DownloadHttp::inicio`] devolvido diz onde o fluxo realmente começa — se
175    /// não for igual a `inicio`, a origem não honrou a retomada e o chamador deve
176    /// recomeçar do zero em vez de emendar.
177    ///
178    /// A implementação padrão ignora `inicio`/`versao` e baixa do começo, para
179    /// clientes que não suportam `Range`; quem retoma detecta o `inicio` devolvido
180    /// (`0`) diferente do pedido e recomeça.
181    async fn baixar_a_partir_de(
182        &self,
183        url: &str,
184        inicio: u64,
185        versao: Option<&str>,
186    ) -> Result<DownloadHttp, ErroDeIngestao> {
187        let _ = (inicio, versao);
188        self.baixar(url).await
189    }
