1//! Decodificador do bronze para fontes que gravam um **CSV "solto"** no raw (não zipado).
2//!
3//! Serve, por exemplo, para as despesas do Senado (CEAPS, Latin-1, com uma linha de
4//! preâmbulo antes do cabeçalho) e para as votações da Câmara (UTF-8 com BOM). O bronze só
5//! reencoda para UTF-8 e põe em colunas — a interpretação dos campos é do silver.
6
7use crate::csv_zip::{
8    partes_de_csv, partes_de_csv_em_fluxo, partes_de_csv_windows1252,
9    partes_de_csv_windows1252_em_fluxo,
10};
11use crate::decodificador::{Decodificador, ParteBronze};
12use crate::escritor::ErroDeRefino;
13
14/// Decodifica um CSV solto em uma parte Parquet, com o nome lógico e o separador da fonte.
15///
16/// Quando chamado via `decodificar_arquivo`, opera em fluxo (lê do disco sem carregar
17/// o CSV inteiro em memória). Ver #164.
18pub struct DecodificadorCsv {
19    nome: &'static str,
20    separador: u8,
21    linhas_a_pular: usize,
22    windows1252: bool,
23}
24
25impl DecodificadorCsv {
26    /// Cria o decodificador. `nome` é o nome lógico da tabela registrado na procedência.
27    pub const fn novo(nome: &'static str, separador: u8, linhas_a_pular: usize) -> Self {
28        Self {
29            nome,
30            separador,
31            linhas_a_pular,
32            windows1252: false,
33        }
34    }
35
36    /// Cria um decodificador para uma fonte cujo CSV é declarado Windows-1252.
37    pub const fn novo_windows1252(
38        nome: &'static str,
39        separador: u8,
40        linhas_a_pular: usize,
41    ) -> Self {
42        Self {
43            nome,
44            separador,
45            linhas_a_pular,
46            windows1252: true,
47        }
48    }
49}
50
51impl Decodificador for DecodificadorCsv {
52    fn decodificar(
53        &self,
54        bytes: &[u8],
55        _tipo_de_conteudo: &str,
56    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
57        if self.windows1252 {
58            partes_de_csv_windows1252(bytes, self.nome, self.separador, self.linhas_a_pular)
59        } else {
60            partes_de_csv(bytes, self.nome, self.separador, self.linhas_a_pular)
61        }
62    }
63
64    fn decodificar_arquivo(
65        &self,
66        caminho: &std::path::Path,
67        _tipo_de_conteudo: &str,
68    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
69        let arquivo = std::fs::File::open(caminho)?;
70        if self.windows1252 {
71            partes_de_csv_windows1252_em_fluxo(
72                arquivo,
73                self.nome,
74                self.separador,
75                self.linhas_a_pular,
76            )
77        } else {
78            partes_de_csv_em_fluxo(arquivo, self.nome, self.separador, self.linhas_a_pular)
79        }
80    }
81}

83#[cfg(test)]
84mod testes {
85    use super::*;
86    use crate::csv_zip::testes::ler_parte;
87
88    fn c(s: &str) -> Option<String> {
89        Some(s.to_owned())
90    }
91
92    #[test]
93    fn decodifica_csv_solto_pulando_o_preambulo() {
94        // Como o CEAPS do Senado: uma linha de preâmbulo antes do cabeçalho real.
95        let csv = "\"ULTIMA ATUALIZACAO\";\"hoje\"\nano;valor\n2014;52,38\n";
96        let partes = DecodificadorCsv::novo("despesas", b';', 1)
97            .decodificar(csv.as_bytes(), "application/octet-stream")
98            .unwrap();
99        assert_eq!(partes.len(), 1);
100        assert_eq!(partes[0].nome, "despesas");
101        let (colunas, linhas) = ler_parte(&partes[0]);
102        assert_eq!(colunas, vec!["ano", "valor"]);
103        assert_eq!(linhas, vec![vec![c("2014"), c("52,38")]]);
104    }
105
106    #[test]
107    fn remove_o_bom_de_utf8_da_primeira_coluna() {
108        // Como as votações da Câmara: UTF-8 com BOM no início.
109        let mut csv = vec![0xEF, 0xBB, 0xBF];
110        csv.extend_from_slice("idVotacao;orientacao\n1;Sim\n".as_bytes());
111        let partes = DecodificadorCsv::novo("votacoes", b';', 0)
112            .decodificar(&csv, "application/octet-stream")
113            .unwrap();
114        let (colunas, linhas) = ler_parte(&partes[0]);
115        assert_eq!(colunas, vec!["idVotacao", "orientacao"]);
116        assert_eq!(linhas, vec![vec![c("1"), c("Sim")]]);
117    }
118
119    #[test]
120    fn decodifica_windows1252_mesmo_com_prefixo_ascii() {
121        let csv = b"pais;valor\n";
122        let mut bytes = csv.to_vec();
123        bytes.extend_from_slice(b"Sao Paulo;1\nCaf\xe9;2\n");
124        let partes = DecodificadorCsv::novo_windows1252("dados", b';', 0)
125            .decodificar(&bytes, "text/csv")
126            .unwrap();
127        let (colunas, linhas) = ler_parte(&partes[0]);
128        assert_eq!(colunas, ["pais", "valor"]);
129        assert_eq!(linhas[1], vec![c("Café"), c("2")]);
130    }
131}
