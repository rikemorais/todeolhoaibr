19/// Tarefa Silver para o recurso Bronze `deputados`.
20pub struct TarefaDeSilverCamaraDeputados {
21    bronze: Arc<dyn ArmazenamentoRaw>,
22    snapshot: Arc<SnapshotBronze>,
23    recurso: String,
24}
25
26impl TarefaDeSilverCamaraDeputados {
27    /// Cria a tarefa para o recurso e snapshot selecionados.
28    pub fn nova(
29        bronze: Arc<dyn ArmazenamentoRaw>,
30        snapshot: Arc<SnapshotBronze>,
31        recurso: impl Into<String>,
32    ) -> Self {
33        Self {
34            bronze,
35            snapshot,
36            recurso: recurso.into(),
37        }
38    }
39
40    /// Indica se a Silver ainda precisa acompanhar o snapshot Bronze.
41    pub async fn pendente(&self, silver: &dyn ArmazenamentoRaw) -> Result<bool, ErroDeIngestao> {
42        let versao = format!("{}|regra={REGRA}", self.snapshot.manifesto().id_execucao);
43        Ok(
44            !matches!(estado_na_raw(silver, "camara", &self.recurso).await?,
45            Some(ref estado) if estado.completa && estado.integra && estado.versao.as_deref() == Some(versao.as_str())),
46        )
47    }
48}

50#[async_trait]
51impl Tarefa for TarefaDeSilverCamaraDeputados {
52    fn nome(&self) -> &str {
53        &self.recurso
54    }
55    async fn executar(
56        &self,
57        contexto: &ContextoDeExecucao,
58    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
59        publicar_camara_deputados(
60            self.bronze.as_ref(),
61            contexto.armazenamento.as_ref(),
62            &self.snapshot,
63            &self.recurso,
64            contexto.iniciada_em,
65            &contexto.id_execucao,
66        )
67        .await
68    }
69}

71/// Publica a fotografia vigente do cadastro de deputados.
72pub async fn publicar_camara_deputados(
73    bronze: &dyn ArmazenamentoRaw,
74    silver: &dyn ArmazenamentoRaw,
75    snapshot: &SnapshotBronze,
76    recurso: &str,
77    iniciada_em: DateTime<Utc>,
78    id_execucao: &str,
79) -> Result<ResumoDaTarefa, ErroDeIngestao> {
80    publicar_transformacao(
81        bronze,
82        silver,
83        snapshot,
84        ConfiguracaoDePublicacao {
85            fonte: "camara",
86            recurso,
87            regra: REGRA,
88            esquema: Some(esquema_camara_deputados()),
89            iniciada_em,
90            id_execucao,
91        },
92        |lote| transformar_camara_deputados(lote).map_err(|erro| erro.to_string()),
93    )
94    .await
95}

116    #[async_trait]
117    impl ArmazenamentoRaw for Memoria {
118        async fn preparar(&self) -> Result<(), ErroDeIngestao> {
119            Ok(())
120        }
121        async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao> {
122            self.0.lock().unwrap().insert(objeto.chave, objeto.conteudo);
123            Ok(())
124        }
125        async fn gravar_fluxo(
126            &self,
127            chave: &str,
128            _tipo: &str,
129            fluxo: &mut (dyn AsyncRead + Send + Unpin),
130        ) -> Result<(), ErroDeIngestao> {
131            let mut dados = Vec::new();
132            fluxo
133                .read_to_end(&mut dados)
134                .await
135                .map_err(|erro| ErroDeIngestao::Armazenamento {
136                    chave: chave.into(),
137                    detalhe: erro.to_string(),
138                })?;
139            self.0.lock().unwrap().insert(chave.into(), dados);
140            Ok(())
141        }
142        async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao> {
143            Ok(self.0.lock().unwrap().get(chave).cloned())
144        }
145    }

183        let mut bytes = Vec::new();

187        writer.close().unwrap();

191    #[tokio::test]
192    async fn publica_e_reutiliza_a_versao_bronze() {
193        let bronze = Memoria::default();
194        let silver = Memoria::default();
195        let iniciado = Utc::now();
196        let local = LocalizacaoRaw::nova("camara", "deputados", iniciado, "bronze-1");
197        let bytes = parquet();
198        let parte = ParteRaw {
199            chave: local.parte(1, "parquet"),
200            url: "https://dadosabertos.camara.leg.br/api/v2/deputados".into(),
201            bytes: bytes.len() as u64,
202            sha256: hex::encode(Sha256::digest(&bytes)),
203            tipo_de_conteudo: "application/vnd.apache.parquet".into(),
204            versao: Some("raw-1".into()),
205        };
206        let manifesto = ManifestoRaw {
207            fonte: "camara".into(),
208            recurso: "deputados".into(),
209            id_execucao: "bronze-1".into(),
210            url_inicial: "https://dadosabertos.camara.leg.br/api/v2/deputados".into(),
211            iniciada_em: iniciado,
212            concluida_em: Utc::now(),
213            partes: vec![parte.clone()],
214        };
215        for objeto in [
216            ObjetoRaw {
217                chave: parte.chave,
218                conteudo: bytes,
219                tipo_de_conteudo: "application/vnd.apache.parquet".into(),
220            },
221            ObjetoRaw {
222                chave: local.manifesto(),
223                conteudo: serde_json::to_vec(&manifesto).unwrap(),
224                tipo_de_conteudo: "application/json".into(),
225            },
226            ObjetoRaw {
227                chave: LocalizacaoRaw::ultima_versao("camara", "deputados"),
228                conteudo: serde_json::to_vec(&UltimaVersao {
229                    versao: Some("bronze-1".into()),
230                    manifesto: local.manifesto(),
231                    iniciada_em: Some(iniciado),
232                })
233                .unwrap(),
234                tipo_de_conteudo: "application/json".into(),
235            },
236        ] {
237            bronze.gravar(objeto).await.unwrap();
238        }
239        let snapshot = SnapshotBronze::vigente(&bronze, "camara", "deputados")
240            .await
241            .unwrap()
242            .unwrap();
243        let tarefa = TarefaDeSilverCamaraDeputados::nova(
244            Arc::new(Memoria::default()),
245            Arc::new(snapshot),
246            "deputados",
247        );
248        assert_eq!(tarefa.nome(), "deputados");
249        assert!(tarefa.pendente(&silver).await.unwrap());
250        let primeiro = publicar_camara_deputados(
251            &bronze,
252            &silver,
253            &tarefa.snapshot,
254            "deputados",
255            Utc::now(),
256            "silver-1",
257        )
258        .await
259        .unwrap();
260        assert_eq!(primeiro.partes, 1);
261        assert!(!tarefa.pendente(&silver).await.unwrap());
262    }
