13/// Uma parte por vez em memória: limita o consumo, seja qual for o tamanho do arquivo.
14/// O `put_object_stream` do rust-s3 não serve: dimensiona a concorrência pela memória
15/// livre da máquina, ignorando o limite do container.
16const TAMANHO_DA_PARTE: usize = 16 * 1024 * 1024;
17const TENTATIVAS_DE_ESCRITA: u32 = 4;
18const ESPERA_BASE: std::time::Duration = std::time::Duration::from_millis(200);
19/// Timeout curto para ponteiros e outros metadados pequenos. Transferências de
20/// dados continuam usando o timeout generoso configurado no bucket principal.
21const TIMEOUT_CONTROLE_S3: Duration = Duration::from_secs(15);

45impl ArmazenamentoS3 {
46    async fn repetir<T, F, Fut>(
47        &self,
48        chave: &str,
49        rotulo: &str,
50        operacao: F,
51    ) -> Result<T, ErroDeIngestao>
52    where
53        F: FnMut() -> Fut,
54        Fut: Future<Output = Result<T, S3Error>>,
55    {
56        self.repetir_s3(chave, rotulo, operacao)
57            .await
58            .map_err(|erro| self.erro(chave, erro))
59    }
60
61    async fn repetir_s3<T, F, Fut>(
62        &self,
63        chave: &str,
64        rotulo: &str,
65        mut operacao: F,
66    ) -> Result<T, S3Error>
67    where
68        F: FnMut() -> Fut,
69        Fut: Future<Output = Result<T, S3Error>>,
70    {
71        for tentativa in 1..=TENTATIVAS_DE_ESCRITA {
72            match operacao().await {
73                Ok(resultado) => return Ok(resultado),
74                Err(erro) if eh_transitorio(&erro) && tentativa < TENTATIVAS_DE_ESCRITA => {
75                    let exponencial = ESPERA_BASE.saturating_mul(1 << (tentativa - 1));
76                    let espera = exponencial + jitter(tentativa);
77                    warn!(%chave, %erro, tentativa, ?espera, %rotulo, "operação S3 falhou; tentando novamente");
78                    tokio::time::sleep(espera).await;
79                }
80                Err(erro) => return Err(erro),
81            }
82        }
83        unreachable!("ao menos uma tentativa é feita")
84    }

86    /// Conecta ao serviço S3 com a configuração dada.
87    pub fn novo(configuracao: ConfiguracaoS3) -> Result<Self, ErroDeIngestao> {
88        let erro = |detalhe: String| ErroDeIngestao::Armazenamento {
89            chave: configuracao.bucket.clone(),
90            detalhe,
91        };
92
93        let regiao = Region::Custom {
94            region: configuracao.regiao.clone(),
95            endpoint: configuracao.endpoint.clone(),
96        };
97        let credenciais = Credentials::new(
98            Some(&configuracao.chave_de_acesso),
99            Some(&configuracao.chave_secreta),
100            None,
101            None,
102            None,
103        )
104        .map_err(|e| erro(e.to_string()))?;
105        // Timeout generoso: baixar/subir objetos de vários GB (ZIPs do TSE) passa do padrão.
106        let bucket = Bucket::new(&configuracao.bucket, regiao.clone(), credenciais.clone())
107            .map_err(|e| erro(e.to_string()))?
108            .with_path_style()
109            .with_request_timeout(Duration::from_secs(1800))
110            .map_err(|e| erro(e.to_string()))?;
111
112        Ok(Self {
113            bucket,
114            regiao,
115            credenciais,
116        })
117    }

150    async fn preparar(&self) -> Result<(), ErroDeIngestao> {
151        let nome = self.bucket.name();
152        if self
153            .bucket
154            .exists()
155            .await
156            .map_err(|e| self.erro(&nome, e))?
157        {
158            return Ok(());
159        }
160
161        let criacao = Bucket::create_with_path_style(
162            &nome,
163            self.regiao.clone(),
164            self.credenciais.clone(),
165            BucketConfiguration::default(),
166        )
167        .await;
168
169        match criacao {
170            Ok(_) => {
171                info!(bucket = %nome, "bucket criado");
172                Ok(())
173            }
174            // Outro processo criou o bucket entre a verificação e a criação.
175            Err(S3Error::HttpFailWithBody(409, _)) => Ok(()),
176            Err(erro) => Err(self.erro(&nome, erro)),
177        }
178    }

184                &objeto.conteudo,
185                &objeto.tipo_de_conteudo,
186            )
187        })
188        .await?;
189        Ok(())
190    }

192    async fn gravar_condicional(
193        &self,
194        objeto: ObjetoRaw,
195        conteudo_vigente: Option<&[u8]>,
196    ) -> Result<bool, ErroDeIngestao> {
197        let bucket = self.bucket_de_controle(&objeto.chave)?;
198        let atual = self
199            .repetir_s3(&objeto.chave, "GET condicional", || {
200                bucket.get_object(&objeto.chave)
201            })
202            .await;
203        let (nome_cabecalho, valor_cabecalho) = match atual {
204            Ok(atual) => {
205                let Some(esperado) = conteudo_vigente else {
206                    return Ok(false);
207                };
208                if atual.as_slice() != esperado {
209                    return Ok(false);
210                }
211                let etag = atual
212                    .headers()
213                    .into_iter()
214                    .find(|(nome, _)| nome.eq_ignore_ascii_case("etag"))
215                    .map(|(_, valor)| valor)
216                    .ok_or_else(|| ErroDeIngestao::Armazenamento {
217                        chave: objeto.chave.clone(),
218                        detalhe: "resposta S3 sem ETag para gravação condicional".into(),
219                    })?;
220                ("if-match", etag)
221            }
222            Err(S3Error::HttpFailWithBody(404, _)) if conteudo_vigente.is_none() => {
223                ("if-none-match", "*".to_owned())
224            }
225            Err(S3Error::HttpFailWithBody(404, _)) => return Ok(false),
226            Err(erro) => return Err(self.erro(&objeto.chave, erro)),
227        };
228
229        let resultado = self
230            .repetir_s3(&objeto.chave, "PUT condicional", || async {
231                bucket
232                    .put_object_builder(&objeto.chave, &objeto.conteudo)
233                    .with_content_type(&objeto.tipo_de_conteudo)
234                    .with_header(nome_cabecalho, valor_cabecalho.clone())?
235                    .execute()
236                    .await
237            })
238            .await;
239
240        match resultado {
241            Ok(_) => Ok(true),
242            Err(erro @ S3Error::HttpFailWithBody(409 | 412, _)) => {
243                match self.ler_controle(&objeto.chave).await {
244                    Ok(Some(atual)) if atual == objeto.conteudo => Ok(true),
245                    Ok(_) => Ok(false),
246                    Err(_) => Err(self.erro(&objeto.chave, erro)),
247                }
248            }
249            Err(erro) => match self.ler_controle(&objeto.chave).await {
250                Ok(Some(atual)) if atual == objeto.conteudo => Ok(true),
251                Ok(atual) if atual.as_deref() != conteudo_vigente => Ok(false),
252                Ok(_) | Err(_) => Err(self.erro(&objeto.chave, erro)),
253            },
254        }
255    }

267    async fn gravar_fluxo(
268        &self,
269        chave: &str,
270        tipo_de_conteudo: &str,
271        fluxo: &mut (dyn AsyncRead + Send + Unpin),
272    ) -> Result<(), ErroDeIngestao> {
273        let primeira = ler_parte(fluxo).await.map_err(|e| self.erro(chave, e))?;
274        if primeira.len() < TAMANHO_DA_PARTE {
275            self.repetir(chave, "PUT", || {
276                self.bucket
277                    .put_object_with_content_type(chave, &primeira, tipo_de_conteudo)
278            })
279            .await?;
280            return Ok(());
281        }
282
283        let envio = self
284            .repetir(chave, "POST initiate-multipart", || {
285                self.bucket
286                    .initiate_multipart_upload(chave, tipo_de_conteudo)
287            })
288            .await?;
289
290        let resultado = self
291            .enviar_partes(chave, tipo_de_conteudo, &envio.upload_id, primeira, fluxo)
292            .await;
293        if resultado.is_err() {
294            // Sem isso, as partes já enviadas ficariam ocupando espaço no volume.
295            if let Err(erro) = self.bucket.abort_upload(chave, &envio.upload_id).await {
296                warn!(%chave, %erro, "não foi possível cancelar o envio");
297            }
298        }
299        resultado
300    }

302    async fn descartar_envios_incompletos(&self) -> Result<usize, ErroDeIngestao> {
303        let nome = self.bucket.name();
304        let paginas = self
305            .bucket
306            .list_multiparts_uploads(None, None)
307            .await
308            .map_err(|e| self.erro(&nome, e))?;
309
310        let mut descartados = 0;
311        for envio in paginas.iter().flat_map(|pagina| &pagina.uploads) {
312            self.bucket
313                .abort_upload(&envio.key, &envio.id)
314                .await
315                .map_err(|e| self.erro(&envio.key, e))?;
316            descartados += 1;
317        }
318        Ok(descartados)
319    }

341    async fn ler_fluxo(
342        &self,
343        chave: &str,
344        escrita: &mut (dyn tokio::io::AsyncWrite + Send + Unpin),
345    ) -> Result<bool, ErroDeIngestao> {
346        match self.bucket.get_object_to_writer(chave, escrita).await {
347            Ok(200) => Ok(true),
348            Ok(404) => Ok(false),
349            Ok(status) => Err(self.erro(
350                chave,
351                S3Error::HttpFailWithBody(status, "status inesperado".into()),
352            )),
353            Err(S3Error::HttpFailWithBody(404, _)) => Ok(false),
354            Err(erro) => Err(self.erro(chave, erro)),
355        }
356    }

358    /// Um `HEAD`, sem baixar o objeto: a pré-checagem confere o tamanho de
359    /// arquivos grandes sem carregá-los na memória (ver `docs/adr/0012`).
360    async fn tamanho(&self, chave: &str) -> Result<Option<u64>, ErroDeIngestao> {
361        match self.bucket.head_object(chave).await {
362            Ok((cabecalho, _)) => Ok(cabecalho
363                .content_length
364                .and_then(|bytes| u64::try_from(bytes).ok())),
365            Err(S3Error::HttpFailWithBody(404, _)) => Ok(None),
366            Err(erro) => Err(self.erro(chave, erro)),
367        }
368    }

371impl ArmazenamentoS3 {
372    async fn enviar_partes(
373        &self,
374        chave: &str,
375        tipo_de_conteudo: &str,
376        id_do_envio: &str,
377        primeira: Vec<u8>,
378        fluxo: &mut (dyn AsyncRead + Send + Unpin),
379    ) -> Result<(), ErroDeIngestao> {
380        let mut partes = Vec::new();
381        let mut atual = primeira;
382        loop {
383            let parte = self
384                .bucket
385                .put_multipart_chunk(
386                    atual,
387                    chave,
388                    partes.len() as u32 + 1,
389                    id_do_envio,
390                    tipo_de_conteudo,
391                )
392                .await
393                .map_err(|e| self.erro(chave, e))?;
394            partes.push(parte);
395
396            atual = ler_parte(fluxo).await.map_err(|e| self.erro(chave, e))?;
397            if atual.is_empty() {
398                break;
399            }
400        }
401
402        self.repetir(chave, "POST complete-multipart", || {
403            self.bucket
404                .complete_multipart_upload(chave, id_do_envio, partes.clone())
405        })
406        .await?;
407        Ok(())
408    }
409}

411/// Lê até completar uma parte; menos que isso só no fim do fluxo.
412async fn ler_parte(fluxo: &mut (dyn AsyncRead + Send + Unpin)) -> std::io::Result<Vec<u8>> {
413    let mut parte = Vec::with_capacity(TAMANHO_DA_PARTE);
414    fluxo
415        .take(TAMANHO_DA_PARTE as u64)
416        .read_to_end(&mut parte)
417        .await?;
418    Ok(parte)
419}

421fn eh_transitorio(erro: &S3Error) -> bool {
422    match erro {
423        S3Error::HttpFail => true,
424        S3Error::HttpFailWithBody(status, detalhe) => {
425            (500..=599).contains(status)
426                || *status == 408
427                || *status == 429
428                || detalhe
429                    .to_ascii_lowercase()
430                    .contains("end of file before message length reached")
431        }
432        S3Error::Reqwest(erro) => {
433            erro.is_timeout()
434                || erro.is_connect()
435                || erro.is_body()
436                || erro
437                    .to_string()
438                    .to_ascii_lowercase()
439                    .contains("end of file before message length reached")
440        }
441        S3Error::Io(erro) => matches!(
442            erro.kind(),
443            std::io::ErrorKind::TimedOut
444                | std::io::ErrorKind::ConnectionReset
445                | std::io::ErrorKind::ConnectionAborted
446                | std::io::ErrorKind::UnexpectedEof
447        ),
448        _ => false,
449    }
450}

452fn jitter(tentativa: u32) -> std::time::Duration {
453    static SEQUENCIA: AtomicU64 = AtomicU64::new(0x9e37_79b9_7f4a_7c15);
454    let agora = std::time::SystemTime::now()
455        .duration_since(std::time::UNIX_EPOCH)
456        .map_or(0, |duracao| duracao.subsec_nanos() as u64);
457    let valor = SEQUENCIA
458        .fetch_add(0x9e37_79b9_7f4a_7c15, Ordering::Relaxed)
459        .wrapping_add(agora)
460        .wrapping_add(u64::from(tentativa));
461    std::time::Duration::from_millis(valor % 200)
462}

521    /// Simula somente o protocolo HTTP que o SDK usa nos cenários exercitados.
522    /// Cada resposta fecha a conexão para manter o teste independente de estado.
523    fn servidor_s3(respostas: Vec<RespostaS3>) -> (String, JoinHandle<Vec<String>>) {
524        let listener = TcpListener::bind("127.0.0.1:0").expect("bind loopback");
525        listener
526            .set_nonblocking(true)
527            .expect("listener não bloqueante");
528        let endpoint = format!("http://{}", listener.local_addr().unwrap());
529        let servidor = std::thread::spawn(move || {
530            let mut requisicoes = Vec::new();
531            for resposta in respostas {
532                let prazo = Instant::now() + Duration::from_secs(10);
533                let mut stream = loop {
534                    match listener.accept() {
535                        Ok((stream, _)) => break stream,
536                        Err(erro) if erro.kind() == std::io::ErrorKind::WouldBlock => {
537                            if Instant::now() >= prazo {
538                                return requisicoes;
539                            }
540                            std::thread::sleep(Duration::from_millis(5));
541                        }
542                        Err(erro) => panic!("falha ao aceitar conexão do SDK: {erro}"),
543                    }
544                };
545                stream
546                    .set_read_timeout(Some(Duration::from_secs(10)))
547                    .unwrap();
548
549                let mut leitor = BufReader::new(stream.try_clone().unwrap());
550                let requisicao = ler_requisicao(&mut leitor, &mut stream).unwrap();
551                requisicoes.push(requisicao);
552
553                let texto_status = match resposta.status {
554                    200 => "OK",
555                    204 => "No Content",
556                    404 => "Not Found",
557                    412 => "Precondition Failed",
558                    503 => "Service Unavailable",
559                    status => panic!("status de mock não previsto: {status}"),
560                };
561                let tamanho_corpo = resposta
562                    .cabecalhos
563                    .lines()
564                    .find_map(|linha| {
565                        let (nome, valor) = linha.split_once(':')?;
566                        nome.eq_ignore_ascii_case("content-length")
567                            .then(|| valor.trim().to_owned())
568                    })
569                    .unwrap_or_else(|| resposta.corpo.len().to_string());
570                let content_length = if resposta
571                    .cabecalhos
572                    .lines()
573                    .any(|linha| linha.to_ascii_lowercase().starts_with("content-length:"))
574                {
575                    String::new()
576                } else {
577                    format!("Content-Length: {tamanho_corpo}\r\n")
578                };
579                let cabecalho = format!(
580                    "HTTP/1.1 {} {}\r\n{}Connection: close\r\n{}\r\n",
581                    resposta.status, texto_status, content_length, resposta.cabecalhos,
582                );
583                stream.write_all(cabecalho.as_bytes()).unwrap();
584                stream.write_all(resposta.corpo.as_bytes()).unwrap();
585                stream.flush().unwrap();
586            }
587            requisicoes
588        });
589
590        (endpoint, servidor)
591    }

593    /// Faz dois GET observarem a mesma versão antes de aceitar os PUTs concorrentes.
594    fn servidor_s3_cas_concorrente() -> (String, JoinHandle<Vec<String>>) {
595        let listener = TcpListener::bind("127.0.0.1:0").expect("bind loopback");
596        listener
597            .set_nonblocking(true)
598            .expect("listener não bloqueante");
599        let endpoint = format!("http://{}", listener.local_addr().unwrap());
600        let servidor = std::thread::spawn(move || {
601            fn aceitar(listener: &TcpListener) -> TcpStream {
602                let prazo = Instant::now() + Duration::from_secs(10);
603                loop {
604                    match listener.accept() {
605                        Ok((stream, _)) => return stream,
606                        Err(erro) if erro.kind() == std::io::ErrorKind::WouldBlock => {
607                            assert!(Instant::now() < prazo, "timeout aceitando requisição S3");
608                            std::thread::sleep(Duration::from_millis(5));
609                        }
610                        Err(erro) => panic!("falha ao aceitar conexão do SDK: {erro}"),
611                    }
612                }
613            }
614            fn responder(mut stream: TcpStream, status: u16, etag: &str, corpo: &str) {
615                let motivo = if status == 200 {
616                    "OK"
617                } else {
618                    "Precondition Failed"
619                };
620                let resposta = format!(
621                    "HTTP/1.1 {status} {motivo}\r\nContent-Length: {}\r\nETag: {etag}\r\nConnection: close\r\n\r\n{corpo}",
622                    corpo.len()
623                );
624                stream.write_all(resposta.as_bytes()).unwrap();
625                stream.flush().unwrap();
626            }
627
628            let mut requisicoes = Vec::new();
629            for _ in 0..2 {
630                let mut stream = aceitar(&listener);
631                let mut leitor = BufReader::new(stream.try_clone().unwrap());
632                requisicoes.push(ler_requisicao(&mut leitor, &mut stream).unwrap());
633                responder(stream, 200, "\"cas-v1\"", "ponteiro antigo");
634            }
635
636            let mut etag_atual = "\"cas-v1\"";
637            for _ in 0..2 {
638                let mut stream = aceitar(&listener);
639                let mut leitor = BufReader::new(stream.try_clone().unwrap());
640                let requisicao = ler_requisicao(&mut leitor, &mut stream).unwrap();
641                let condicao = valor_do_cabecalho(&requisicao, "if-match").map(str::to_owned);
642                requisicoes.push(requisicao);
643                if condicao.as_deref() == Some(etag_atual) {
644                    etag_atual = "\"cas-v2\"";
645                    responder(stream, 200, etag_atual, "");
646                } else {
647                    responder(stream, 412, etag_atual, "conflito");
648                }
649            }
650
651            let mut stream = aceitar(&listener);
652            let mut leitor = BufReader::new(stream.try_clone().unwrap());
653            requisicoes.push(ler_requisicao(&mut leitor, &mut stream).unwrap());
654            responder(stream, 200, etag_atual, "ponteiro vencedor");
655            requisicoes
656        });
657
658        (endpoint, servidor)
659    }

661    fn ler_requisicao(
662        leitor: &mut BufReader<TcpStream>,
663        stream: &mut TcpStream,
664    ) -> std::io::Result<String> {
665        let mut texto = String::new();
666        let mut linha = String::new();
667        leitor.read_line(&mut linha)?;
668        texto.push_str(&linha);
669        let mut tamanho = 0u64;
670        let mut chunked = false;
671        let mut espera = false;
672        loop {
673            linha.clear();
674            leitor.read_line(&mut linha)?;
675            if linha == "\r\n" || linha.is_empty() {
676                break;
677            }
678            let (nome, valor) = linha.split_once(':').unwrap_or(("", ""));
679            match nome.to_ascii_lowercase().as_str() {
680                "content-length" => tamanho = valor.trim().parse().unwrap_or(0),
681                "transfer-encoding" => chunked = valor.to_ascii_lowercase().contains("chunked"),
682                "expect" => espera = valor.trim().eq_ignore_ascii_case("100-continue"),
683                _ => {}
684            }
685            texto.push_str(&linha);
686        }
687        if espera {
688            stream.write_all(b"HTTP/1.1 100 Continue\r\n\r\n")?;
689        }
690
691        if chunked {
692            loop {
693                linha.clear();
694                leitor.read_line(&mut linha)?;
695                let tamanho_chunk =
696                    u64::from_str_radix(linha.trim().split(';').next().unwrap_or("0"), 16)
697                        .unwrap_or(0);
698                if tamanho_chunk == 0 {
699                    loop {
700                        linha.clear();
701                        leitor.read_line(&mut linha)?;
702                        if linha == "\r\n" || linha.is_empty() {
703                            break;
704                        }
705                    }
706                    break;
707                }
708                std::io::copy(
709                    &mut (&mut *leitor).take(tamanho_chunk),
710                    &mut std::io::sink(),
711                )?;
712                let mut crlf = [0; 2];
713                leitor.read_exact(&mut crlf)?;
714            }
715        } else if tamanho > 0 {
716            std::io::copy(&mut (&mut *leitor).take(tamanho), &mut std::io::sink())?;
717        }
718        Ok(texto)
719    }

721    fn valor_do_cabecalho<'a>(requisicao: &'a str, nome: &str) -> Option<&'a str> {
722        requisicao.lines().find_map(|linha| {
723            let (nome_cabecalho, valor) = linha.split_once(':')?;
724            nome_cabecalho
725                .eq_ignore_ascii_case(nome)
726                .then_some(valor.trim())
727        })
728    }
