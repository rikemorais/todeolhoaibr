10/// Conjunto anual presente no recurso Bronze.
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub enum ConjuntoCamaraVotacoes {
13    /// Cadastro das proposições.
14    Proposicoes,
15    /// Autores de proposições.
16    ProposicoesAutores,
17    /// Eventos de votação.
18    Votacoes,
19    /// Votos individuais.
20    VotacoesVotos,
21    /// Objetos possíveis das votações.
22    VotacoesObjetos,
23    /// Proposições afetadas pelas votações.
24    VotacoesProposicoes,
25    /// Orientações de bancada.
26    VotacoesOrientacoes,
27}
28
29impl ConjuntoCamaraVotacoes {
30    /// Resolve o conjunto a partir do caminho do recurso Bronze.
31    pub fn de_recurso(recurso: &str) -> Option<Self> {
32        let segmento = recurso.strip_prefix("votacoes/")?.split('/').next()?;
33        Some(match segmento {
34            "proposicoes" => Self::Proposicoes,
35            "proposicoesAutores" => Self::ProposicoesAutores,
36            "votacoes" => Self::Votacoes,
37            "votacoesVotos" => Self::VotacoesVotos,
38            "votacoesObjetos" => Self::VotacoesObjetos,
39            "votacoesProposicoes" => Self::VotacoesProposicoes,
40            "votacoesOrientacoes" => Self::VotacoesOrientacoes,
41            _ => return None,
42        })
43    }
44
45    /// Colunas da chave candidata validada para o recorte de 2024.
46    pub fn chave(self) -> &'static [&'static str] {
47        match self {
48            Self::Proposicoes | Self::Votacoes => &["id"],
49            Self::ProposicoesAutores => &["idProposicao", "ordemAssinatura"],
50            Self::VotacoesVotos => &["idVotacao", "deputado_id"],
51            Self::VotacoesObjetos | Self::VotacoesProposicoes => &["idVotacao", "proposicao_id"],
52            Self::VotacoesOrientacoes => &["idVotacao", "siglaBancada"],
53        }
54    }
55
56    /// Segmento usado na versão da regra de qualidade.
57    pub fn nome(self) -> &'static str {
58        match self {
59            Self::Proposicoes => "proposicoes",
60            Self::ProposicoesAutores => "proposicoes-autores",
61            Self::Votacoes => "votacoes",
62            Self::VotacoesVotos => "votacoes-votos",
63            Self::VotacoesObjetos => "votacoes-objetos",
64            Self::VotacoesProposicoes => "votacoes-proposicoes",
65            Self::VotacoesOrientacoes => "votacoes-orientacoes",
66        }
67    }
68}

70/// Erro de validação de um lote Bronze da Câmara.
71#[derive(Debug, Error, PartialEq, Eq)]
72pub enum ErroCamaraVotacoesSilver {
73    /// Coluna de chave ausente ou tipo Bronze diferente de texto.
74    #[error("schema Bronze incompatível com o conjunto {conjunto}")]
75    Schema {
76        /// Nome do conjunto anual.
77        conjunto: String,
78    },
79    /// A chave está nula ou vazia.
80    #[error("chave nula ou vazia na linha {linha} do conjunto {conjunto}")]
81    ChaveInvalida {
82        /// Nome do conjunto anual.
83        conjunto: String,
84        /// Índice da linha no lote, começando em zero.
85        linha: usize,
86    },
87    /// A chave se repetiu no mesmo recurso anual.
88    #[error("chave duplicada na linha {linha} do conjunto {conjunto}")]
89    ChaveDuplicada {
90        /// Nome do conjunto anual.
91        conjunto: String,
92        /// Índice da linha no lote, começando em zero.
93        linha: usize,
94    },
95}
96
97/// Valida as chaves sem colapsar eventos nem descartar atributos da origem.
98pub fn validar_camara_votacoes(
99    conjunto: ConjuntoCamaraVotacoes,
100    lote: &RecordBatch,
101    chaves_vistas: &mut HashSet<Vec<String>>,
102) -> Result<RecordBatch, ErroCamaraVotacoesSilver> {
103    let nome = conjunto.nome().to_owned();
104    let indices = conjunto
105        .chave()
106        .iter()
107        .map(|campo| {
108            let indice =
109                lote.schema()
110                    .index_of(campo)
111                    .map_err(|_| ErroCamaraVotacoesSilver::Schema {
112                        conjunto: nome.clone(),
113                    })?;
114            if lote.schema().field(indice).data_type() != &DataType::Utf8 {
115                return Err(ErroCamaraVotacoesSilver::Schema {
116                    conjunto: nome.clone(),
117                });
118            }
119            lote.column(indice)
120                .as_any()
121                .downcast_ref::<StringArray>()
122                .map(|_| indice)
123                .ok_or_else(|| ErroCamaraVotacoesSilver::Schema {
124                    conjunto: nome.clone(),
125                })
126        })
127        .collect::<Result<Vec<_>, _>>()?;
128
129    for linha in 0..lote.num_rows() {
130        let mut chave = Vec::with_capacity(indices.len());
131        for &indice in &indices {
132            let valores = lote
133                .column(indice)
134                .as_any()
135                .downcast_ref::<StringArray>()
136                .expect("column type checked above");
137            if valores.is_null(linha) || valores.value(linha).trim().is_empty() {
138                return Err(ErroCamaraVotacoesSilver::ChaveInvalida {
139                    conjunto: nome,
140                    linha,
141                });
142            }
143            chave.push(valores.value(linha).to_owned());
144        }
145        if !chaves_vistas.insert(chave) {
146            return Err(ErroCamaraVotacoesSilver::ChaveDuplicada {
147                conjunto: nome,
148                linha,
149            });
150        }
151    }
152    Ok(lote.clone())
153}

155#[cfg(test)]
156mod testes {
157    use super::*;
158    use arrow::array::StringArray;
159    use std::sync::Arc;
160
161    #[test]
162    fn resolve_todos_os_sete_conjuntos_e_suas_chaves() {
163        let esperados = [
164            ("proposicoes", &["id"][..]),
165            (
166                "proposicoesAutores",
167                &["idProposicao", "ordemAssinatura"][..],
168            ),
169            ("votacoes", &["id"][..]),
170            ("votacoesVotos", &["idVotacao", "deputado_id"][..]),
171            ("votacoesObjetos", &["idVotacao", "proposicao_id"][..]),
172            ("votacoesProposicoes", &["idVotacao", "proposicao_id"][..]),
173            ("votacoesOrientacoes", &["idVotacao", "siglaBancada"][..]),
174        ];
175        for (nome, chave) in esperados {
176            let conjunto =
177                ConjuntoCamaraVotacoes::de_recurso(&format!("votacoes/{nome}/ano=2024")).unwrap();
178            assert_eq!(conjunto.chave(), chave);
179        }
180        assert!(ConjuntoCamaraVotacoes::de_recurso("votacoes/desconhecido/ano=2024").is_none());
181    }
182
183    #[test]
184    fn preserva_linhas_e_detecta_repeticao_entre_lotes() {
185        let lote = RecordBatch::try_from_iter(vec![
186            ("idVotacao", Arc::new(StringArray::from(vec!["V1"])) as _),
187            ("deputado_id", Arc::new(StringArray::from(vec!["D1"])) as _),
188            ("tipoVoto", Arc::new(StringArray::from(vec!["Sim"])) as _),
189        ])
190        .unwrap();
191        let conjunto = ConjuntoCamaraVotacoes::VotacoesVotos;
192        let mut vistos = HashSet::new();
193        let saida = validar_camara_votacoes(conjunto, &lote, &mut vistos).unwrap();
194        assert_eq!(saida, lote);
195        assert_eq!(
196            validar_camara_votacoes(conjunto, &lote, &mut vistos),
197            Err(ErroCamaraVotacoesSilver::ChaveDuplicada {
198                conjunto: "votacoes-votos".into(),
199                linha: 0,
200            })
201        );
202    }
