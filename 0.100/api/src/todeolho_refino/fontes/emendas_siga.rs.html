10const NOME_TABELA: &str = "favorecidos-pagamento";
11const LINHAS_POR_LOTE: usize = 10_000;
12const COLUNAS: [&str; 8] = [
13    "Favorecido do Pagamento - Município/UF",
14    "Favorecido do Pagamento (CPF/CNPJ)",
15    "Favorecido do Pagamento (Natureza Subgrupo)",
16    "Favorecido do Pagamento (Tipo)",
17    "Favorecido do Pagamento (UF)",
18    "Autor",
19    "Emenda (Número/Ano)",
20    "Pago + RP (Favorecido lista OB)",
21];

23/// Converte o XLSX do SIGA em Parquet preservando o cabeçalho e as células como texto.
24pub struct DecodificadorEmendasSiga;
25
26impl Decodificador for DecodificadorEmendasSiga {
27    fn decodificar(
28        &self,
29        bytes: &[u8],
30        _tipo_de_conteudo: &str,
31    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
32        let workbook = open_workbook_from_rs(Cursor::new(bytes)).map_err(|erro| {
33            ErroDeRefino::Decodificacao(format!("XLSX inválido do SIGA Brasil: {erro}"))
34        })?;
35        converter(workbook)
36    }
37
38    fn decodificar_arquivo(
39        &self,
40        caminho: &std::path::Path,
41        _tipo_de_conteudo: &str,
42    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
43        let arquivo = std::fs::File::open(caminho)?;
44        let workbook = open_workbook_from_rs(arquivo).map_err(|erro| {
45            ErroDeRefino::Decodificacao(format!("XLSX inválido do SIGA Brasil: {erro}"))
46        })?;
47        converter(workbook)
48    }
49}

51fn converter<R: Read + Seek>(mut workbook: Xlsx<R>) -> Result<Vec<ParteBronze>, ErroDeRefino> {
52    let planilha = workbook
53        .worksheet_range_at(0)
54        .ok_or_else(|| ErroDeRefino::Decodificacao("XLSX do SIGA sem planilhas".into()))?
55        .map_err(|erro| {
56            ErroDeRefino::Decodificacao(format!("não foi possível ler a planilha SIGA: {erro}"))
57        })?;
58    let mut linhas = planilha.rows();
59    let cabecalho = linhas
60        .next()
61        .ok_or_else(|| ErroDeRefino::Decodificacao("planilha SIGA vazia".into()))?;
62    let colunas = cabecalho
63        .iter()
64        .map(ToString::to_string)
65        .collect::<Vec<_>>();
66    if colunas.iter().map(String::as_str).ne(COLUNAS) {
67        return Err(ErroDeRefino::Decodificacao(format!(
68            "schema XLSX do SIGA mudou; esperado: {}",
69            COLUNAS.join(" | ")
70        )));
71    }
72
73    let temporario = tempfile::NamedTempFile::new()?;
74    let mut escritor = EscritorParquet::novo_em(temporario.reopen()?, &colunas)?;
75    let mut lote = Vec::with_capacity(LINHAS_POR_LOTE);
76    for linha in linhas {
77        if linha.iter().all(|celula| matches!(celula, Data::Empty)) {
78            continue;
79        }
80        let valores = (0..COLUNAS.len())
81            .map(|indice| {
82                linha
83                    .get(indice)
84                    .filter(|celula| !matches!(celula, Data::Empty))
85                    .map(ToString::to_string)
86            })
87            .collect::<Vec<_>>();
88        lote.push(valores);
89        if lote.len() == LINHAS_POR_LOTE {
90            escritor.escrever_lote(&lote)?;
91            lote.clear();
92        }
93    }
94    escritor.escrever_lote(&lote)?;
95    escritor.fechar()?;
96
97    Ok(vec![ParteBronze::de_arquivo(
98        NOME_TABELA,
99        temporario.into_temp_path(),
100    )])
101}

112    fn xlsx(headers: &[&str], rows: &[&[&str]]) -> Vec<u8> {
113        let mut bytes = Cursor::new(Vec::new());
114        {
115            let mut zip = ZipWriter::new(&mut bytes);
116            let options = SimpleFileOptions::default();
117            for (name, xml) in [
118                (
119                    "[Content_Types].xml",
120                    r#"<?xml version="1.0" encoding="UTF-8"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/><Default Extension="xml" ContentType="application/xml"/><Override PartName="/xl/workbook.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml"/><Override PartName="/xl/worksheets/sheet1.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml"/></Types>"#,
121                ),
122                (
123                    "_rels/.rels",
124                    r#"<?xml version="1.0" encoding="UTF-8"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="xl/workbook.xml"/></Relationships>"#,
125                ),
126                (
127                    "xl/workbook.xml",
128                    r#"<?xml version="1.0" encoding="UTF-8"?><workbook xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"><sheets><sheet name="Dados" sheetId="1" r:id="rId1"/></sheets></workbook>"#,
129                ),
130                (
131                    "xl/_rels/workbook.xml.rels",
132                    r#"<?xml version="1.0" encoding="UTF-8"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet" Target="worksheets/sheet1.xml"/></Relationships>"#,
133                ),
134            ] {
135                zip.start_file(name, options).unwrap();
136                zip.write_all(xml.as_bytes()).unwrap();
137            }
138            zip.start_file("xl/worksheets/sheet1.xml", options).unwrap();
139            let mut xml = String::from(
140                r#"<?xml version="1.0" encoding="UTF-8"?><worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main"><sheetData>"#,
141            );
142            for (linha, valores) in std::iter::once(headers)
143                .chain(rows.iter().copied())
144                .enumerate()
145            {
146                xml.push_str(&format!(r#"<row r="{}">"#, linha + 1));
147                for (indice, valor) in valores.iter().enumerate() {
148                    let mut coluna = indice + 1;
149                    let mut letra = String::new();
150                    while coluna > 0 {
151                        let resto = (coluna - 1) % 26;
152                        letra.insert(0, char::from(b'A' + resto as u8));
153                        coluna = (coluna - 1) / 26;
154                    }
155                    xml.push_str(&format!(
156                        r#"<c r="{letra}{}" t="inlineStr"><is><t>{valor}</t></is></c>"#,
157                        linha + 1
158                    ));
159                }
160                xml.push_str("</row>");
161            }
162            xml.push_str("</sheetData></worksheet>");
163            zip.write_all(xml.as_bytes()).unwrap();
164            zip.finish().unwrap();
165        }
166        bytes.into_inner()
167    }

169    #[test]
170    fn preserva_o_schema_e_os_valores_do_xlsx_em_parquet() {
171        let valores = [
172            "MUNICIPIO DE BONFIM-BONFIM-RR",
173            "DADO_PESSOAL_TESTE",
174            "ÓRGÃO PÚBLICO MUNICIPAL",
175            "PESSOA JURIDICA",
176            "RR",
177            "AUTOR DE TESTE",
178            "30300001-2016",
179            "2157073",
180        ];
181        let refs = COLUNAS.to_vec();
182        let bytes = xlsx(&refs, &[&valores]);
183
184        let partes = DecodificadorEmendasSiga
185            .decodificar(
186                &bytes,
187                "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
188            )
189            .unwrap();
190        let (colunas, linhas) = ler_parte(&partes[0]);
191
192        assert_eq!(partes[0].nome, NOME_TABELA);
193        assert_eq!(colunas, COLUNAS);
194        assert_eq!(linhas.len(), 1);
195        assert_eq!(linhas[0][0].as_deref(), Some(valores[0]));
196        assert_eq!(linhas[0][1].as_deref(), Some(valores[1]));
197        assert_eq!(linhas[0][7].as_deref(), Some(valores[7]));
198    }
