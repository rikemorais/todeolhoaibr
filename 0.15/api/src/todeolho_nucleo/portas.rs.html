19/// Destino dos dados brutos.
20#[async_trait]
21pub trait ArmazenamentoRaw: Send + Sync {
22    /// Garante que o destino existe. Precisa ser idempotente.
23    async fn preparar(&self) -> Result<(), ErroDeIngestao>;
24
25    /// Grava um objeto por inteiro na camada Raw.
26    async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao>;
27
28    /// Grava sem carregar tudo na memória. O objeto só fica visível ao final da transferência.
29    async fn gravar_fluxo(
30        &self,
31        chave: &str,
32        tipo_de_conteudo: &str,
33        fluxo: &mut (dyn AsyncRead + Send + Unpin),
34    ) -> Result<(), ErroDeIngestao>;
35
36    /// Remove um objeto publicado na camada Raw.
37    async fn remover(&self, chave: &str) -> Result<(), ErroDeIngestao> {
38        Err(ErroDeIngestao::Armazenamento {
39            chave: chave.to_owned(),
40            detalhe: "remoção não implementada pelo adaptador".into(),
41        })
42    }
43
44    /// Apaga transferências que nunca terminaram, como as de um processo interrompido,
45    /// para que não ocupem disco. Uma ingestão em andamento perde a transferência atual e
46    /// a recomeça, então nenhum dado se perde.
47    async fn descartar_envios_incompletos(&self) -> Result<usize, ErroDeIngestao> {
48        Ok(0)
49    }
50
51    /// Conteúdo do objeto, ou `None` se ele não existir.
52    async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao>;
53
54    /// Tamanho do objeto em bytes, sem baixá-lo (um `HEAD`), ou `None` se não
55    /// existir. A implementação padrão lê o objeto; as de infraestrutura devem
56    /// sobrescrever com um `HEAD` para não carregar arquivos grandes na memória.
57    async fn tamanho(&self, chave: &str) -> Result<Option<u64>, ErroDeIngestao> {
58        Ok(self.ler(chave).await?.map(|conteudo| conteudo.len() as u64))
59    }
60}

71/// Resultado de um GET condicional.
72#[derive(Debug, Clone)]
73pub enum RespostaCondicional {
74    /// 304: o conteúdo não mudou desde o validador enviado.
75    NaoModificado,
76    /// A origem devolveu um corpo novo.
77    Modificado {
78        /// Tipo de conteúdo (`Content-Type`) do novo corpo.
79        tipo_de_conteudo: String,
80        /// Validador do novo conteúdo (ETag), quando informado.
81        versao: Option<String>,
82        /// Corpo novo.
83        corpo: Vec<u8>,
84    },
85}

99/// Resposta bem-sucedida cujo corpo ainda será lido aos poucos.
100pub struct DownloadHttp {
101    /// Tipo de conteúdo (`Content-Type`) do arquivo.
102    pub tipo_de_conteudo: String,
103    /// Identifica a versão do arquivo na origem (ETag), quando informada.
104    pub versao: Option<String>,
105    /// Deslocamento, em bytes, onde este fluxo começa no arquivo. `0` num
106    /// download do início; numa retomada (`206 Partial Content`) é o início do
107    /// `Content-Range`. Quem retoma compara com o offset pedido: se não bater, a
108    /// origem não honrou a retomada e o download deve recomeçar do zero.
109    pub inicio: u64,
110    /// Tamanho total do arquivo em bytes, quando a origem informa
111    /// (`Content-Length` de um `200`, ou o total do `Content-Range` de um `206`).
112    pub tamanho_total: Option<u64>,
113    /// Fluxo do corpo, lido sob demanda.
114    pub fluxo: FluxoDeBytes,
115}

117/// Acesso às APIs públicas. Falhas transitórias são tratadas pela implementação.
118#[async_trait]
119pub trait ClienteHttp: Send + Sync {
120    /// GET simples: devolve o corpo inteiro em memória.
121    async fn obter(&self, url: &str, aceita: &str) -> Result<RespostaHttp, ErroDeIngestao>;
122
123    /// GET condicional: envia o validador (If-None-Match) e devolve
124    /// `NaoModificado` em caso de 304. A implementação padrão ignora o validador
125    /// e sempre baixa, para fontes que não suportam requisição condicional.
126    async fn obter_condicional(
127        &self,
128        url: &str,
129        aceita: &str,
130        _validador: Option<&str>,
131    ) -> Result<RespostaCondicional, ErroDeIngestao> {
132        let resposta = self.obter(url, aceita).await?;
133        Ok(RespostaCondicional::Modificado {
134            tipo_de_conteudo: resposta.tipo_de_conteudo,
135            versao: None,
136            corpo: resposta.corpo,
137        })
138    }
139
140    /// Consulta o arquivo na origem, sem baixá-lo.
141    async fn informacoes(&self, url: &str) -> Result<InformacoesDoArquivo, ErroDeIngestao>;
142
143    /// Baixa um arquivo como fluxo, para gravá-lo sem carregá-lo na memória.
144    async fn baixar(&self, url: &str) -> Result<DownloadHttp, ErroDeIngestao>;
145
146    /// Baixa a partir de `inicio` (em bytes; `0` é do começo), para **retomar**
147    /// um download interrompido sem perder nem duplicar byte. `versao` é o
148    /// validador (ETag) da primeira resposta: a implementação deve enviar
149    /// `If-Range`, de modo a só retomar se o arquivo não mudou. O
150    /// [`DownloadHttp::inicio`] devolvido diz onde o fluxo realmente começa — se
151    /// não for igual a `inicio`, a origem não honrou a retomada e o chamador deve
152    /// recomeçar do zero em vez de emendar.
153    ///
154    /// A implementação padrão ignora `inicio`/`versao` e baixa do começo, para
155    /// clientes que não suportam `Range`; quem retoma detecta o `inicio` devolvido
156    /// (`0`) diferente do pedido e recomeça.
157    async fn baixar_a_partir_de(
158        &self,
159        url: &str,
160        inicio: u64,
161        versao: Option<&str>,
162    ) -> Result<DownloadHttp, ErroDeIngestao> {
163        let _ = (inicio, versao);
164        self.baixar(url).await
165    }
