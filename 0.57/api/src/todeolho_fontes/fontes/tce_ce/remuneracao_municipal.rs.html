1//! Totais agregados das folhas municipais enviados ao SIM do TCE-CE.
2//!
3//! A fonte ingere somente `/folhas_pagamentos`, cujo contrato contém totais por
4//! folha/unidade e não os campos nominais de `/agentes_publicos_folha`.
5
6use std::time::Duration;
7
8use serde::Deserialize;
9use todeolho_nucleo::{FonteDeDados, LimiteHttp, PerfilHttp, TarefaPlanejada};
10
11use crate::ingestao_http::TarefaDeIngestaoHttp;
12use crate::paginacao::EstrategiaDePaginacao;
13
14const NOME: &str = "remuneracao-municipal-tce-ce";
15const BASE: &str = "https://api-dados-abertos.tce.ce.gov.br/sim/folhas_pagamentos";
16const COMPETENCIA: &str = "2025-12";
17const REFERENCIA: u32 = 202512;
18const EXERCICIO: u32 = 202500;
19const TAMANHO_PAGINA: usize = 1000;
20const PRIMEIRO_CODIGO_SIM: u16 = 2;
21const ULTIMO_CODIGO_SIM: u16 = 185;
22
23#[derive(Deserialize)]
24struct EnvelopeFolhas {
25    #[serde(default)]
26    links: Vec<Link>,
27}
28
29#[derive(Deserialize)]
30struct Link {
31    rel: String,
32    href: String,
33}
34
35/// Paginação por `links.next`, resolvendo a URL relativa documentada pela API.
36struct PaginacaoFolhas;
37
38impl EstrategiaDePaginacao for PaginacaoFolhas {
39    fn proxima_url(&self, corpo: &[u8]) -> Result<Option<String>, String> {
40        let envelope: EnvelopeFolhas = serde_json::from_slice(corpo)
41            .map_err(|erro| format!("resposta de folhas inválida: {erro}"))?;
42        envelope
43            .links
44            .into_iter()
45            .find(|link| link.rel == "next")
46            .map(|link| normalizar_link(&link.href))
47            .transpose()
48    }
49}
50
51fn url_inicial(codigo_municipio: u16) -> String {
52    format!(
53        "{BASE}?codigo_municipio={codigo_municipio:03}&exercicio_orcamento={EXERCICIO}&data_referencia_doc={REFERENCIA}&$format=json&$count={TAMANHO_PAGINA}&$start_index=0"
54    )
55}
56
57fn normalizar_link(href: &str) -> Result<String, String> {
58    if href.starts_with('?') {
59        return Ok(format!("{BASE}{href}"));
60    }
61    if href.starts_with(&format!("{BASE}?")) {
62        return Ok(href.to_owned());
63    }
64    Err(format!(
65        "link de paginação fora do endpoint esperado: {href}"
66    ))
67}

69/// Ingestão dos totais agregados do TCE-CE para os 184 municípios com código
70/// IBGE, na competência 2025-12. Não consulta o endpoint nominal de agentes.
71pub struct FonteRemuneracaoMunicipalTceCe;
72
73impl FonteDeDados for FonteRemuneracaoMunicipalTceCe {
74    fn nome(&self) -> &str {
75        NOME
76    }
77
78    fn descricao(&self) -> &str {
79        "TCE-CE - totais agregados de folhas municipais (competência 2025-12)"
80    }
81
82    fn sob_demanda(&self) -> bool {
83        true
84    }
85
86    fn perfil_http(&self) -> PerfilHttp {
87        PerfilHttp {
88            credencial: None,
89            limite: Some(LimiteHttp {
90                host: "api-dados-abertos.tce.ce.gov.br",
91                requisicoes: 5,
92                intervalo: Duration::from_secs(1),
93            }),
94        }
95    }
96
97    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
98        (PRIMEIRO_CODIGO_SIM..=ULTIMO_CODIGO_SIM)
99            .map(|codigo| {
100                let municipio = format!("{codigo:03}");
101                let recurso =
102                    format!("folhas-pagamentos/municipio={municipio}/competencia={COMPETENCIA}");
103                let tarefa = TarefaDeIngestaoHttp::paginada(
104                    NOME,
105                    &recurso,
106                    url_inicial(codigo),
107                    PaginacaoFolhas,
108                )
109                .com_tamanho_tipico(250_000)
110                .com_limite_de_paginas(20_000);
111
112                TarefaPlanejada::nova(tarefa).com_etiquetas([
113                    ("municipio_sim", municipio.as_str()),
114                    ("competencia", COMPETENCIA),
115                ])
116            })
117            .collect()
118    }
119}

121#[cfg(test)]
122mod testes {
123    use std::collections::BTreeSet;
124
125    use super::*;
126
127    #[test]
128    fn cria_tarefa_particionada_para_cada_municipio_validado() {
129        let tarefas = FonteRemuneracaoMunicipalTceCe.criar_tarefas();
130        assert_eq!(tarefas.len(), 184);
131
132        let codigos: BTreeSet<_> = tarefas
133            .iter()
134            .map(|tarefa| tarefa.etiquetas["municipio_sim"].as_str())
135            .collect();
136        assert_eq!(codigos.len(), 184);
137        assert!(codigos.contains("057"));
138        assert!(codigos.contains("185"));
139        assert!(
140            tarefas
141                .iter()
142                .all(|tarefa| tarefa.etiquetas["competencia"] == COMPETENCIA)
143        );
144    }
145
146    #[test]
147    fn url_consulta_a_folha_agregada_do_municipio_e_competencia() {
148        assert_eq!(
149            url_inicial(57),
150            "https://api-dados-abertos.tce.ce.gov.br/sim/folhas_pagamentos?codigo_municipio=057&exercicio_orcamento=202500&data_referencia_doc=202512&$format=json&$count=1000&$start_index=0"
151        );
152    }
153
154    #[test]
155    fn segue_link_relativo_e_rejeita_host_inesperado() {
156        let estrategia = PaginacaoFolhas;
157        assert_eq!(
158            estrategia
159                .proxima_url(br#"{"elements":[],"links":[{"rel":"next","href":"?$start_index=1000&codigo_municipio=057"}]}"#)
160                .unwrap()
161                .as_deref(),
162            Some("https://api-dados-abertos.tce.ce.gov.br/sim/folhas_pagamentos?$start_index=1000&codigo_municipio=057")
163        );
164        assert!(normalizar_link("https://example.com/steal").is_err());
165        assert_eq!(
166            estrategia
167                .proxima_url(br#"{"elements":[],"links":[]}"#)
168                .unwrap(),
169            None
170        );
171    }
172
173    #[test]
174    fn fonte_e_sob_demanda_para_nao_entrar_na_ingestao_completa() {
175        assert!(FonteRemuneracaoMunicipalTceCe.sob_demanda());
176        assert_eq!(
177            FonteRemuneracaoMunicipalTceCe
178                .perfil_http()
179                .limite
180                .unwrap()
181                .requisicoes,
182            5
183        );
184    }
