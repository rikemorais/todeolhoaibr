1//! Contrato Silver para linhas dos agregados da API do IBGE.

3use std::sync::Arc;
4
5use arrow::array::{Array, Decimal128Array, RecordBatch, StringArray};
6use arrow::datatypes::{DataType, Field, Schema};
7use thiserror::Error;

9const COLUNAS_TEXTO: [&str; 10] = [
10    "variavel_id",
11    "variavel",
12    "unidade",
13    "classificacoes",
14    "nivel_id",
15    "nivel_nome",
16    "localidade_id",
17    "localidade_nome",
18    "periodo",
19    "valor",
20];
21
22/// Erros encontrados ao validar o schema Bronze dos agregados IBGE.
23#[derive(Debug, Error, PartialEq, Eq)]
24pub enum ErroIbgeSilver {
25    /// O lote não contém as colunas textuais esperadas pelo contrato.
26    #[error(
27        "schema Bronze IBGE incompatível: esperadas dez colunas Utf8 do decodificador de agregados"
28    )]
29    Schema,
30}
31
32/// Schema versionado do contrato `ibge.agregado.observacao.v1`.
33///
34/// Os identificadores, classificações e períodos permanecem texto. O valor
35/// original também é preservado; `valor_decimal` é nulo quando o conteúdo não
36/// é um decimal simples com ponto e até nove casas.
37pub fn esquema_ibge_agregados() -> Arc<Schema> {
38    let mut campos: Vec<Field> = COLUNAS_TEXTO[..9]
39        .iter()
40        .map(|nome| Field::new(*nome, DataType::Utf8, true))
41        .collect();
42    campos.push(Field::new("valor_original", DataType::Utf8, true));
43    campos.push(Field::new(
44        "valor_decimal",
45        DataType::Decimal128(38, 9),
46        true,
47    ));
48    Arc::new(Schema::new(campos))
49}
50
51/// Converte dimensões textuais do Bronze e adiciona a representação decimal opcional.
52///
53/// O texto original nunca é descartado. Códigos de situação ou valores fora do
54/// formato decimal são mantidos em `valor_original` e recebem decimal nulo;
55/// interpretar esses códigos depende de metadados específicos de cada tabela.
56pub fn transformar_ibge_agregados(lote: &RecordBatch) -> Result<RecordBatch, ErroIbgeSilver> {
57    let schema = lote.schema();
58    let indices: Vec<usize> = COLUNAS_TEXTO
59        .iter()
60        .map(|nome| schema.index_of(nome).map_err(|_| ErroIbgeSilver::Schema))
61        .collect::<Result<_, _>>()?;
62    if indices
63        .iter()
64        .any(|indice| schema.field(*indice).data_type() != &DataType::Utf8)
65    {
66        return Err(ErroIbgeSilver::Schema);
67    }
68
69    let arrays: Vec<&StringArray> = indices
70        .iter()
71        .map(|indice| {
72            lote.column(*indice)
73                .as_any()
74                .downcast_ref::<StringArray>()
75                .ok_or(ErroIbgeSilver::Schema)
76        })
77        .collect::<Result<_, _>>()?;
78    let valores = arrays[9];
79    let decimais: Vec<Option<i128>> = (0..lote.num_rows())
80        .map(|linha| {
81            (!valores.is_null(linha))
82                .then(|| decimal_escalado(valores.value(linha)))
83                .flatten()
84        })
85        .collect();
86
87    let mut saida: Vec<Arc<dyn Array>> = arrays[..9]
88        .iter()
89        .map(|array| Arc::new((*array).clone()) as Arc<dyn Array>)
90        .collect();
91    saida.push(Arc::new(valores.clone()));
92    saida.push(Arc::new(
93        Decimal128Array::from(decimais)
94            .with_precision_and_scale(38, 9)
95            .map_err(|_| ErroIbgeSilver::Schema)?,
96    ));
97    RecordBatch::try_new(esquema_ibge_agregados(), saida).map_err(|_| ErroIbgeSilver::Schema)
98}

100fn decimal_escalado(texto: &str) -> Option<i128> {
101    let (negativo, numero) = texto
102        .strip_prefix('-')
103        .map_or((false, texto), |resto| (true, resto));
104    let (inteiro, fracao) = numero.split_once('.').unwrap_or((numero, ""));
105    if inteiro.is_empty()
106        || !inteiro.bytes().all(|byte| byte.is_ascii_digit())
107        || !fracao.bytes().all(|byte| byte.is_ascii_digit())
108        || inteiro.len() > 29
109        || fracao.len() > 9
110    {
111        return None;
112    }
113    let escala = format!("{fracao:0<9}");
114    let magnitude = format!("{inteiro}{escala}").parse::<i128>().ok()?;
115    Some(if negativo { -magnitude } else { magnitude })
116}
