1//! Decodificador do bronze para fontes que gravam um **CSV "solto"** no raw (não zipado).
2//!
3//! Serve, por exemplo, para as despesas do Senado (CEAPS, Latin-1, com uma linha de
4//! preâmbulo antes do cabeçalho) e para as votações da Câmara (UTF-8 com BOM). O bronze só
5//! reencoda para UTF-8 e põe em colunas — a interpretação dos campos é do silver.
6
7use crate::csv_zip::{partes_de_csv, partes_de_csv_em_fluxo};
8use crate::decodificador::{Decodificador, ParteBronze};
9use crate::escritor::ErroDeRefino;
10
11/// Decodifica um CSV solto em uma parte Parquet, com o nome lógico e o separador da fonte.
12///
13/// Quando chamado via `decodificar_arquivo`, opera em fluxo (lê do disco sem carregar
14/// o CSV inteiro em memória). Ver #164.
15pub struct DecodificadorCsv {
16    nome: &'static str,
17    separador: u8,
18    linhas_a_pular: usize,
19}
20
21impl DecodificadorCsv {
22    /// Cria o decodificador. `nome` é o nome lógico da tabela registrado na procedência.
23    pub const fn novo(nome: &'static str, separador: u8, linhas_a_pular: usize) -> Self {
24        Self {
25            nome,
26            separador,
27            linhas_a_pular,
28        }
29    }
30}
31
32impl Decodificador for DecodificadorCsv {
33    fn decodificar(
34        &self,
35        bytes: &[u8],
36        _tipo_de_conteudo: &str,
37    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
38        partes_de_csv(bytes, self.nome, self.separador, self.linhas_a_pular)
39    }
40
41    fn decodificar_arquivo(
42        &self,
43        caminho: &std::path::Path,
44        _tipo_de_conteudo: &str,
45    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
46        let arquivo = std::fs::File::open(caminho)?;
47        partes_de_csv_em_fluxo(arquivo, self.nome, self.separador, self.linhas_a_pular)
48    }
49}

51#[cfg(test)]
52mod testes {
53    use super::*;
54    use crate::csv_zip::testes::ler_parte;
55
56    fn c(s: &str) -> Option<String> {
57        Some(s.to_owned())
58    }
59
60    #[test]
61    fn decodifica_csv_solto_pulando_o_preambulo() {
62        // Como o CEAPS do Senado: uma linha de preâmbulo antes do cabeçalho real.
63        let csv = "\"ULTIMA ATUALIZACAO\";\"hoje\"\nano;valor\n2014;52,38\n";
64        let partes = DecodificadorCsv::novo("despesas", b';', 1)
65            .decodificar(csv.as_bytes(), "application/octet-stream")
66            .unwrap();
67        assert_eq!(partes.len(), 1);
68        assert_eq!(partes[0].nome, "despesas");
69        let (colunas, linhas) = ler_parte(&partes[0]);
70        assert_eq!(colunas, vec!["ano", "valor"]);
71        assert_eq!(linhas, vec![vec![c("2014"), c("52,38")]]);
72    }
73
74    #[test]
75    fn remove_o_bom_de_utf8_da_primeira_coluna() {
76        // Como as votações da Câmara: UTF-8 com BOM no início.
77        let mut csv = vec![0xEF, 0xBB, 0xBF];
78        csv.extend_from_slice("idVotacao;orientacao\n1;Sim\n".as_bytes());
79        let partes = DecodificadorCsv::novo("votacoes", b';', 0)
80            .decodificar(&csv, "application/octet-stream")
81            .unwrap();
82        let (colunas, linhas) = ler_parte(&partes[0]);
83        assert_eq!(colunas, vec!["idVotacao", "orientacao"]);
84        assert_eq!(linhas, vec![vec![c("1"), c("Sim")]]);
85    }
