1//! O contrato [`FonteDeDados`]: cada API pública ingerida implementa este trait,
2//! que decide quais tarefas existem; montar e executar o grafo é igual para todas.
3
4use crate::dag::DagDeIngestao;
5use crate::erros::ErroDeDag;
6use crate::tarefa::TarefaPlanejada;
7
8/// Credencial HTTP declarada sem expor o valor do segredo ao domínio.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub struct CredencialHttp {
11    /// Host autorizado a receber a credencial, sem esquema ou porta.
12    pub host: &'static str,
13    /// Variável de ambiente que contém o segredo.
14    pub variavel: &'static str,
15    /// Nome do cabeçalho enviado à API.
16    pub cabecalho: &'static str,
17    /// Prefixo antes do segredo, por exemplo `Bearer `; vazio para chave direta.
18    pub prefixo: &'static str,
19}
20
21/// Máximo de requisições permitidas para um host em um intervalo.
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub struct LimiteHttp {
24    /// Host ao qual o limite se aplica, sem esquema ou porta.
25    pub host: &'static str,
26    /// Número máximo de requisições no intervalo.
27    pub requisicoes: u32,
28    /// Duração do intervalo.
29    pub intervalo: std::time::Duration,
30}
31
32/// Requisitos HTTP de uma fonte. O padrão é acesso anônimo sem limite.
33#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
34pub struct PerfilHttp {
35    /// Credencial opcional da fonte.
36    pub credencial: Option<CredencialHttp>,
37    /// Limite opcional da API.
38    pub limite: Option<LimiteHttp>,
39    /// User-Agent alternativo; quando `Some`, sobrescreve o padrão do projeto.
40    pub agente: Option<&'static str>,
41}
42
43/// Uma API pública ingerida pelo projeto (o "criador" do Factory Method).
44///
45/// Cada fonte concreta só decide *quais* tarefas existem, em [`criar_tarefas`].
46/// Validar o grafo e executá-lo é igual para todas, então novas fontes entram sem
47/// alterar o executor. Ver `docs/adr/0004`.
48///
49/// [`criar_tarefas`]: FonteDeDados::criar_tarefas
50pub trait FonteDeDados: Send + Sync {
51    /// Identificador estável: é usado na CLI e no caminho dos dados na camada Raw.
52    fn nome(&self) -> &str;
53
54    /// Texto curto que descreve a fonte, exibido na CLI e no menu de seleção.
55    fn descricao(&self) -> &str;
56
57    /// Fontes muito grandes só rodam quando pedidas pelo nome, nunca em `--todas`.
58    fn sob_demanda(&self) -> bool {
59        false
60    }
61
62    /// Declara autenticação e limite de taxa sem carregar o segredo no domínio.
63    fn perfil_http(&self) -> PerfilHttp {
64        PerfilHttp::default()
65    }
66
67    /// Factory Method: cria as tarefas concretas desta fonte.
68    fn criar_tarefas(&self) -> Vec<TarefaPlanejada>;
69
70    /// Operação comum do criador: monta a DAG validada a partir do Factory Method.
71    fn montar_dag(&self) -> Result<DagDeIngestao, ErroDeDag> {
72        DagDeIngestao::construir(self.nome(), self.criar_tarefas())
73    }
74}
