1//! Recuperação de partes Raw órfãs: objetos gravados durante ingestões que não
2//! concluíram (erro ambíguo no manifesto, cancelamento abrupto, ou rollback
3//! incompleto). A limpeza só remove objetos comprovadamente desprotegidos — sem
4//! manifesto — e nunca atinge partes de ingestões vigentes.
5//!
6//! Ver issue #269 e PR #278 (revisão).
7

17const CONCORRENCIA: usize = 4;
18const DEADLINE: Duration = Duration::from_secs(30);

20/// Resultado da verificação de manifesto após erro ambíguo.
21#[derive(Debug)]
22pub enum ResultadoVerificacao {
23    /// Manifesto existe e é válido — partes protegidas, nenhuma remoção feita.
24    ManifestoConfirmado,
25    /// Manifesto ausente — partes removidas (falhas individuais no vetor).
26    ManifestoAusente(Vec<FalhaDeRemocao>),
27    /// Manifesto presente mas ilegível — partes removidas por precaução.
28    ManifestoIlegivel(Vec<FalhaDeRemocao>),
29}
30
31/// Verifica se o manifesto de uma execução realmente foi gravado e, se não,
32/// remove as partes que ficariam desprotegidas.
33///
34/// Cenário: a gravação do manifesto recebeu erro de rede, mas o servidor pode
35/// ter feito o commit. Esta função verifica o estado real antes de apagar.
36pub async fn verificar_manifesto_ou_limpar(
37    armazenamento: &dyn ArmazenamentoRaw,
38    chave_manifesto: &str,
39    partes_novas: &[String],
40) -> Result<ResultadoVerificacao, ErroDeIngestao> {
41    match armazenamento.ler(chave_manifesto).await? {
42        Some(bytes) => {
43            if serde_json::from_slice::<ManifestoRaw>(&bytes).is_ok() {
44                info!(
45                    manifesto = %chave_manifesto,
46                    "manifesto confirmado após erro ambíguo; partes protegidas"
47                );
48                return Ok(ResultadoVerificacao::ManifestoConfirmado);
49            }
50            warn!(
51                manifesto = %chave_manifesto,
52                "manifesto ilegível após erro ambíguo; removendo partes órfãs"
53            );
54            let falhas =
55                remover_e_logar(armazenamento, partes_novas, "verificação de manifesto").await;
56            Ok(ResultadoVerificacao::ManifestoIlegivel(falhas))
57        }
58        None => {
59            info!(
60                manifesto = %chave_manifesto,
61                "manifesto ausente após erro ambíguo; removendo partes órfãs"
62            );
63            let falhas =
64                remover_e_logar(armazenamento, partes_novas, "verificação de manifesto").await;
65            Ok(ResultadoVerificacao::ManifestoAusente(falhas))
66        }
67    }
68}

70/// Varre um prefixo no armazenamento e remove objetos que não pertencem a
71/// nenhum manifesto vigente nem à execução corrente.
72///
73/// `id_execucao_atual` exclui partes da ingestão em andamento para evitar
74/// colisão quando duas execuções do mesmo recurso se sobrepõem.
75///
76/// Se um manifesto listado não puder ser lido ou desserializado, todas as
77/// chaves sob a mesma execução são consideradas protegidas (princípio
78/// conservador: na dúvida, não apagar).
79pub async fn limpar_orfas_do_recurso(
80    armazenamento: &dyn ArmazenamentoRaw,
81    prefixo_recurso: &str,
82    id_execucao_atual: &str,
83) -> Result<(usize, Vec<FalhaDeRemocao>), ErroDeIngestao> {
84    let chaves = armazenamento.listar_prefixo(prefixo_recurso).await?;
85    if chaves.is_empty() {
86        return Ok((0, Vec::new()));
87    }
88
89    let mut protegidas = HashSet::new();
90
91    // Protege tudo da execução corrente.
92    for chave in &chaves {
93        if chave.contains(&format!("execucao={id_execucao_atual}/")) {
94            protegidas.insert(chave.clone());
95        }
96    }
97
98    for chave in &chaves {
99        if chave.ends_with("/_manifesto.json") {
100            protegidas.insert(chave.clone());
101            match armazenamento.ler(chave).await {
102                Ok(Some(bytes)) => match serde_json::from_slice::<ManifestoRaw>(&bytes) {
103                    Ok(manifesto) => {
104                        for parte in &manifesto.partes {
105                            protegidas.insert(parte.chave.clone());
106                        }
107                    }
108                    Err(_) => {
109                        // Manifesto ilegível: proteger TODAS as chaves sob a mesma
110                        // execução para não apagar dados referenciados.
111                        warn!(
112                            manifesto = %chave,
113                            "manifesto ilegível; protegendo toda a execução"
114                        );
115                        let prefixo_exec = prefixo_da_execucao(chave);
116                        for outra in &chaves {
117                            if outra.starts_with(&prefixo_exec) {
118                                protegidas.insert(outra.clone());
119                            }
120                        }
121                    }
122                },
123                Ok(None) | Err(_) => {
124                    // Listou mas não conseguiu ler: proteger a execução inteira.
125                    warn!(
126                        manifesto = %chave,
127                        "falha ao ler manifesto listado; protegendo toda a execução"
128                    );
129                    let prefixo_exec = prefixo_da_execucao(chave);
130                    for outra in &chaves {
131                        if outra.starts_with(&prefixo_exec) {
132                            protegidas.insert(outra.clone());
133                        }
134                    }
135                }
136            }
137        }
138        if chave.ends_with("/_ultima_versao.json") {
139            protegidas.insert(chave.clone());
140        }
141    }
142
143    let orfas: Vec<String> = chaves
144        .into_iter()
145        .filter(|chave| !protegidas.contains(chave))
146        .collect();
147
148    if orfas.is_empty() {
149        return Ok((0, Vec::new()));
150    }
151
152    let total = orfas.len();
153    info!(
154        prefixo = %prefixo_recurso,
155        orfas = total,
156        "encontradas partes órfãs; iniciando limpeza"
157    );
158
159    let falhas = remover_e_logar(armazenamento, &orfas, "varredura").await;
160    Ok((total, falhas))
161}

163/// Extrai o prefixo da execução de uma chave de manifesto.
164/// Ex.: `ibge/estados/data=.../execucao=abc/_manifesto.json` → `ibge/estados/data=.../execucao=abc/`
165fn prefixo_da_execucao(chave_manifesto: &str) -> String {
166    match chave_manifesto.rfind('/') {
167        Some(pos) => chave_manifesto[..=pos].to_owned(),
168        None => chave_manifesto.to_owned(),
169    }
170}
171
172async fn remover_e_logar(
173    armazenamento: &dyn ArmazenamentoRaw,
174    chaves: &[String],
175    contexto: &str,
176) -> Vec<FalhaDeRemocao> {
177    let falhas = armazenamento
178        .remover_lote(chaves, CONCORRENCIA, DEADLINE)
179        .await;
180    for falha in &falhas {
181        warn!(
182            chave = %falha.chave,
183            detalhe = %falha.detalhe,
184            contexto = %contexto,
185            "falha ao remover parte órfã"
186        );
187    }
188    falhas
189}

202    #[derive(Default)]
203    struct ArmazenamentoEmMemoria {
204        objetos: Mutex<HashMap<String, Vec<u8>>>,
205    }
206
207    impl ArmazenamentoEmMemoria {
208        fn com(objetos: impl IntoIterator<Item = (String, Vec<u8>)>) -> Self {
209            Self {
210                objetos: Mutex::new(objetos.into_iter().collect()),
211            }
212        }
213
214        fn contem(&self, chave: &str) -> bool {
215            self.objetos.lock().unwrap().contains_key(chave)
216        }
217    }
218
219    #[async_trait]
220    impl ArmazenamentoRaw for ArmazenamentoEmMemoria {
221        async fn preparar(&self) -> Result<(), ErroDeIngestao> {
222            Ok(())
223        }
224
225        async fn gravar(&self, objeto: ObjetoRaw) -> Result<(), ErroDeIngestao> {
226            self.objetos
227                .lock()
228                .unwrap()
229                .insert(objeto.chave, objeto.conteudo);
230            Ok(())
231        }
232
233        async fn gravar_fluxo(
234            &self,
235            _: &str,
236            _: &str,
237            _: &mut (dyn AsyncRead + Send + Unpin),
238        ) -> Result<(), ErroDeIngestao> {
239            unreachable!()
240        }
241
242        async fn remover(&self, chave: &str) -> Result<(), ErroDeIngestao> {
243            self.objetos.lock().unwrap().remove(chave);
244            Ok(())
245        }
246
247        async fn listar_prefixo(&self, prefixo: &str) -> Result<Vec<String>, ErroDeIngestao> {
248            Ok(self
249                .objetos
250                .lock()
251                .unwrap()
252                .keys()
253                .filter(|chave| chave.starts_with(prefixo))
254                .cloned()
255                .collect())
256        }
257
258        async fn ler(&self, chave: &str) -> Result<Option<Vec<u8>>, ErroDeIngestao> {
259            Ok(self.objetos.lock().unwrap().get(chave).cloned())
260        }
261    }
262
263    fn json<T: serde::Serialize>(valor: &T) -> Vec<u8> {
264        serde_json::to_vec(valor).unwrap()
265    }

267    fn parte(chave: &str) -> ParteRaw {
268        ParteRaw {
269            chave: chave.to_owned(),
270            url: "https://origem/dados".to_owned(),
271            bytes: 100,
272            sha256: "abc".to_owned(),
273            tipo_de_conteudo: "application/json".to_owned(),
274            versao: None,
275        }
276    }

287        }
