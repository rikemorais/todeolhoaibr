4use crate::ingestao_arquivo::TarefaDeArquivo;

6const NOME: &str = "senado-despesas";
7
8/// A Raw guarda as despesas sob a fonte `senado` (`raw/senado/despesas/...`),
9/// embora no catálogo elas sejam uma fonte separada, para poderem ser `sob_demanda`.
10const FONTE_RAW: &str = "senado";
11/// Arquivos anuais da CEAPS (Cota para o Exercício da Atividade Parlamentar dos Senadores),
12/// publicados pela Transparência do Senado em CSV nacional por ano.
13const BASE_CEAPS: &str = "https://www.senado.leg.br/transparencia/LAI/verba";
14/// Primeiro ano com arquivo anual da CEAPS publicado.
15const PRIMEIRO_ANO_CEAPS: u16 = 2008;
16
17fn ultimo_ano() -> u16 {
18    Utc::now().year() as u16
19}
20
21fn tarefas_ate(ano_final: u16) -> Vec<TarefaPlanejada> {
22    (PRIMEIRO_ANO_CEAPS..=ano_final)
23        .map(|ano| {
24            let mut tarefa = TarefaDeArquivo::nova(
25                format!("despesas_{ano}"),
26                FONTE_RAW,
27                format!("despesas/ano={ano}"),
28                format!("{BASE_CEAPS}/despesa_ceaps_{ano}.csv"),
29            );
30            if ano == ano_final {
31                tarefa = tarefa.aceitar_ausencia();
32            }
33            TarefaPlanejada::nova(tarefa).com_etiquetas([("ano", ano.to_string().as_str())])
34        })
35        .collect()
36}

38/// Despesas da CEAPS (Cota para o Exercício da Atividade Parlamentar dos Senadores).
39///
40/// Equivalente da CEAP da Câmara para a casa alta. Um arquivo CSV por ano, nacional,
41/// baixado como arquivo grande. O arquivo é guardado opaco na Raw ([ADR 0014]); a UF é
42/// coluna do CSV, então o filtro aqui é só por ano, e a deduplicação por chave composta
43/// (a CEAPS não tem ID único por despesa) fica para o refino ([ADR 0015]). Fonte separada
44/// da `senado` para ser `sob_demanda` e não entrar em `--todas`. Ver issue #26.
45///
46/// O layout das colunas difere do da Câmara (ver a issue), mas isso é problema do refino:
47/// a ingestão preserva o CSV como veio.
48///
49/// [ADR 0014]: ../../../docs/adr/0014-raw-opaca-e-refino-em-estagio-separado.md
50/// [ADR 0015]: ../../../docs/adr/0015-arquitetura-de-dados-em-medalhao.md
51pub struct FonteSenadoDespesas;
52
53impl FonteDeDados for FonteSenadoDespesas {
54    fn nome(&self) -> &str {
55        NOME
56    }
57
58    fn descricao(&self) -> &str {
59        "Senado Federal - despesas da CEAPS, um arquivo por ano desde 2008"
60    }
61
62    /// Os arquivos anuais em lote não devem entrar em `--todas` por acidente.
63    fn sob_demanda(&self) -> bool {
64        true
65    }
66
67    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
68        tarefas_ate(ultimo_ano())
69    }
70}

72#[cfg(test)]
73mod testes {
74    use std::collections::BTreeSet;
75
76    use super::*;

78    /// Anos cobertos pelas tarefas, lidos do nome `despesas_<ano>`.
79    fn anos() -> BTreeSet<u16> {
80        FonteSenadoDespesas
81            .criar_tarefas()
82            .iter()
83            .map(|t| {
84                t.tarefa
85                    .nome()
86                    .strip_prefix("despesas_")
87                    .and_then(|ano| ano.parse().ok())
88                    .expect("nome no formato despesas_<ano>")
89            })
90            .collect()
91    }
92
93    #[test]
94    fn cobre_exatamente_um_arquivo_por_ano() {
95        // Conjunto exato (não só a contagem e os extremos): pega ano faltando no meio.
96        let esperado: BTreeSet<u16> = (PRIMEIRO_ANO_CEAPS..=ultimo_ano()).collect();
97        assert_eq!(anos(), esperado);
98    }
99
100    #[test]
101    fn inclui_o_ano_novo_sem_edicao_manual() {
102        let anos: BTreeSet<_> = tarefas_ate(2027)
103            .into_iter()
104            .map(|tarefa| tarefa.etiquetas["ano"].clone())
105            .collect();
106        assert!(anos.contains("2027"));
107        assert!(!anos.contains("2028"));
108    }
109
110    #[test]
111    fn a_etiqueta_ano_bate_com_o_ano_do_nome() {
112        for tarefa in FonteSenadoDespesas.criar_tarefas() {
113            let ano = tarefa
114                .tarefa
115                .nome()
116                .strip_prefix("despesas_")
117                .expect("nome no formato despesas_<ano>");
118            // Não basta a etiqueta existir: seu valor precisa ser o ano do arquivo,
119            // senão o filtro `--filtro ano=` selecionaria a tarefa errada.
120            assert_eq!(
121                tarefa.etiquetas.get("ano").map(String::as_str),
122                Some(ano),
123                "etiqueta `ano` diverge do nome em {}",
124                tarefa.tarefa.nome()
125            );
126        }
127    }
128
129    #[test]
130    fn a_url_aponta_para_o_csv_anual_da_ceaps() {
131        // Trava o padrão da URL: um ano específico cai no arquivo certo do Senado.
132        let tarefa = tarefas_ate(2024)
133            .into_iter()
134            .find(|t| t.etiquetas["ano"] == "2024")
135            .expect("tarefa de 2024");
136        assert_eq!(
137            tarefa.tarefa.nome(),
138            "despesas_2024",
139            "o nome identifica o ano do arquivo"
140        );
141    }
142
143    #[test]
144    fn nao_entra_em_todas_por_acidente() {
145        assert!(FonteSenadoDespesas.sob_demanda());
146    }
