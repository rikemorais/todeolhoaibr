1use std::time::Duration;
2
3use todeolho_nucleo::{FonteDeDados, LimiteHttp, PerfilHttp, TarefaPlanejada};
4
5use crate::ingestao_arquivo::TarefaDeArquivo;
6
7const NOME: &str = "cvm";
8
9const BASE_DFP: &str = "https://dados.cvm.gov.br/dados/CIA_ABERTA/DOC/DFP/DADOS";
10const BASE_ITR: &str = "https://dados.cvm.gov.br/dados/CIA_ABERTA/DOC/ITR/DADOS";
11
12const PRIMEIRO_ANO_DFP: u16 = 2010;
13const PRIMEIRO_ANO_ITR: u16 = 2011;
14const ULTIMO_ANO: u16 = 2026;
15
16/// Demonstrações financeiras padronizadas (DFP, anual) e informações
17/// trimestrais (ITR) das companhias abertas, publicadas pela CVM.
18///
19/// Cobre apenas empresas de **capital aberto** (listadas em bolsa). Cada ano
20/// é um ZIP com ~19 CSVs (BPA, BPP, DRE, DFC, DMPL, DRA, DVA + metadados),
21/// em Latin-1 com `;` como separador.
22pub struct FonteCvm;
23
24impl FonteDeDados for FonteCvm {
25    fn nome(&self) -> &str {
26        NOME
27    }
28
29    fn descricao(&self) -> &str {
30        "CVM — demonstrações financeiras de companhias abertas (DFP/ITR), de 2010 a 2026"
31    }
32
33    fn sob_demanda(&self) -> bool {
34        true
35    }
36
37    fn perfil_http(&self) -> PerfilHttp {
38        PerfilHttp {
39            redirecionamento: todeolho_nucleo::RedirecionamentoHttp::Seguir,
40            credencial: None,
41            limite: Some(LimiteHttp {
42                host: "dados.cvm.gov.br",
43                requisicoes: 2,
44                intervalo: Duration::from_secs(1),
45            }),
46            agente: None,
47            aceitar_certificado_invalido: false,
48        }
49    }
50
51    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
52        let dfp = (PRIMEIRO_ANO_DFP..=ULTIMO_ANO).map(|ano| {
53            let mut tarefa = TarefaDeArquivo::nova(
54                format!("dfp_{ano}"),
55                NOME,
56                format!("dfp/ano={ano}"),
57                format!("{BASE_DFP}/dfp_cia_aberta_{ano}.zip"),
58            );
59            if ano == ULTIMO_ANO {
60                tarefa = tarefa.aceitar_ausencia();
61            }
62            TarefaPlanejada::nova(tarefa)
63                .com_etiquetas([("tipo", "dfp"), ("ano", ano.to_string().as_str())])
64        });
65
66        let itr = (PRIMEIRO_ANO_ITR..=ULTIMO_ANO).map(|ano| {
67            let mut tarefa = TarefaDeArquivo::nova(
68                format!("itr_{ano}"),
69                NOME,
70                format!("itr/ano={ano}"),
71                format!("{BASE_ITR}/itr_cia_aberta_{ano}.zip"),
72            );
73            if ano == ULTIMO_ANO {
74                tarefa = tarefa.aceitar_ausencia();
75            }
76            TarefaPlanejada::nova(tarefa)
77                .com_etiquetas([("tipo", "itr"), ("ano", ano.to_string().as_str())])
78        });
79
80        dfp.chain(itr).collect()
81    }
82}

84#[cfg(test)]
85mod testes {
86    use super::*;
87
88    #[test]
89    fn gera_tarefas_dfp_e_itr() {
90        let tarefas = FonteCvm.criar_tarefas();
91        let total_dfp = (ULTIMO_ANO - PRIMEIRO_ANO_DFP + 1) as usize;
92        let total_itr = (ULTIMO_ANO - PRIMEIRO_ANO_ITR + 1) as usize;
93        assert_eq!(tarefas.len(), total_dfp + total_itr);
94    }
95
96    #[test]
97    fn primeira_tarefa_dfp_e_2010() {
98        let tarefas = FonteCvm.criar_tarefas();
99        assert_eq!(tarefas.first().unwrap().tarefa.nome(), "dfp_2010");
100    }
101
102    #[test]
103    fn ultima_tarefa_itr_e_o_ano_corrente() {
104        let tarefas = FonteCvm.criar_tarefas();
105        assert_eq!(
106            tarefas.last().unwrap().tarefa.nome(),
107            format!("itr_{ULTIMO_ANO}")
108        );
109    }
110
111    #[test]
112    fn url_dfp_aponta_para_o_zip_anual() {
113        let tarefas = FonteCvm.criar_tarefas();
114        let t = tarefas
115            .iter()
116            .find(|t| t.tarefa.nome() == "dfp_2024")
117            .unwrap();
118        assert_eq!(t.etiquetas.get("tipo").map(String::as_str), Some("dfp"));
119        assert_eq!(t.etiquetas.get("ano").map(String::as_str), Some("2024"));
120    }
121
122    #[test]
123    fn url_itr_aponta_para_o_zip_anual() {
124        let tarefas = FonteCvm.criar_tarefas();
125        let t = tarefas
126            .iter()
127            .find(|t| t.tarefa.nome() == "itr_2024")
128            .unwrap();
129        assert_eq!(t.etiquetas.get("tipo").map(String::as_str), Some("itr"));
130    }
131
132    #[test]
133    fn sob_demanda_e_true() {
134        assert!(FonteCvm.sob_demanda());
135    }
136
137    #[test]
138    fn nao_exige_credencial() {
139        assert!(FonteCvm.perfil_http().credencial.is_none());
140    }
141
142    #[test]
143    fn host_do_limite_e_dados_cvm_gov_br() {
144        let limite = FonteCvm.perfil_http().limite.expect("limite da CVM");

146    }
