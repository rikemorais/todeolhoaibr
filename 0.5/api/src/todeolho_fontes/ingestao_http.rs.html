20/// Uma página resolvida durante a caminhada: o corpo (preciso para seguir a
21/// paginação), seus metadados, e se ela pode ser reaproveitada sem reescrever.
22struct PaginaResolvida {
23    corpo: Vec<u8>,
24    tipo_de_conteudo: String,
25    versao: Option<String>,
26    sha256: String,
27    /// `Some` quando a página não mudou: a parte anterior é reaproveitada tal
28    /// qual (mesma chave na Raw), sem gravar uma cópia nova.
29    reuso: Option<ParteRaw>,
30    /// A ingestão precisa reescrever o manifesto por causa desta página (corpo
31    /// novo, ou validador atualizado sobre o mesmo corpo).
32    mudou: bool,
33}
34
35const ACEITA_JSON: &str = "application/json";
36const LIMITE_DE_PAGINAS: usize = 1_000;

84    async fn gravar_json<T: serde::Serialize>(
85        &self,
86        contexto: &ContextoDeExecucao,
87        chave: String,
88        valor: &T,
89    ) -> Result<(), ErroDeIngestao> {
90        let conteudo =
91            serde_json::to_vec_pretty(valor).map_err(|erro| ErroDeIngestao::Manifesto {
92                recurso: self.recurso.clone(),
93                detalhe: erro.to_string(),
94            })?;
95        contexto
96            .armazenamento
97            .gravar(ObjetoRaw {
98                chave,
99                conteudo,
100                tipo_de_conteudo: ACEITA_JSON.to_owned(),
101            })
102            .await
103    }

105    /// Resolve o corpo de uma página e decide se ela pode ser reaproveitada.
106    ///
107    /// Em um 304 valida a parte guardada (tamanho e sha256) antes de confiar
108    /// nela; se estiver ausente, truncada ou corrompida, rebaixa a página sem
109    /// validador. Em um 200 com corpo idêntico ao guardado, reaproveita a chave
110    /// em vez de gravar outra cópia (ver docs/adr/0012).
111    async fn resolver_pagina(
112        &self,
113        contexto: &ContextoDeExecucao,
114        url: &str,
115        anterior: Option<&ParteRaw>,
116    ) -> Result<PaginaResolvida, ErroDeIngestao> {
117        let validador = anterior.and_then(|parte| parte.versao.as_deref());
118        match contexto
119            .http
120            .obter_condicional(url, ACEITA_JSON, validador)
121            .await?
122        {
123            RespostaCondicional::NaoModificado => {
124                let anterior = anterior.expect("304 só ocorre quando havia validador anterior");
125                match contexto.armazenamento.ler(&anterior.chave).await? {
126                    Some(corpo)
127                        if corpo.len() as u64 == anterior.bytes
128                            && hex::encode(Sha256::digest(&corpo)) == anterior.sha256 =>
129                    {
130                        Ok(PaginaResolvida {
131                            corpo,
132                            tipo_de_conteudo: anterior.tipo_de_conteudo.clone(),
133                            versao: anterior.versao.clone(),
134                            sha256: anterior.sha256.clone(),
135                            reuso: Some(anterior.clone()),
136                            mudou: false,
137                        })
138                    }
139                    // Parte ausente ou que não confere com o manifesto: não dá para
140                    // reaproveitar um 304; rebaixa o corpo de verdade e reingere.
141                    _ => self.rebaixar(contexto, url).await,
142                }
143            }
144            RespostaCondicional::Modificado {
145                tipo_de_conteudo,
146                versao,
147                corpo,
148            } => Ok(self.classificar_modificado(anterior, tipo_de_conteudo, versao, corpo)),
149        }
150    }
151
152    /// Rebaixa a página sem validador (um GET incondicional), usado quando a parte
153    /// guardada que um 304 reaproveitaria está ausente ou corrompida.
154    async fn rebaixar(
155        &self,
156        contexto: &ContextoDeExecucao,
157        url: &str,
158    ) -> Result<PaginaResolvida, ErroDeIngestao> {
159        match contexto
160            .http
161            .obter_condicional(url, ACEITA_JSON, None)
162            .await?
163        {
164            RespostaCondicional::Modificado {
165                tipo_de_conteudo,
166                versao,
167                corpo,
168            } => Ok(self.classificar_modificado(None, tipo_de_conteudo, versao, corpo)),
169            RespostaCondicional::NaoModificado => {
170                Err(self
171                    .erro_de_paginacao(format!("origem respondeu 304 sem validador para {url}")))
172            }
173        }
174    }
175
176    /// Classifica um 200: se o corpo bate com a parte anterior (sha256 e tamanho),
177    /// reaproveita a chave — só reescreve o manifesto se o validador mudou.
178    fn classificar_modificado(
179        &self,
180        anterior: Option<&ParteRaw>,
181        tipo_de_conteudo: String,
182        versao: Option<String>,
183        corpo: Vec<u8>,
184    ) -> PaginaResolvida {
185        let sha256 = hex::encode(Sha256::digest(&corpo));
186        if let Some(anterior) = anterior
187            && anterior.sha256 == sha256
188            && anterior.bytes == corpo.len() as u64
189        {
190            let versao_final = versao.or_else(|| anterior.versao.clone());
191            let mut parte = anterior.clone();
192            let mudou = versao_final != parte.versao;
193            parte.versao = versao_final.clone();
194            return PaginaResolvida {
195                corpo,
196                tipo_de_conteudo: anterior.tipo_de_conteudo.clone(),
197                versao: versao_final,
198                sha256,
199                reuso: Some(parte),
200                mudou,
201            };
202        }
203        PaginaResolvida {
204            corpo,
205            tipo_de_conteudo,
206            versao,
207            sha256,
208            reuso: None,
209            mudou: true,
210        }
211    }
212}

215impl Tarefa for TarefaDeIngestaoHttp {
216    fn nome(&self) -> &str {
217        &self.recurso
218    }

220    async fn bytes_a_baixar(
221        &self,
222        contexto: &ContextoDeExecucao,
223    ) -> Result<Option<u64>, ErroDeIngestao> {
224        // Uma rodada sobre uma ingestão completa e íntegra não baixa dado novo no
225        // caso comum (páginas reaproveitadas via requisição condicional), então
226        // não pede espaço novo. Só a primeira ingestão — ou uma reparação, quando
227        // o estado está ausente/incompleto — usa a estimativa típica.
228        let estado =
229            estado_na_raw(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso).await?;
230        match estado {
231            Some(estado) if estado.completa && estado.integra => Ok(Some(0)),
232            _ => Ok(self.tamanho_tipico),
233        }
234    }
235
236    async fn executar(
237        &self,
238        contexto: &ContextoDeExecucao,
239    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
240        let local = LocalizacaoRaw::nova(
241            &self.fonte,
242            &self.recurso,
243            contexto.iniciada_em,
244            &contexto.id_execucao,
245        );
246
247        // Estado anterior na Raw, para o motivo da reingestão (ausente, incompleta
248        // ou desatualizada) refletir o que de fato havia guardado.
249        let estado_anterior =
250            estado_na_raw(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso).await?;
251
252        // Partes da ingestão anterior, por URL, para a requisição condicional e
253        // para reaproveitar, sem reescrever, o que não mudou.
254        let anteriores: HashMap<String, ParteRaw> =
255            manifesto_vigente(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso)
256                .await?
257                .map(|manifesto| {
258                    manifesto
259                        .partes
260                        .into_iter()
261                        .map(|parte| (parte.url.clone(), parte))
262                        .collect()
263                })
264                .unwrap_or_default();
265
266        // As partes guardam só metadados (pequenos); o corpo de cada página é
267        // gravado na hora e liberado, para não acumular toda a paginação na
268        // memória do contêiner (ver docs/adr/0012).
269        let mut partes: Vec<ParteRaw> = Vec::new();
270        let mut visitadas = HashSet::new();
271        let mut proxima = Some(self.url_inicial.clone());
272        let mut alguma_mudou = false;
273        let mut numero = 0usize;
274
275        while let Some(url) = proxima.take() {
276            numero += 1;
277            if numero > self.limite_de_paginas {
278                return Err(self.erro_de_paginacao(format!(
279                    "limite de {} páginas excedido",
280                    self.limite_de_paginas
281                )));
282            }
283            if !visitadas.insert(url.clone()) {
284                return Err(self.erro_de_paginacao(format!("página repetida: {url}")));
285            }
286
287            let anterior = anteriores.get(&url);
288            let resolvida = self.resolver_pagina(contexto, &url, anterior).await?;
289
290            proxima = self
291                .paginacao
292                .proxima_url(&resolvida.corpo)
293                .map_err(|detalhe| self.erro_de_paginacao(detalhe))?;
294
295            let PaginaResolvida {
296                corpo,
297                tipo_de_conteudo,
298                versao,
299                sha256,
300                reuso,
301                mudou,
302            } = resolvida;
303            if mudou {
304                alguma_mudou = true;
305            }
306
307            let parte = match reuso {
308                // Página inalterada: reaproveita a parte anterior tal qual (mesma
309                // chave), sem gravar uma cópia nova.
310                Some(parte) => parte,
311                None => {
312                    let chave = local.parte(numero, extensao(&tipo_de_conteudo));
313                    let parte = ParteRaw {
314                        chave: chave.clone(),
315                        url: url.clone(),
316                        bytes: corpo.len() as u64,
317                        sha256,
318                        tipo_de_conteudo: tipo_de_conteudo.clone(),
319                        versao,
320                    };
321                    contexto
322                        .armazenamento
323                        .gravar(ObjetoRaw {
324                            chave,
325                            conteudo: corpo,
326                            tipo_de_conteudo,
327                        })
328                        .await?;
329                    parte
330                }
331            };
332            partes.push(parte);
333        }
334
335        // Nada mudou e o conjunto de páginas é o mesmo: a ingestão anterior segue
336        // válida, nada a reescrever.
337        let mesmo_conjunto = partes.len() == anteriores.len()
338            && partes
339                .iter()
340                .all(|parte| anteriores.contains_key(&parte.url));
341        if !alguma_mudou && mesmo_conjunto && !partes.is_empty() {
342            return Ok(ResumoDaTarefa {
343                reaproveitada: true,
344                ..ResumoDaTarefa::default()
345            });
346        }
347
348        let motivo = match &estado_anterior {
349            None => MotivoDeIngestao::AusenteNaRaw,
350            Some(estado) if !estado.completa || !estado.integra => {
351                MotivoDeIngestao::IncompletaOuCorrompida
352            }
353            Some(_) => MotivoDeIngestao::Desatualizada,
354        };
355        let resumo = ResumoDaTarefa {
356            partes: partes.len(),
357            bytes: partes.iter().map(|parte| parte.bytes).sum(),
358            reaproveitada: false,
359            motivo: Some(motivo),
360        };
361
362        let manifesto = ManifestoRaw {
363            fonte: self.fonte.clone(),
364            recurso: self.recurso.clone(),
365            id_execucao: contexto.id_execucao.clone(),
366            url_inicial: self.url_inicial.clone(),
367            iniciada_em: contexto.iniciada_em,
368            concluida_em: Utc::now(),
369            partes,
370        };
371        self.gravar_json(contexto, local.manifesto(), &manifesto)
372            .await?;
373
374        // Ponteiro para a próxima rodada localizar este manifesto. Sem validador
375        // de recurso: a requisição condicional é por página (ver docs/adr/0012).
376        // A publicação não deixa uma rodada mais antiga sobrescrever uma mais
377        // nova (ver issue #16).
378        let ultima = UltimaVersao {
379            versao: None,
380            manifesto: local.manifesto(),
381            iniciada_em: Some(contexto.iniciada_em),
382        };
383        let publicou = publicar_ponteiro(
384            contexto.armazenamento.as_ref(),
385            &self.fonte,
386            &self.recurso,
387            &ultima,
388        )
389        .await?;
390        if !publicou {
391            warn!(
392                recurso = %self.recurso,
393                "outra rodada já publicou um ponteiro mais novo; este manifesto não é o vigente"
394            );
395        }
396
397        Ok(resumo)
398    }

401pub(crate) fn extensao(tipo_de_conteudo: &str) -> &'static str {
402    let tipo = tipo_de_conteudo.to_ascii_lowercase();
403    if tipo.contains("json") {
404        "json"
405    } else if tipo.contains("xml") {
406        "xml"
407    } else if tipo.contains("zip") {
408        "zip"
409    } else if tipo.contains("csv") {
410        "csv"
411    } else {
412        "bin"
413    }
414}

601        }

604    /// Entrega o corpo com ETag `v1` na primeira vez e responde 304 quando o
605    /// validador `v1` volta na requisição condicional.

607    struct HttpCondicional {
608        chamadas: Mutex<u32>,
609    }

611    #[async_trait]
612    impl ClienteHttp for HttpCondicional {
613        async fn obter(&self, _: &str, _: &str) -> Result<RespostaHttp, ErroDeIngestao> {
614            unreachable!("usa obter_condicional")
615        }
616
617        async fn informacoes(&self, _: &str) -> Result<InformacoesDoArquivo, ErroDeIngestao> {
618            unreachable!("a ingestão paginada não consulta arquivos")
619        }
620
621        async fn baixar(&self, _: &str) -> Result<DownloadHttp, ErroDeIngestao> {
622            unreachable!("a ingestão paginada não baixa arquivos")
623        }
624
625        async fn obter_condicional(
626            &self,
627            _: &str,
628            _: &str,
629            validador: Option<&str>,
630        ) -> Result<RespostaCondicional, ErroDeIngestao> {
631            *self.chamadas.lock().unwrap() += 1;
632            if validador == Some("v1") {
633                Ok(RespostaCondicional::NaoModificado)
634            } else {
635                Ok(RespostaCondicional::Modificado {
636                    tipo_de_conteudo: "application/json".to_owned(),
637                    versao: Some("v1".to_owned()),
638                    corpo: UNICA.as_bytes().to_vec(),
639                })
640            }
641        }
642    }

698        );

700        assert_eq!(*http.chamadas.lock().unwrap(), 3);
