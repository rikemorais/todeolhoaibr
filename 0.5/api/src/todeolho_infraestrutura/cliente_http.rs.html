17const AGENTE: &str = concat!(
18    "todeolhoai/",
19    env!("CARGO_PKG_VERSION"),
20    " (ingestão de dados públicos)"
21);
22const TEMPO_DE_CONEXAO: Duration = Duration::from_secs(30);
23/// Sem receber nenhum byte por este tempo, a conexão é considerada perdida.
24const TEMPO_SEM_DADOS: Duration = Duration::from_secs(120);
25const TEMPO_LIMITE: Duration = Duration::from_secs(90);
26/// Generoso para o maior arquivo (cerca de 860 MB) em uma conexão lenta.
27const TEMPO_LIMITE_DE_DOWNLOAD: Duration = Duration::from_secs(6 * 60 * 60);

29/// Quantas vezes tentar uma requisição transitória e quanto esperar no início.
30#[derive(Debug, Clone, Copy)]
31pub struct PoliticaDeRetentativas {
32    /// Número **total** de tentativas, já contando a primeira (ex.: `5` = a
33    /// requisição inicial mais até 4 repetições).
34    pub tentativas: u32,
35    /// Espera antes da primeira retentativa; dobra a cada falha.
36    pub espera_inicial: Duration,
37}
38
39impl Default for PoliticaDeRetentativas {
40    fn default() -> Self {
41        Self {
42            tentativas: 5,
43            espera_inicial: Duration::from_secs(2),
44        }
45    }
46}
47
48impl PoliticaDeRetentativas {
49    /// Espera antes da nova tentativa: dobra a cada falha.
50    fn espera(&self, falhas: u32) -> Duration {
51        self.espera_inicial
52            .saturating_mul(2u32.saturating_pow(falhas.saturating_sub(1)))
53    }
54}

77    /// Configura uma fonte. `segredo` vem do ambiente e nunca é incluído em erros ou logs.
78    pub fn com_perfil(
79        politica: PoliticaDeRetentativas,
80        perfil: PerfilHttp,
81        segredo: Option<&str>,
82    ) -> Result<Self, ErroDeIngestao> {
83        let mut construtor = reqwest::Client::builder()
84            .user_agent(AGENTE)
85            .connect_timeout(TEMPO_DE_CONEXAO)
86            .read_timeout(TEMPO_SEM_DADOS);
87
88        let host_da_credencial = if let Some(credencial) = perfil.credencial {
89            let valor = segredo
90                .filter(|valor| !valor.trim().is_empty())
91                .ok_or_else(|| {
92                    erro_de_configuracao(format!(
93                        "variável de ambiente ausente ou vazia: {}",
94                        credencial.variavel
95                    ))
96                })?;
97            let nome = header::HeaderName::from_bytes(credencial.cabecalho.as_bytes())
98                .map_err(|_| erro_de_configuracao("nome de cabeçalho HTTP inválido".into()))?;
99            let mut valor =
100                header::HeaderValue::from_str(&format!("{}{}", credencial.prefixo, valor))
101                    .map_err(|_| {
102                        erro_de_configuracao("valor de credencial HTTP inválido".into())
103                    })?;
104            valor.set_sensitive(true);
105            let mut cabecalhos = header::HeaderMap::new();
106            cabecalhos.insert(nome, valor);
107            // Mesmo um redirect válido nunca transporta o segredo para outro host.
108            construtor = construtor
109                .default_headers(cabecalhos)
110                .redirect(reqwest::redirect::Policy::none());
111            Some(credencial.host.to_ascii_lowercase())
112        } else {
113            None
114        };
115
116        let limite = perfil
117            .limite
118            .map(|limite| {
119                if limite.requisicoes == 0 || limite.intervalo.is_zero() || limite.host.is_empty() {
120                    return Err(erro_de_configuracao("limite HTTP inválido".into()));
121                }
122                Ok((
123                    limite.host.to_ascii_lowercase(),
124                    (limite.intervalo / limite.requisicoes).max(Duration::from_nanos(1)),
125                ))
126            })
127            .transpose()?;
128
129        let cliente = construtor.build().map_err(|erro| ErroDeIngestao::Http {
130            url: String::new(),
131            detalhe: erro.to_string(),
132        })?;
133        if let Some((host, intervalo)) = limite {
134            registrar_limite(&host, intervalo);
135        }
136        Ok(Self {
137            cliente,
138            politica,
139            host_da_credencial,
140        })
141    }

143    async fn enviar(&self, requisicao: RequestBuilder) -> Result<Response, Falha> {
144        classificar_resposta(self.enviar_bruto(requisicao).await)
145    }

147    async fn enviar_bruto(&self, requisicao: RequestBuilder) -> Result<Response, Falha> {
148        let requisicao = requisicao
149            .build()
150            .map_err(|erro| Falha::definitiva(descrever(erro)))?;
151        let host = requisicao
152            .url()
153            .host_str()
154            .unwrap_or_default()
155            .to_ascii_lowercase();
156        if let Some(autorizado) = &self.host_da_credencial
157            && &host != autorizado
158        {
159            return Err(Falha::definitiva(format!(
160                "host HTTP não autorizado para a credencial: {host}"
161            )));
162        }
163        aguardar_limite(&host).await;
164        self.cliente
165            .execute(requisicao)
166            .await
167            .map_err(|erro| Falha::transitoria(descrever(erro)))
168    }
169
170    async fn tentar_obter(&self, url: &str, aceita: &str) -> Result<RespostaHttp, Falha> {
171        let resposta = self
172            .enviar(
173                self.cliente
174                    .get(url)
175                    .timeout(TEMPO_LIMITE)
176                    .header(header::ACCEPT, aceita),
177            )
178            .await?;
179
180        let tipo_de_conteudo = tipo_de_conteudo(&resposta);
181        let corpo = resposta
182            .bytes()
183            .await
184            .map_err(|erro| Falha::transitoria(descrever(erro)))?;
185
186        Ok(RespostaHttp {
187            tipo_de_conteudo,
188            corpo: corpo.to_vec(),
189        })
190    }

192    async fn tentar_condicional(
193        &self,
194        url: &str,
195        aceita: &str,
196        validador: Option<&str>,
197    ) -> Result<RespostaCondicional, Falha> {
198        let mut requisicao = self
199            .cliente
200            .get(url)
201            .timeout(TEMPO_LIMITE)
202            .header(header::ACCEPT, aceita);
203        if let Some(validador) = validador {
204            requisicao = requisicao.header(header::IF_NONE_MATCH, validador);
205        }
206
207        let resposta = self.enviar_bruto(requisicao).await?;
208
209        let status = resposta.status();
210        if status == StatusCode::NOT_MODIFIED {
211            return Ok(RespostaCondicional::NaoModificado);
212        }
213        verificar_status(status)?;
214
215        let tipo_de_conteudo = tipo_de_conteudo(&resposta);
216        let versao = etag(&resposta);
217        let corpo = resposta
218            .bytes()
219            .await
220            .map_err(|erro| Falha::transitoria(descrever(erro)))?;
221
222        Ok(RespostaCondicional::Modificado {
223            tipo_de_conteudo,
224            versao,
225            corpo: corpo.to_vec(),
226        })
227    }

229    /// Repete a operação enquanto a falha for transitória.
230    async fn com_retentativas<T, F>(
231        &self,
232        url: &str,
233        mut operacao: impl FnMut() -> F,
234    ) -> Result<T, ErroDeIngestao>
235    where
236        F: Future<Output = Result<T, Falha>>,
237    {
238        let mut falhas = 0;
239        loop {
240            let falha = match operacao().await {
241                Ok(resultado) => return Ok(resultado),
242                Err(falha) => falha,
243            };
244
245            falhas += 1;
246            if let Some(status) = falha.status_de_autenticacao {
247                return Err(ErroDeIngestao::Autenticacao {
248                    url: url.to_owned(),
249                    status,
250                });
251            }
252            if !falha.transitoria || falhas >= self.politica.tentativas {
253                return Err(ErroDeIngestao::Http {
254                    url: url.to_owned(),
255                    detalhe: format!("{} (após {falhas} tentativa(s))", falha.detalhe),
256                });
257            }
258
259            let espera = self.politica.espera(falhas);
260            warn!(%url, detalhe = %falha.detalhe, ?espera, "falha transitória; tentando novamente");
261            tokio::time::sleep(espera).await;
262        }
263    }
264}

283    async fn informacoes(&self, url: &str) -> Result<InformacoesDoArquivo, ErroDeIngestao> {
284        let resposta = self
285            .com_retentativas(url, || {
286                self.enviar(self.cliente.head(url).timeout(TEMPO_LIMITE))
287            })
288            .await;
289
290        // Sem essas informações, a estimativa fica desconhecida e o arquivo é baixado de novo.
291        Ok(match resposta {
292            Ok(resposta) => InformacoesDoArquivo {
293                versao: etag(&resposta),
294                tamanho: cabecalho(&resposta, header::CONTENT_LENGTH)
295                    .and_then(|tamanho| tamanho.parse().ok()),
296            },
297            Err(erro @ ErroDeIngestao::Autenticacao { .. }) => return Err(erro),
298            Err(erro) => {
299                warn!(%erro, "não foi possível consultar as informações do arquivo");
300                InformacoesDoArquivo::default()
301            }
302        })
303    }

309    async fn baixar_a_partir_de(
310        &self,
311        url: &str,
312        inicio: u64,
313        versao: Option<&str>,
314    ) -> Result<DownloadHttp, ErroDeIngestao> {
315        let resposta = self
316            .com_retentativas(url, || {
317                let mut requisicao = self.cliente.get(url).timeout(TEMPO_LIMITE_DE_DOWNLOAD);
318                if inicio > 0 {
319                    // Retomada: pede do offset em diante e só aceita se o arquivo
320                    // não mudou (If-Range com o ETag da primeira resposta). If-Range
321                    // só aceita validador FORTE; um ETag fraco (`W/"..."`) faria a
322                    // origem devolver o arquivo inteiro (200), impedindo a retomada.
323                    requisicao = requisicao.header(header::RANGE, format!("bytes={inicio}-"));
324                    if let Some(validador) = versao
325                        && !validador.trim_start().starts_with("W/")
326                    {
327                        requisicao = requisicao.header(header::IF_RANGE, validador);
328                    }
329                }
330                self.enviar(requisicao)
331            })
332            .await?;
333
334        // `206` traz o início e o total no Content-Range; `200` (inclusive quando
335        // a origem ignora o Range) começa do zero, com o total no Content-Length.
336        let (inicio_real, tamanho_total) = if resposta.status() == StatusCode::PARTIAL_CONTENT {
337            faixa_de_conteudo(&resposta).unwrap_or((0, None))
338        } else {
339            (
340                0,
341                cabecalho(&resposta, header::CONTENT_LENGTH).and_then(|total| total.parse().ok()),
342            )
343        };
344
345        Ok(DownloadHttp {
346            tipo_de_conteudo: tipo_de_conteudo(&resposta),
347            versao: etag(&resposta),
348            inicio: inicio_real,
349            tamanho_total,
350            fluxo: Box::new(StreamReader::new(
351                resposta
352                    .bytes_stream()
353                    .map_err(|erro| io::Error::other(descrever(erro))),
354            )),
355        })
356    }
357}

359/// Lê o `Content-Range: bytes INICIO-FIM/TOTAL` de uma resposta `206` e devolve
360/// `(inicio, total)`. `total` é `None` quando a origem manda `*`.
361fn faixa_de_conteudo(resposta: &Response) -> Option<(u64, Option<u64>)> {
362    let valor = cabecalho(resposta, header::CONTENT_RANGE)?;
363    let resto = valor.strip_prefix("bytes ")?;
364    let (faixa, total) = resto.split_once('/')?;
365    let inicio = faixa.split('-').next()?.trim().parse().ok()?;
366    Some((inicio, total.trim().parse().ok()))
367}

369fn classificar_resposta(resposta: Result<Response, Falha>) -> Result<Response, Falha> {
370    let resposta = resposta?;
371    verificar_status(resposta.status())?;
372    Ok(resposta)
373}
374
375fn verificar_status(status: StatusCode) -> Result<(), Falha> {
376    if status.is_success() {
377        return Ok(());
378    }
379    if status == StatusCode::UNAUTHORIZED || status == StatusCode::FORBIDDEN {
380        return Err(Falha::autenticacao(status.as_u16()));
381    }
382    let detalhe = format!("status {status}");
383    Err(if eh_transitorio(status) {
384        Falha::transitoria(detalhe)
385    } else {
386        Falha::definitiva(detalhe)
387    })
388}

397async fn aguardar_limite(host: &str) {
398    if let Some(instante) = reservar_limite(host) {
399        tokio::time::sleep_until(instante).await;
400    }
401}
402
403fn registrar_limite(host: &str, intervalo: Duration) {
404    let mut limites = LIMITES_POR_HOST
405        .get_or_init(|| Mutex::new(HashMap::new()))
406        .lock()
407        .unwrap_or_else(std::sync::PoisonError::into_inner);
408    let agora = tokio::time::Instant::now();
409    let estado = limites.entry(host.to_owned()).or_insert(EstadoDoHost {
410        proxima: agora,
411        intervalo_entre_requisicoes: intervalo,
412    });
413    estado.intervalo_entre_requisicoes = estado.intervalo_entre_requisicoes.max(intervalo);
414}
415
416fn reservar_limite(host: &str) -> Option<tokio::time::Instant> {
417    let mut limites = LIMITES_POR_HOST
418        .get_or_init(|| Mutex::new(HashMap::new()))
419        .lock()
420        .unwrap_or_else(std::sync::PoisonError::into_inner);
421    let estado = limites.get_mut(host)?;
422    let instante = estado.proxima.max(tokio::time::Instant::now());
423    estado.proxima = instante + estado.intervalo_entre_requisicoes;
424    Some(instante)
425}

475/// Inclui a causa de origem, sem a URL, que já consta na mensagem final.
476fn descrever(erro: reqwest::Error) -> String {
477    let erro = erro.without_url();
478    let mut descricao = erro.to_string();
479    let mut causa = Error::source(&erro);
480    while let Some(atual) = causa {
481        descricao.push_str(": ");
482        descricao.push_str(&atual.to_string());
483        causa = atual.source();
484    }
485    descricao
486}

488fn eh_transitorio(status: StatusCode) -> bool {
489    status.is_server_error()
490        || status == StatusCode::TOO_MANY_REQUESTS
491        || status == StatusCode::REQUEST_TIMEOUT
492}

612    /// `Range`/`If-Range` e lê o `Content-Range` de um `206`, entregando o resto.
613    #[tokio::test]
614    async fn baixar_a_partir_de_envia_range_e_le_content_range() {
615        use std::io::{Read, Write};
616        use std::net::TcpListener;
617
618        use tokio::io::AsyncReadExt;
619
620        const CORPO: &[u8] = b"0123456789abcdefghij"; // 20 bytes
621        let total = CORPO.len();
622        let inicio = 5usize;
623
624        let ouvinte = TcpListener::bind("127.0.0.1:0").unwrap();
625        let porta = ouvinte.local_addr().unwrap().port();
626        let (enviar_requisicao, receber_requisicao) = std::sync::mpsc::channel::<String>();
627
628        let servidor = std::thread::spawn(move || {
629            let (mut conexao, _) = ouvinte.accept().unwrap();
630            let mut buf = [0u8; 2048];
631            let lidos = conexao.read(&mut buf).unwrap();
632            enviar_requisicao
633                .send(String::from_utf8_lossy(&buf[..lidos]).into_owned())
634                .unwrap();
635
636            let resto = &CORPO[inicio..];
637            let cabecalho = format!(
638                "HTTP/1.1 206 Partial Content\r\nContent-Type: application/zip\r\n\
639                 ETag: \"etag\"\r\nContent-Range: bytes {inicio}-{fim}/{total}\r\n\
640                 Content-Length: {len}\r\nConnection: close\r\n\r\n",
641                fim = total - 1,
642                len = resto.len(),
643            );
644            conexao.write_all(cabecalho.as_bytes()).unwrap();
645            conexao.write_all(resto).unwrap();
646            conexao.flush().unwrap();
647        });
648
649        let cliente = ClienteHttpComRetentativas::novo(PoliticaDeRetentativas::default()).unwrap();
650        let url = format!("http://127.0.0.1:{porta}/arquivo.zip");
651        let download = cliente
652            .baixar_a_partir_de(&url, inicio as u64, Some("\"etag\""))
653            .await
654            .unwrap();
655
656        // O adaptador leu o Content-Range: começa em `inicio`, com o total certo.
657        assert_eq!(download.inicio, inicio as u64);
658        assert_eq!(download.tamanho_total, Some(total as u64));
659        assert_eq!(download.versao.as_deref(), Some("\"etag\""));
660
661        let mut corpo = Vec::new();
662        let mut fluxo = download.fluxo;
663        fluxo.read_to_end(&mut corpo).await.unwrap();
664        assert_eq!(corpo, &CORPO[inicio..]);
665
666        // E enviou os cabeçalhos de retomada (nome do cabeçalho é case-insensitive).
667        let requisicao = receber_requisicao.recv().unwrap().to_lowercase();
668        assert!(requisicao.contains("range: bytes=5-"), "{requisicao}");
669        assert!(requisicao.contains("if-range: \"etag\""), "{requisicao}");
670
671        servidor.join().unwrap();
672    }
