10/// A Raw guarda as votações sob a mesma fonte `camara` (`raw/camara/votacoes/...`),
11/// embora no catálogo sejam uma fonte separada, para poderem ser `sob_demanda`.
12const FONTE_RAW: &str = "camara";
13/// Arquivos anuais em lote dos Dados Abertos da Câmara.
14const BASE_ARQUIVOS: &str = "https://dadosabertos.camara.leg.br/arquivos";
15
16fn ultimo_ano() -> u16 {
17    Utc::now().year() as u16
18}
19
20fn tarefas_ate(ano_final: u16) -> Vec<TarefaPlanejada> {
21    CONJUNTOS
22        .iter()
23        .flat_map(|&(conjunto, primeiro_ano)| {
24            (primeiro_ano..=ano_final).map(move |ano| {
25                let mut tarefa = TarefaDeArquivo::nova(
26                    format!("{conjunto}_{ano}"),
27                    FONTE_RAW,
28                    format!("votacoes/{conjunto}/ano={ano}"),
29                    format!("{BASE_ARQUIVOS}/{conjunto}/csv/{conjunto}-{ano}.csv"),
30                );
31                if ano == ano_final {
32                    tarefa = tarefa.aceitar_ausencia();
33                }
34                TarefaPlanejada::nova(tarefa)
35                    .com_etiquetas([("ano", ano.to_string().as_str()), ("conjunto", conjunto)])
36            })
37        })
38        .collect()
39}

41/// Conjuntos (datasets) anuais ingeridos e o primeiro ano publicado de cada um.
42///
43/// O escopo começa em **2001**, o primeiro ano com voto nominal (`votacoesVotos`):
44/// é a janela onde "quem votou em quê" existe. `proposicoes`/`votacoes` têm arquivos
45/// mais antigos (desde ~1990), mas sem os votos não servem ao objetivo desta fonte.
46/// `votacoesOrientacoes` (orientação de bancada) só começa em 2003.
47const CONJUNTOS: &[(&str, u16)] = &[
48    ("proposicoes", 2001),
49    ("proposicoesAutores", 2001),
50    ("votacoes", 2001),
51    ("votacoesVotos", 2001),
52    ("votacoesObjetos", 2001),
53    ("votacoesProposicoes", 2001),
54    ("votacoesOrientacoes", 2003),
55];
56
57/// Proposições e votações nominais da Câmara dos Deputados — "quais projetos foram
58/// votados e quem votou em quê".
59///
60/// Em vez da API REST (`/votacoes/{id}/votos`, que exigiria um fan-out por id de
61/// votação), usa os **arquivos anuais em lote** dos Dados Abertos
62/// (`/arquivos/{conjunto}/csv/{conjunto}-{ano}.csv`): um download por (conjunto, ano),
63/// via [`TarefaDeArquivo`], guardado **opaco** na Raw ([ADR 0014]). O voto nominal de
64/// cada deputado vem no conjunto `votacoesVotos`; o cruzamento entre proposição,
65/// votação, voto e deputado é trabalho do refino ([ADR 0015]).
66///
67/// Fonte separada da `camara` (API leve) para ser `sob_demanda` e não entrar em
68/// `--todas`. O dado mora em `raw/camara/votacoes/`. Ver issue #86.
69///
70/// Documentação dos conjuntos: <https://dadosabertos.camara.leg.br/howtouse/2020-02-28-dados-votacoes.html>
71///
72/// [ADR 0014]: ../../../docs/adr/0014-raw-opaca-e-refino-em-estagio-separado.md
73/// [ADR 0015]: ../../../docs/adr/0015-arquitetura-de-dados-em-medalhao.md
74pub struct FonteCamaraVotacoes;
75
76impl FonteDeDados for FonteCamaraVotacoes {
77    fn nome(&self) -> &str {
78        NOME
79    }
80
81    fn descricao(&self) -> &str {
82        "Câmara dos Deputados - proposições e votações nominais, um arquivo por ano desde 2001"
83    }
84
85    /// São dezenas de arquivos anuais em lote: não devem entrar em `--todas` por acidente.
86    fn sob_demanda(&self) -> bool {
87        true
88    }
89
90    fn perfil_http(&self) -> todeolho_nucleo::PerfilHttp {
91        api::perfil_http()
92    }
93
94    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
95        tarefas_ate(ultimo_ano())
96    }
97}

105    /// Pares (conjunto, ano) cobertos pelas tarefas, lidos do nome `<conjunto>_<ano>`.
106    fn pares() -> BTreeSet<(String, u16)> {
107        FonteCamaraVotacoes
108            .criar_tarefas()
109            .iter()
110            .map(|t| {
111                let nome = t.tarefa.nome();
112                let (conjunto, ano) = nome
113                    .rsplit_once('_')
114                    .expect("nome no formato <conjunto>_<ano>");
115                (conjunto.to_owned(), ano.parse().expect("ano numérico"))
116            })
117            .collect()
118    }

120    #[test]
121    fn cobre_exatamente_cada_conjunto_do_seu_primeiro_ano_ate_o_ultimo() {
122        let esperado: BTreeSet<(String, u16)> = CONJUNTOS
123            .iter()
124            .flat_map(|&(conjunto, primeiro)| {
125                (primeiro..=ultimo_ano()).map(move |ano| (conjunto.to_owned(), ano))
126            })
127            .collect();
128        // A contagem bruta (antes de virar `BTreeSet`) pega uma duplicata de
129        // `(conjunto, ano)` que o conjunto esconderia.
130        assert_eq!(
131            FonteCamaraVotacoes.criar_tarefas().len(),
132            esperado.len(),
133            "número de tarefas diverge do esperado (duplicata ou lacuna)"
134        );
135        assert_eq!(pares(), esperado);
136    }
137
138    #[test]
139    fn cobre_os_sete_conjuntos_esperados() {
140        // Lista explícita e independente de `CONJUNTOS`: se um dataset exigido
141        // for removido de `CONJUNTOS`, este teste falha (incluindo `votacoesVotos`,
142        // o voto nominal, sem o qual a fonte perde o sentido).
143        let esperados: BTreeSet<String> = [
144            "proposicoes",
145            "proposicoesAutores",
146            "votacoes",
147            "votacoesVotos",
148            "votacoesObjetos",
149            "votacoesProposicoes",
150            "votacoesOrientacoes",
151        ]
152        .iter()
153        .map(|s| s.to_string())
154        .collect();
155        let gerados: BTreeSet<String> = pares().into_iter().map(|(conjunto, _)| conjunto).collect();
156        assert_eq!(gerados, esperados);
157    }

159    #[test]
160    fn inclui_o_ano_novo_em_todos_os_conjuntos() {
161        let tarefas = tarefas_ate(2027);
162        let atuais: BTreeSet<_> = tarefas
163            .iter()
164            .filter(|tarefa| tarefa.etiquetas["ano"] == "2027")
165            .map(|tarefa| tarefa.etiquetas["conjunto"].as_str())
166            .collect();
167        assert_eq!(atuais.len(), CONJUNTOS.len());
168        assert!(
169            !tarefas
170                .iter()
171                .any(|tarefa| tarefa.etiquetas["ano"] == "2028")
172        );
173    }

175    #[test]
176    fn as_etiquetas_batem_com_o_nome() {
177        for tarefa in FonteCamaraVotacoes.criar_tarefas() {
178            let nome = tarefa.tarefa.nome();
179            let (conjunto, ano) = nome
180                .rsplit_once('_')
181                .expect("nome no formato <conjunto>_<ano>");
182            // As etiquetas guiam `--filtro`: se divergirem do nome, o filtro
183            // selecionaria a tarefa errada.
184            assert_eq!(tarefa.etiquetas.get("ano").map(String::as_str), Some(ano));
185            assert_eq!(
186                tarefa.etiquetas.get("conjunto").map(String::as_str),
187                Some(conjunto)
188            );
189        }
190    }
