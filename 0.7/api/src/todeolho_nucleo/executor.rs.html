2//! limitada, e avisa um observador sobre o início e o fim de cada tarefa.

16/// Como uma tarefa terminou.
17#[derive(Debug, Clone, PartialEq, Eq)]
18pub enum EstadoDaTarefa {
19    /// Concluída, com o resumo do que produziu.
20    Concluida(ResumoDaTarefa),
21    /// Falhou, com a mensagem de erro.
22    Falhou(String),
23    /// Não executada porque a tarefa `causa`, da qual ela depende, falhou.
24    Ignorada {
25        /// Nome da tarefa que falhou e causou a ignorada.
26        causa: String,
27    },
28}
29
30/// Resultado de uma tarefa: nome, estado final e duração.
31#[derive(Debug, Clone)]
32pub struct ResultadoDaTarefa {
33    /// Nome da tarefa.
34    pub nome: String,
35    /// Como ela terminou.
36    pub estado: EstadoDaTarefa,
37    /// Quanto tempo levou.
38    pub duracao: Duration,
39}
40
41/// Resultado agregado da execução de uma fonte.
42#[derive(Debug, Clone)]
43pub struct RelatorioDaExecucao {
44    /// Fonte executada.
45    pub fonte: String,
46    /// Resultado de cada tarefa.
47    pub resultados: Vec<ResultadoDaTarefa>,
48}
49
50impl RelatorioDaExecucao {
51    /// `true` se todas as tarefas concluíram (nenhuma falhou nem foi ignorada).
52    pub fn sucesso(&self) -> bool {
53        self.resultados
54            .iter()
55            .all(|r| matches!(r.estado, EstadoDaTarefa::Concluida(_)))
56    }
57}

59/// Recebe os acontecimentos de uma execução (padrão Observer).
60///
61/// O executor não sabe quem observa nem para quê. Os métodos não devolvem erro de
62/// propósito: quem observa não pode interferir na ingestão.
63#[async_trait]
64pub trait ObservadorDeExecucao: Send + Sync {
65    /// Avisado quando uma tarefa começa.
66    async fn tarefa_iniciada(&self, fonte: &str, tarefa: &str);
67    /// Avisado quando uma tarefa termina, com o resultado.
68    async fn tarefa_finalizada(&self, fonte: &str, resultado: &ResultadoDaTarefa);
69}
70
71/// Executa uma DAG nível a nível, com concorrência limitada dentro de cada nível.
72///
73/// A falha de uma tarefa não interrompe as demais: apenas seus descendentes são
74/// ignorados. O Dagron só fornece o grafo; a execução é nossa (ver `docs/adr/0003`).
75pub struct ExecutorDeDag {
76    concorrencia: usize,
77}
78
79impl ExecutorDeDag {
80    /// Cria o executor; `concorrencia` é o máximo de tarefas simultâneas por
81    /// nível (no mínimo 1).
82    pub fn novo(concorrencia: usize) -> Self {
83        Self {
84            concorrencia: concorrencia.max(1),
85        }
86    }
87
88    /// Executa a DAG nível a nível e devolve o relatório com o resultado de cada
89    /// tarefa. A falha de uma tarefa ignora apenas seus descendentes.
90    pub async fn executar(
91        &self,
92        dag: &DagDeIngestao,
93        contexto: &ContextoDeExecucao,
94        observador: &dyn ObservadorDeExecucao,
95    ) -> Result<RelatorioDaExecucao, ErroDeDag> {
96        let mut resultados = Vec::new();
97        let mut causa_de_ignorar: HashMap<String, String> = HashMap::new();
98
99        for nivel in dag.niveis()? {
100            let mut a_executar = Vec::new();
101            for nome in nivel {
102                match causa_de_ignorar.get(&nome) {
103                    Some(causa) => {
104                        warn!(fonte = dag.fonte(), tarefa = %nome, %causa, "tarefa ignorada");
105                        let resultado = ResultadoDaTarefa {
106                            nome,
107                            estado: EstadoDaTarefa::Ignorada {
108                                causa: causa.clone(),
109                            },
110                            duracao: Duration::ZERO,
111                        };
112                        observador.tarefa_finalizada(dag.fonte(), &resultado).await;
113                        resultados.push(resultado);
114                    }
115                    None => a_executar.push((dag.tarefa(&nome)?, nome)),
116                }
117            }
118
119            let concluidas: Vec<ResultadoDaTarefa> = stream::iter(a_executar)
120                .map(|(tarefa, nome)| async move {
121                    info!(fonte = dag.fonte(), tarefa = %nome, "tarefa iniciada");
122                    observador.tarefa_iniciada(dag.fonte(), &nome).await;
123                    let inicio = Instant::now();
124                    let estado = match tarefa.executar(contexto).await {
125                        Ok(resumo) => {
126                            info!(
127                                fonte = dag.fonte(),
128                                tarefa = %nome,
129                                partes = resumo.partes,
130                                bytes = resumo.bytes,
131                                "tarefa concluída"
132                            );
133                            EstadoDaTarefa::Concluida(resumo)
134                        }
135                        Err(erro) => {
136                            error!(fonte = dag.fonte(), tarefa = %nome, %erro, "tarefa falhou");
137                            EstadoDaTarefa::Falhou(erro.to_string())
138                        }
139                    };
140                    let resultado = ResultadoDaTarefa {
141                        nome,
142                        estado,
143                        duracao: inicio.elapsed(),
144                    };
145                    observador.tarefa_finalizada(dag.fonte(), &resultado).await;
146                    resultado
147                })
148                .buffer_unordered(self.concorrencia)
149                .collect()
150                .await;
151
152            for resultado in &concluidas {
153                if matches!(resultado.estado, EstadoDaTarefa::Falhou(_)) {
154                    for descendente in dag.descendentes(&resultado.nome)? {
155                        causa_de_ignorar
156                            .entry(descendente)
157                            .or_insert_with(|| resultado.nome.clone());
158                    }
159                }
160            }
161            resultados.extend(concluidas);
162        }
163
164        Ok(RelatorioDaExecucao {
165            fonte: dag.fonte().to_owned(),
166            resultados,
167        })
168    }
169}

281    struct TarefaFalsa {
282        nome: &'static str,
283        falha: bool,
284        ordem: Arc<Mutex<Vec<&'static str>>>,
285    }
286
287    #[async_trait]
288    impl Tarefa for TarefaFalsa {
289        fn nome(&self) -> &str {
290            self.nome
291        }
292
293        async fn executar(&self, _: &ContextoDeExecucao) -> Result<ResumoDaTarefa, ErroDeIngestao> {
294            self.ordem.lock().unwrap().push(self.nome);
295            if self.falha {
296                return Err(ErroDeIngestao::Http {
297                    url: "http://exemplo".into(),
298                    detalhe: "fora do ar".into(),
299                });
300            }
301            Ok(ResumoDaTarefa::default())
302        }
303    }

357    async fn falha_ignora_somente_os_descendentes() {
358        let cenario = Cenario::novo();
359        let dag = DagDeIngestao::construir(
360            "teste",
361            vec![
362                cenario.tarefa("quebrada", true),
363                cenario.tarefa("filha", false).depois_de(["quebrada"]),
364                cenario.tarefa("neta", false).depois_de(["filha"]),
365                cenario.tarefa("independente", false),
366            ],
367        )
368        .unwrap();
369
370        let relatorio = ExecutorDeDag::novo(2)
371            .executar(&dag, &contexto(), &ObservadorDeTeste::default())
372            .await
373            .unwrap();
374
375        assert!(!relatorio.sucesso());
376        assert!(matches!(
377            estado(&relatorio, "quebrada"),
378            EstadoDaTarefa::Falhou(_)
379        ));
380        assert!(matches!(
381            estado(&relatorio, "independente"),
382            EstadoDaTarefa::Concluida(_)
383        ));
384        for nome in ["filha", "neta"] {
385            assert_eq!(
386                estado(&relatorio, nome),
387                &EstadoDaTarefa::Ignorada {
388                    causa: "quebrada".into()
389                }
390            );
391        }
392        assert!(!cenario.ordem.lock().unwrap().contains(&"filha"));
393    }
