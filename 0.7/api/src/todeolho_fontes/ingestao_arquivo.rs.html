24const TENTATIVAS: u32 = 3;
25const ESPERA_ENTRE_TENTATIVAS: Duration = Duration::from_secs(15);
26/// Quantas vezes retomar um mesmo arquivo (reconexões) antes de desistir desta
27/// transferência. Alto porque a origem da Câmara derruba arquivos grandes com
28/// frequência; estourar o limite faz a tarefa recomeçar o arquivo do zero.
29const LIMITE_DE_RETOMADAS: u32 = 20;
30/// Espera base entre retomadas de um fluxo caído; cresce um pouco a cada falha.
31const ESPERA_ENTRE_RETOMADAS: Duration = Duration::from_secs(2);
32
33/// Baixa um arquivo grande para a camada Raw sem carregá-lo na memória.
34///
35/// Não baixa de novo o que não mudou na origem: compara a versão (ETag) com a da última
36/// ingestão concluída. Isso também permite retomar uma carga longa que foi interrompida.
37pub struct TarefaDeArquivo {
38    nome: String,
39    fonte: String,
40    recurso: String,
41    url: String,
42    espera_entre_tentativas: Duration,
43    espera_entre_retomadas: Duration,
44    aceitar_ausencia: bool,
45}
46
47struct ArquivoGravado {
48    parte: ParteRaw,
49    versao: Option<String>,
50}

53    /// `recurso` pode conter subpastas, como `votacao_secao/ano=2024/uf=AC`.
54    pub fn nova(nome: String, fonte: &str, recurso: String, url: String) -> Self {
55        Self {
56            nome,
57            fonte: fonte.to_owned(),
58            recurso,
59            url,
60            espera_entre_tentativas: ESPERA_ENTRE_TENTATIVAS,
61            espera_entre_retomadas: ESPERA_ENTRE_RETOMADAS,
62            aceitar_ausencia: false,
63        }
64    }

66    /// Um arquivo do ano corrente pode ainda não ter sido publicado. Só nesse caso
67    /// o 404 é aceito, e apenas enquanto não houver versão anterior na Raw.

73    async fn transferir(
74        &self,
75        contexto: &ContextoDeExecucao,
76        local: &LocalizacaoRaw,
77        tamanho_informado: Option<u64>,
78    ) -> Result<ArquivoGravado, ErroDeIngestao> {
79        let primeiro = contexto.http.baixar_a_partir_de(&self.url, 0, None).await?;
80        // A primeira resposta tem de começar do início: um 206 fora de posição
81        // seria um pedaço tratado como o arquivo inteiro.
82        if primeiro.inicio != 0 {
83            return Err(ErroDeIngestao::Http {
84                url: self.url.clone(),
85                detalhe: format!(
86                    "a origem respondeu um pedaço (offset {}) em vez do arquivo inteiro",
87                    primeiro.inicio
88                ),
89            });
90        }
91        let tipo_de_conteudo = primeiro.tipo_de_conteudo.clone();
92        let versao = primeiro.versao.clone();
93        // Total autoritativo: o do GET ou, se ele omitir, o que o HEAD informou.
94        // É o que a guarda final e a checagem de completude usam.
95        let total_esperado = primeiro.tamanho_total.or(tamanho_informado);
96        let chave = local.parte(1, extensao(&tipo_de_conteudo));
97
98        // O fluxo se retoma via Range quando a conexão cai, sem perder nem
99        // duplicar byte, entregando ao `gravar_fluxo` um fluxo contínuo.
100        let fluxo = FluxoRetomavel::novo(
101            contexto.http.clone(),
102            self.url.clone(),
103            primeiro,
104            self.espera_entre_retomadas,
105            total_esperado,
106        );
107        let mut leitor = LeitorComResumo::novo(Box::new(fluxo));
108
109        contexto
110            .armazenamento
111            .gravar_fluxo(&chave, &tipo_de_conteudo, &mut leitor)
112            .await?;
113
114        // Garantia final: só publicamos se o tamanho bater com o esperado.
115        if let Some(total) = total_esperado
116            && leitor.bytes != total
117        {
118            contexto.armazenamento.remover(&chave).await?;
119            return Err(ErroDeIngestao::Http {
120                url: self.url.clone(),
121                detalhe: format!(
122                    "transferência incompleta: {} de {total} bytes",
123                    leitor.bytes
124                ),
125            });
126        }
127
128        Ok(ArquivoGravado {
129            parte: ParteRaw {
130                chave,
131                url: self.url.clone(),
132                bytes: leitor.bytes,
133                sha256: hex::encode(leitor.hash.finalize()),
134                tipo_de_conteudo,
135                versao: versao.clone(),
136            },
137            versao,
138        })
139    }

142        ErroDeIngestao::Manifesto {
143            recurso: self.recurso.clone(),
144            detalhe: erro.to_string(),
145        }
146    }

155    /// O que já está atualizado na Raw não será baixado de novo, então não ocupa espaço novo.
156    async fn bytes_a_baixar(
157        &self,
158        contexto: &ContextoDeExecucao,
159    ) -> Result<Option<u64>, ErroDeIngestao> {
160        let informacoes = contexto.http.informacoes(&self.url).await?;
161        let estado =
162            estado_na_raw(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso).await?;
163        match decidir(&informacoes, estado.as_ref()) {
164            DecisaoDeIngestao::Reaproveitar => Ok(Some(0)),
165            DecisaoDeIngestao::Ingerir(_) => Ok(informacoes.tamanho),
166        }
167    }
168
169    async fn executar(
170        &self,
171        contexto: &ContextoDeExecucao,
172    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
173        let local = LocalizacaoRaw::nova(
174            &self.fonte,
175            &self.recurso,
176            contexto.iniciada_em,
177            &contexto.id_execucao,
178        );
179        let informacoes = contexto.http.informacoes(&self.url).await?;
180        let estado =
181            estado_na_raw(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso).await?;
182        let motivo = match decidir(&informacoes, estado.as_ref()) {
183            DecisaoDeIngestao::Reaproveitar => {
184                return Ok(ResumoDaTarefa {
185                    reaproveitada: true,
186                    ..ResumoDaTarefa::default()
187                });
188            }
189            DecisaoDeIngestao::Ingerir(motivo) => motivo,
190        };
191
192        // A conexão pode cair no meio de um arquivo grande. Nada fica visível na Raw
193        // antes do fim da transferência, então basta recomeçar.
194        let mut falhas = 0;
195        let arquivo = loop {
196            match self.transferir(contexto, &local, informacoes.tamanho).await {
197                Ok(arquivo) => break arquivo,
198                Err(erro) => {
199                    if matches!(erro, ErroDeIngestao::NaoEncontrado { .. }) {
200                        if self.aceitar_ausencia && estado.is_none() {
201                            warn!(tarefa = %self.nome, "arquivo do ano corrente ainda não publicado; tarefa ignorada");
202                            return Ok(ResumoDaTarefa::default());
203                        }
204                        return Err(erro);
205                    }
206                    if matches!(erro, ErroDeIngestao::Autenticacao { .. }) {
207                        return Err(erro);
208                    }
209                    falhas += 1;
210                    if falhas >= TENTATIVAS {
211                        return Err(erro);
212                    }
213                    warn!(tarefa = %self.nome, %erro, "transferência falhou; recomeçando o arquivo");
214                    tokio::time::sleep(self.espera_entre_tentativas).await;
215                }
216            }
217        };
218
219        let resumo = ResumoDaTarefa {
220            partes: 1,
221            bytes: arquivo.parte.bytes,
222            reaproveitada: false,
223            motivo: Some(motivo),
224        };
225
226        let manifesto = ManifestoRaw {
227            fonte: self.fonte.clone(),
228            recurso: self.recurso.clone(),
229            id_execucao: contexto.id_execucao.clone(),
230            url_inicial: self.url.clone(),
231            iniciada_em: contexto.iniciada_em,
232            concluida_em: Utc::now(),
233            partes: vec![arquivo.parte],
234        };
235        gravar_json(
236            contexto,
237            local.manifesto(),
238            serde_json::to_vec_pretty(&manifesto).map_err(|e| self.erro_de_manifesto(e))?,
239        )
240        .await?;
241
242        // Grava o ponteiro mesmo sem validador (versao `None`): é o que permite
243        // a pré-checagem por tamanho reaproveitar numa próxima rodada. A
244        // publicação não deixa uma rodada mais antiga sobrescrever uma mais nova
245        // (ver issue #16).
246        let ultima = UltimaVersao {
247            versao: arquivo.versao,
248            manifesto: local.manifesto(),
249            iniciada_em: Some(contexto.iniciada_em),
250        };
251        let publicou = publicar_ponteiro(
252            contexto.armazenamento.as_ref(),
253            &self.fonte,
254            &self.recurso,
255            &ultima,
256        )
257        .await?;
258        if !publicou {
259            warn!(
260                tarefa = %self.nome,
261                "outra rodada já publicou um ponteiro mais novo; este manifesto não é o vigente"
262            );
263        }
264
265        Ok(resumo)
266    }
267}

284/// Calcula tamanho e hash enquanto os bytes passam.
285struct LeitorComResumo {
286    fluxo: FluxoDeBytes,
287    hash: Sha256,
288    bytes: u64,
289}
290
291impl LeitorComResumo {
292    fn novo(fluxo: FluxoDeBytes) -> Self {
293        Self {
294            fluxo,
295            hash: Sha256::new(),
296            bytes: 0,
297        }
298    }
299}
300
301impl AsyncRead for LeitorComResumo {
302    fn poll_read(
303        self: Pin<&mut Self>,
304        contexto: &mut Context<'_>,
305        destino: &mut ReadBuf<'_>,
306    ) -> Poll<io::Result<()>> {
307        let este = self.get_mut();
308        let antes = destino.filled().len();
309        let resultado = Pin::new(&mut este.fluxo).poll_read(contexto, destino);
310        if let Poll::Ready(Ok(())) = resultado {
311            let novos = &destino.filled()[antes..];
312            este.hash.update(novos);
313            este.bytes += novos.len() as u64;
314        }
315        resultado
316    }
317}

319/// Fluxo de download que se **retoma** via Range quando a conexão cai, sem perder
320/// nem duplicar byte: ao cair, pede `bytes=<entregue>-` e continua exatamente do
321/// ponto certo, entregando ao `gravar_fluxo` um fluxo contínuo. Se a origem não
322/// honrar a retomada (responder `200`, começar noutro offset, ou mudar o ETag/tamanho),
323/// devolve erro — e a tarefa recomeça o arquivo do zero, nunca emenda versões diferentes.
324struct FluxoRetomavel {
325    http: Arc<dyn ClienteHttp>,
326    url: String,
327    /// Validador (ETag) da primeira resposta; exigido igual em cada retomada.
328    versao: Option<String>,
329    /// Tamanho total anunciado pela origem, quando conhecido.
330    tamanho_total: Option<u64>,
331    /// Bytes já entregues ao chamador — é o offset exato de uma retomada.
332    entregue: u64,
333    /// Quantas reconexões já foram feitas nesta transferência.
334    retomadas: u32,
335    /// Fluxo atualmente em leitura.
336    fluxo: FluxoDeBytes,
337    /// Espera base entre retomadas.
338    espera: Duration,
339    /// Reconexão em andamento (buscando o próximo pedaço via Range).
340    reconectando: Option<Reconexao>,
341}
342
343/// Uma reconexão em andamento: busca o próximo pedaço do download via Range.
344type Reconexao = Pin<Box<dyn Future<Output = Result<DownloadHttp, ErroDeIngestao>> + Send>>;
345
346/// Um ETag fraco (`W/"..."`) não é validador forte: a origem não o aceita no
347/// `If-Range` e ele não prova identidade byte a byte, então não serve para retomar.
348fn validador_forte(etag: &str) -> bool {
349    !etag.trim_start().starts_with("W/")
350}
351
352impl FluxoRetomavel {
353    fn novo(
354        http: Arc<dyn ClienteHttp>,
355        url: String,
356        primeiro: DownloadHttp,
357        espera: Duration,
358        tamanho_total: Option<u64>,
359    ) -> Self {
360        Self {
361            http,
362            url,
363            // Só um ETag forte serve de validador para retomar; sem ele, uma queda
364            // recomeça o arquivo do zero (nunca emendamos sem garantia de identidade).
365            versao: primeiro.versao.filter(|etag| validador_forte(etag)),
366            tamanho_total,
367            entregue: 0,
368            retomadas: 0,
369            fluxo: primeiro.fluxo,
370            espera,
371            reconectando: None,
372        }
373    }
374
375    /// Agenda uma reconexão do offset já entregue. Erra se estourar o limite.
376    fn agendar_reconexao(&mut self) -> io::Result<()> {
377        // Sem validador forte não dá para garantir que a origem serviria o MESMO
378        // arquivo numa nova conexão: retomar poderia emendar versões diferentes do
379        // mesmo tamanho. Falha, e a tarefa recomeça o arquivo do zero.
380        if self.versao.is_none() {
381            return Err(io::Error::other(
382                "origem sem validador forte (ETag): retomada insegura, recomeçando do zero",
383            ));
384        }
385        self.retomadas += 1;
386        if self.retomadas > LIMITE_DE_RETOMADAS {
387            return Err(io::Error::other(format!(
388                "excedeu {LIMITE_DE_RETOMADAS} retomadas do download"
389            )));
390        }
391        warn!(
392            url = %self.url,
393            entregue = self.entregue,
394            "conexão caiu; retomando o download do offset exato"
395        );
396        let http = self.http.clone();
397        let url = self.url.clone();
398        let inicio = self.entregue;
399        let versao = self.versao.clone();
400        let espera = self.espera.saturating_mul(self.retomadas.min(5));
401        self.reconectando = Some(Box::pin(async move {
402            tokio::time::sleep(espera).await;
403            http.baixar_a_partir_de(&url, inicio, versao.as_deref())
404                .await
405        }));
406        Ok(())
407    }
408}

410impl AsyncRead for FluxoRetomavel {
411    fn poll_read(
412        self: Pin<&mut Self>,
413        contexto: &mut Context<'_>,
414        destino: &mut ReadBuf<'_>,
415    ) -> Poll<io::Result<()>> {
416        let este = self.get_mut();
417        loop {
418            // Reconexão em andamento: espera o novo fluxo e valida a integridade.
419            if let Some(futuro) = este.reconectando.as_mut() {
420                match futuro.as_mut().poll(contexto) {
421                    Poll::Pending => return Poll::Pending,
422                    Poll::Ready(Ok(download)) => {
423                        este.reconectando = None;
424                        // A origem precisa recomeçar EXATAMENTE onde paramos.
425                        if download.inicio != este.entregue {
426                            return Poll::Ready(Err(io::Error::other(format!(
427                                "a origem não retomou no offset exato: pedi {}, veio {}",
428                                este.entregue, download.inicio
429                            ))));
430                        }
431                        // E precisa ser o MESMO arquivo (ETag/tamanho inalterados).
432                        if download.versao != este.versao {
433                            return Poll::Ready(Err(io::Error::other(
434                                "o arquivo mudou na origem durante o download (ETag divergente)",
435                            )));
436                        }
437                        // Se não conhecíamos o total, adota o que a retomada informa
438                        // (passa a valer para a checagem de EOF e a guarda final).
439                        if este.tamanho_total.is_none() {
440                            este.tamanho_total = download.tamanho_total;
441                        }
442                        if let (Some(antes), Some(agora)) =
443                            (este.tamanho_total, download.tamanho_total)
444                            && antes != agora
445                        {
446                            return Poll::Ready(Err(io::Error::other(
447                                "o tamanho do arquivo mudou na origem durante o download",
448                            )));
449                        }
450                        este.fluxo = download.fluxo;
451                    }
452                    Poll::Ready(Err(_)) => {
453                        // Falha ao reconectar: tenta de novo do mesmo offset (até o
454                        // limite) antes de desistir e recomeçar o arquivo do zero.
455                        este.reconectando = None;
456                        este.agendar_reconexao()?;
457                    }
458                }
459            }
460
461            let antes = destino.filled().len();
462            match Pin::new(&mut este.fluxo).poll_read(contexto, destino) {
463                Poll::Pending => return Poll::Pending,
464                Poll::Ready(Ok(())) => {
465                    let lidos = destino.filled().len() - antes;
466                    if lidos > 0 {
467                        este.entregue += lidos as u64;
468                        return Poll::Ready(Ok(()));
469                    }
470                    // Fim do fluxo atual: só é o fim de verdade se entregamos tudo.
471                    match este.tamanho_total {
472                        Some(total) if este.entregue < total => este.agendar_reconexao()?,
473                        _ => return Poll::Ready(Ok(())),
474                    }
475                }
476                Poll::Ready(Err(_)) => {
477                    // Erro de transporte após já termos entregue tudo que a origem
478                    // anunciou não é perda: a transferência está completa.
479                    if let Some(total) = este.tamanho_total
480                        && este.entregue >= total
481                    {
482                        return Poll::Ready(Ok(()));
483                    }
484                    este.agendar_reconexao()?;
485                }
486            }
487        }
488    }
489}

536            chave: &str,
537            _: &str,

541            if *self.limitar_fluxo.lock().unwrap() {
542                (&mut *fluxo)
543                    .take(5)
544                    .read_to_end(&mut conteudo)

546                    .map_err(|erro| ErroDeIngestao::Armazenamento {
547                        chave: chave.to_owned(),
548                        detalhe: erro.to_string(),
549                    })?;
550            } else {
551                fluxo.read_to_end(&mut conteudo).await.map_err(|erro| {
552                    ErroDeIngestao::Armazenamento {
553                        chave: chave.to_owned(),
554                        detalhe: erro.to_string(),
555                    }
556                })?;
557            }
558            self.objetos
559                .lock()
560                .unwrap()
561                .insert(chave.to_owned(), conteudo);
562            Ok(())
563        }

566    /// HTTP falso que serve `CONTEUDO` por Range e pode cortar a conexão no meio,
567    /// simulando as quedas da origem nos arquivos grandes.
568    ///
569    /// `cortes` é consumido a cada download: `Some(n)` serve só `n` bytes a partir
570    /// do offset pedido e depois **cai**; `None` (ou fila vazia) serve até o fim.
571    /// Registra em `pedidos` cada offset solicitado, para os testes conferirem que
572    /// a retomada começa no ponto exato — sem perder nem duplicar byte.
573    struct HttpFalso {
574        conteudo: Vec<u8>,
575        versao: Mutex<Option<String>>,
576        /// ETag devolvido numa retomada (`inicio > 0`): simula mudança no meio.
577        versao_na_retomada: Mutex<Option<String>>,
578        /// Se `true`, responde sempre do zero (ignora o `Range`).
579        ignora_range: bool,
580        /// Se `true`, toda requisição falha de imediato (origem indisponível).
581        sempre_falha: bool,
582        ausente: Mutex<bool>,
583        cortes: Mutex<VecDeque<Option<usize>>>,
584        pedidos: Mutex<Vec<u64>>,
585        downloads: Mutex<u32>,
586    }
587
588    impl HttpFalso {
589        fn novo(versao: Option<&str>) -> Self {
590            Self {
591                conteudo: CONTEUDO.to_vec(),
592                versao: Mutex::new(versao.map(str::to_owned)),
593                versao_na_retomada: Mutex::new(None),
594                ignora_range: false,
595                sempre_falha: false,
596                ausente: Mutex::new(false),
597                cortes: Mutex::new(VecDeque::new()),
598                pedidos: Mutex::new(Vec::new()),
599                downloads: Mutex::new(0),
600            }
601        }
602
603        /// A cada download, serve `n` bytes (a partir do offset) e cai a conexão.
604        fn com_cortes(self, cortes: impl IntoIterator<Item = usize>) -> Self {
605            *self.cortes.lock().unwrap() = cortes.into_iter().map(Some).collect();
606            self
607        }
608    }

610    #[async_trait]
611    impl ClienteHttp for HttpFalso {
612        async fn obter(&self, _: &str, _: &str) -> Result<RespostaHttp, ErroDeIngestao> {
613            unreachable!("a tarefa de arquivo não usa respostas em memória")
614        }
615
616        async fn informacoes(&self, _: &str) -> Result<InformacoesDoArquivo, ErroDeIngestao> {
617            if *self.ausente.lock().unwrap() {
618                return Ok(InformacoesDoArquivo::default());
619            }
620            Ok(InformacoesDoArquivo {
621                versao: self.versao.lock().unwrap().clone(),
622                tamanho: Some(self.conteudo.len() as u64),
623            })
624        }
625
626        async fn baixar(&self, url: &str) -> Result<DownloadHttp, ErroDeIngestao> {
627            self.baixar_a_partir_de(url, 0, None).await
628        }
629
630        async fn baixar_a_partir_de(
631            &self,
632            url: &str,
633            inicio: u64,
634            _versao: Option<&str>,
635        ) -> Result<DownloadHttp, ErroDeIngestao> {
636            *self.downloads.lock().unwrap() += 1;
637            self.pedidos.lock().unwrap().push(inicio);
638
639            if *self.ausente.lock().unwrap() {
640                return Err(ErroDeIngestao::NaoEncontrado {
641                    url: url.to_owned(),
642                });
643            }
644
645            if self.sempre_falha {
646                return Err(ErroDeIngestao::Http {
647                    url: url.to_owned(),
648                    detalhe: "origem indisponível".into(),
649                });
650            }
651
652            // A origem pode ignorar o Range e responder do zero.
653            let inicio_real = if self.ignora_range {
654                0
655            } else {
656                inicio as usize
657            };
658            let corte = self.cortes.lock().unwrap().pop_front().flatten();
659            let resto = &self.conteudo[inicio_real..];
660            // `Some(n)`: serve `n` bytes (limitado ao que resta) e CAI a conexão —
661            // mesmo quando `n` cobre o resto, para simular queda após entregar tudo.
662            let (dados, cai) = match corte {
663                Some(n) => (resto[..n.min(resto.len())].to_vec(), true),
664                None => (resto.to_vec(), false),
665            };
666
667            // Numa retomada pode mudar o ETag, simulando arquivo alterado no meio.
668            let versao = if inicio > 0 {
669                self.versao_na_retomada
670                    .lock()
671                    .unwrap()
672                    .clone()
673                    .or_else(|| self.versao.lock().unwrap().clone())
674            } else {
675                self.versao.lock().unwrap().clone()
676            };
677
678            Ok(DownloadHttp {
679                tipo_de_conteudo: "application/zip".into(),
680                versao,
681                inicio: inicio_real as u64,
682                tamanho_total: Some(self.conteudo.len() as u64),
683                fluxo: Box::new(FatiaComFinal { dados, pos: 0, cai }),
684            })
685        }
686    }
687
688    /// Entrega `dados` e, no fim, cai (erro, se `cai`) ou termina (EOF).
689    struct FatiaComFinal {
690        dados: Vec<u8>,
691        pos: usize,
692        cai: bool,
693    }
694
695    impl AsyncRead for FatiaComFinal {
696        fn poll_read(
697            self: Pin<&mut Self>,
698            _: &mut Context<'_>,
699            destino: &mut ReadBuf<'_>,
700        ) -> Poll<io::Result<()>> {
701            let este = self.get_mut();
702            if este.pos < este.dados.len() {
703                let n = (este.dados.len() - este.pos).min(destino.remaining());
704                destino.put_slice(&este.dados[este.pos..este.pos + n]);
705                este.pos += n;
706                return Poll::Ready(Ok(()));
707            }
708            if este.cai {
709                este.cai = false;
710                Poll::Ready(Err(io::Error::other("conexão perdida")))
711            } else {
712                Poll::Ready(Ok(()))
713            }
714        }
715    }
