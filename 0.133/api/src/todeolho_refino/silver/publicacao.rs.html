32/// Identidade e contrato da tabela produzida por uma transformação Silver.
33pub(crate) struct ConfiguracaoDePublicacao<'a> {
34    pub fonte: &'a str,
35    pub recurso: &'a str,
36    /// Recurso de saída quando a transformação materializa uma relação derivada.
37    pub recurso_destino: Option<&'a str>,
38    pub regra: &'a str,
39    pub esquema: Option<SchemaRef>,
40    pub iniciada_em: DateTime<Utc>,
41    pub id_execucao: &'a str,
42}

210async fn publicar_transformacao_sql_interno(
211    bronze: &dyn ArmazenamentoRaw,
212    silver: &dyn ArmazenamentoRaw,
213    snapshot: &SnapshotBronze,
214    configuracao: ConfiguracaoDePublicacao<'_>,
215    sql: &str,
216    executor: &ConfiguracaoExecutorSql,
217    opcoes: OpcoesDeTransformacao<'_>,
218) -> Result<ResumoDaTarefa, ErroDeIngestao> {
219    let ConfiguracaoDePublicacao {
220        fonte,
221        recurso,
222        recurso_destino,
223        regra,
224        esquema,
225        iniciada_em,
226        id_execucao,
227    } = configuracao;
228    let recurso_destino = recurso_destino.unwrap_or(recurso);
229    if snapshot.manifesto().fonte != opcoes.fonte_bronze.unwrap_or(fonte)
230        || snapshot.manifesto().recurso != recurso
231    {
232        return Err(erro_manifesto(
233            recurso,
234            format!("snapshot não pertence ao recurso {fonte} solicitado"),
235        ));
236    }
237    let versao_silver = format!("{}|regra={regra}", snapshot.manifesto().id_execucao);
238    if let Some(estado) = estado_na_raw(silver, fonte, recurso_destino).await?
239        && estado.completa
240        && estado.integra
241        && estado.versao.as_deref() == Some(versao_silver.as_str())
242        && (opcoes.limites_rejeicao.is_none()
243            || quarentena_disponivel(silver, fonte, recurso_destino).await?)
244    {
245        return Ok(ResumoDaTarefa {
246            reaproveitada: true,
247            ..ResumoDaTarefa::default()
248        });
249    }
250
251    let entrada = match opcoes.json_original_aninhado {
252        Some(true) => {
253            preparar_snapshot_com_json_original(bronze, snapshot, executor.limite_entrada_bytes)
254                .await
255        }
256        Some(false) => {
257            preparar_snapshot_com_json_string_original(
258                bronze,
259                snapshot,
260                executor.limite_entrada_bytes,
261            )
262            .await
263        }
264        None => preparar_snapshot(bronze, snapshot, executor.limite_entrada_bytes).await,
265    }
266    .map_err(|erro| {
267        let detalhe = match erro {
268            ErroExecutorSql::Snapshot(_) => {
269                "falha ao validar ou baixar parte do snapshot Bronze para SQL"
270            }
271            ErroExecutorSql::Configuracao(_) => "snapshot Bronze excede os limites de entrada SQL",
272            ErroExecutorSql::Parquet(_) | ErroExecutorSql::Io(_) => {
273                "falha ao preparar arquivos temporários do snapshot Bronze"
274            }
275            ErroExecutorSql::DataFusion(_) => "falha ao preparar a consulta DataFusion",
276        };
277        erro_manifesto(recurso, detalhe)
278    })?;
279    let diretorio_spill = entrada.diretorio_execucao().join("spill");
280    std::fs::create_dir(&diretorio_spill).map_err(|_| {
281        erro_manifesto(
282            recurso,
283            "falha ao preparar diretório temporário do DataFusion",
284        )
285    })?;
286    let sql = entrada
287        .expandir_placeholders_sql(sql)
288        .map_err(|erro| erro_manifesto(recurso, erro))?;
289    let mut fluxo = executar_sql_com_filtros(
290        entrada.diretorio_bronze(),
291        &diretorio_spill,
292        &sql,
293        executor,
294        opcoes.filtros_emendas,
295    )
296    .await
297    .map_err(|erro| {
298        erro_manifesto(
299            recurso,
300            format!("falha ao planejar contrato SQL Silver: {erro}"),
301        )
302    })?;
303    let esquema_saida = fluxo.schema();
304    validar_linhagem(&esquema_saida).map_err(|detalhe| erro_manifesto(recurso, detalhe))?;
305    let esquema_saida = esquema_publicado(&esquema_saida, opcoes.limites_rejeicao.is_some())
306        .map_err(|detalhe| erro_manifesto(recurso, detalhe))?;
307    if esquema
308        .as_ref()
309        .is_some_and(|esperado| !schemas_compativeis(esperado, &esquema_saida))
310    {
311        let esquema_esperado = esquema.as_ref().expect("schema checked above");
312        return Err(erro_manifesto(
313            recurso,
314            format!(
315                "schema SQL {esquema_saida:?} não corresponde ao schema declarado {esquema_esperado:?}"
316            ),
317        ));
318    }
319
320    let local = LocalizacaoRaw::nova(fonte, recurso_destino, iniciada_em, id_execucao);
321    let arquivo = tempfile::Builder::new()
322        .prefix("todeolho-silver-")
323        .suffix(".parquet")
324        .tempfile_in(entrada.diretorio_execucao())
325        .map_err(|_| erro_manifesto(recurso, "falha ao criar Parquet Silver temporário"))?;
326    let esquema_parquet = esquema.clone().unwrap_or(esquema_saida);
327    let mut escritor = EscritorParquetTipado::novo_em(
328        arquivo
329            .reopen()
330            .map_err(|_| erro_manifesto(recurso, "falha ao abrir Parquet Silver temporário"))?,
331        esquema_parquet,
332    )
333    .map_err(|_| erro_manifesto(recurso, "falha ao iniciar escritor Parquet Silver"))?;
334    let mut arquivo_diagnosticos = opcoes
335        .limites_rejeicao
336        .map(|_| {
337            tempfile::Builder::new()
338                .prefix("todeolho-rejeicoes-")
339                .suffix(".jsonl")
340                .tempfile_in(entrada.diretorio_execucao())
341        })
342        .transpose()
343        .map_err(|_| erro_manifesto(recurso, "falha ao criar arquivo de diagnósticos Silver"))?;
344    let mut rejeicoes_por_motivo = BTreeMap::new();
345    let mut conversoes_por_motivo = BTreeMap::new();
346    let mut copias_removidas = 0u64;
347    let mut linhas_saida = 0u64;
348    while let Some(lote) = fluxo.next().await {
349        let lote = lote.map_err(|erro| {
350            erro_manifesto(
351                recurso,
352                format!("falha na execução do contrato SQL Silver: {erro}"),
353            )
354        })?;
355        let lote = if let Some(limites) = opcoes.limites_rejeicao {
356            separar_rejeicoes(
357                lote,
358                regra,
359                limites,
360                &mut rejeicoes_por_motivo,
361                &mut conversoes_por_motivo,
362                &mut copias_removidas,
363                arquivo_diagnosticos
364                    .as_mut()
365                    .expect("arquivo criado")
366                    .as_file_mut(),
367            )
368            .map_err(|detalhe| erro_manifesto(recurso, detalhe))?
369        } else {
370            lote
371        };
372        let lote = match &esquema {
373            Some(esquema) => normalizar_schema_lote(lote, esquema)
374                .map_err(|detalhe| erro_manifesto(recurso, detalhe))?,
375            None => lote,
376        };
377        linhas_saida = linhas_saida
378            .checked_add(lote.num_rows() as u64)
379            .ok_or_else(|| erro_manifesto(recurso, "contagem de linhas Silver excede u64"))?;
380        escritor
381            .escrever_lote(&lote)
382            .map_err(|_| erro_manifesto(recurso, "resultado SQL diverge do schema de saída"))?;
383    }
384    drop(fluxo);
385    escritor
386        .fechar()
387        .map_err(|_| erro_manifesto(recurso, "falha ao finalizar Parquet Silver"))?;
388
389    let bytes = arquivo
390        .as_file()
391        .metadata()
392        .map_err(|_| erro_manifesto(recurso, "falha ao medir Parquet Silver"))?
393        .len();
394    let mut hash = Sha256::new();
395    let mut leitura = arquivo
396        .reopen()
397        .map_err(|_| erro_manifesto(recurso, "falha ao ler Parquet Silver temporário"))?;
398    let mut bloco = [0u8; 64 * 1024];
399    loop {
400        let lidos = leitura
401            .read(&mut bloco)
402            .map_err(|_| erro_manifesto(recurso, "falha ao calcular hash do Parquet Silver"))?;
403        if lidos == 0 {
404            break;
405        }
406        hash.update(&bloco[..lidos]);
407    }
408    let chave = local.parte(1, "parquet");
409    let mut fluxo_upload = tokio::fs::File::from_std(
410        arquivo
411            .reopen()
412            .map_err(|_| erro_manifesto(recurso, "falha ao reabrir Parquet Silver"))?,
413    );
414    silver
415        .gravar_fluxo(&chave, TIPO_PARQUET, &mut fluxo_upload)
416        .await?;
417
418    let linhas_rejeitadas = rejeicoes_por_motivo
419        .values()
420        .try_fold(0u64, |total, quantidade| total.checked_add(*quantidade))
421        .ok_or_else(|| erro_manifesto(recurso, "contagem total de rejeições excede u64"))?;
422    let mut quarentena = None;
423    if linhas_rejeitadas > 0 {
424        let diagnosticos = arquivo_diagnosticos.as_mut().expect("arquivo criado");
425        diagnosticos
426            .as_file_mut()
427            .flush()
428            .map_err(|_| erro_manifesto(recurso, "falha ao finalizar diagnósticos Silver"))?;
429        let bytes = diagnosticos
430            .as_file()
431            .metadata()
432            .map_err(|_| erro_manifesto(recurso, "falha ao medir diagnósticos Silver"))?
433            .len();
434        let mut hash = Sha256::new();
435        let mut leitura = diagnosticos
436            .reopen()
437            .map_err(|_| erro_manifesto(recurso, "falha ao ler diagnósticos Silver"))?;
438        loop {
439            let lidos = leitura.read(&mut bloco).map_err(|_| {
440                erro_manifesto(recurso, "falha ao calcular hash dos diagnósticos Silver")
441            })?;
442            if lidos == 0 {
443                break;
444            }
445            hash.update(&bloco[..lidos]);
446        }
447        let chave_quarentena =
448            format!("_quarentena/{fonte}/{recurso_destino}/execucao={id_execucao}/rejeicoes.jsonl");
449        let mut fluxo = tokio::fs::File::from_std(
450            diagnosticos
451                .reopen()
452                .map_err(|_| erro_manifesto(recurso, "falha ao reabrir diagnósticos Silver"))?,
453        );
454        silver
455            .gravar_fluxo(&chave_quarentena, "application/x-ndjson", &mut fluxo)
456            .await?;
457        quarentena = Some(serde_json::json!({
458            "chave": chave_quarentena,
459            "bytes": bytes,
460            "sha256": hex::encode(hash.finalize()),
461        }));
462    }
463
464    let guarda = bloquear_publicacao(fonte, recurso).await;
465    if !snapshot
466        .ainda_vigente(bronze, opcoes.fonte_bronze.unwrap_or(fonte), recurso)
467        .await
468        .map_err(|erro| erro_manifesto(recurso, erro))?
469    {
470        return Err(erro_manifesto(
471            recurso,
472            "ponteiro Bronze mudou durante a transformação; Silver não publicada",
473        ));
474    }
475
476    let manifesto = ManifestoRaw {
477        fonte: fonte.into(),
478        recurso: recurso_destino.into(),
479        id_execucao: id_execucao.into(),
480        url_inicial: format!("bronze/{recurso}"),
481        iniciada_em,
482        concluida_em: Utc::now(),
483        partes: vec![ParteRaw {
484            chave,
485            url: recurso.into(),
486            bytes,
487            sha256: hex::encode(hash.finalize()),
488            tipo_de_conteudo: TIPO_PARQUET.into(),
489            versao: Some(format!(
490                "{}|regra={regra}",
491                snapshot.manifesto().id_execucao
492            )),
493        }],
494    };
495    let mut manifesto_json =
496        serde_json::to_value(&manifesto).map_err(|erro| erro_manifesto(recurso, erro))?;
497    manifesto_json["qualidade"] = serde_json::json!({
498        "versao": 2,
499        "manifesto_bronze": snapshot.chave_manifesto(),
500        "regra": regra,
501        "linhas_bronze_preparadas": entrada.linhas_bronze(),
502        "linhas_saida": linhas_saida,
503        "linhas_rejeitadas": linhas_rejeitadas,
504        "rejeicoes_por_motivo": rejeicoes_por_motivo,
505        "conversoes_opcionais_por_motivo": conversoes_por_motivo,
506        "copias_removidas": copias_removidas,
507        "politica_rejeicao": if opcoes.limites_rejeicao.is_some() { "limites_absolutos" } else { "falha_total" },
508        "quarentena": quarentena,
509    });
510    silver
511        .gravar(ObjetoRaw {
512            chave: local.manifesto(),
513            conteudo: serde_json::to_vec_pretty(&manifesto_json)
514                .map_err(|erro| erro_manifesto(recurso, erro))?,
515            tipo_de_conteudo: "application/json".into(),
516        })
517        .await?;
518    let publicou = publicar_ponteiro_sob_guarda(
519        silver,
520        fonte,
521        recurso_destino,
522        &UltimaVersao {
523            versao: Some(versao_silver),
524            manifesto: local.manifesto(),
525            iniciada_em: Some(iniciada_em),
526        },
527        &guarda,
528    )
529    .await?;
530    if !publicou {
531        return Err(erro_manifesto(
532            recurso,
533            "ponteiro Silver mais recente preservado; esta execução não foi publicada",
534        ));
535    }
536    Ok(ResumoDaTarefa {
537        partes: 1,
538        bytes,
539        ..ResumoDaTarefa::default()
540    })
541}

543/// Confere a presença e o tamanho da quarentena apontada pelo manifesto vigente.
544pub(crate) async fn quarentena_disponivel(
545    silver: &dyn ArmazenamentoRaw,
546    fonte: &str,
547    recurso: &str,
548) -> Result<bool, ErroDeIngestao> {
549    let Some(ponteiro) = silver
550        .ler_controle(&LocalizacaoRaw::ultima_versao(fonte, recurso))
551        .await?
552    else {
553        return Ok(false);
554    };
555    let Ok(ponteiro) = serde_json::from_slice::<UltimaVersao>(&ponteiro) else {
556        return Ok(false);
557    };
558    let Some(manifesto) = silver.ler_controle(&ponteiro.manifesto).await? else {
559        return Ok(false);
560    };
561    let Ok(manifesto) = serde_json::from_slice::<serde_json::Value>(&manifesto) else {
562        return Ok(false);
563    };
564    if manifesto["qualidade"]["politica_rejeicao"] != "limites_absolutos" {
565        return Ok(false);
566    }
567    let quarentena = &manifesto["qualidade"]["quarentena"];
568    if quarentena.is_null() {
569        return Ok(manifesto["qualidade"]["linhas_rejeitadas"] == 0);
570    }
571    let (Some(chave), Some(bytes), Some(sha256)) = (
572        quarentena["chave"].as_str(),
573        quarentena["bytes"].as_u64(),
574        quarentena["sha256"].as_str(),
575    ) else {
576        return Ok(false);
577    };
578    if silver.tamanho(chave).await? != Some(bytes) {
579        return Ok(false);
580    }
581    let mut arquivo = tempfile::tempfile().map_err(|erro| erro_manifesto(recurso, erro))?;
582    let mut escrita = tokio::fs::File::from_std(
583        arquivo
584            .try_clone()
585            .map_err(|erro| erro_manifesto(recurso, erro))?,
586    );
587    if !silver.ler_fluxo(chave, &mut escrita).await? {
588        return Ok(false);
589    }
590    escrita
591        .flush()
592        .await
593        .map_err(|erro| erro_manifesto(recurso, erro))?;
594    drop(escrita);
595    arquivo
596        .rewind()
597        .map_err(|erro| erro_manifesto(recurso, erro))?;
598    let mut hash = Sha256::new();
599    let mut total = 0u64;
600    let mut bloco = [0u8; 64 * 1024];
601    loop {
602        let lidos = arquivo
603            .read(&mut bloco)
604            .map_err(|erro| erro_manifesto(recurso, erro))?;
605        if lidos == 0 {
606            break;
607        }
608        total = total
609            .checked_add(lidos as u64)
610            .ok_or_else(|| erro_manifesto(recurso, "tamanho da quarentena excede u64"))?;
611        hash.update(&bloco[..lidos]);
612    }
613    Ok(total == bytes && hex::encode(hash.finalize()) == sha256)
614}

616/// Compara o contrato SQL sem exigir que o planejador prove nulabilidade a partir do Parquet.
617fn schemas_compativeis(esperado: &SchemaRef, recebido: &SchemaRef) -> bool {
618    esperado.metadata() == recebido.metadata()
619        && esperado.fields().len() == recebido.fields().len()
620        && esperado
621            .fields()
622            .iter()
623            .zip(recebido.fields())
624            .all(|(esperado, recebido)| {
625                esperado.name() == recebido.name()
626                    && esperado.data_type() == recebido.data_type()
627                    && esperado.metadata() == recebido.metadata()
628            })
629}

631/// Colunas internas de diagnóstico não fazem parte do Parquet Silver publicado.
632fn esquema_publicado(recebido: &SchemaRef, com_rejeicao: bool) -> Result<SchemaRef, &'static str> {
633    if !com_rejeicao {
634        return Ok(recebido.clone());
635    }
636    let motivo = recebido
637        .field_with_name(COL_REJEICAO)
638        .map_err(|_| "contrato com rejeição deve projetar _tdoa_rejeicao_motivo")?;
639    if motivo.data_type() != &DataType::Utf8 {
640        return Err("_tdoa_rejeicao_motivo deve ser Utf8");
641    }
642    let url = recebido
643        .field_with_name(COL_URL)
644        .map_err(|_| "contrato com rejeição deve projetar _tdoa_origem_url")?;
645    if url.data_type() != &DataType::Utf8 {
646        return Err("_tdoa_origem_url deve ser Utf8");
647    }
648    if let Ok(conversoes) = recebido.field_with_name(COL_CONVERSOES)
649        && conversoes.data_type() != &DataType::Utf8
650    {
651        return Err("_tdoa_conversoes_motivos deve ser Utf8");
652    }
653    if let Ok(copia) = recebido.field_with_name(COL_COPIA)
654        && copia.data_type() != &DataType::Boolean
655    {
656        return Err("_tdoa_copia_removida deve ser Boolean");
657    }
658    let campos = recebido
659        .fields()
660        .iter()
661        .filter(|campo| {
662            campo.name() != COL_REJEICAO
663                && campo.name() != COL_URL
664                && campo.name() != COL_CONVERSOES
665                && campo.name() != COL_COPIA
666        })
667        .cloned()
668        .collect::<Vec<_>>();
669    Ok(std::sync::Arc::new(Schema::new(campos)))
670}

672fn separar_rejeicoes(
673    lote: RecordBatch,
674    regra: &str,
675    limites: &BTreeMap<String, u64>,
676    contagens: &mut BTreeMap<String, u64>,
677    conversoes: &mut BTreeMap<String, u64>,
678    copias_removidas: &mut u64,
679    diagnosticos: &mut std::fs::File,
680) -> Result<RecordBatch, String> {
681    let esquema = lote.schema();
682    let indice_motivo = esquema.index_of(COL_REJEICAO).map_err(|e| e.to_string())?;
683    let motivos = lote
684        .column(indice_motivo)
685        .as_any()
686        .downcast_ref::<StringArray>()
687        .ok_or("motivo de rejeição não é Utf8")?;
688    let motivos_conversao = match esquema.index_of(COL_CONVERSOES) {
689        Ok(indice) => Some(
690            lote.column(indice)
691                .as_any()
692                .downcast_ref::<StringArray>()
693                .ok_or("motivos de conversão não são Utf8")?,
694        ),
695        Err(_) => None,
696    };
697    let copias = match esquema.index_of(COL_COPIA) {
698        Ok(indice) => Some(
699            lote.column(indice)
700                .as_any()
701                .downcast_ref::<BooleanArray>()
702                .ok_or("indicador de cópia não é Boolean")?,
703        ),
704        Err(_) => None,
705    };
706    let texto = |nome: &str| -> Result<&StringArray, String> {
707        let indice = esquema.index_of(nome).map_err(|e| e.to_string())?;
708        lote.column(indice)
709            .as_any()
710            .downcast_ref::<StringArray>()
711            .ok_or_else(|| format!("{nome} não é Utf8"))
712    };
713    let manifestos = texto(COL_MANIFESTO)?;
714    let partes = texto(COL_PARTE)?;
715    let urls = texto(COL_URL)?;
716    let indice_ordinal = esquema.index_of(COL_ORDINAL).map_err(|e| e.to_string())?;
717    let ordinais = lote
718        .column(indice_ordinal)
719        .as_any()
720        .downcast_ref::<UInt64Array>()
721        .ok_or("ordinal não é UInt64")?;
722    let mut validas = Vec::with_capacity(lote.num_rows());
723    for linha in 0..lote.num_rows() {
724        if let Some(copias) = copias
725            && !copias.is_null(linha)
726            && copias.value(linha)
727        {
728            if !motivos.is_null(linha) {
729                return Err("linha não pode ser cópia removida e rejeição".into());
730            }
731            if manifestos.is_null(linha)
732                || partes.is_null(linha)
733                || urls.is_null(linha)
734                || ordinais.is_null(linha)
735            {
736                return Err("cópia removida não tem referência Bronze completa".into());
737            }
738            *copias_removidas = copias_removidas
739                .checked_add(1)
740                .ok_or("contagem de cópias removidas excede u64")?;
741            validas.push(false);
742            continue;
743        }
744        if motivos.is_null(linha) {
745            if let Some(motivos_conversao) = motivos_conversao
746                && !motivos_conversao.is_null(linha)
747            {
748                let nomes: Vec<String> = serde_json::from_str(motivos_conversao.value(linha))
749                    .map_err(|e| format!("motivos de conversão inválidos: {e}"))?;
750                for nome in nomes {
751                    if nome.is_empty() {
752                        return Err("motivo de conversão vazio".into());
753                    }
754                    let contagem = conversoes.entry(nome).or_default();
755                    *contagem = contagem
756                        .checked_add(1)
757                        .ok_or("contagem de conversões excede u64")?;
758                }
759            }
760            validas.push(true);
761            continue;
762        }
763        let motivo = motivos.value(linha);
764        let limite = limites
765            .get(motivo)
766            .ok_or_else(|| format!("motivo de rejeição não declarado: {motivo}"))?;
767        let contagem = contagens.entry(motivo.to_owned()).or_default();
768        *contagem = contagem
769            .checked_add(1)
770            .ok_or("contagem de rejeições excede u64")?;
771        if *contagem > *limite {
772            return Err(format!("rejeições por {motivo} excedem o limite {limite}"));
773        }
774        if manifestos.is_null(linha)
775            || partes.is_null(linha)
776            || urls.is_null(linha)
777            || ordinais.is_null(linha)
778        {
779            return Err("linha rejeitada não tem referência Bronze completa".into());
780        }
781        serde_json::to_writer(
782            &mut *diagnosticos,
783            &serde_json::json!({
784                "motivo": motivo,
785                "regra": regra,
786                "manifesto_bronze": manifestos.value(linha),
787                "parte_bronze": partes.value(linha),
788                "tabela_bronze": urls.value(linha),
789                "ordinal_bronze": ordinais.value(linha),
790            }),
791        )
792        .map_err(|e| e.to_string())?;
793        diagnosticos.write_all(b"\n").map_err(|e| e.to_string())?;
794        validas.push(false);
795    }
796    let filtrado =
797        filter_record_batch(&lote, &BooleanArray::from(validas)).map_err(|e| e.to_string())?;
798    let indices = esquema
799        .fields()
800        .iter()
801        .enumerate()
802        .filter(|(_, campo)| {
803            campo.name() != COL_REJEICAO
804                && campo.name() != COL_URL
805                && campo.name() != COL_CONVERSOES
806                && campo.name() != COL_COPIA
807        })
808        .map(|(indice, _)| indice)
809        .collect::<Vec<_>>();
810    filtrado.project(&indices).map_err(|e| e.to_string())
811}

813/// Restaura a nulabilidade declarada e impede que SQL publique nulos proibidos pelo contrato.
814fn normalizar_schema_lote(
815    lote: RecordBatch,
816    esperado: &SchemaRef,
817) -> Result<RecordBatch, &'static str> {
818    if esperado
819        .fields()
820        .iter()
821        .zip(lote.columns())
822        .any(|(campo, coluna)| !campo.is_nullable() && coluna.null_count() > 0)
823    {
824        return Err("resultado SQL contém nulos em campos obrigatórios do contrato Silver");
825    }
826    RecordBatch::try_new(esperado.clone(), lote.columns().to_vec())
827        .map_err(|_| "resultado SQL não pode usar a nulabilidade declarada pelo contrato")
828}

830fn validar_linhagem(esquema: &SchemaRef) -> Result<(), String> {
831    for (nome, tipo) in [
832        (COL_MANIFESTO, arrow::datatypes::DataType::Utf8),
833        (COL_PARTE, arrow::datatypes::DataType::Utf8),
834        (COL_ORDINAL, arrow::datatypes::DataType::UInt64),
835    ] {
836        match esquema.field_with_name(nome) {
837            Ok(campo) if campo.data_type() == &tipo => {}
838            _ => {
839                return Err(format!(
840                    "contrato SQL Silver deve projetar {nome} ({tipo:?})"
841                ));
842            }
843        }
844    }
845    Ok(())
846}
