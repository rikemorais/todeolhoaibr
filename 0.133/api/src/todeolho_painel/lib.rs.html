18const LIMITE_DE_EXECUCOES: usize = 200;
19const POR_PAGINA_PADRAO: usize = 20;
20const POR_PAGINA_MAXIMO: usize = 100;
21const PAGINA: &str = include_str!("pagina.html");
22/// A página não carrega nada externo e só conversa com o próprio painel.
23const POLITICA_DE_CONTEUDO: &str = "default-src 'none'; style-src 'unsafe-inline'; \
24     script-src 'unsafe-inline'; connect-src 'self'; base-uri 'none'; form-action 'none'";

26#[derive(Serialize)]
27struct ExecucaoVisivel {
28    situacao: SituacaoDaExecucao,
29    #[serde(flatten)]
30    registro: RegistroDeExecucao,
31}
32
33#[derive(Serialize)]
34struct RespostaPaginada {
35    dados: Vec<ExecucaoVisivel>,
36    total: usize,
37    pagina: usize,
38    por_pagina: usize,
39    total_paginas: usize,
40    ha_andamento: bool,
41}
42
43#[derive(Deserialize)]
44struct FiltroDeExecucoes {
45    situacao: Option<String>,
46    pagina: Option<usize>,
47    por_pagina: Option<usize>,
48}
49
50#[derive(Serialize)]
51struct RespostaDeErro {
52    erro: &'static str,
53}

55/// Monta as rotas do painel.
56pub fn rotas(consulta: Arc<dyn ConsultaDeExecucoes>) -> Router {
57    Router::new()
58        .route("/", get(pagina))
59        .route("/api/execucoes", get(execucoes))
60        .layer(middleware::map_response(cabecalhos_de_seguranca))
61        .with_state(consulta)
62}
63
64/// Serve o painel até o processo ser encerrado.
65pub async fn servir(
66    endereco: SocketAddr,
67    consulta: Arc<dyn ConsultaDeExecucoes>,
68) -> std::io::Result<()> {
69    let ouvinte = tokio::net::TcpListener::bind(endereco).await?;
70    info!(%endereco, "painel disponível");
71    axum::serve(ouvinte, rotas(consulta)).await
72}

74async fn pagina() -> Html<&'static str> {

78fn parse_situacao(valor: &str) -> Option<SituacaoDaExecucao> {
79    match valor {
80        "sucesso" => Some(SituacaoDaExecucao::Sucesso),
81        "com_falhas" => Some(SituacaoDaExecucao::ComFalhas),
82        "em_andamento" => Some(SituacaoDaExecucao::EmAndamento),
83        "interrompida" => Some(SituacaoDaExecucao::Interrompida),
84        _ => None,
85    }
86}

88async fn execucoes(
89    State(consulta): State<Arc<dyn ConsultaDeExecucoes>>,
90    Query(filtro): Query<FiltroDeExecucoes>,
91) -> Response {
92    match consulta.recentes(LIMITE_DE_EXECUCOES).await {
93        Ok(registros) => {
94            let agora = Utc::now();
95            let mut visiveis: Vec<ExecucaoVisivel> = registros
96                .into_iter()
97                .map(|registro| ExecucaoVisivel {
98                    situacao: registro.situacao(agora),
99                    registro,
100                })
101                .collect();
102
103            let ha_andamento = visiveis
104                .iter()
105                .any(|e| e.situacao == SituacaoDaExecucao::EmAndamento);
106
107            if let Some(ref valor) = filtro.situacao {
108                match parse_situacao(valor) {
109                    Some(situacao) => visiveis.retain(|e| e.situacao == situacao),
110                    None => {
111                        return (
112                            StatusCode::BAD_REQUEST,
113                            Json(RespostaDeErro {
114                                erro: "Valor de situação inválido.",
115                            }),
116                        )
117                            .into_response();
118                    }
119                }
120            }
121
122            let total = visiveis.len();
123            let por_pagina = filtro
124                .por_pagina
125                .unwrap_or(POR_PAGINA_PADRAO)
126                .clamp(1, POR_PAGINA_MAXIMO);
127            let total_paginas = total.div_ceil(por_pagina).max(1);
128            let pagina_num = filtro.pagina.unwrap_or(1).clamp(1, total_paginas);
129            let inicio = (pagina_num - 1) * por_pagina;
130            let dados: Vec<ExecucaoVisivel> =
131                visiveis.into_iter().skip(inicio).take(por_pagina).collect();
132
133            Json(RespostaPaginada {
134                dados,
135                total,
136                pagina: pagina_num,
137                por_pagina,
138                total_paginas,
139                ha_andamento,
140            })
141            .into_response()
142        }
143        Err(erro) => {
144            error!(%erro, "não foi possível consultar o histórico");
145            (
146                StatusCode::SERVICE_UNAVAILABLE,
147                Json(RespostaDeErro {
148                    erro: "Não foi possível consultar o armazenamento.",
149                }),
150            )
151                .into_response()
152        }
153    }
154}

156async fn cabecalhos_de_seguranca(mut resposta: Response) -> Response {
157    let cabecalhos = resposta.headers_mut();
158    cabecalhos.insert(
159        header::CONTENT_SECURITY_POLICY,
160        HeaderValue::from_static(POLITICA_DE_CONTEUDO),
161    );
162    cabecalhos.insert(
163        header::X_CONTENT_TYPE_OPTIONS,
164        HeaderValue::from_static("nosniff"),
165    );
166    cabecalhos.insert(header::CACHE_CONTROL, HeaderValue::from_static("no-store"));
167    resposta
168}

183    #[async_trait]
184    impl ConsultaDeExecucoes for ConsultaFalsa {
185        async fn recentes(&self, _: usize) -> Result<Vec<RegistroDeExecucao>, ErroDeIngestao> {
186            self.0.clone().map_err(|()| ErroDeIngestao::Armazenamento {
187                chave: "execucoes".into(),
188                detalhe: "segredo interno".into(),
189            })
190        }

193    async fn obter(consulta: ConsultaFalsa, caminho: &str) -> (StatusCode, String, Option<String>) {
194        let resposta = rotas(Arc::new(consulta))
195            .oneshot(Request::get(caminho).body(Body::empty()).unwrap())
196            .await
197            .unwrap();
198        let status = resposta.status();
199        let politica = resposta
200            .headers()
201            .get(header::CONTENT_SECURITY_POLICY)
202            .map(|valor| valor.to_str().unwrap().to_owned());
203        let corpo = to_bytes(resposta.into_body(), usize::MAX).await.unwrap();
204        (status, String::from_utf8(corpo.to_vec()).unwrap(), politica)
205    }

207    fn registro(id: &str, atraso: TimeDelta) -> RegistroDeExecucao {
208        let instante = Utc::now() - atraso;
209        RegistroDeExecucao {
210            id: id.into(),
211            camada: todeolho_nucleo::Camada::Raw,
212            iniciada_em: instante,
213            atualizada_em: instante,
214            concluida_em: None,
215            fontes: Vec::new(),
216        }
217    }
218
219    fn registro_concluido(id: &str, atraso: TimeDelta) -> RegistroDeExecucao {
220        let instante = Utc::now() - atraso;
221        RegistroDeExecucao {
222            id: id.into(),
223            camada: todeolho_nucleo::Camada::Raw,
224            iniciada_em: instante,
225            atualizada_em: instante,
226            concluida_em: Some(instante + TimeDelta::seconds(10)),
227            fontes: Vec::new(),
228        }
229    }
