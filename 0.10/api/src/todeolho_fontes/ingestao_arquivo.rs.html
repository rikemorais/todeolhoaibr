24const TENTATIVAS: u32 = 3;
25const ESPERA_ENTRE_TENTATIVAS: Duration = Duration::from_secs(15);
26/// Quantas vezes retomar um mesmo arquivo (reconexões) antes de desistir desta
27/// transferência. Alto porque a origem da Câmara derruba arquivos grandes com
28/// frequência; estourar o limite faz a tarefa recomeçar o arquivo do zero.
29const LIMITE_DE_RETOMADAS: u32 = 20;
30/// Espera base entre retomadas de um fluxo caído; cresce um pouco a cada falha.
31const ESPERA_ENTRE_RETOMADAS: Duration = Duration::from_secs(2);
32
33/// Baixa um arquivo grande para a camada Raw sem carregá-lo na memória.
34///
35/// Não baixa de novo o que não mudou na origem: compara a versão (ETag) com a da última
36/// ingestão concluída. Isso também permite retomar uma carga longa que foi interrompida.
37pub struct TarefaDeArquivo {
38    nome: String,
39    fonte: String,
40    recurso: String,
41    url: String,
42    espera_entre_tentativas: Duration,
43    espera_entre_retomadas: Duration,
44    aceitar_ausencia: bool,
45}
46
47struct ArquivoGravado {
48    parte: ParteRaw,
49    versao: Option<String>,
50}

73    async fn transferir(
74        &self,
75        contexto: &ContextoDeExecucao,
76        local: &LocalizacaoRaw,
77        tamanho_informado: Option<u64>,
78    ) -> Result<ArquivoGravado, ErroDeIngestao> {
79        let primeiro = contexto.http.baixar_a_partir_de(&self.url, 0, None).await?;
80        // A primeira resposta tem de começar do início: um 206 fora de posição
81        // seria um pedaço tratado como o arquivo inteiro.
82        if primeiro.inicio != 0 {
83            return Err(ErroDeIngestao::Http {
84                url: self.url.clone(),
85                detalhe: format!(
86                    "a origem respondeu um pedaço (offset {}) em vez do arquivo inteiro",
87                    primeiro.inicio
88                ),
89            });
90        }
91        let tipo_de_conteudo = primeiro.tipo_de_conteudo.clone();
92        let versao = primeiro.versao.clone();
93        // Total autoritativo: o do GET ou, se ele omitir, o que o HEAD informou.
94        // É o que a guarda final e a checagem de completude usam.
95        let total_esperado = primeiro.tamanho_total.or(tamanho_informado);
96        let chave = local.parte(1, extensao(&tipo_de_conteudo));
97
98        // Anuncia o total ao acompanhador de progresso, para o painel calcular o
99        // percentual enquanto o download acontece.
100        if let (Some(progresso), Some(total)) = (&contexto.progresso, total_esperado) {
101            progresso.definir_total(total);
102        }
103
104        // O fluxo se retoma via Range quando a conexão cai, sem perder nem
105        // duplicar byte, entregando ao `gravar_fluxo` um fluxo contínuo.
106        let fluxo = FluxoRetomavel::novo(
107            contexto.http.clone(),
108            self.url.clone(),
109            primeiro,
110            self.espera_entre_retomadas,
111            total_esperado,
112        );
113        let mut leitor = LeitorComResumo::novo(Box::new(fluxo), contexto.progresso.clone());
114
115        contexto
116            .armazenamento
117            .gravar_fluxo(&chave, &tipo_de_conteudo, &mut leitor)
118            .await?;
119
120        // Garantia final: só publicamos se o tamanho bater com o esperado.
121        if let Some(total) = total_esperado
122            && leitor.bytes != total
123        {
124            contexto.armazenamento.remover(&chave).await?;
125            return Err(ErroDeIngestao::Http {
126                url: self.url.clone(),
127                detalhe: format!(
128                    "transferência incompleta: {} de {total} bytes",
129                    leitor.bytes
130                ),
131            });
132        }
133
134        Ok(ArquivoGravado {
135            parte: ParteRaw {
136                chave,
137                url: self.url.clone(),
138                bytes: leitor.bytes,
139                sha256: hex::encode(leitor.hash.finalize()),
140                tipo_de_conteudo,
141                versao: versao.clone(),
142            },
143            versao,
144        })
145    }

147    fn erro_de_manifesto(&self, erro: serde_json::Error) -> ErroDeIngestao {
148        ErroDeIngestao::Manifesto {
149            recurso: self.recurso.clone(),
150            detalhe: erro.to_string(),
151        }
152    }

161    /// O que já está atualizado na Raw não será baixado de novo, então não ocupa espaço novo.
162    async fn bytes_a_baixar(
163        &self,
164        contexto: &ContextoDeExecucao,
165    ) -> Result<Option<u64>, ErroDeIngestao> {
166        let informacoes = contexto.http.informacoes(&self.url).await?;
167        let estado =
168            estado_na_raw(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso).await?;
169        match decidir(&informacoes, estado.as_ref()) {
170            DecisaoDeIngestao::Reaproveitar => Ok(Some(0)),
171            DecisaoDeIngestao::Ingerir(_) => Ok(informacoes.tamanho),
172        }
173    }
174
175    async fn executar(
176        &self,
177        contexto: &ContextoDeExecucao,
178    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
179        let local = LocalizacaoRaw::nova(
180            &self.fonte,
181            &self.recurso,
182            contexto.iniciada_em,
183            &contexto.id_execucao,
184        );
185        let informacoes = contexto.http.informacoes(&self.url).await?;
186        let estado =
187            estado_na_raw(contexto.armazenamento.as_ref(), &self.fonte, &self.recurso).await?;
188        let motivo = match decidir(&informacoes, estado.as_ref()) {
189            DecisaoDeIngestao::Reaproveitar => {
190                return Ok(ResumoDaTarefa {
191                    reaproveitada: true,
192                    ..ResumoDaTarefa::default()
193                });
194            }
195            DecisaoDeIngestao::Ingerir(motivo) => motivo,
196        };
197
198        // A conexão pode cair no meio de um arquivo grande. Nada fica visível na Raw
199        // antes do fim da transferência, então basta recomeçar.
200        let mut falhas = 0;
201        let arquivo = loop {
202            match self.transferir(contexto, &local, informacoes.tamanho).await {
203                Ok(arquivo) => break arquivo,
204                Err(erro) => {
205                    if matches!(erro, ErroDeIngestao::NaoEncontrado { .. }) {
206                        if self.aceitar_ausencia && estado.is_none() {
207                            warn!(tarefa = %self.nome, "arquivo do ano corrente ainda não publicado; tarefa ignorada");
208                            return Ok(ResumoDaTarefa::default());
209                        }
210                        return Err(erro);
211                    }
212                    if matches!(erro, ErroDeIngestao::Autenticacao { .. }) {
213                        return Err(erro);
214                    }
215                    falhas += 1;
216                    if falhas >= TENTATIVAS {
217                        return Err(erro);
218                    }
219                    warn!(tarefa = %self.nome, %erro, "transferência falhou; recomeçando o arquivo");
220                    tokio::time::sleep(self.espera_entre_tentativas).await;
221                }
222            }
223        };
224
225        let resumo = ResumoDaTarefa {
226            partes: 1,
227            bytes: arquivo.parte.bytes,
228            reaproveitada: false,
229            motivo: Some(motivo),
230        };
231
232        let manifesto = ManifestoRaw {
233            fonte: self.fonte.clone(),
234            recurso: self.recurso.clone(),
235            id_execucao: contexto.id_execucao.clone(),
236            url_inicial: self.url.clone(),
237            iniciada_em: contexto.iniciada_em,
238            concluida_em: Utc::now(),
239            partes: vec![arquivo.parte],
240        };
241        gravar_json(
242            contexto,
243            local.manifesto(),
244            serde_json::to_vec_pretty(&manifesto).map_err(|e| self.erro_de_manifesto(e))?,
245        )
246        .await?;
247
248        // Grava o ponteiro mesmo sem validador (versao `None`): é o que permite
249        // a pré-checagem por tamanho reaproveitar numa próxima rodada. A
250        // publicação não deixa uma rodada mais antiga sobrescrever uma mais nova
251        // (ver issue #16).
252        let ultima = UltimaVersao {
253            versao: arquivo.versao,
254            manifesto: local.manifesto(),
255            iniciada_em: Some(contexto.iniciada_em),
256        };
257        let publicou = publicar_ponteiro(
258            contexto.armazenamento.as_ref(),
259            &self.fonte,
260            &self.recurso,
261            &ultima,
262        )
263        .await?;
264        if !publicou {
265            warn!(
266                tarefa = %self.nome,
267                "outra rodada já publicou um ponteiro mais novo; este manifesto não é o vigente"
268            );
269        }
270
271        Ok(resumo)
272    }
273}

290/// Calcula tamanho e hash enquanto os bytes passam, e reporta o avanço ao
291/// acompanhador de progresso, quando há um.
292struct LeitorComResumo {
293    fluxo: FluxoDeBytes,
294    hash: Sha256,
295    bytes: u64,
296    progresso: Option<Arc<ProgressoDaTarefa>>,
297}
298
299impl LeitorComResumo {
300    fn novo(fluxo: FluxoDeBytes, progresso: Option<Arc<ProgressoDaTarefa>>) -> Self {
301        Self {
302            fluxo,
303            hash: Sha256::new(),
304            bytes: 0,
305            progresso,
306        }
307    }
308}
309
310impl AsyncRead for LeitorComResumo {
311    fn poll_read(
312        self: Pin<&mut Self>,
313        contexto: &mut Context<'_>,
314        destino: &mut ReadBuf<'_>,
315    ) -> Poll<io::Result<()>> {
316        let este = self.get_mut();
317        let antes = destino.filled().len();
318        let resultado = Pin::new(&mut este.fluxo).poll_read(contexto, destino);
319        if let Poll::Ready(Ok(())) = resultado {
320            let novos = &destino.filled()[antes..];
321            este.hash.update(novos);
322            este.bytes += novos.len() as u64;
323            if let Some(progresso) = &este.progresso {
324                progresso.avancar(novos.len() as u64);
325            }
326        }
327        resultado
328    }
329}

331/// Fluxo de download que se **retoma** via Range quando a conexão cai, sem perder
332/// nem duplicar byte: ao cair, pede `bytes=<entregue>-` e continua exatamente do
333/// ponto certo, entregando ao `gravar_fluxo` um fluxo contínuo. Se a origem não
334/// honrar a retomada (responder `200`, começar noutro offset, ou mudar o ETag/tamanho),
335/// devolve erro — e a tarefa recomeça o arquivo do zero, nunca emenda versões diferentes.
336struct FluxoRetomavel {
337    http: Arc<dyn ClienteHttp>,
338    url: String,
339    /// Validador (ETag) da primeira resposta; exigido igual em cada retomada.
340    versao: Option<String>,
341    /// Tamanho total anunciado pela origem, quando conhecido.
342    tamanho_total: Option<u64>,
343    /// Bytes já entregues ao chamador — é o offset exato de uma retomada.
344    entregue: u64,
345    /// Quantas reconexões já foram feitas nesta transferência.
346    retomadas: u32,
347    /// Fluxo atualmente em leitura.
348    fluxo: FluxoDeBytes,
349    /// Espera base entre retomadas.
350    espera: Duration,
351    /// Reconexão em andamento (buscando o próximo pedaço via Range).
352    reconectando: Option<Reconexao>,
353}
354
355/// Uma reconexão em andamento: busca o próximo pedaço do download via Range.
356type Reconexao = Pin<Box<dyn Future<Output = Result<DownloadHttp, ErroDeIngestao>> + Send>>;
357
358/// Um ETag fraco (`W/"..."`) não é validador forte: a origem não o aceita no
359/// `If-Range` e ele não prova identidade byte a byte, então não serve para retomar.
360fn validador_forte(etag: &str) -> bool {
361    !etag.trim_start().starts_with("W/")
362}
363
364impl FluxoRetomavel {
365    fn novo(
366        http: Arc<dyn ClienteHttp>,
367        url: String,
368        primeiro: DownloadHttp,
369        espera: Duration,
370        tamanho_total: Option<u64>,
371    ) -> Self {
372        Self {
373            http,
374            url,
375            // Só um ETag forte serve de validador para retomar; sem ele, uma queda
376            // recomeça o arquivo do zero (nunca emendamos sem garantia de identidade).
377            versao: primeiro.versao.filter(|etag| validador_forte(etag)),
378            tamanho_total,
379            entregue: 0,
380            retomadas: 0,
381            fluxo: primeiro.fluxo,
382            espera,
383            reconectando: None,
384        }
385    }
386
387    /// Agenda uma reconexão do offset já entregue. Erra se estourar o limite.
388    fn agendar_reconexao(&mut self) -> io::Result<()> {
389        // Sem validador forte não dá para garantir que a origem serviria o MESMO
390        // arquivo numa nova conexão: retomar poderia emendar versões diferentes do
391        // mesmo tamanho. Falha, e a tarefa recomeça o arquivo do zero.
392        if self.versao.is_none() {
393            return Err(io::Error::other(
394                "origem sem validador forte (ETag): retomada insegura, recomeçando do zero",
395            ));
396        }
397        self.retomadas += 1;
398        if self.retomadas > LIMITE_DE_RETOMADAS {
399            return Err(io::Error::other(format!(
400                "excedeu {LIMITE_DE_RETOMADAS} retomadas do download"
401            )));
402        }
403        warn!(
404            url = %self.url,
405            entregue = self.entregue,
406            "conexão caiu; retomando o download do offset exato"
407        );
408        let http = self.http.clone();
409        let url = self.url.clone();
410        let inicio = self.entregue;
411        let versao = self.versao.clone();
412        let espera = self.espera.saturating_mul(self.retomadas.min(5));
413        self.reconectando = Some(Box::pin(async move {
414            tokio::time::sleep(espera).await;
415            http.baixar_a_partir_de(&url, inicio, versao.as_deref())
416                .await
417        }));
418        Ok(())
419    }
420}

422impl AsyncRead for FluxoRetomavel {
423    fn poll_read(
424        self: Pin<&mut Self>,
425        contexto: &mut Context<'_>,
426        destino: &mut ReadBuf<'_>,
427    ) -> Poll<io::Result<()>> {
428        let este = self.get_mut();
429        loop {
430            // Reconexão em andamento: espera o novo fluxo e valida a integridade.
431            if let Some(futuro) = este.reconectando.as_mut() {
432                match futuro.as_mut().poll(contexto) {
433                    Poll::Pending => return Poll::Pending,
434                    Poll::Ready(Ok(download)) => {
435                        este.reconectando = None;
436                        // A origem precisa recomeçar EXATAMENTE onde paramos.
437                        if download.inicio != este.entregue {
438                            return Poll::Ready(Err(io::Error::other(format!(
439                                "a origem não retomou no offset exato: pedi {}, veio {}",
440                                este.entregue, download.inicio
441                            ))));
442                        }
443                        // E precisa ser o MESMO arquivo (ETag/tamanho inalterados).
444                        if download.versao != este.versao {
445                            return Poll::Ready(Err(io::Error::other(
446                                "o arquivo mudou na origem durante o download (ETag divergente)",
447                            )));
448                        }
449                        // Se não conhecíamos o total, adota o que a retomada informa
450                        // (passa a valer para a checagem de EOF e a guarda final).
451                        if este.tamanho_total.is_none() {
452                            este.tamanho_total = download.tamanho_total;
453                        }
454                        if let (Some(antes), Some(agora)) =
455                            (este.tamanho_total, download.tamanho_total)
456                            && antes != agora
457                        {
458                            return Poll::Ready(Err(io::Error::other(
459                                "o tamanho do arquivo mudou na origem durante o download",
460                            )));
461                        }
462                        este.fluxo = download.fluxo;
463                    }
464                    Poll::Ready(Err(_)) => {
465                        // Falha ao reconectar: tenta de novo do mesmo offset (até o
466                        // limite) antes de desistir e recomeçar o arquivo do zero.
467                        este.reconectando = None;
468                        este.agendar_reconexao()?;
469                    }
470                }
471            }
472
473            let antes = destino.filled().len();
474            match Pin::new(&mut este.fluxo).poll_read(contexto, destino) {
475                Poll::Pending => return Poll::Pending,
476                Poll::Ready(Ok(())) => {
477                    let lidos = destino.filled().len() - antes;
478                    if lidos > 0 {
479                        este.entregue += lidos as u64;
480                        return Poll::Ready(Ok(()));
481                    }
482                    // Fim do fluxo atual: só é o fim de verdade se entregamos tudo.
483                    match este.tamanho_total {
484                        Some(total) if este.entregue < total => este.agendar_reconexao()?,
485                        _ => return Poll::Ready(Ok(())),
486                    }
487                }
488                Poll::Ready(Err(_)) => {
489                    // Erro de transporte após já termos entregue tudo que a origem
490                    // anunciou não é perda: a transferência está completa.
491                    if let Some(total) = este.tamanho_total
492                        && este.entregue >= total
493                    {
494                        return Poll::Ready(Ok(()));
495                    }
496                    este.agendar_reconexao()?;
497                }
498            }
499        }
500    }
501}

578    /// HTTP falso que serve `CONTEUDO` por Range e pode cortar a conexão no meio,
579    /// simulando as quedas da origem nos arquivos grandes.
580    ///
581    /// `cortes` é consumido a cada download: `Some(n)` serve só `n` bytes a partir
582    /// do offset pedido e depois **cai**; `None` (ou fila vazia) serve até o fim.
583    /// Registra em `pedidos` cada offset solicitado, para os testes conferirem que
584    /// a retomada começa no ponto exato — sem perder nem duplicar byte.
585    struct HttpFalso {
586        conteudo: Vec<u8>,
587        versao: Mutex<Option<String>>,
588        /// ETag devolvido numa retomada (`inicio > 0`): simula mudança no meio.
589        versao_na_retomada: Mutex<Option<String>>,
590        /// Se `true`, responde sempre do zero (ignora o `Range`).
591        ignora_range: bool,
592        /// Se `true`, toda requisição falha de imediato (origem indisponível).
593        sempre_falha: bool,
594        ausente: Mutex<bool>,
595        cortes: Mutex<VecDeque<Option<usize>>>,
596        pedidos: Mutex<Vec<u64>>,
597        downloads: Mutex<u32>,
598    }
599
600    impl HttpFalso {
601        fn novo(versao: Option<&str>) -> Self {
602            Self {
603                conteudo: CONTEUDO.to_vec(),
604                versao: Mutex::new(versao.map(str::to_owned)),
605                versao_na_retomada: Mutex::new(None),
606                ignora_range: false,
607                sempre_falha: false,
608                ausente: Mutex::new(false),
609                cortes: Mutex::new(VecDeque::new()),
610                pedidos: Mutex::new(Vec::new()),
611                downloads: Mutex::new(0),
612            }
613        }
614
615        /// A cada download, serve `n` bytes (a partir do offset) e cai a conexão.
616        fn com_cortes(self, cortes: impl IntoIterator<Item = usize>) -> Self {
617            *self.cortes.lock().unwrap() = cortes.into_iter().map(Some).collect();
618            self
619        }
620    }

622    #[async_trait]
623    impl ClienteHttp for HttpFalso {
624        async fn obter(&self, _: &str, _: &str) -> Result<RespostaHttp, ErroDeIngestao> {
625            unreachable!("a tarefa de arquivo não usa respostas em memória")
626        }
627
628        async fn informacoes(&self, _: &str) -> Result<InformacoesDoArquivo, ErroDeIngestao> {
629            if *self.ausente.lock().unwrap() {
630                return Ok(InformacoesDoArquivo::default());
631            }
632            Ok(InformacoesDoArquivo {
633                versao: self.versao.lock().unwrap().clone(),
634                tamanho: Some(self.conteudo.len() as u64),
635            })
636        }
637
638        async fn baixar(&self, url: &str) -> Result<DownloadHttp, ErroDeIngestao> {
639            self.baixar_a_partir_de(url, 0, None).await
640        }
641
642        async fn baixar_a_partir_de(
643            &self,
644            url: &str,
645            inicio: u64,
646            _versao: Option<&str>,
647        ) -> Result<DownloadHttp, ErroDeIngestao> {
648            *self.downloads.lock().unwrap() += 1;
649            self.pedidos.lock().unwrap().push(inicio);
650
651            if *self.ausente.lock().unwrap() {
652                return Err(ErroDeIngestao::NaoEncontrado {
653                    url: url.to_owned(),
654                });
655            }
656
657            if self.sempre_falha {
658                return Err(ErroDeIngestao::Http {
659                    url: url.to_owned(),
660                    detalhe: "origem indisponível".into(),
661                });
662            }
663
664            // A origem pode ignorar o Range e responder do zero.
665            let inicio_real = if self.ignora_range {
666                0
667            } else {
668                inicio as usize
669            };
670            let corte = self.cortes.lock().unwrap().pop_front().flatten();
671            let resto = &self.conteudo[inicio_real..];
672            // `Some(n)`: serve `n` bytes (limitado ao que resta) e CAI a conexão —
673            // mesmo quando `n` cobre o resto, para simular queda após entregar tudo.
674            let (dados, cai) = match corte {
675                Some(n) => (resto[..n.min(resto.len())].to_vec(), true),
676                None => (resto.to_vec(), false),
677            };
678
679            // Numa retomada pode mudar o ETag, simulando arquivo alterado no meio.
680            let versao = if inicio > 0 {
681                self.versao_na_retomada
682                    .lock()
683                    .unwrap()
684                    .clone()
685                    .or_else(|| self.versao.lock().unwrap().clone())
686            } else {
687                self.versao.lock().unwrap().clone()
688            };
689
690            Ok(DownloadHttp {
691                tipo_de_conteudo: "application/zip".into(),
692                versao,
693                inicio: inicio_real as u64,
694                tamanho_total: Some(self.conteudo.len() as u64),
695                fluxo: Box::new(FatiaComFinal { dados, pos: 0, cai }),
696            })
697        }
698    }
699
700    /// Entrega `dados` e, no fim, cai (erro, se `cai`) ou termina (EOF).
701    struct FatiaComFinal {
702        dados: Vec<u8>,
703        pos: usize,
704        cai: bool,
705    }
706
707    impl AsyncRead for FatiaComFinal {
708        fn poll_read(
709            self: Pin<&mut Self>,
710            _: &mut Context<'_>,
711            destino: &mut ReadBuf<'_>,
712        ) -> Poll<io::Result<()>> {
713            let este = self.get_mut();
714            if este.pos < este.dados.len() {
715                let n = (este.dados.len() - este.pos).min(destino.remaining());
716                destino.put_slice(&este.dados[este.pos..este.pos + n]);
717                este.pos += n;
718                return Poll::Ready(Ok(()));
719            }
720            if este.cai {
721                este.cai = false;
722                Poll::Ready(Err(io::Error::other("conexão perdida")))
723            } else {
724                Poll::Ready(Ok(()))
725            }
726        }
727    }
