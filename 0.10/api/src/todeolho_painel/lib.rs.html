18const LIMITE_DE_EXECUCOES: usize = 50;
19const PAGINA: &str = include_str!("pagina.html");
20/// A página não carrega nada externo e só conversa com o próprio painel.
21const POLITICA_DE_CONTEUDO: &str = "default-src 'none'; style-src 'unsafe-inline'; \
22     script-src 'unsafe-inline'; connect-src 'self'; base-uri 'none'; form-action 'none'";
23
24#[derive(Serialize)]
25struct ExecucaoVisivel {
26    situacao: SituacaoDaExecucao,
27    #[serde(flatten)]
28    registro: RegistroDeExecucao,
29}
30
31#[derive(Serialize)]
32struct RespostaDeErro {
33    erro: &'static str,
34}
35
36/// Monta as rotas do painel.
37pub fn rotas(consulta: Arc<dyn ConsultaDeExecucoes>) -> Router {
38    Router::new()
39        .route("/", get(pagina))
40        .route("/api/execucoes", get(execucoes))
41        .layer(middleware::map_response(cabecalhos_de_seguranca))
42        .with_state(consulta)
43}
44
45/// Serve o painel até o processo ser encerrado.
46pub async fn servir(
47    endereco: SocketAddr,
48    consulta: Arc<dyn ConsultaDeExecucoes>,
49) -> std::io::Result<()> {
50    let ouvinte = tokio::net::TcpListener::bind(endereco).await?;
51    info!(%endereco, "painel disponível");
52    axum::serve(ouvinte, rotas(consulta)).await
53}
54
55async fn pagina() -> Html<&'static str> {
56    Html(PAGINA)
57}
58
59async fn execucoes(State(consulta): State<Arc<dyn ConsultaDeExecucoes>>) -> Response {
60    match consulta.recentes(LIMITE_DE_EXECUCOES).await {
61        Ok(registros) => {
62            let agora = Utc::now();
63            let visiveis: Vec<ExecucaoVisivel> = registros
64                .into_iter()
65                .map(|registro| ExecucaoVisivel {
66                    situacao: registro.situacao(agora),
67                    registro,
68                })
69                .collect();
70            Json(visiveis).into_response()
71        }
72        Err(erro) => {
73            error!(%erro, "não foi possível consultar o histórico");
74            (
75                StatusCode::SERVICE_UNAVAILABLE,
76                Json(RespostaDeErro {
77                    erro: "Não foi possível consultar o armazenamento.",
78                }),
79            )
80                .into_response()
81        }
82    }
83}
84
85async fn cabecalhos_de_seguranca(mut resposta: Response) -> Response {
86    let cabecalhos = resposta.headers_mut();
87    cabecalhos.insert(
88        header::CONTENT_SECURITY_POLICY,
89        HeaderValue::from_static(POLITICA_DE_CONTEUDO),
90    );
91    cabecalhos.insert(
92        header::X_CONTENT_TYPE_OPTIONS,
93        HeaderValue::from_static("nosniff"),
94    );
95    cabecalhos.insert(header::CACHE_CONTROL, HeaderValue::from_static("no-store"));
96    resposta
97}

110    struct ConsultaFalsa(Result<Vec<RegistroDeExecucao>, ()>);
111
112    #[async_trait]
113    impl ConsultaDeExecucoes for ConsultaFalsa {
114        async fn recentes(&self, _: usize) -> Result<Vec<RegistroDeExecucao>, ErroDeIngestao> {
115            self.0.clone().map_err(|()| ErroDeIngestao::Armazenamento {
116                chave: "execucoes".into(),
117                detalhe: "segredo interno".into(),
118            })
119        }
120    }
121
122    async fn obter(consulta: ConsultaFalsa, caminho: &str) -> (StatusCode, String, Option<String>) {
123        let resposta = rotas(Arc::new(consulta))
124            .oneshot(Request::get(caminho).body(Body::empty()).unwrap())
125            .await
126            .unwrap();
127        let status = resposta.status();
128        let politica = resposta
129            .headers()
130            .get(header::CONTENT_SECURITY_POLICY)
131            .map(|valor| valor.to_str().unwrap().to_owned());
132        let corpo = to_bytes(resposta.into_body(), usize::MAX).await.unwrap();
133        (status, String::from_utf8(corpo.to_vec()).unwrap(), politica)
134    }
135
136    fn registro(id: &str, atraso: TimeDelta) -> RegistroDeExecucao {
137        let instante = Utc::now() - atraso;
138        RegistroDeExecucao {
139            id: id.into(),
140            iniciada_em: instante,
141            atualizada_em: instante,
142            concluida_em: None,
143            fontes: Vec::new(),
144        }
145    }

149        let (status, corpo, politica) = obter(ConsultaFalsa(Ok(Vec::new())), "/").await;

151        assert_eq!(status, StatusCode::OK);

156    #[tokio::test]
157    async fn api_calcula_a_situacao_de_cada_execucao() {
158        let consulta = ConsultaFalsa(Ok(vec![
159            registro("viva", TimeDelta::seconds(5)),
160            registro("parada", TimeDelta::minutes(10)),

162
163        let (status, corpo, _) = obter(consulta, "/api/execucoes").await;
164
165        assert_eq!(status, StatusCode::OK);
166        let execucoes: serde_json::Value = serde_json::from_str(&corpo).unwrap();
167        assert_eq!(execucoes[0]["id"], "viva");
168        assert_eq!(execucoes[0]["situacao"], "em_andamento");
169        assert_eq!(execucoes[1]["situacao"], "interrompida");
170    }
