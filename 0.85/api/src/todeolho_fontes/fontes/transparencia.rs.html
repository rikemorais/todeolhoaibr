30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31enum Programa {
32    BolsaFamilia,
33    AuxilioBrasil,
34    NovoBolsaFamilia,

38impl Programa {
39    fn slug(self) -> &'static str {
40        match self {
41            Self::BolsaFamilia => "bolsa-familia",
42            Self::AuxilioBrasil => "auxilio-brasil",
43            Self::NovoBolsaFamilia => "novo-bolsa-familia",
44            Self::Bpc => "bpc",
45        }
46    }
47
48    fn endpoint(self) -> &'static str {
49        match self {
50            Self::BolsaFamilia => "bolsa-familia-sacado-beneficiario-por-municipio",
51            Self::AuxilioBrasil => "auxilio-brasil-sacado-beneficiario-por-municipio",
52            Self::NovoBolsaFamilia => "novo-bolsa-familia-sacado-beneficiario-por-municipio",
53            Self::Bpc => "bpc-beneficiario-por-municipio",
54        }
55    }
56
57    fn competencia_inicial(self) -> (i32, u32) {
58        match self {
59            Self::BolsaFamilia => PRIMEIRA_COMPETENCIA_BOLSA,
60            Self::AuxilioBrasil => PRIMEIRA_COMPETENCIA_AUXILIO,
61            Self::NovoBolsaFamilia => PRIMEIRA_COMPETENCIA_NOVO_BOLSA,
62            Self::Bpc => PRIMEIRA_COMPETENCIA_BPC,
63        }
64    }
65
66    fn competencia_final(self) -> (i32, u32) {
67        match self {
68            Self::BolsaFamilia => ULTIMA_COMPETENCIA_BOLSA,
69            Self::AuxilioBrasil => ULTIMA_COMPETENCIA_AUXILIO,
70            Self::NovoBolsaFamilia | Self::Bpc => ultima_competencia_publicada(),
71        }
72    }
73
74    fn url(self, competencia: &str, municipio: &str, pagina: usize) -> String {
75        format!(
76            "{BASE_API}/{}?mesAno={}&codigoIbge={municipio}&pagina={pagina}",
77            self.endpoint(),
78            competencia.replace('-', "")
79        )
80    }
81}

83/// Mês de referência conservador: evita solicitar o mês corrente e o imediatamente
84/// anterior, que ainda pode estar em publicação no Portal.
85fn ultima_competencia_publicada() -> (i32, u32) {
86    let mes = Utc::now().date_naive().with_day(1).expect("dia válido");
87    let mes = mes
88        .checked_sub_months(chrono::Months::new(2))
89        .unwrap_or(mes);
90    (mes.year(), mes.month())
91}

93fn competencias(inicio: (i32, u32), fim: (i32, u32)) -> Vec<String> {
94    let mut atual = NaiveDate::from_ymd_opt(inicio.0, inicio.1, 1).expect("competência válida");
95    let limite = NaiveDate::from_ymd_opt(fim.0, fim.1, 1).expect("competência válida");
96    let mut resultado = Vec::new();
97    while atual <= limite {
98        resultado.push(format!("{}-{:02}", atual.year(), atual.month()));
99        let Some(proximo) = atual.checked_add_months(chrono::Months::new(1)) else {
100            break;
101        };
102        atual = proximo;
103    }
104    resultado
105}

119/// A API retorna um array por página, sem total ou link de próxima página. Uma página
120/// vazia encerra a sequência; o limite superior evita laços causados por erro da origem.
121struct PaginaPortal {
122    url_base: String,
123    proxima: AtomicUsize,
124}
125
126impl PaginaPortal {
127    fn nova(url_base: String) -> Self {
128        Self {
129            url_base,
130            proxima: AtomicUsize::new(2),
131        }
132    }
133}
134
135impl EstrategiaDePaginacao for PaginaPortal {
136    fn proxima_url(&self, corpo: &[u8]) -> Result<Option<String>, String> {
137        if corpo.is_empty() {
138            return Ok(None);
139        }
140        let itens: Vec<serde_json::Value> = serde_json::from_slice(corpo)
141            .map_err(|erro| format!("resposta do Portal não é um array JSON: {erro}"))?;
142        if itens.is_empty() {
143            return Ok(None);
144        }
145        let pagina = self.proxima.fetch_add(1, Ordering::Relaxed);
146        Ok(Some(
147            self.url_base
148                .replace("pagina=1", &format!("pagina={pagina}")),
149        ))
150    }
151}

153struct TarefaProgramaCompetencia {
154    nome: String,
155    programa: Programa,
156    competencia: String,
157}
158
159impl TarefaProgramaCompetencia {
160    fn nova(programa: Programa, competencia: &str) -> Self {
161        Self {
162            nome: format!("{}-{competencia}", programa.slug()),
163            programa,
164            competencia: competencia.to_owned(),
165        }
166    }
167
168    async fn municipios(
169        &self,
170        contexto: &todeolho_nucleo::ContextoDeExecucao,
171    ) -> Result<Vec<MunicipioIbge>, ErroDeIngestao> {
172        let estado = estado_na_raw(contexto.armazenamento.as_ref(), "ibge", "municipios").await?;
173        if !matches!(estado, Some(ref estado) if estado.completa && estado.integra) {
174            return Err(ErroDeIngestao::Manifesto {
175                recurso: "ibge/municipios".to_owned(),
176                detalhe: "cadastro IBGE ausente, incompleto ou corrompido na Raw; execute a fonte `ibge` antes desta fonte".to_owned(),
177            });
178        }
179        let manifesto = manifesto_vigente(contexto.armazenamento.as_ref(), "ibge", "municipios")
180            .await?
181            .ok_or_else(|| ErroDeIngestao::Manifesto {
182                recurso: "ibge/municipios".to_owned(),
183                detalhe: "cadastro IBGE ausente na Raw; execute a fonte `ibge` antes desta fonte"
184                    .to_owned(),
185            })?;
186        let mut municipios = Vec::new();
187        for parte in manifesto.partes {
188            let bytes = contexto
189                .armazenamento
190                .ler(&parte.chave)
191                .await?
192                .ok_or_else(|| ErroDeIngestao::Manifesto {
193                    recurso: "ibge/municipios".to_owned(),
194                    detalhe: format!("parte do cadastro IBGE ausente: {}", parte.chave),
195                })?;
196            let mut pagina: Vec<MunicipioIbge> =
197                serde_json::from_slice(&bytes).map_err(|erro| ErroDeIngestao::Manifesto {
198                    recurso: "ibge/municipios".to_owned(),
199                    detalhe: format!("cadastro IBGE inválido: {erro}"),
200                })?;
201            municipios.append(&mut pagina);
202        }
203        if municipios.is_empty() {
204            return Err(ErroDeIngestao::Manifesto {
205                recurso: "ibge/municipios".to_owned(),
206                detalhe: "cadastro IBGE não contém municípios".to_owned(),
207            });
208        }
209        Ok(municipios)
210    }
211}

213#[async_trait]
214impl Tarefa for TarefaProgramaCompetencia {
215    fn nome(&self) -> &str {
216        &self.nome
217    }
218
219    async fn executar(
220        &self,
221        contexto: &todeolho_nucleo::ContextoDeExecucao,
222    ) -> Result<ResumoDaTarefa, ErroDeIngestao> {
223        let municipios = self.municipios(contexto).await?;
224        let mut resumo = ResumoDaTarefa {
225            reaproveitada: true,
226            ..ResumoDaTarefa::default()
227        };
228
229        for municipio in municipios {
230            let codigo = municipio.id.to_string();
231            let recurso = recurso(self.programa, municipio.id, &self.competencia);
232            let url = self.programa.url(&self.competencia, &codigo, 1);
233            let tarefa = TarefaDeIngestaoHttp::paginada(
234                FONTE_RAW,
235                &recurso,
236                url.clone(),
237                PaginaPortal::nova(url),
238            )
239            .com_tamanho_tipico(TAMANHO_TIPICO)
240            .com_limite_de_paginas(LIMITE_PAGINAS);
241            let parcial = tarefa.executar(contexto).await?;
242            resumo.partes += parcial.partes;
243            resumo.bytes += parcial.bytes;
244            resumo.reaproveitada &= parcial.reaproveitada;
245            if resumo.motivo.is_none() {
246                resumo.motivo = parcial.motivo;
247            }
248        }
249        Ok(resumo)
250    }
251
252    async fn bytes_a_baixar(
253        &self,
254        _contexto: &todeolho_nucleo::ContextoDeExecucao,
255        _precisao: PrecisaoDaEstimativa,
256    ) -> Result<Option<u64>, ErroDeIngestao> {
257        Ok(None)
258    }
259}

261/// Benefícios sociais em partições mensais e municipais, na Raw do Portal da Transparência.
262pub struct FonteTransparenciaBeneficios;
263
264impl FonteDeDados for FonteTransparenciaBeneficios {
265    fn nome(&self) -> &str {
266        NOME
267    }
268
269    fn descricao(&self) -> &str {
270        "Portal da Transparência - Bolsa Família, Auxílio Brasil e BPC por município e competência"
271    }
272
273    fn sob_demanda(&self) -> bool {
274        true
275    }
276
277    fn perfil_http(&self) -> PerfilHttp {
278        PerfilHttp {
279            credencial: Some(CredencialHttp {
280                host: HOST_API,
281                variavel: "PORTAL_TRANSPARENCIA_TOKEN",
282                cabecalho: "chave-api-dados",
283                prefixo: "",
284            }),
285            limite: Some(LimiteHttp {
286                host: HOST_API,
287                requisicoes: 90,
288                intervalo: Duration::from_secs(60),
289            }),
290            agente: None,
291        }
292    }
293
294    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
295        let programas = [
296            Programa::BolsaFamilia,
297            Programa::AuxilioBrasil,
298            Programa::NovoBolsaFamilia,
299            Programa::Bpc,
300        ];
301        programas
302            .into_iter()
303            .flat_map(|programa| {
304                competencias(programa.competencia_inicial(), programa.competencia_final())
305                    .into_iter()
306                    .map(move |competencia| {
307                        TarefaPlanejada::nova(TarefaProgramaCompetencia::nova(
308                            programa,
309                            &competencia,
310                        ))
311                        .com_etiquetas([
312                            ("programa", programa.slug()),
313                            ("competencia", competencia.as_str()),
314                        ])
315                    })
316            })
317            .collect()
318    }
319}
