1use chrono::{Datelike, Utc};
2use todeolho_nucleo::{FonteDeDados, TarefaPlanejada};
3
4use crate::ingestao_http::TarefaDeIngestaoHttp;
5
6use super::api;
7
8const NOME: &str = "senado-parlamentares";
9const FONTE_RAW: &str = "senado";
10const BASE: &str = "https://legis.senado.leg.br/dadosabertos/senador/lista/legislatura";
11/// A 49ª legislatura (1991–1995) coincide com o início da série de votações nominais.
12const PRIMEIRA_LEGISLATURA: u16 = 49;
13
14fn legislatura_atual() -> u16 {
15    // Legislaturas do Senado têm quatro anos; a 57ª começou em 2023.
16    57 + ((Utc::now().year() - 2023) / 4) as u16
17}
18
19/// Catálogo de parlamentares por legislatura, com identificação partidária,
20/// mandatos e UF conforme publicados pela API oficial.
21pub struct FonteSenadoParlamentares;
22
23impl FonteDeDados for FonteSenadoParlamentares {
24    fn nome(&self) -> &str {
25        NOME
26    }
27
28    fn descricao(&self) -> &str {
29        "Senado Federal - senadores, partidos, UFs e mandatos por legislatura desde 1991"
30    }
31
32    fn perfil_http(&self) -> todeolho_nucleo::PerfilHttp {
33        api::perfil_http()
34    }
35
36    fn sob_demanda(&self) -> bool {
37        true
38    }
39
40    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
41        (PRIMEIRA_LEGISLATURA..=legislatura_atual())
42            .map(|legislatura| {
43                let url = format!("{BASE}/{legislatura}");
44                TarefaPlanejada::nova(TarefaDeIngestaoHttp::sem_paginacao(
45                    FONTE_RAW,
46                    &format!("parlamentares/legislatura={legislatura}"),
47                    url,
48                ))
49                .com_etiquetas([("legislatura", legislatura.to_string().as_str())])
50            })
51            .collect()
52    }
53}
