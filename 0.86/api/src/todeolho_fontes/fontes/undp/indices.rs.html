1//! IDH global e índices compostos do Human Development Report (UNDP).

3//! O CSV completo traz HDI, GDI, GII, PHDI, MPI e seus componentes por país

7use todeolho_nucleo::{FonteDeDados, TarefaPlanejada};

9use crate::ingestao_http::TarefaDeIngestaoHttp;

11use super::api;

13const NOME: &str = "undp";
14const URL_INDICES: &str = "https://hdr.undp.org/sites/default/files/2025_HDR/HDR25_Composite_indices_complete_time_series.csv";
15
16/// IDH global e índices compostos de desenvolvimento humano (UNDP/HDRO).
17///
18/// O CSV completo traz todos os índices compostos (HDI, GDI, GII, PHDI, MPI)
19/// e seus componentes por país e ano (1990-2023), gravado opaco na Raw
20/// ([ADR 0014]). O recorte por país e a normalização são do refino ([ADR 0015]).
21/// Ver issue #64.
22///
23/// Catálogo: <https://hdr.undp.org/data-center/documentation-and-downloads>
24///
25/// [ADR 0014]: ../../../docs/adr/0014-raw-opaca-e-refino-em-estagio-separado.md
26/// [ADR 0015]: ../../../docs/adr/0015-arquitetura-de-dados-em-medalhao.md
27pub struct FonteUndp;
28
29impl FonteDeDados for FonteUndp {
30    fn nome(&self) -> &str {
31        NOME
32    }
33
34    fn descricao(&self) -> &str {
35        "UNDP Human Development Report - IDH global e índices compostos por país"
36    }
37
38    fn perfil_http(&self) -> todeolho_nucleo::PerfilHttp {
39        api::perfil_http()
40    }
41
42    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
43        vec![
44            TarefaPlanejada::nova(
45                TarefaDeIngestaoHttp::sem_paginacao(
46                    NOME,
47                    "indices-compostos",
48                    URL_INDICES.to_owned(),
49                )
50                .com_tamanho_tipico(2_100_000),
51            )
52            .com_etiquetas([("recurso", "indices-compostos")]),
53        ]
54    }
55}
56
57#[cfg(test)]
58mod testes {
59    use super::*;
60
61    #[test]
62    fn cria_uma_unica_tarefa_com_recurso_e_etiqueta() {
63        let tarefas = FonteUndp.criar_tarefas();
64        assert_eq!(tarefas.len(), 1);
65        let tarefa = &tarefas[0];
66        assert_eq!(tarefa.tarefa.nome(), "indices-compostos");
67        assert_eq!(
68            tarefa.etiquetas.get("recurso").map(String::as_str),
69            Some("indices-compostos")
70        );
71    }
72
73    #[test]
74    fn a_url_aponta_para_o_csv_do_hdr_2025() {
75        assert!(URL_INDICES.starts_with("https://hdr.undp.org/"));
76        assert!(URL_INDICES.ends_with(".csv"));
77    }
78
79    #[test]
80    fn nao_exige_credencial() {
81        let perfil = FonteUndp.perfil_http();
82        assert!(perfil.credencial.is_none());
83    }
84
85    #[test]
86    fn declara_o_limite_do_host_do_undp() {
87        let limite = FonteUndp.perfil_http().limite.expect("limite do UNDP");
88        assert_eq!(limite.host, "hdr.undp.org");
89    }
