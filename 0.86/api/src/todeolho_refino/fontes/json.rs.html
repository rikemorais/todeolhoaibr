1//! Decodificador do bronze para fontes que gravam **JSON de API** no raw.
2//!
3//! Cobre os casos de "array de registros": no topo (`bcb`), sob uma chave (`value` no
4//! ipeadata, `dados` nas APIs da Câmara) ou um array de objetos (listas do IBGE, processos e
5//! votações do Senado). Cada registro vira uma linha; os campos viram colunas de **texto**,
6//! e valores aninhados (objeto/array) são guardados como **JSON compacto** — o bronze
7//! mantém a fidelidade, a interpretação é do silver.
8
9use serde_json::Value;
10
11use crate::decodificador::{Decodificador, ParteBronze};
12use crate::escritor::{ErroDeRefino, Tabela};
13
14/// Decodifica um JSON de API em uma parte Parquet, a partir do array de registros.
15pub struct DecodificadorJson {
16    nome: &'static str,
17    /// Caminho de chaves até o array de registros; vazio = array no topo.
18    caminho: &'static [&'static str],
19}
20
21impl DecodificadorJson {
22    /// Para JSON cujo array de registros está no **topo** (ex.: `bcb`).
23    pub const fn novo(nome: &'static str) -> Self {
24        Self { nome, caminho: &[] }
25    }
26
27    /// Para JSON cujo array está sob um **caminho** de chaves (ex.: `["value"]`, `["dados"]`,
28    /// ou aninhado como `["ListaParlamentarLegislatura", "Parlamentares", "Parlamentar"]`).
29    pub const fn com_caminho(nome: &'static str, caminho: &'static [&'static str]) -> Self {
30        Self { nome, caminho }
31    }
32}
33
34/// Converte um valor JSON em texto: escalares viram o texto natural; objeto/array viram
35/// JSON compacto; `null` vira ausência (nulo no Parquet).
36fn texto_de(valor: &Value) -> Option<String> {
37    match valor {
38        Value::Null => None,
39        Value::String(s) => Some(s.clone()),
40        Value::Bool(b) => Some(b.to_string()),
41        Value::Number(n) => Some(n.to_string()),
42        outro => Some(outro.to_string()),
43    }
44}
45
46impl Decodificador for DecodificadorJson {
47    fn versao_regra(&self, fonte: &str, recurso: &str) -> Option<&'static str> {
48        // A versão antiga escreveu Parquet sem colunas para listas vazias de processos do
49        // Senado. Reprocessa esses recursos uma vez para apontar a manifestos sem partes.
50        (self.nome == "processos" && fonte == "senado" && recurso.starts_with("processos/ano="))
51            .then_some("json-processos-sem-parquet-v2")
52    }
53
54    fn decodificar(
55        &self,
56        bytes: &[u8],
57        _tipo_de_conteudo: &str,
58    ) -> Result<Vec<ParteBronze>, ErroDeRefino> {
59        let raiz: Value = serde_json::from_slice(bytes)
60            .map_err(|erro| ErroDeRefino::Decodificacao(format!("JSON inválido: {erro}")))?;
61        let mut atual = Some(&raiz);
62        for chave in self.caminho {
63            atual = atual.and_then(|v| match chave.parse::<usize>() {
64                Ok(indice) if v.is_array() => v.get(indice),
65                _ => v.get(chave),
66            });
67        }
68        let Some(Value::Array(registros)) = atual else {
69            let onde = if self.caminho.is_empty() {
70                String::new()
71            } else {
72                format!(" em '{}'", self.caminho.join("."))
73            };
74            return Err(ErroDeRefino::Decodificacao(format!(
75                "esperava um array de registros{onde}"
76            )));
77        };
78
79        // Colunas: a união das chaves dos registros, na ordem de primeira aparição.
80        let mut colunas: Vec<String> = Vec::new();
81        for registro in registros {
82            if let Value::Object(mapa) = registro {
83                for chave in mapa.keys() {
84                    if !colunas.iter().any(|c| c == chave) {
85                        colunas.push(chave.clone());
86                    }
87                }
88            }
89        }
90
91        // Sem colunas (array vazio): não gera Parquet — um Parquet de 0 colunas é ilegível.
92        if colunas.is_empty() {
93            return Ok(Vec::new());
94        }
95
96        let mut tabela = Tabela::nova(colunas.clone());
97        for registro in registros {
98            let objeto = registro.as_object();
99            let linha: Vec<Option<String>> = colunas
100                .iter()
101                .map(|coluna| objeto.and_then(|mapa| mapa.get(coluna)).and_then(texto_de))
102                .collect();
103            tabela = tabela.com_linha(linha);
104        }
105
106        Ok(vec![ParteBronze::de_tabela(self.nome, &tabela)?])
107    }
108}

110#[cfg(test)]
111mod testes {
112    use super::*;
113    use crate::csv_zip::testes::ler_parte;
114
115    fn c(s: &str) -> Option<String> {
116        Some(s.to_owned())
117    }
118
119    #[test]
120    fn decodifica_array_no_topo() {
121        let json =
122            br#"[{"data":"01/06/1989","valor":"19.68"},{"data":"01/07/1989","valor":"35.90"}]"#;
123        let partes = DecodificadorJson::novo("serie")
124            .decodificar(json, "application/json")
125            .unwrap();
126        let (colunas, linhas) = ler_parte(&partes[0]);
127        assert_eq!(colunas, vec!["data", "valor"]);
128        assert_eq!(
129            linhas,
130            vec![
131                vec![c("01/06/1989"), c("19.68")],
132                vec![c("01/07/1989"), c("35.90")]
133            ]
134        );
135    }
136
137    #[test]
138    fn decodifica_array_sob_uma_chave_com_numeros_e_nulos() {
139        let json = br#"{"value":[{"a":5.78,"b":"x"},{"a":null,"b":"y","c":true}]}"#;
140        let partes = DecodificadorJson::com_caminho("v", &["value"])
141            .decodificar(json, "application/json")
142            .unwrap();
143        let (colunas, linhas) = ler_parte(&partes[0]);
144        assert_eq!(colunas, vec!["a", "b", "c"]);
145        assert_eq!(
146            linhas,
147            vec![vec![c("5.78"), c("x"), None], vec![None, c("y"), c("true")]]
148        );
149    }
150
151    #[test]
152    fn aninhados_viram_json_compacto() {
153        let json = br#"[{"id":11,"regiao":{"id":1,"sigla":"N"}}]"#;
154        let partes = DecodificadorJson::novo("estados")
155            .decodificar(json, "application/json")
156            .unwrap();
157        let (_, linhas) = ler_parte(&partes[0]);
158        assert_eq!(linhas, vec![vec![c("11"), c(r#"{"id":1,"sigla":"N"}"#)]]);
159    }
160
161    #[test]
162    fn decodifica_array_em_caminho_profundo() {
163        let json = br#"{"raiz":{"meio":{"itens":[{"a":"1"},{"a":"2"}]}}}"#;
164        let partes = DecodificadorJson::com_caminho("x", &["raiz", "meio", "itens"])
165            .decodificar(json, "application/json")
166            .unwrap();
167        let (colunas, linhas) = ler_parte(&partes[0]);
168        assert_eq!(colunas, vec!["a"]);
169        assert_eq!(linhas, vec![vec![c("1")], vec![c("2")]]);
170    }

172    #[test]
173    fn decodifica_array_por_indice_numerico() {
174        let json =
175            br#"[{"page":1,"total":2},[{"pais":"BR","valor":"100"},{"pais":"US","valor":"200"}]]"#;
176        let partes = DecodificadorJson::com_caminho("indicador", &["1"])
177            .decodificar(json, "application/json")
178            .unwrap();
179        let (colunas, linhas) = ler_parte(&partes[0]);
180        assert_eq!(colunas, vec!["pais", "valor"]);
181        assert_eq!(
182            linhas,
