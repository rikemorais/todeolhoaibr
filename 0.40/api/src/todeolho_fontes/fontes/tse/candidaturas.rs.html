1use chrono::{Datelike, Utc};
2use todeolho_nucleo::{FonteDeDados, TarefaPlanejada};
3
4use crate::ingestao_arquivo::TarefaDeArquivo;
5
6const NOME: &str = "tse-candidaturas";
7
8/// A Raw guarda os dados sob a fonte `tse` (`raw/tse/...`), embora no catálogo sejam
9/// uma fonte separada, para poderem ser `sob_demanda`.
10const FONTE_RAW: &str = "tse";
11const BASE: &str = "https://cdn.tse.jus.br/estatistica/sead/odsele";
12
13/// Primeira eleição com o registro de candidaturas publicado em dados abertos.
14const PRIMEIRA_ELEICAO_CANDIDATURAS: u16 = 1994;
15/// Primeira eleição com a declaração de bens dos candidatos publicada.
16const PRIMEIRA_ELEICAO_BENS: u16 = 2006;
17
18/// Última eleição com arquivo publicado: o maior ano par até hoje (eleições são de dois
19/// em dois anos), então um ano ímpar cai na eleição anterior.
20fn ultima_eleicao() -> u16 {
21    let ano = Utc::now().year() as u16;
22    if ano.is_multiple_of(2) { ano } else { ano - 1 }
23}
24
25/// Monta a tarefa de um ano de um conjunto. `tipo` vira etiqueta; `pasta`/`prefixo` montam
26/// a URL (`{BASE}/{pasta}/{prefixo}_{ano}.zip`).
27fn tarefa(tipo: &str, pasta: &str, prefixo: &str, ano: u16, ultima: u16) -> TarefaPlanejada {
28    let mut arquivo = TarefaDeArquivo::nova(
29        format!("{tipo}_{ano}"),
30        FONTE_RAW,
31        format!("{tipo}/ano={ano}"),
32        format!("{BASE}/{pasta}/{prefixo}_{ano}.zip"),
33    );
34    if ano == ultima {
35        // A eleição corrente pode ainda não ter o arquivo publicado no começo do ano.
36        arquivo = arquivo.aceitar_ausencia();
37    }
38    TarefaPlanejada::nova(arquivo)
39        .com_etiquetas([("ano", ano.to_string().as_str()), ("tipo", tipo)])
40}
41
42fn tarefas_ate(ultima: u16) -> Vec<TarefaPlanejada> {
43    let candidaturas = (PRIMEIRA_ELEICAO_CANDIDATURAS..=ultima)
44        .step_by(2)
45        .map(|ano| {
46            tarefa(
47                "candidaturas",
48                "consulta_cand",
49                "consulta_cand",
50                ano,
51                ultima,
52            )
53        });
54    let bens = (PRIMEIRA_ELEICAO_BENS..=ultima)
55        .step_by(2)
56        .map(|ano| tarefa("bens", "bem_candidato", "bem_candidato", ano, ultima));
57    candidaturas.chain(bens).collect()
58}

60/// Candidaturas e bens declarados dos candidatos (TSE).
61///
62/// Dois conjuntos por eleição (de dois em dois anos), nacionais, baixados como arquivos
63/// grandes e guardados opacos na Raw ([ADR 0014]): o **registro de candidaturas** (cargo,
64/// partido, UF, município — desde 1994) e os **bens declarados** de cada candidato (tipo e
65/// valor — desde 2006). O filtro é por `ano` da eleição e `tipo`; o recorte por UF e a
66/// evolução patrimonial vêm das colunas dos CSVs, então ficam para o refino ([ADR 0015]).
67/// Fonte separada da `tse` (votação por seção) para rodar sozinha. Ver issue #58.
68///
69/// Dados abertos: <https://dadosabertos.tse.jus.br/dataset/?groups=candidatos>
70///
71/// [ADR 0014]: ../../../docs/adr/0014-raw-opaca-e-refino-em-estagio-separado.md
72/// [ADR 0015]: ../../../docs/adr/0015-arquitetura-de-dados-em-medalhao.md
73pub struct FonteTseCandidaturas;
74
75impl FonteDeDados for FonteTseCandidaturas {
76    fn nome(&self) -> &str {
77        NOME
78    }
79
80    fn descricao(&self) -> &str {
81        "TSE - candidaturas (desde 1994) e bens declarados dos candidatos (desde 2006), por eleição"
82    }
83
84    /// Arquivos grandes por eleição: não devem entrar em `--todas` por acidente.
85    fn sob_demanda(&self) -> bool {
86        true
87    }
88
89    fn criar_tarefas(&self) -> Vec<TarefaPlanejada> {
90        tarefas_ate(ultima_eleicao())
91    }
92}

94#[cfg(test)]
95mod testes {
96    use std::collections::BTreeSet;
97
98    use super::*;
99
100    fn anos_do_tipo(ultima: u16, tipo: &str) -> BTreeSet<u16> {
101        tarefas_ate(ultima)
102            .iter()
103            .filter(|t| t.etiquetas.get("tipo").map(String::as_str) == Some(tipo))
104            .map(|t| t.etiquetas["ano"].parse().expect("ano numérico"))
105            .collect()
106    }
107
108    #[test]
109    fn candidaturas_cobrem_de_1994_ate_a_ultima_eleicao() {
110        assert_eq!(
111            anos_do_tipo(2026, "candidaturas"),
112            (PRIMEIRA_ELEICAO_CANDIDATURAS..=2026).step_by(2).collect()
113        );
114    }
115
116    #[test]
117    fn bens_cobrem_de_2006_ate_a_ultima_eleicao() {
118        assert_eq!(
119            anos_do_tipo(2026, "bens"),
120            (PRIMEIRA_ELEICAO_BENS..=2026).step_by(2).collect()
121        );
122    }
123
124    #[test]
125    fn nao_inclui_anos_sem_eleicao() {
126        let anos = anos_do_tipo(2026, "candidaturas");
127        assert!(anos.contains(&2022));
128        assert!(!anos.contains(&2023), "2023 não é ano de eleição");
129    }
130
131    #[test]
132    fn ano_impar_cai_na_eleicao_par_anterior() {
133        assert_eq!(
134            *anos_do_tipo(2027, "candidaturas").iter().max().unwrap(),
135            2026
136        );
137    }
138
139    #[test]
140    fn cada_tarefa_tem_ano_e_tipo_coerentes_com_o_nome() {
141        for tarefa in FonteTseCandidaturas.criar_tarefas() {
142            let nome = tarefa.tarefa.nome();
143            let ano = tarefa.etiquetas.get("ano").expect("etiqueta ano");
144            let tipo = tarefa.etiquetas.get("tipo").expect("etiqueta tipo");
145            assert_eq!(
146                nome,
147                format!("{tipo}_{ano}"),
148                "nome incoerente com ano/tipo"
149            );
150        }
151    }
152
153    #[test]
154    fn nao_entra_em_todas_por_acidente() {
155        assert!(FonteTseCandidaturas.sob_demanda());
156    }
157}
